def choose_move(observation):
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    obstacles = set(tuple(p) for p in (observation.get("obstacles", []) or []))
    resources = observation.get("resources", []) or []

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def man(a, b):
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

    legal = []
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            nx, ny = sx + dx, sy + dy
            if inb(nx, ny) and (nx, ny) not in obstacles:
                legal.append((dx, dy))

    if not legal:
        return [0, 0]

    res_cells = []
    for r in resources:
        if len(r) >= 2:
            x, y = int(r[0]), int(r[1])
            if inb(x, y) and (x, y) not in obstacles:
                res_cells.append((x, y))

    if not res_cells:
        return [0, 0]

    opp_pos = (ox, oy)
    best = None
    for dx, dy in legal:
        nx, ny = sx + dx, sy + dy
        self_here = (nx, ny)

        best_target = None
        best_target_key = (-10**9, 10**9)

        for cell in res_cells:
            sd = man(self_here, cell)
            od = man(opp_pos, cell)
            margin = od - sd  # prefer cells we can reach sooner than opponent
            key = (margin, sd)
            if key > best_target_key:
                best_target_key = key
                best_target = cell

        margin, sd = best_target_key
        # tie-break: prefer larger margin, then smaller self distance, then prefer staying diagonal towards target
        tx, ty = best_target
        step_dx = 0 if tx == nx else (1 if tx > nx else -1)
        step_dy = 0 if ty == ny else (1 if ty > ny else -1)
        alignment = (dx == step_dx) + (dy == step_dy)  # 0..2

        move_key = (margin, -sd, alignment, -abs(dx) - abs(dy))
        if best is None or move_key > best:
            best = move_key, [dx, dy]

    return best[1]
