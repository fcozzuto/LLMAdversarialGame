def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    resources = observation.get("resources", []) or []
    if not resources:
        return [0, 0]
    obstacles = set((int(p[0]), int(p[1])) for p in (observation.get("obstacles", []) or []))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def man(x1, y1, x2, y2):
        dx = x1 - x2
        if dx < 0:
            dx = -dx
        dy = y1 - y2
        if dy < 0:
            dy = -dy
        return dx + dy

    def valid(x, y):
        return inb(x, y) and (x, y) not in obstacles

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    # Pick resource to contest: minimize (self_dist - opp_dist); tie-break by smaller self_dist, then fixed ordering.
    best = None
    for rx, ry in resources:
        sd = man(sx, sy, rx, ry)
        od = man(ox, oy, rx, ry)
        key = (sd - od, sd, rx, ry)
        if best is None or key < best[0]:
            best = (key, rx, ry)
    _, tx, ty = best

    # Choose move minimizing our distance advantage toward target, with obstacle avoidance and slight tie-break.
    best_move = (10**9, None)
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not valid(nx, ny):
            continue
        ndt = man(nx, ny, tx, ty)
        odt = man(ox, oy, tx, ty)
        # prefer being closer to target than opponent (or at least reduce our disadvantage)
        adv = (ndt - odt)
        # secondary: prefer smaller distance to some contested resource (one-step look)
        sec = 0
        # check a couple of nearest resources deterministically
        for rx, ry in resources[:min(6, len(resources))]:
            sec = min(sec, man(nx, ny, rx, ry)) if sec else man(nx, ny, rx, ry)
        key = (adv, ndt, sec, dx, dy)
        if key < best_move[0]:
            best_move = (key, [dx, dy])

    # If all moves invalid (rare), stay.
    return best_move[1] if best_move[1] is not None else [0, 0]
