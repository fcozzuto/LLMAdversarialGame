def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)

    def ti(v):
        try:
            return int(v)
        except:
            return 0

    sp = observation.get("self_position", [0, 0]) or [0, 0]
    op = observation.get("opponent_position", [0, 0]) or [0, 0]
    sx, sy = ti(sp[0]), ti(sp[1])
    ox, oy = ti(op[0]), ti(op[1])

    obstacles = set()
    for p in observation.get("obstacles", []) or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = ti(p[0]), ti(p[1])
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

    resources = observation.get("resources", []) or []
    if not resources:
        return [0, 0]

    # Pick best target by who can arrive earlier; deterministic tie-breaks.
    best = None
    best_adv = None
    for r in resources:
        if not (isinstance(r, (list, tuple)) and len(r) >= 2):
            continue
        x, y = ti(r[0]), ti(r[1])
        if not (0 <= x < w and 0 <= y < h) or (x, y) in obstacles:
            continue
        sd = abs(x - sx) + abs(y - sy)
        od = abs(x - ox) + abs(y - oy)
        adv = (od - sd)
        if best is None or adv > best_adv or (adv == best_adv and (sd < best[2] or (sd == best[2] and (x + y) > (best[0] + best[1])))):
            best = (x, y)
            best_adv = adv

    if best is None:
        return [0, 0]
    tx, ty = best

    # Choose move that maximizes same advantage after stepping, while avoiding obstacles.
    moves = []
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            nx, ny = sx + dx, sy + dy
            if 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obstacles:
                moves.append((dx, dy))

    if not moves:
        return [0, 0]

    best_move = None
    best_val = None
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        sd2 = abs(tx - nx) + abs(ty - ny)
        od2 = abs(tx - ox) + abs(ty - oy)
        # Main objective: maximize arrival advantage; secondary: get closer; tertiary: push toward center.
        val = (od2 - sd2) * 1000 - sd2
        center_bias = -abs(nx - (w - 1) / 2.0) - abs(ny - (h - 1) / 2.0)
        val = val + int(center_bias * 2)
        if best_move is None or val > best_val:
            best_val = val
            best_move = (dx, dy)

    return [int(best_move[0]), int(best_move[1])]
