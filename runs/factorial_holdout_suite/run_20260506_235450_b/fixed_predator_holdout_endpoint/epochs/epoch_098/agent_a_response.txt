def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)

    def ti(v):
        try:
            return int(v)
        except:
            return 0

    sp = observation.get("self_position") or [0, 0]
    op = observation.get("opponent_position") or [0, 0]
    sx, sy = ti(sp[0]), ti(sp[1])
    ox, oy = ti(op[0]), ti(op[1])

    obstacles = set()
    for p in observation.get("obstacles") or []:
        x = y = None
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = ti(p[0]), ti(p[1])
        elif isinstance(p, dict):
            if "x" in p and "y" in p:
                x, y = ti(p["x"]), ti(p["y"])
        if x is not None and y is not None and 0 <= x < w and 0 <= y < h:
            obstacles.add((x, y))

    resources = []
    for r in observation.get("resources") or []:
        x = y = None
        if isinstance(r, (list, tuple)) and len(r) >= 2:
            x, y = ti(r[0]), ti(r[1])
        elif isinstance(r, dict):
            if "x" in r and "y" in r:
                x, y = ti(r["x"]), ti(r["y"])
        if x is not None and y is not None and 0 <= x < w and 0 <= y < h and (x, y) not in obstacles:
            resources.append((x, y))
    if not resources:
        return [0, 0]

    def cheb(ax, ay, bx, by):
        dx = ax - bx
        dy = ay - by
        return abs(dx) if abs(dx) > abs(dy) else abs(dy)

    def nearest_dist(px, py):
        md = 10**9
        for rx, ry in resources:
            d = cheb(px, py, rx, ry)
            if d < md:
                md = d
        return md

    moves = [(-1,-1),(-1,0),(-1,1),(0,-1),(0,0),(0,1),(1,-1),(1,0),(1,1)]
    best = None
    best_score = -10**18
    opp_best = nearest_dist(ox, oy)

    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not (0 <= nx < w and 0 <= ny < h) or (nx, ny) in obstacles:
            continue
        my_d = nearest_dist(nx, ny)
        my_immediate = 1 if my_d == 0 else 0
        my_one = 1 if my_d == 1 else 0
        # Assume opponent continues to pursue nearest resource; choose moves that make us faster and them slower.
        # Approximate their "effective" nearest distance by how much our move changes contested progress (heuristic).
        # If we move closer to their nearest resource, reduce opponent distance; else increase it slightly.
        opp_d = opp_best
        # Simple deterrence: if we end up closer to their current closest resource, they likely can still grab it.
        # Otherwise, we drift away from their chase path.
        target = None
        mind = 10**9
        for rx, ry in resources:
            d = cheb(ox, oy, rx, ry)
            if d < mind:
                mind = d
                target = (rx, ry)
        if target is not None:
            tx, ty = target
            my_to_target = cheb(nx, ny, tx, ty)
            opp_to_target = cheb(ox, oy, tx, ty)
            # if we get very close, opponent likely must go around/us, increasing their effective distance
            if my_to_target < opp_to_target:
                opp_d = opp_d + 1
