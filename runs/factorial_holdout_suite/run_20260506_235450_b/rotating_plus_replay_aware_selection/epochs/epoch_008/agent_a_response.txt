def choose_move(observation):
    w = observation.get("grid_width", 8)
    h = observation.get("grid_height", 8)
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    resources = observation.get("resources", []) or []
    obstacles = set(tuple(p) for p in (observation.get("obstacles", []) or []))

    def inb(x, y): return 0 <= x < w and 0 <= y < h
    def man(x1, y1, x2, y2): return abs(x1 - x2) + abs(y1 - y2)

    moves = [(-1, -1), (-1, 0), (-1, 1),
             (0, -1), (0, 0), (0, 1),
             (1, -1), (1, 0), (1, 1)]
    legal = []
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if inb(nx, ny) and (nx, ny) not in obstacles:
            legal.append((dx, dy, nx, ny))
    if not legal:
        return [0, 0]
    if not resources:
        # Go toward the center to avoid corner trapping.
        tx, ty = (w - 1) // 2, (h - 1) // 2
        best = None
        for dx, dy, nx, ny in legal:
            d = man(nx, ny, tx, ty)
            if best is None or d < best[0]:
                best = (d, dx, dy)
        return [best[1], best[2]]

    # Pick a target resource we can reach with an advantage; otherwise contest.
    best_gain = None
    best_dist = None
    best_res = None
    for rx, ry in resources:
        if (rx, ry) in obstacles:
            continue
        ds = man(sx, sy, rx, ry)
        do = man(ox, oy, rx, ry)
        gain = ds - do  # negative means we're closer
        if best_gain is None:
            best_gain, best_dist, best_res = gain, ds, (rx, ry)
            continue
        # Primary: prefer smallest ds-do; Secondary: prefer shorter ds.
        if gain < best_gain or (gain == best_gain and ds < best_dist):
            best_gain, best_dist, best_res = gain, ds, (rx, ry)

    # If opponent is clearly closer to every available resource, switch to denial: target opponent's closest.
    if best_res is None:
        return [0, 0]
    if best_gain > 0 and len(resources) > 0:
        # Contention: minimize opponent distance to the chosen resource (i.e., intercept).
        opp_best = None
        opp_dist = None
        for rx, ry in resources:
            if (rx, ry) in obstacles:
                continue
            do = man(ox, oy, rx, ry)
            if opp_dist is None or do < opp_dist or (do == opp_dist and man(sx, sy, rx, ry) < man(sx, sy, opp_best[0], opp_best[1])):
                opp_dist = do
                opp_best = (rx, ry)
        if opp_best is not None:
            best_res = opp_best

    rx, ry = best_res
    # Evaluate moves by resulting advantage on the target, with a slight safety term.
    best = None
    for dx, dy, nx, ny in legal:
        ns = man(nx, ny, rx, ry)
        no = man(ox, oy, rx, ry)  # opponent doesn't move in our evaluation
        advantage = (ns - no)
        # Also discourage stepping toward giving opponent an even better contest point:
        # compute our best "negative advantage" over all resources after move.
        best_after = None
        for rr_x, rr_y in resources:
            if (rr_x, rr_y) in obstacles:
                continue
            dss = man(nx, ny, rr_x, rr_y)
            noo = man(ox, oy, rr_x, rr_y)
            g = dss - noo
            if best_after is None or g < best_after:
                best_after = g
        safety = best_after if best_after is not None else 0
        key = (advantage, safety, ns)  # lexicographic: smallest is best
        if best is None or key < best[0]:
            best = (key, dx, dy)
    return [best[1], best[2]]

    obstacles = set()
    for p in obs_list:
        try:
            obstacles.add((p[0], p[1]))
        except Exception:
            pass

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    if not legal:
        return [0, 0]

    def dist(x1, y1, x2, y2):
        return abs(x1 - x2) + abs(y1 - y2)

            priority = 0 if ds <= do else 1
