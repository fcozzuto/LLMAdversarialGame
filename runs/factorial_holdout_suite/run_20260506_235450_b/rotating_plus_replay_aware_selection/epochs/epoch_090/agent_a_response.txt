def choose_move(observation):
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    sx, sy = observation.get("self_position", (0, 0))
    ox, oy = observation.get("opponent_position", (0, 0))
    resources = observation.get("resources", []) or []
    obstacles = set()
    for p in (observation.get("obstacles", []) or []):
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            obstacles.add((int(p[0]), int(p[1])))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def mdist(x1, y1, x2, y2):
        dx = x1 - x2
        if dx < 0: dx = -dx
        dy = y1 - y2
        if dy < 0: dy = -dy
        return dx + dy

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    legal = []
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if inb(nx, ny) and (nx, ny) not in obstacles:
            legal.append((dx, dy, nx, ny))
    if not legal:
        return [0, 0]
    if not resources:
        # Drift toward nearest corner away from opponent
        corners = [(0, 0), (0, h - 1), (w - 1, 0), (w - 1, h - 1)]
        cx, cy = max(corners, key=lambda c: (mdist(c[0], c[1], sx, sy), mdist(c[0], c[1], ox, oy)))
        best = None
        bestv = -10**18
        for dx, dy, nx, ny in legal:
            v = mdist(nx, ny, cx, cy) * -1
            if v > bestv:
                bestv = v
                best = (dx, dy)
        return [best[0], best[1]]

    # Deterministic ordering for tie-breaking
    res = sorted(((int(r[0]), int(r[1])) for r in resources), key=lambda p: (p[0], p[1]))
    moves.sort(key=lambda t: (t[2], t[3], t[0], t[1]))  # deterministic preference

    best_move = (0, 0)
    best_val = -10**18

    for dx, dy, nx, ny in legal:
        # If stepping onto a resource, highly prefer it
        on_resource = 1 if (nx, ny) in set(res) else 0
        worst_opponent_penalty = 0
        best_resource_val = -10**18
        for rx, ry in res:
            ds = mdist(nx, ny, rx, ry)
            do = mdist(ox, oy, rx, ry)
            # Positive if we are earlier; negative if opponent can beat us
            # Also prefer closer targets overall.
            lead = do - ds  # positive means we arrive earlier
            v = 50 * on_resource + 40 * (1 if (rx, ry) == (nx, ny) else 0)
            v += 12 * lead
            v += -3 * ds
            # Discourage moving into areas where opponent is already better by a lot
            if lead <= 0:
                v += -25 - 5 * (-lead)
            # Mildly prefer resources not extremely far from us
            if ds > 0:
                v += 2 / (ds)
            # Opponent proximity to our position adds deterrence
            worst_opponent_penalty = max(worst_opponent_penalty, -mdist(nx, ny, ox, oy))
            if v > best_resource_val:
                best_resource_val = v
        # Final evaluation includes small term to keep moving toward resources rather than oscillating
        val = best_resource_val + 0.5 * (-mdist(nx, ny, ox, oy)) + 0.2 * on_resource + 0.01 * (-mdist(nx, ny, res[0][0], res[0][1]))
        if val > best_val:
            best_val = val
            best_move = (dx, dy)

    return [int(best_move[0]), int(best_move[1])]
