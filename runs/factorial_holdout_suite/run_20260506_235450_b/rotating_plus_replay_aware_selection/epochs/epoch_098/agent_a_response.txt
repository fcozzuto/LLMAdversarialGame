def choose_move(observation):
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    sx, sy = observation.get("self_position", (0, 0))
    ox, oy = observation.get("opponent_position", (0, 0))
    resources = observation.get("resources", []) or []
    obs_list = observation.get("obstacles", []) or []
    obstacles = set()
    for p in obs_list:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            obstacles.add((int(p[0]), int(p[1])))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def dist(x1, y1, x2, y2):
        dx = x1 - x2
        if dx < 0: dx = -dx
        dy = y1 - y2
        if dy < 0: dy = -dy
        return dx + dy

    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)
    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    legal = []
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if inb(nx, ny) and (nx, ny) not in obstacles:
            legal.append((dx, dy, nx, ny))
    if not legal:
        return [0, 0]
    if not resources:
        # drift toward opponent side to reduce blocking
        tx = 0 if sx > (w - 1) // 2 else (w - 1)
        ty = (h - 1) if sy < h // 2 else 0
        best = (10**9, 0, 0)
        for dx, dy, nx, ny in legal:
            d = dist(nx, ny, tx, ty)
            if d < best[0]:
                best = (d, dx, dy)
        return [best[1], best[2]]

    # Score moves by best resource we can secure soon; heavily penalize if opponent can beat us.
    best_dx, best_dy = 0, 0
    best_val = -10**18
    for dx, dy, nx, ny in legal:
        my_to_best = 10**9
        opp_to_best = 10**9
        win_margin = -10**9
        for r in resources:
            rx, ry = int(r[0]), int(r[1])
            myd = dist(nx, ny, rx, ry)
            opd = dist(ox, oy, rx, ry)
            if opd < myd:
                # opponent closer: avoid unless very good otherwise
                continue
            if myd < my_to_best or (myd == my_to_best and opd < opp_to_best):
                my_to_best = myd
                opp_to_best = opd
        # Compute value based on closest non-contested resource; if none, allow contest tie-breaking.
        if my_to_best == 10**9:
            # pick resource minimizing (myd - opd) to reduce losses
            best_delta = 10**9
            tie_my = 10**9
            tie_opd = 10**9
            for r in resources:
                rx, ry = int(r[0]), int(r[1])
                myd = dist(nx, ny, rx, ry)
                opd = dist(ox, oy, rx, ry)
                delta = myd - opd
                if delta < best_delta or (delta == best_delta and myd < tie_my) or (delta == best_delta and myd == tie_my and opd < tie_opd):
                    best_delta = delta
                    tie_my = myd
                    tie_opd = opd
            my_to_best, opp_to_best = tie_my, tie_opd
        win_margin = opp_to_best - my_to_best  # positive means we arrive earlier/equal-ish
        # Prefer earlier and winning; slight tie-break to centralize and avoid staying.
        val = 1000 * win_margin - 10 * my_to_best
        if (dx == 0 and dy == 0):
            val -= 3
        # Nudge toward reducing distance to opponent to contest
        val -= 0.2 * dist(nx, ny, ox, oy)
        # Nudge away from edges to avoid getting stuck
        val -= 0.02 * (abs(nx - (w - 1) / 2) + abs(ny - (h - 1) / 2))
        if val > best_val:
            best_val = val
            best_dx, best_dy = dx, dy

    return [int(best_dx), int(best_dy)]
