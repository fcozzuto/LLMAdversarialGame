def choose_move(observation):
    w = observation.get("grid_width", 8)
    h = observation.get("grid_height", 8)
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    resources = observation.get("resources", []) or []
    obstacles = observation.get("obstacles", []) or []

    obs = set()
    for p in obstacles:
        if p and len(p) >= 2 and isinstance(p[0], int) and isinstance(p[1], int):
            obs.add((p[0], p[1]))

    def cheb(ax, ay, bx, by):
        dx = ax - bx
        if dx < 0: dx = -dx
        dy = ay - by
        if dy < 0: dy = -dy
        return dx if dx > dy else dy

    def free(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obs

    def best_step(px, py):
        if not resources:
            return (px, py)
        best = (px, py, 10**18)
        for dxm in (-1, 0, 1):
            for dym in (-1, 0, 1):
                if dxm == 0 and dym == 0:
                    nx, ny = px, py
                else:
                    nx, ny = px + dxm, py + dym
                if not free(nx, ny):
                    continue
                best_res = 10**18
                for rx, ry in resources:
                    d = cheb(nx, ny, rx, ry)
                    if d < best_res:
                        best_res = d
                # Favor exact pickup then approach
                tie = 0 if any(nx == rx and ny == ry for rx, ry in resources) else 1
                val = best_res * 2 + tie
                if val < best[2] or (val == best[2] and (nx, ny) < (best[0], best[1])):
                    best = (nx, ny, val)
        return (best[0], best[1])

    moves = [(-1,-1),(-1,0),(-1,1),(0,-1),(0,0),(0,1),(1,-1),(1,0),(1,1)]
    if not resources:
        return [0, 0]

    # Predict opponent's greedy next position to attempt interception/denial.
    nx_op, ny_op = best_step(ox, oy)

    best_move = (0, 0)
    best_score = -10**18
    for dxm, dym in moves:
        nx, ny = sx + dxm, sy + dym
        if not free(nx, ny):
            continue

        # Immediate pickup dominates.
        onto = 1 if any(nx == rx and ny == ry for rx, ry in resources) else 0

        # Distance to closest resource (pickup race).
        d_self = 10**18
        closest = (0, 0)
        for rx, ry in resources:
            d = cheb(nx, ny, rx, ry)
            if d < d_self:
                d_self = d
                closest = (rx, ry)

        # Try to block where opponent is likely to be soon.
        d_op = cheb(nx, ny, nx_op, ny_op)

        # Also discourage moving away from remaining resources.
        # Use a mild penalty for increasing distance from the chosen closest target.
        d_curr = 10**18
        for rx, ry in resources:
            d = cheb(sx, sy, rx, ry)
            if d < d_curr:
                d_curr = d

        # Deterministic tie-breaker favors lower coordinates.
        score = 0
        score += onto * 1000000
        score += -d_self * 200
        score += -d_op * 60
        score += -(cheb(sx, sy, closest[0], closest[1]) - cheb(nx, ny, closest[0], closest[1])) * 20
        score += -(dxm + 2 * dym)  # deterministic small bias

        if score > best_score or (score == best_score and (nx, ny) < (sx + best_move[0], sy + best_move[1])):
            best_score = score
            best_move = (dxm, dym)

    return [int(best_move[0]), int(best_move[1])]
