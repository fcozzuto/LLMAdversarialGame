def choose_move(observation):
    w = observation.get("grid_width", 8)
    h = observation.get("grid_height", 8)
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    resources = observation.get("resources", []) or []
    obstacles = observation.get("obstacles", []) or []
    obs = set()
    for p in obstacles:
        if p and len(p) >= 2:
            x, y = p[0], p[1]
            if isinstance(x, int) and isinstance(y, int):
                obs.add((x, y))

    def free(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obs

    if not resources:
        return [0, 0]

    def cheb(ax, ay, bx, by):
        dx = ax - bx
        if dx < 0: dx = -dx
        dy = ay - by
        if dy < 0: dy = -dy
        return dx if dx > dy else dy

    def dist_self_to(r):
        return cheb(sx, sy, r[0], r[1])

    def dist_opp_to(r):
        return cheb(ox, oy, r[0], r[1])

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    best_adv = -10**9
    best_r = resources[0]
    feasible_equal = []
    for r in resources:
        sd = dist_self_to(r)
        od = dist_opp_to(r)
        adv = od - sd
        if adv >= 0:
            feasible_equal.append((adv, sd, r))
        if adv > best_adv or (adv == best_adv and sd < dist_self_to(best_r)):
            best_adv, best_r = adv, r

    if feasible_equal:
        feasible_equal.sort(key=lambda t: (-t[0], t[1], t[2][0], t[2][1]))
        target = feasible_equal[0][2]
    else:
        target = best_r

    tx, ty = target[0], target[1]
    dx0 = 0 if tx == sx else (1 if tx > sx else -1)
    dy0 = 0 if ty == sy else (1 if ty > sy else -1)
    prefer = (sx + dx0, sy + dy0)

    def next_value(nx, ny):
        if not free(nx, ny):
            return -10**12
        # If we would land on a resource, prioritize strongly.
        imm = 0
        for r in resources:
            if nx == r[0] and ny == r[1]:
                imm = 1
                break
        # Otherwise race for the best available target, but bias to not be behind.
        best = -10**9
        for r in resources:
            sd = cheb(nx, ny, r[0], r[1])
            od = dist_opp_to(r)
            adv = od - sd
            if adv > best:
                best = adv
        # Slight tie-break: closer is better.
        return imm * 10**6 + best * 100 - cheb(nx, ny, tx, ty)

    best_move = (0, 0)
    best_val = -10**18
    for dxm, dym in moves:
        nx, ny = sx + dxm, sy + dym
        if not free(nx, ny):
            continue
        val = next_value(nx, ny)
        # Encourage the direct greedy step if it's feasible.
        if (nx, ny) == prefer:
            val += 25
        if val > best_val:
            best_val = val
            best_move = (dxm, dym)

    return [best_move[0], best_move[1]]
