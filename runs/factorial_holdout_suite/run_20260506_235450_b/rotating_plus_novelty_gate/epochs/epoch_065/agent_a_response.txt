def choose_move(observation):
    w = observation.get("grid_width", 8)
    h = observation.get("grid_height", 8)
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    resources = observation.get("resources", []) or []
    obstacles = observation.get("obstacles", []) or []
    turns_remaining = observation.get("turns_remaining", 0)

    obs = set()
    for p in obstacles:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = p[0], p[1]
            if isinstance(x, int) and isinstance(y, int) and 0 <= x < w and 0 <= y < h:
                obs.add((x, y))

    def blocked(x, y):
        return (x, y) in obs or not (0 <= x < w and 0 <= y < h)

    def cheb(ax, ay, bx, by):
        dx = ax - bx
        if dx < 0:
            dx = -dx
        dy = ay - by
        if dy < 0:
            dy = -dy
        return dx if dx > dy else dy

    def obstacle_penalty(ax, ay, bx, by):
        # Check a simple monotone path (x first then y) for obstacle contact.
        x, y = ax, ay
        sxg = 0 if bx == ax else (1 if bx > ax else -1)
        syg = 0 if by == ay else (1 if by > ay else -1)
        pen = 0
        step = cheb(ax, ay, bx, by)
        for _ in range(step + 1):
            if (x, y) in obs:
                pen += 2
                break
            if x != bx:
                x += sxg
            else:
                if y != by:
                    y += syg
            if x == bx and y == by:
                if (x, y) in obs:
                    pen += 2
                break
        return pen

    if not resources:
        return [0, 0]

    moves = []
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            nx, ny = sx + dx, sy + dy
            if blocked(nx, ny):
                continue
            moves.append((dx, dy))

    # If all moves blocked, stay.
    if not moves:
        return [0, 0]

    best_dx, best_dy = 0, 0
    best_val = -10**18
    # Prefer collecting earlier; also adapt to opponent being closer.
    time_weight = 1.0 + (1.0 if turns_remaining and turns_remaining < 8 else 0.0)

    for dx, dy in moves:
        myx, myy = sx + dx, sy + dy
        # Pick best resource for our next step; evaluate by relative advantage.
        local_best = -10**18
        for r in resources:
            if not (isinstance(r, (list, tuple)) and len(r) >= 2):
                continue
            rx, ry = r[0], r[1]
            if not (isinstance(rx, int) and isinstance(ry, int)):
                continue
            if blocked(rx, ry):
                continue
            du = cheb(myx, myy, rx, ry)
            dov = cheb(ox, oy, rx, ry)
            pen = obstacle_penalty(myx, myy, rx, ry)
            # Higher is better: we want to be strictly sooner than opponent.
            val = (dov - du) * 12 - du * time_weight - pen
            # Slight preference for nearer resources to avoid dithering.
            val += -0.1 * cheb(myx, myy, rx, ry)
            if val > local_best:
                local_best = val
        if local_best > best_val:
            best_val = local_best
            best_dx, best_dy = dx, dy

    return [int(best_dx), int(best_dy)]
