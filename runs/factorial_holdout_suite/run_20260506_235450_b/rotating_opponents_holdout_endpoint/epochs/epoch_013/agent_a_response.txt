def choose_move(observation):
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    resources = observation.get("resources") or []
    obstacles = observation.get("obstacles") or []
    obs = set()
    for a in obstacles:
        if isinstance(a, (list, tuple)) and len(a) >= 2:
            obs.add((int(a[0]), int(a[1])))

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obs

    def bfs_dist(start, goal):
        if start == goal:
            return 0
        if not inb(goal[0], goal[1]) or not inb(start[0], start[1]):
            return 10**9
        from collections import deque
        q = deque([start])
        dist = {start: 0}
        while q:
            x, y = q.popleft()
            d = dist[(x, y)]
            for dx, dy in moves:
                nx, ny = x + dx, y + dy
                if not inb(nx, ny):
                    continue
                np = (nx, ny)
                if np in dist:
                    continue
                nd = d + 1
                if np == goal:
                    return nd
                dist[np] = nd
                q.append(np)
        return 10**9

    # Pick target resource with strongest (opponent slower than us), obstacle-aware via BFS.
    best_r = None
    best_key = None
    for r in resources:
        rx, ry = int(r[0]), int(r[1])
        sd = bfs_dist((sx, sy), (rx, ry))
        od = bfs_dist((ox, oy), (rx, ry))
        # If unreachable, ignore (very large distances).
        if sd >= 10**8 and od >= 10**8:
            continue
        # Deterministic tie-break using coordinates.
        adv = od - sd  # positive => we closer
        # Encourage not only advantage but also faster completion.
        key = (-adv, sd, od, rx, ry)  # minimizing negative advantage => maximizing advantage
        if best_key is None or key < best_key:
            best_key = key
            best_r = (rx, ry)

    if best_r is None:
        return [0, 0]

    tx, ty = best_r
    my_d = bfs_dist((sx, sy), (tx, ty))
    # Choose move maximizing improvement to target and slight denial via increasing opponent distance.
    best_move = [0, 0]
    best_key = None
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny):
            continue
        nd_self = bfs_dist((nx, ny), (tx, ty))
        nd_opp = bfs_dist((ox, oy), (tx, ty))
        # Opponent distance unchanged for current turn; denial achieved by stealing: prioritize lowering our distance.
        # If tied, prefer moving such that our distance is smaller than opponent's.
        opp_d_now = nd_opp
        key = (nd_self, - (opp_d_now - nd_self), dx, dy)  # smaller nd_self, then larger advantage
        if best_key is None or key < best_key:
            best_key = key
            best_move = [dx, dy]

    return best_move
