def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position") or [0, 0]
    ox, oy = observation.get("opponent_position") or [w - 1, h - 1]
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    obstacles = set()
    for a in observation.get("obstacles") or []:
        if isinstance(a, dict) and "x" in a and "y" in a:
            x, y = int(a["x"]), int(a["y"])
        elif isinstance(a, (list, tuple)) and len(a) >= 2:
            x, y = int(a[0]), int(a[1])
        else:
            continue
        if 0 <= x < w and 0 <= y < h:
            obstacles.add((x, y))

    resources = []
    for r in observation.get("resources") or []:
        if isinstance(r, dict) and "x" in r and "y" in r:
            x, y = int(r["x"]), int(r["y"])
        elif isinstance(r, (list, tuple)) and len(r) >= 2:
            x, y = int(r[0]), int(r[1])
        else:
            continue
        if 0 <= x < w and 0 <= y < h and (x, y) not in obstacles:
            resources.append((x, y))

    def cheb(x1, y1, x2, y2):
        dx = x1 - x2
        if dx < 0: dx = -dx
        dy = y1 - y2
        if dy < 0: dy = -dy
        return dx if dx > dy else dy

    if resources:
        best = None
        for tx, ty in resources:
            my_d = cheb(sx, sy, tx, ty)
            op_d = cheb(ox, oy, tx, ty)
            adv = my_d - op_d
            key = (adv, my_d, tx, ty)
            if best is None or key < best[0]:
                best = (key, (tx, ty))
        _, (tx, ty) = best
    else:
        # No visible resources: retreat to safer corner-ish if opponent is close, otherwise drift.
        tx, ty = (0, 0) if cheb(sx, sy, 0, 0) <= cheb(sx, sy, w - 1, h - 1) else (w - 1, h - 1)

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    best_m = (10**9, None)
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if nx < 0 or nx >= w or ny < 0 or ny >= h:
            continue
        if (nx, ny) in obstacles:
            continue
        d_to_t = cheb(nx, ny, tx, ty)
        d_to_o = cheb(nx, ny, ox, oy)
        # Prefer moves that reduce distance to our chosen resource and keep distance from opponent slightly.
        # If equal, prefer progressing in dx then dy deterministically.
        score = d_to_t * 10 + (0 if resources and (nx, ny) == (tx, ty) else 1) + (0 if d_to_o > 1 else 2)
        score = score + (1 if nx > sx else 0) * 0 + (0 if ny > sy else 0) * 0
        key = (score, nx, ny, dx, dy)
        if key < best_m[0]:
            best_m = (key, (dx, dy))

    return [int(best_m[1][0]), int(best_m[1][1])]

        if d < best_dist or (d == best_dist and (tx < best_tx or (tx == best_tx and ty < best_ty))):

    # Greedy one-step heuristic: get closer to target, avoid opponent, prefer landing on resource.
