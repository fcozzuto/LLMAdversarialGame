def choose_move(observation):
    w = int(observation.get("grid_width") or 8)
    h = int(observation.get("grid_height") or 8)
    sp = observation.get("self_position") or [0, 0]
    ap = observation.get("opponent_position") or [w - 1, h - 1]
    sx, sy = int(sp[0]), int(sp[1])
    ox, oy = int(ap[0]), int(ap[1])

    obstacles = set()
    for a in observation.get("obstacles") or []:
        if isinstance(a, (list, tuple)) and len(a) >= 2:
            x, y = int(a[0]), int(a[1])
        else:
            x, y = int(getattr(a, "x", -1)), int(getattr(a, "y", -1))
        if 0 <= x < w and 0 <= y < h:
            obstacles.add((x, y))

    resources = []
    for r in observation.get("resources") or []:
        if isinstance(r, (list, tuple)) and len(r) >= 2:
            x, y = int(r[0]), int(r[1])
        else:
            x, y = int(getattr(r, "x", -1)), int(getattr(r, "y", -1))
        if 0 <= x < w and 0 <= y < h and (x, y) not in obstacles:
            resources.append((x, y))
    if not resources:
        return [0, 0]

    def bfs(start):
        inf = 10**9
        dist = [[inf] * h for _ in range(w)]
        if start in obstacles:
            return dist
        sx0, sy0 = start
        dist[sx0][sy0] = 0
        q = [(sx0, sy0)]
        head = 0
        while head < len(q):
            x, y = q[head]
            head += 1
            nd = dist[x][y] + 1
            for dx in (-1, 0, 1):
                for dy in (-1, 0, 1):
                    if dx == 0 and dy == 0:
                        continue
                    nx, ny = x + dx, y + dy
                    if 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obstacles:
                        if nd < dist[nx][ny]:
                            dist[nx][ny] = nd
                            q.append((nx, ny))
        return dist

    sd = bfs((sx, sy))
    od = bfs((ox, oy))

    best_cell = None
    best_key = None
    for rx, ry in resources:
        a = sd[rx][ry]
        b = od[rx][ry]
        if a >= 10**9:
            continue
        key = (b - a, -a)  # prioritize being closer than opponent; then shorter
        if best_key is None or key > best_key or (key == best_key and (rx, ry) < best_cell):
            best_key = key
            best_cell = (rx, ry)
    if best_cell is None:
        return [0, 0]

    tx, ty = best_cell
    cur = sd[tx][ty]
    candidates = []
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            nx, ny = sx + dx, sy + dy
            if not (0 <= nx < w and 0 <= ny < h) or (nx, ny) in obstacles:
                continue
            nd = sd[nx][ny]
            if nd > cur:
                continue
            opp_adv = od[tx][ty] - od[nx][ny]
            candidates.append(((cur - nd, opp_adv, -abs(tx - nx) - abs(ty - ny)), dx, dy))
    if not candidates:
        return [0, 0]
    candidates.sort(reverse=True)
    _, dx, dy = candidates[0]
    return [dx, dy]
