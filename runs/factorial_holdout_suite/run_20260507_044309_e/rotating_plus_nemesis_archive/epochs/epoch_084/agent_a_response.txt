def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sp = observation.get("self_position") or [0, 0]
    op = observation.get("opponent_position") or [0, 0]
    sx, sy = int(sp[0]), int(sp[1])
    ox, oy = int(op[0]), int(op[1])

    obstacles = set()
    for b in observation.get("obstacles") or []:
        if b and len(b) >= 2:
            bx, by = int(b[0]), int(b[1])
            if 0 <= bx < w and 0 <= by < h:
                obstacles.add((bx, by))

    resources = []
    for r in observation.get("resources") or []:
        if r and len(r) >= 2:
            rx, ry = int(r[0]), int(r[1])
            if 0 <= rx < w and 0 <= ry < h and (rx, ry) not in obstacles:
                resources.append((rx, ry))

    def cheb(ax, ay, bx, by):
        dx = ax - bx
        if dx < 0: dx = -dx
        dy = ay - by
        if dy < 0: dy = -dy
        return dx if dx > dy else dy

    def valid(nx, ny):
        return 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obstacles

    # fallback: go to a stable target near our side
    if not resources:
        tx = 0 if sx > (w - 1) // 2 else w - 1
        ty = (h - 1) // 2
        dx = 0 if tx == sx else (1 if tx > sx else -1)
        dy = 0 if ty == sy else (1 if ty > sy else -1)
        nx, ny = sx + dx, sy + dy
        if valid(nx, ny):
            return [dx, dy]
        # try axis move
        for ddx, ddy in [dx, 0], [0, dy], [0, 0]:
            nx2, ny2 = sx + ddx, sy + ddy
            if valid(nx2, ny2):
                return [ddx, ddy]
        return [0, 0]

    best = None
    for rx, ry in resources:
        ds = cheb(sx, sy, rx, ry)
        do = cheb(ox, oy, rx, ry)
        # prefer closer, but heavily avoid resources where opponent can arrive no later
        penalty = 0
        if do <= ds:
            penalty = (ds - do) * 3 + 6  # always non-negative, deters contest
            if penalty < 6:
                penalty = 6
        # slight tie-break favoring higher y then x deterministically
        score = ds + penalty - (ry * 0.01 + rx * 0.001)
        key = (score, rx, ry)
        if best is None or key < best[0]:
            best = (key, rx, ry)

    _, tx, ty = best[0], best[1], best[2]

    moves = []
    for ddx in (-1, 0, 1):
        for ddy in (-1, 0, 1):
            if ddx == 0 and ddy == 0:
                moves.append((ddx, ddy))
            else:
                moves.append((ddx, ddy))
    # prioritize moves reducing chebyshev distance to target with opponent-aware tie-break
    best_move = None
    best_rank = None
    for ddx, ddy in moves:
        nx, ny = sx + ddx, sy + ddy
        if not valid(nx, ny):
            continue
        d1 = cheb(nx, ny, tx, ty)
        d2 = cheb(ox, oy, tx, ty)
        # if we can reduce distance while opponent isn't advantaged, prefer
        contest = 0
        if d2 <= d1:
            contest = (d1 - d2) * 3 + 6
        rank = (d1 + contest, ddx * ddx + ddy * ddy, nx, ny)
        if best_rank is None or rank < best_rank:
            best_rank = rank
            best_move = (ddx, ddy)

    if best_move is None:
        return [0, 0]
    return [int(best_move[0]), int(best_move[1])]

                if dx == 0 and dy == 0:

                    yield dx, dy

        # Prefer moving toward nearest resource; slight preference to reduce distance to opponent if needed
