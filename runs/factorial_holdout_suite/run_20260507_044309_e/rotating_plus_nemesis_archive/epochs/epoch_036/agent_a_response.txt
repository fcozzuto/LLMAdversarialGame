def choose_move(observation):
    w = int(observation.get("grid_width") or 8)
    h = int(observation.get("grid_height") or 8)
    sx, sy = observation.get("self_position") or [0, 0]
    ox, oy = observation.get("opponent_position") or [0, 0]
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    blocked = set()
    for o in observation.get("obstacles") or []:
        try:
            x, y = int(o[0]), int(o[1])
            if 0 <= x < w and 0 <= y < h:
                blocked.add((x, y))
        except:
            pass

    def valid(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in blocked

    resources = []
    for r in observation.get("resources") or []:
        try:
            x, y = int(r[0]), int(r[1])
            if valid(x, y):
                resources.append((x, y))
        except:
            pass

    moves = []
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            nx, ny = sx + dx, sy + dy
            if valid(nx, ny):
                moves.append((dx, dy, nx, ny))
    if not moves:
        return [0, 0]

    if not resources:
        best = None
        bestv = None
        for dx, dy, nx, ny in moves:
            v = abs(nx - ox) + abs(ny - oy) + 0.01 * (abs(nx - sx) + abs(ny - sy))
            if bestv is None or v < bestv:
                bestv = v
                best = (dx, dy)
        return [int(best[0]), int(best[1])]

    def dist(a, b, c, d):
        return abs(a - c) + abs(b - d)

    # Choose a resource considering turn-by-turn race: minimize (our_dist - opp_dist*0.9),
    # but prefer closer overall to avoid indecision.
    best_res = None
    best_rv = None
    for rx, ry in resources:
        d1 = dist(sx, sy, rx, ry)
        d2 = dist(ox, oy, rx, ry)
        gap = d1 - 0.9 * d2  # if gap < 0, we're likely faster
        rv = gap + 0.03 * d1
        if best_rv is None or rv < best_rv:
            best_rv = rv
            best_res = (rx, ry)

    tx, ty = best_res

    # Move to reduce our distance to target; if opponent is clearly closer to target,
    # still try to reduce the gap faster by focusing on minimizing (our_dist - opp_dist).
    opp_closer = dist(ox, oy, tx, ty) + 0.5 < dist(sx, sy, tx, ty)

    best_move = None
    best_mv = None
    for dx, dy, nx, ny in moves:
        my = dist(nx, ny, tx, ty)
        op = dist(ox, oy, tx, ty)
        if opp_closer:
            mv = (my - op) + 0.02 * (abs(nx - sx) + abs(ny - sy))
        else:
            mv = my + 0.02 * (abs(nx - sx) + abs(ny - sy))
        # Secondary: avoid stepping into squares with fewer available exits (simple obstacle pressure)
        exits = 0
        for ddx in (-1, 0, 1):
            for ddy in (-1, 0, 1):
                if ddx == 0 and ddy == 0:
                    continue
                xx, yy = nx + ddx, ny + ddy
                if valid(xx, yy):
                    exits += 1
        mv += 0.05 * max(0, 8 - exits)
        if best_mv is None or mv < best_mv:
            best_mv = mv
            best_move = (dx, dy)

    return [int(best_move[0]), int(best_move[1])]

            pass

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    best_val = None

    def man(a, b, c, d):
