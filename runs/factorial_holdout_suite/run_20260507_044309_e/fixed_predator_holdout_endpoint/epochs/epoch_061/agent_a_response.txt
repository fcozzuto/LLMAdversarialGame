def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sp = observation.get("self_position", (0, 0)) or (0, 0)
    op = observation.get("opponent_position", (0, 0)) or (0, 0)
    sx, sy = int(sp[0]), int(sp[1])
    ox, oy = int(op[0]), int(op[1])

    obstacles = set()
    for p in (observation.get("obstacles", None) or []):
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            px, py = int(p[0]), int(p[1])
            if 0 <= px < w and 0 <= py < h:
                obstacles.add((px, py))

    resources = []
    for r in (observation.get("resources", None) or []):
        if isinstance(r, (list, tuple)) and len(r) >= 2:
            rx, ry = int(r[0]), int(r[1])
            if 0 <= rx < w and 0 <= ry < h and (rx, ry) not in obstacles:
                resources.append((rx, ry))
    if not resources:
        return [0, 0]

    def man(a, b):
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

    best = None
    best_val = None
    for rx, ry in resources:
        sd = man((sx, sy), (rx, ry))
        od = man((ox, oy), (rx, ry))
        # If we can beat, strongly prefer it; otherwise choose the resource that hurts their timing most.
        if sd <= od:
            val = 1000 - 20 * sd - od
        else:
            val = 500 + 30 * (od - sd) - 5 * od
        key = (val, -sd, -od, -rx, -ry)
        if best is None or key > best_val:
            best = (rx, ry)
            best_val = key

    tx, ty = best
    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    best_move = None
    best_score = None
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not (0 <= nx < w and 0 <= ny < h) or (nx, ny) in obstacles:
            continue
        d_to = man((nx, ny), (tx, ty))
        # Extra bias: if we are behind on the chosen target, head directly toward it to deny.
        behind = 1 if man((sx, sy), (tx, ty)) > man((ox, oy), (tx, ty)) else 0
        # Small tie-break to avoid oscillation: prefer moves that progress roughly toward the target.
        prog = (0 if tx == sx else (1 if (nx - sx) * (tx - sx) >= 0 else -1)) + (0 if ty == sy else (1 if (ny - sy) * (ty - sy) >= 0 else -1))
        score = (-d_to, -behind * d_to, -prog, dx, dy)
        if best_score is None or score > best_score:
            best_score = score
