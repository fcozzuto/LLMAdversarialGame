def choose_move(observation):
    w = observation["grid_width"]
    h = observation["grid_height"]
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    resources = observation.get("resources", [])
    obstacles = set()
    for p in observation.get("obstacles", []):
        obstacles.add((p[0], p[1]))

    def man(ax, ay, bx, by):
        dx = ax - bx
        if dx < 0: dx = -dx
        dy = ay - by
        if dy < 0: dy = -dy
        return dx + dy

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obstacles

    if not resources:
        return [0, 0]

    moves = [(-1,-1),(-1,0),(-1,1),(0,-1),(0,0),(0,1),(1,-1),(1,0),(1,1)]
    best_score = None
    best_move = [0, 0]
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny):
            continue

        # Select/pressure resources where we are (currently) likely to arrive first.
        # Score: sum of advantage, with a small tie-break toward closeness and away from opponent.
        score = 0
        min_our = None
        min_opp = None
        for rx, ry in resources:
            if (rx, ry) in obstacles:
                continue
            our_d = man(nx, ny, rx, ry)
            opp_d = man(ox, oy, rx, ry)
            adv = opp_d - our_d  # positive => we are closer
            if adv > 0:
                score += 20 * adv
            else:
                score += adv  # still consider contested ones but much less
            if min_our is None or our_d < min_our:
                min_our = our_d
            if min_opp is None or opp_d < min_opp:
                min_opp = opp_d

        # Encourage keeping distance from opponent when we have no clear advantage.
        # Also mildly prefer moving closer to the closest resource.
        if min_our is None:
            min_our = 0
        if min_opp is None:
            min_opp = 0
        opp_dist = man(nx, ny, ox, oy)
        score += (-2 * opp_dist)
        score += (2 * (min_opp - min_our))

        if best_score is None or score > best_score:
            best_score = score
            best_move = [dx, dy]
        elif score == best_score:
            # Deterministic tie-break: smallest manhattan to nearest resource, then smallest dx,dy lexicographically.
            cur_min_our = None
            cur_min_opp = None
            for rx, ry in resources:
                if (rx, ry) in obstacles:
                    continue
                our_d = man(nx, ny, rx, ry)
                opp_d = man(ox, oy, rx, ry)
                if cur_min_our is None or our_d < cur_min_our:
                    cur_min_our = our_d
                if cur_min_opp is None or opp_d < cur_min_opp:
                    cur_min_opp = opp_d
            cand_key = (cur_min_our if cur_min_our is not None else 10**9, dx, dy)
            bestx, besty = best_move
            bx, by = sx + bestx, sy + besty
            best_min = None
            for rx, ry in resources:
                if (rx, ry) in obstacles:
                    continue
                d = man(bx, by, rx, ry)
                if best_min is None or d < best_min:
                    best_min = d
            best_key = (best_min if best_min is not None else 10**9, bestx, besty)
            if cand_key < best_key:
                best_move = [dx, dy]

    return [best_move[0], best_move[1]]
