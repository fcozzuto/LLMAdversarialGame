def choose_move(observation):
    w = observation["grid_width"]; h = observation["grid_height"]
    sx, sy = observation["self_position"]; ox, oy = observation["opponent_position"]
    obstacles = {(p[0], p[1]) for p in (observation.get("obstacles", []) or [])}
    resources = observation.get("resources", []) or []

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obstacles

    def dist(x1, y1, x2, y2):
        dx = x1 - x2; dx = -dx if dx < 0 else dx
        dy = y1 - y2; dy = -dy if dy < 0 else dy
        return dx + dy

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    valid = []
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if inb(nx, ny):
            valid.append((dx, dy, nx, ny))
    if not valid:
        return [0, 0]

    if not resources:
        tx, ty = w // 2, h // 2
        best = min(valid, key=lambda m: dist(m[2], m[3], tx, ty))
        return [best[0], best[1]]

    # Predict opponent target as nearest resource (nearest_resource archetype).
    opp_near = min(resources, key=lambda r: dist(ox, oy, r[0], r[1]))
    remaining = [r for r in resources if r != opp_near]
    if not remaining:
        remaining = resources

    # Choose next best resource for us: maximize (our speed advantage).
    best_r = None
    best_adv = -10**9
    for rx, ry in remaining:
        ad = dist(ox, oy, rx, ry) - dist(sx, sy, rx, ry)
        # Small tie-break: prefer slightly closer resources to reduce risk.
        key = (ad, -dist(sx, sy, rx, ry))
        if best_r is None or key > (best_adv, -dist(sx, sy, best_r[0], best_r[1])):
            best_adv = ad; best_r = (rx, ry)
    tx, ty = best_r
