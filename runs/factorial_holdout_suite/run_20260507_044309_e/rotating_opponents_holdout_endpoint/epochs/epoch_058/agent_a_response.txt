def choose_move(observation):
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    resources = observation.get("resources") or []
    obstacles_in = observation.get("obstacles") or []

    obstacles = set()
    for p in obstacles_in:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            obstacles.add((int(p[0]), int(p[1])))

    def inside(x, y):
        return 0 <= x < w and 0 <= y < h

    def blocked(x, y):
        return (x, y) in obstacles

    def king_dist(x1, y1, x2, y2):
        dx = x1 - x2
        dy = y1 - y2
        if dx < 0:
            dx = -dx
        if dy < 0:
            dy = -dy
        return dx if dx >= dy else dy

    valid = []
    for r in resources:
        if isinstance(r, (list, tuple)) and len(r) >= 2:
            tx, ty = int(r[0]), int(r[1])
            if inside(tx, ty) and not blocked(tx, ty):
                valid.append((tx, ty))
    if not valid:
        return [0, 0]

    # Predict opponent's next target: nearest resource to opponent (king distance).
    opp_best = None
    opp_bd = None
    for tx, ty in valid:
        d = king_dist(ox, oy, tx, ty)
        if opp_best is None or d < opp_bd or (d == opp_bd and (tx, ty) < opp_best):
            opp_best = (tx, ty)
            opp_bd = d
    txo, tyo = opp_best

    my_doing_wins = king_dist(sx, sy, txo, tyo) <= opp_bd
    if my_doing_wins:
        target = (txo, tyo)
    else:
        # Choose resource maximizing lead: (opp_dist - my_dist).
        target = None
        best_gain = None
        for tx, ty in valid:
            myd = king_dist(sx, sy, tx, ty)
            opd = king_dist(ox, oy, tx, ty)
            gain = opd - myd
            if target is None or gain > best_gain or (gain == best_gain and (tx, ty) < target):
                target = (tx, ty)
                best_gain = gain

    tx, ty = target

    deltas = []
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            if dx == 0 and dy == 0:
                deltas.append((0, 0, 99))
                continue
            score = 0
            if dx != 0 and dy != 0:
                score -= 2  # prefer diagonal
            if dx == 0 or dy == 0:
                score -= 1  # then straight
            score += (1 if dx == 0 else 0) + (1 if dy == 0 else 0)
            deltas.append((dx, dy, score))
    deltas.sort(key=lambda t: t[2])

    best_move = (0, 0)
    best_d = None
    for dx, dy, _ in deltas:
        nx, ny = sx + dx, sy + dy
        if not inside(nx, ny) or blocked(nx, ny):
            continue
        d = king_dist(nx, ny, tx, ty)
        if best_d is None or d < best_d:
            best_d = d
            best_move = (dx, dy)
        elif d == best_d:
            if (dx, dy) < best_move:
                best_move = (dx, dy)

    return [int(best_move[0]), int(best_move[1])]
