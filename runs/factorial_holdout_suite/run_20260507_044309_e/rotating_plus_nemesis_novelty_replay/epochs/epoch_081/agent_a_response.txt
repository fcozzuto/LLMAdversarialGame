def choose_move(observation):
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    w = observation.get("grid_width", 8)
    h = observation.get("grid_height", 8)
    resources = observation.get("resources") or []
    obstacles_raw = observation.get("obstacles") or []
    obstacles = obstacles_raw if isinstance(obstacles_raw, set) else set(tuple(p) for p in obstacles_raw)

    def in_bounds(x, y):
        return 0 <= x < w and 0 <= y < h

    def cheb(ax, ay, bx, by):
        dx = ax - bx
        if dx < 0: dx = -dx
        dy = ay - by
        if dy < 0: dy = -dy
        return dx if dx > dy else dy

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    if not resources:
        return [0, 0]

    best_r = None
    best_key = None
    for rx, ry in resources:
        d_self = cheb(sx, sy, rx, ry)
        d_opp = cheb(ox, oy, rx, ry)
        steal = d_opp - d_self
        # Prefer immediate collection, then larger steal margin, then closer overall.
        key = (-(1 if d_self == 0 else 0), -steal, d_self, rx, ry)
        if best_key is None or key < best_key:
            best_key = key
            best_r = (rx, ry)

    tx, ty = best_r
    # Choose a legal move that minimizes distance to target, but avoids letting opponent get a better steal.
    best_move = (10**9, 10**9, 10**9, 0, 0)
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not in_bounds(nx, ny) or (nx, ny) in obstacles:
            continue
        # simulate opponent's likely objective (nearest-by-advantage for them)
        opp_best_steal = -10**9
        opp_best_d = 10**9
        for rx, ry in resources:
            if (rx, ry) in obstacles:
                continue
            d_o = cheb(ox, oy, rx, ry)
            d_a = cheb(nx, ny, rx, ry)
            steal_o = d_o - d_a
            if steal_o > opp_best_steal or (steal_o == opp_best_steal and d_o < opp_best_d):
                opp_best_steal = steal_o
                opp_best_d = d_o

        d_to_target = cheb(nx, ny, tx, ty)
        # primary: reach target quickly; secondary: maximize safety against opponent advantage; final tie: deterministic ordering
        safety = -opp_best_steal
        key = (d_to_target, safety, 0 if (nx, ny) == (tx, ty) else 1, nx, ny)
        if key < (best_move[0], best_move[1], best_move[2], best_move[3], best_move[4]):
