def choose_move(observation):
    w = observation.get("grid_width", 8)
    h = observation.get("grid_height", 8)
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    resources = observation.get("resources") or []
    obstacles = observation.get("obstacles") or []
    obs_set = set((int(x), int(y)) for x, y in obstacles if 0 <= int(x) < w and 0 <= int(y) < h)
    res = [(int(x), int(y)) for x, y in resources if 0 <= int(x) < w and 0 <= int(y) < h]
    if not res:
        return [0, 0]

    def man(x1, y1, x2, y2):
        dx = x1 - x2
        if dx < 0: dx = -dx
        dy = y1 - y2
        if dy < 0: dy = -dy
        return dx + dy

    def nearest_obs_adj(nx, ny):
        for dx in (-1, 0, 1):
            for dy in (-1, 0, 1):
                if dx == 0 and dy == 0:
                    continue
                if (nx + dx, ny + dy) in obs_set:
                    return 1
        return 0

    # pick best and runner-up targets for denial/collection
    best = None
    runner = None
    for rx, ry in res:
        self_d = man(sx, sy, rx, ry)
        opp_d = man(ox, oy, rx, ry)
        adv = opp_d - self_d
        key = (adv, -self_d, -man(ox, oy, rx, ry))
        if best is None or key > best[0]:
            runner = best
            best = (key, rx, ry)
        elif runner is None or key > runner[0]:
            runner = (key, rx, ry)

    tx, ty = best[1], best[2]
    tx2, ty2 = (runner[1], runner[2]) if runner else (tx, ty)

    moves = [(-1,-1),(-1,0),(-1,1),(0,-1),(0,0),(0,1),(1,-1),(1,0),(1,1)]
    best_move = (0, 0, -10**18)

    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not (0 <= nx < w and 0 <= ny < h):
            continue
        if (nx, ny) in obs_set:
            continue

        d1 = man(nx, ny, tx, ty)
        d2 = man(nx, ny, tx2, ty2)
        opp_d1 = man(ox, oy, tx, ty)
        opp_d2 = man(ox, oy, tx2, ty2)
        adv1 = opp_d1 - d1
        adv2 = opp_d2 - d2

        # avoid obstacle clumps and prefer improving advantage + closing on target
        score = 10 * adv1 + 3 * adv2 - d1 - 0.25 * d2 - 1.5 * nearest_obs_adj(nx, ny)

        if score > best_move[2]:
            best_move = (dx, dy, score)

    dx, dy, _ = best_move
    return

def choose_move(observation):

    h = int(observation.get("grid_height", 8))
    sp = observation.get("self_position", [0, 0])

    ox, oy = int(op[0]), int(op[1])

    def man(ax, ay, bx, by):
        ax = ax - bx
        ay = ay - by
        return (ax if ax >= 0 else -ax) + (ay if ay >= 0 else -ay)

    resources = observation.get("resources") or []
    if not resources:
        return [0, 0]

    obs_set = set()
    for p in observation.get("obstacles") or []:
        if p is None or len(p) < 2:
            continue
        x, y = int(p[0]), int(p[1])
        if 0 <= x < w and 0 <= y < h:
            obs_set.add((x, y))

    res = []
    for p in resources:
        if p is None or len(p) < 2:
            continue
        x, y = int(p[0]), int(p[1])
        if 0 <= x < w and 0 <= y < h and (x, y) not in obs_set:
            res.append((x, y))
    if not res:
        return [0, 0]

    best = None
    for rx, ry in res:
        sd = man(sx, sy, rx, ry)
        od = man(ox, oy, rx, ry)
        key = (od - sd, -sd, -rx, -ry)
        if best is None or key > best[0]:
            best = (key, rx, ry)
    _, tx, ty = best

    best_move = [0, 0]
    best_score = None
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            nx, ny = sx + dx, sy + dy
            if not (0 <= nx < w and 0 <= ny < h):

                continue
            d_self = man(nx, ny, tx, ty)
            d_opp = man(ox, oy, tx, ty)
            d_self_now = man(sx, sy, tx, ty)
            step_adv = (d_self_now - d_self)
            score = (step_adv * 10) - (d_self * 2) + (d_opp - d_self) * 0.5
