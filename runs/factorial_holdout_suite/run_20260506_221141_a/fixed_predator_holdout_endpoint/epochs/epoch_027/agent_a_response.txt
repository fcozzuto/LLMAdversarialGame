def choose_move(observation):
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    resources = observation.get("resources") or []
    obstacles = observation.get("obstacles") or []
    w = observation.get("grid_width", 8)
    h = observation.get("grid_height", 8)

    if not resources:
        return [0, 0]

    obs_set = {(int(p[0]), int(p[1])) for p in obstacles}

    def pt(p):
        if isinstance(p, dict):
            if "position" in p:
                p = p["position"]
            return int(p.get("x", p[0])), int(p.get("y", p[1]))
        return int(p[0]), int(p[1])

    def man(x1, y1, x2, y2):
        dx = x1 - x2
        if dx < 0: dx = -dx
        dy = y1 - y2
        if dy < 0: dy = -dy
        return dx + dy

    def in_bounds(x, y):
        return 0 <= x < w and 0 <= y < h

    best = None
    best_val = None
    for r in resources:
        rx, ry = pt(r)
        sd = man(sx, sy, rx, ry)
        od = man(ox, oy, rx, ry)
        lead = od - sd  # positive means we are closer
        # If we can't beat, still try to deny by increasing our advantage as much as possible
        # Also lightly prefer resources not adjacent to opponent (denial vs fixed_pool/pickup races)
        opp_adj = 1 if man(ox, oy, rx, ry) <= 1 else 0
        # Primary: maximize lead; Secondary: minimize our distance; Tertiary: reduce opponent adjacency
        val = (lead, -sd, -opp_adj, -(sd + od))
        if best_val is None or val > best_val:
            best_val = val
            best = (rx, ry)

    tx, ty = best
    dx1 = 0 if tx == sx else (1 if tx > sx else -1)
    dy1 = 0 if ty == sy else (1 if ty > sy else -1)

    # Candidate single-step deltas; try toward target first, then alternatives.
    candidates = []
    candidates.append((dx1, dy1))
    candidates.append((dx1, 0))
    candidates.append((0, dy1))
    candidates.append((dx1, -dy1))
    candidates.append((-dx1, dy1))
    candidates.append((-dx1, -dy1))
    candidates.append((0, 0))

    def step_score(dx, dy):
        nx, ny = sx + dx, sy + dy
        if not in_bounds(nx, ny) or (nx, ny) in obs_set:
            return -10**9
        nd = man(nx, ny, tx, ty)
        # discourage stepping too close to opponent while moving
        opp_d = man(nx, ny, ox, oy)
        return (-nd, opp_d)

    best_step = (0, 0)
    best_step_val = None
    for dx, dy in candidates:
        v = step_score(dx, dy)
        if best_step_val is None or v > best_step_val:
            best_step_val = v
            best_step = (dx, dy)

    return [int(best_step[0]), int(best_step[1])]
