def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sp = observation.get("self_position") or [0, 0]
    op = observation.get("opponent_position") or [0, 0]
    sx, sy = int(sp[0]), int(sp[1])
    ox, oy = int(op[0]), int(op[1])

    resources = observation.get("resources") or []
    obstacles = observation.get("obstacles") or []
    obs = set()
    for p in obstacles:
        if p and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obs.add((x, y))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def dist(a, b):
        return max(abs(a[0] - b[0]), abs(a[1] - b[1]))  # diagonal allowed

    def step_towards(tx, ty, avoid=None):
        best = (0, 0)
        best_score = -10**18
        for dx in (-1, 0, 1):
            for dy in (-1, 0, 1):
                nx, ny = sx + dx, sy + dy
                if dx == 0 and dy == 0:
                    cand_ok = True
                else:
                    cand_ok = inb(nx, ny) and ((nx, ny) not in obs) and (avoid is None or (nx, ny) != avoid)
                if not cand_ok:
                    continue
                # Prefer moves that reduce distance to target; slight preference for staying in bounds.
                d = dist((nx, ny), (tx, ty))
                dself = dist((nx, ny), (sx, sy))
                score = -d * 10 - dself * 0.01
                if score > best_score:
                    best_score = score
                    best = (dx, dy)
        return [best[0], best[1]]

    # If no resources, drift away from opponent (resource_denier tends to camp).
    if not resources:
        dx = 1 if sx <= ox else -1 if sx >= ox else 0
        dy = 1 if sy <= oy else -1 if sy >= oy else 0
        nx, ny = sx + dx, sy + dy
        if inb(nx, ny) and (nx, ny) not in obs:
            return [dx, dy]
        return [0, 0]

    best_target = None
    best_val = -10**18

    # Strategy: if opponent is currently closer to some resource, contest the most valuable of those.
    # Otherwise, take the resource where we have the clearest tempo advantage; penalize long walks.
    for r in resources:
        if not r or len(r) < 2:
            continue
        tx, ty = int(r[0]), int(r[1])
        if not inb(tx, ty) or (tx, ty) in obs:
            continue
        sd = dist((sx, sy), (tx, ty))
        od = dist((ox, oy), (tx, ty))

        # Clear advantage means we should push it; if opponent is closer, we should deny by contesting.
        adv = od - sd  # positive => we are earlier
        # Add a deterministic "value" bias favoring resources farther from opponent when we are behind (deny).
        deny_bias = (sd - od)  # negative when opponent earlier
        # Weight early arrival heavily to be robust against denial camping.
        val = 0
        if adv >= 0:
            val = adv * 200 - sd * 6 - (sd == 0) * 1000
        else:
            # Contested: prioritize resources with biggest opponent closeness advantage and still reachable.
            # Also bias toward cells that make our next steps reduce the opponent's approach.
            val = (-adv) * 160 - sd * 5 - deny_bias * 20

        # Small tie-break: choose lower (x,y) deterministically.
        val = val - (tx * 0.0001 + ty * 0.00001)
        if val > best_val:
            best_val = val
            best_target = (tx, ty)

    if best_target is None:
        return [0, 0]

    tx, ty = best_target

    # If we can reach the target immediately, attempt it; else contest.
    return step_towards(tx, ty)
