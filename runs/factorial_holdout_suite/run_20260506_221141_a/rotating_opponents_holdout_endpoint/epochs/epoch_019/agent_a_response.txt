def choose_move(observation):
    w = observation["grid_width"]; h = observation["grid_height"]
    sx, sy = observation["self_position"]; ox, oy = observation["opponent_position"]
    resources = observation.get("resources") or []
    obstacles = set()
    for p in observation.get("obstacles") or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = p[0], p[1]
            if isinstance(x, int) and isinstance(y, int) and 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

    if not resources:
        dx = 0
        dy = 0
        if ox < sx: dx = -1
        elif ox > sx: dx = 1
        if oy < sy: dy = -1
        elif oy > sy: dy = 1
        nx, ny = sx + dx, sy + dy
        if 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obstacles:
            return [dx, dy]
        return [0, 0]

    deltas = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    def md(a, b, c, d): 
        return abs(a - c) + abs(b - d)

    # Counter sweep-rows: prefer resources with row far from opponent and/or where opponent is much farther than us.
    best = None
    for dx, dy in deltas:
        nx, ny = sx + dx, sy + dy
        if not (0 <= nx < w and 0 <= ny < h) or (nx, ny) in obstacles:
            continue

        best_cell = None
        for rx, ry in resources:
            if (rx, ry) in obstacles: 
                continue
            du = md(nx, ny, rx, ry)
            do = md(nx, ny, ox, oy) + md(ox, oy, rx, ry)  # slightly penalize if opponent is positioned to reach quickly
            row_gap = abs(ry - oy)  # go to rows less likely to be swept
            # Encourage finishing closer and winning contest; discourage opponent access.
            cand = (du, -row_gap, do - du, abs(ry - sy))
            if best_cell is None or cand < best_cell:
                best_cell = cand

        if best_cell is None:
            continue

        # Primary: minimize our distance; Secondary: maximize row separation; Tertiary: minimize opponent advantage
        overall = (best_cell[0], best_cell[1], best_cell[2], best_cell[3], nx, ny, dx, dy)
        if best is None or overall < best:
            best = (overall, dx, dy)

    if best is None:
        return [0, 0]
    return [best[1], best[2]]

                obstacles.add((x, y))

    def man(a, b, c, d):
        return abs(a - c) + abs(b - d)

    best_move = [0, 0]
    best_score = None

    has_resources = bool(resources)
    for dx, dy in deltas:
        nx, ny = sx + dx, sy + dy
        if not (0 <= nx < w and 0 <= ny < h):
            continue
        if (nx, ny) in obstacles:
            continue

        score = 0
        if has_resources:
            md_opp = man(nx, ny, ox, oy)
            score += md_opp
            nr = None
            for rx, ry in resources:
                if (rx, ry) in obstacles:
                    continue
                d = man(nx, ny, rx, ry)
