def choose_move(observation):
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    obstacles = observation.get("obstacles", [])
    obs_set = set((x, y) for x, y in obstacles)
    resources = observation.get("resources", [])
    gw = observation.get("grid_width", 8)
    gh = observation.get("grid_height", 8)

    def in_bounds(x, y):
        return 0 <= x < gw and 0 <= y < gh

    def best_delta_for(target):
        tx, ty = target
        dx = 0 if tx == sx else (1 if tx > sx else -1)
        dy = 0 if ty == sy else (1 if ty > sy else -1)
        return [dx, dy]

    def score_move(dx, dy):
        nx, ny = sx + dx, sy + dy
        if not in_bounds(nx, ny) or (nx, ny) in obs_set:
            return -10**9
        # distance to closest resource
        if resources:
            dmin = None
            for rx, ry in resources:
                dxr = nx - rx
                dyr = ny - ry
                d = dxr * dxr + dyr * dyr
                if dmin is None or d < dmin:
                    dmin = d
            # small bias to intercept opponent
            dox = nx - ox
            doy = ny - oy
            opp_d = dox * dox + doy * doy
            return -dmin - 0.01 * opp_d
        # no resources left: move to opponent to tie-break with points already earned
        dox = nx - ox
        doy = ny - oy
        return - (dox * dox + doy * doy)

    # If resources exist, head toward the nearest one (deterministic tie-break).
    if resources:
        best = None
        best_d = None
        for rx, ry in resources:
            dxr = sx - rx
            dyr = sy - ry
            d = dxr * dxr + dyr * dyr
            if best_d is None or d < best_d or (d == best_d and (rx, ry) < best):
                best_d = d
                best = (rx, ry)
        tx, ty = best
        preferred = best_delta_for((tx, ty))
        candidates = [preferred, [preferred[0], 0], [0, preferred[1]], [0, 0]]
        # add adjacent alternatives to avoid obstacles deterministically
        for ddx in (-1, 0, 1):
            for ddy in (-1, 0, 1):
                if ddx == 0 and ddy == 0:
                    continue
                candidates.append([ddx, ddy])

        seen = set()
        ordered = []
        for d in candidates:
            t = (d[0], d[1])
            if t in seen:
                continue
            seen.add(t)
            if d[0] in (-1, 0, 1) and d[1] in (-1, 0, 1):
                ordered.append(d)

        best_move = None
        best_sc = None
        for d in ordered:
            sc = score_move(d[0], d[1])
            if best_sc is None or sc > best_sc or (sc == best_sc and (d[0], d[1]) < (best_move[0], best_move[1])):
                best_sc = sc
                best_move = d
        return [int(best_move[0]), int(best_move[1])]

    # No resources: move toward opponent corner deterministically.
    dx = 0 if ox == sx else (1 if ox > sx else -1)
    dy = 0 if oy == sy else (1 if oy > sy else -1)
    for cand in ([dx, dy], [dx, 0], [0, dy], [0, 0]):
        nx, ny = sx + cand[0], sy + cand[1]
        if in_bounds(nx, ny) and (nx, ny) not in obs_set:
            return [int(cand[0]), int(cand[1])]
    return [0, 0]
