    def free(x, y):
        return 0 <= x < gw and 0 <= y < gh and (x, y) not in obs

             (0, -1), (0, 0), (0, 1),
             (1, -1), (1, 0), (1, 1)]

    if not resources:
        best = (0, 0, 10**18)

            if not free(nx, ny):

            if score > best[2] or (score == best[2] and (dx, dy) < (best[0], best[1])):
                best = (dx, dy, score)
