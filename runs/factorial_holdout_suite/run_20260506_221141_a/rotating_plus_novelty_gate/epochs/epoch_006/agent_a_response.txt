def choose_move(observation):
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [w - 1, h - 1])
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    resources = observation.get("resources", []) or []
    obstacles = observation.get("obstacles", []) or []
    obs_set = set()
    for p in obstacles:
        try:
            x, y = int(p[0]), int(p[1])
        except:
            continue
        if 0 <= x < w and 0 <= y < h:
            obs_set.add((x, y))

    def cheb(x1, y1, x2, y2):
        dx = x1 - x2
        dy = y1 - y2
        return abs(dx) if abs(dx) > abs(dy) else abs(dy)

    if not resources:
        tx, ty = (w - 1) // 2, (h - 1) // 2
        dx = 0 if tx == sx else (1 if tx > sx else -1)
        dy = 0 if ty == sy else (1 if ty > sy else -1)
        return [dx, dy]

    # If resources exist, choose the move that maximizes our lead to the best contested resource.
    best = None
    for dx, dy in ((-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)):
        nx, ny = sx + dx, sy + dy
        if not (0 <= nx < w and 0 <= ny < h):
            continue
        if (nx, ny) in obs_set:
            continue

        # Evaluate next position vs all resources; primary is lead advantage, secondary closeness.
        best_adv = None
        best_md = None
        best_od = None
        for r in resources:
            rx, ry = int(r[0]), int(r[1])
            if not (0 <= rx < w and 0 <= ry < h) or (rx, ry) in obs_set:
                continue
            md = cheb(nx, ny, rx, ry)
            od = cheb(ox, oy, rx, ry)
            adv = od - md  # positive => we arrive no later than opponent
            cand = (adv, -md, md, od)
            if best_adv is None or cand > best_adv:
                best_adv, best_md, best_od = adv, md, od
        # Encourage taking a resource when we can be at least as fast as opponent.
        # Also penalize giving opponent a large lead.
        if best_adv is None:
            continue
        # Convert to tuple for deterministic ordering
        value = (best_adv, -best_md, best_od, -abs(nx - ox) - abs(ny - oy), -dx, -dy, dx + dy)
        if best is None or value > best:
            best = value
            best_move = [dx, dy]

    if best is None:
        return [0, 0]
    return best_move

    def dist(a, b):
        ax, ay = a
        bx, by = b
        adx = ax - bx
        ady = ay - by
        return abs(adx) if abs(adx) > abs(ady) else abs(ady)  # Chebyshev

    best_move = [0, 0]
    best_val = -10**18

    # Deterministically pick: best "lead" to a resource based on our next position.
    # Primary: minimize our distance to chosen resource minus opponent distance (maximize lead).
