def choose_move(observation):
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [w - 1, h - 1])
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    resources = observation.get("resources", []) or []
    obstacles = observation.get("obstacles", []) or []
    if not resources:
        return [0, 0]

    obs = [[False] * w for _ in range(h)]
    for p in obstacles:
        if p is None or len(p) < 2:
            continue
        x, y = int(p[0]), int(p[1])
        if 0 <= x < w and 0 <= y < h:
            obs[y][x] = True

    def bfs(startx, starty):
        INF = 10**9
        dist = [[INF] * w for _ in range(h)]
        if not (0 <= startx < w and 0 <= starty < h) or obs[starty][startx]:
            return dist
        dist[starty][startx] = 0
        qx = [startx]
        qy = [starty]
        qi = 0
        while qi < len(qx):
            x = qx[qi]
            y = qy[qi]
            qi += 1
            d = dist[y][x] + 1
            nx0 = -1
            for dy in (-1, 0, 1):
                ny = y + dy
                if ny < 0 or ny >= h:
                    continue
                for dx in (-1, 0, 1):
                    if dx == 0 and dy == 0:
                        continue
                    nx = x + dx
                    if nx < 0 or nx >= w or obs[ny][nx]:
                        continue
                    if d < dist[ny][nx]:
                        dist[ny][nx] = d
                        qx.append(nx)
                        qy.append(ny)
        return dist

    ds = bfs(sx, sy)
    do = bfs(ox, oy)

    def valid(x, y):
        return 0 <= x < w and 0 <= y < h and not obs[y][x]

    best = None
    best_key = None
    for p in resources:
        tx, ty = int(p[0]), int(p[1])
        if not valid(tx, ty):
            continue
        d1 = ds[ty][tx]
        d2 = do[ty][tx]
        if d1 >= 10**8:
            continue
        # Prefer resources we can reach earlier; if uncertain, slightly prefer disrupting opponent timing
        # Tie-break by shorter distance and then parity (anti-sweep/anti-straightline bias)
        parity = (tx + ty) & 1
        key = (d2 - d1, -d1, parity)
        if best_key is None or key > best_key:
            best_key = key
            best = (tx, ty)

    if best is None:
        return [0, 0]

    tx, ty = best
    curd = ds[sy][sx]
    moves = []
    for dy in (-1, 0, 1):
        for dx in (-1, 0, 1):
            if dx == 0 and dy == 0:
                nx, ny = sx, sy
            else:
                nx, ny = sx + dx, sy + dy
            if not valid(nx, ny):
                continue
            nd = ds[ny][nx]
            if nd >= 10**8:
                continue
            # Prefer strictly reducing distance to target; otherwise keep best possible progress.
            primary = -(abs(tx - nx) + abs(ty - ny))
            key = (-(nd), primary, (nx + ny) & 1)
            moves.append((key, [dx, dy]))
    if not moves:
        return [0, 0]
    moves.sort(reverse=True, key=lambda t: t[0])
    return moves[0][1]
