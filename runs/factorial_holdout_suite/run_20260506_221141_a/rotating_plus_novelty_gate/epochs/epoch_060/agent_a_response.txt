def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position", [0, 0]) or [0, 0]
    ox, oy = observation.get("opponent_position", [w - 1, h - 1]) or [w - 1, h - 1]
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    blocked = set()
    for p in observation.get("obstacles", []) or []:
        if p and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                blocked.add((x, y))

    resources = []
    for p in observation.get("resources", []) or []:
        if p and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h and (x, y) not in blocked:
                resources.append((x, y))
    if not resources:
        return [0, 0]

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    moves.sort()

    def bfs(start):
        INF = 10**9
        dist = [[INF] * h for _ in range(w)]
        if start in blocked:
            return dist
        from collections import deque
        q = deque([start])
        dist[start[0]][start[1]] = 0
        while q:
            x, y = q.popleft()
            d = dist[x][y] + 1
            for dx, dy in moves:
                nx, ny = x + dx, y + dy
                if 0 <= nx < w and 0 <= ny < h and (nx, ny) not in blocked and d < dist[nx][ny]:
                    dist[nx][ny] = d
                    q.append((nx, ny))
        return dist

    ds = bfs((sx, sy))
    do = bfs((ox, oy))

    best = None
    best_key = None
    for rx, ry in resources:
        dsr = ds[rx][ry]
        dor = do[rx][ry]
        if dsr >= 10**8:
            continue
        # Prefer resources we can reach sooner; then maximize opponent delay (tie-break).
        key = (dsr - dor, dsr, -(dor), rx, ry)
        if best_key is None or key < best_key:
            best_key = key
            best = (rx, ry)

    rx, ry = best
    if (sx, sy) == (rx, ry):
        return [0, 0]

    curd = ds[rx][ry]
    best_step = (0, 0)
    best_step_key = None
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if 0 <= nx < w and 0 <= ny < h and (nx, ny) not in blocked:
            nd = ds[nx][ny]
            if nd < curd:
                # Deterministic descent with preference that also keeps opponent away.
                step_key = (nd - do[nx][ny], nd, dx, dy)
                if best_step_key is None or step_key < best_step_key:
                    best_step_key = step_key
                    best_step = (dx, dy)

    # If no strict descent, do a step that minimizes our distance deterministically.
    if best_step_key is None:
        for dx, dy in moves:
            nx, ny = sx + dx, sy + dy
            if 0 <= nx < w and 0 <= ny < h and (nx, ny) not in blocked:
                step_key = (ds[nx][ny] - do[nx][ny], ds[nx][ny], dx, dy)
                if best_step_key is None or step_key < best_step_key:
                    best_step_key = step_key
                    best_step = (dx, dy)

    return [int(best_step[0]), int(best_step[1])]
