def choose_move(observation):
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    w, h = observation["grid_width"], observation["grid_height"]
    resources = observation.get("resources") or []
    obstacles = observation.get("obstacles") or []
    occ = set((x, y) for x, y in obstacles)

    deltas = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def cheb(ax, ay, bx, by):
        dx = ax - bx
        if dx < 0:
            dx = -dx
        dy = ay - by
        if dy < 0:
            dy = -dy
        return dx if dx > dy else dy

    def score_for_move(nx, ny):
        # Primary: best "capture advantage" against any resource (we closer than opponent).
        best_adv = -10**9
        best_margin = -10**9
        best_r = None
        my_min_d = 10**9
        for rx, ry in resources:
            my_d = cheb(nx, ny, rx, ry)
            op_d = cheb(ox, oy, rx, ry)
            if my_d < my_min_d:
                my_min_d = my_d
            adv = op_d - my_d  # positive -> likely closer than opponent
            # Prefer resources we can actually arrive no later than opponent.
            margin = adv
            if adv > best_adv or (adv == best_adv and margin > best_margin):
                best_adv = adv
                best_margin = margin
                best_r = (rx, ry)
        # Secondary: reduce our distance to nearest resource.
        # Tertiary: slightly avoid moving away from the chosen best resource.
        tie_pen = 0
        if best_r is not None:
            tie_pen = cheb(nx, ny, best_r[0], best_r[1])
        return (best_adv, -my_min_d, -tie_pen)

    if not resources:
        # Deterministic fallback: move toward the farther opponent corner to avoid deadlock and keep distance.
        corners = [(0, 0), (0, h - 1), (w - 1, 0), (w - 1, h - 1)]
        tx, ty = max(corners, key=lambda c: cheb(ox, oy, c[0], c[1]))
        best = (-10**9, [0, 0])
        for dx, dy in deltas:
            nx, ny = sx + dx, sy + dy
            if not inb(nx, ny) or (nx, ny) in occ:
                continue
            s = -cheb(nx, ny, tx, ty)
            if s > best[0]:
                best = (s, [dx, dy])
        return best[1]

    best_val = (-10**18, [0, 0])
    for dx, dy in deltas:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in occ:
            continue
        val = score_for_move(nx, ny)
        if val > best_val[0]:
            best_val = (val, [dx, dy])

    return best_val[1]
