def choose_move(observation):
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    resources = observation.get("resources") or []
    obstacles = observation.get("obstacles") or []
    env = observation.get("environment_name", "resource_collection")

    obs = set()
    for p in obstacles:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obs.add((x, y))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def cheb(x1, y1, x2, y2):
        dx = x1 - x2
        if dx < 0:
            dx = -dx
        dy = y1 - y2
        if dy < 0:
            dy = -dy
        return dx if dx > dy else dy

    res_set = set()
    for r in resources:
        if isinstance(r, (list, tuple)) and len(r) >= 2:
            x, y = int(r[0]), int(r[1])
            if inb(x, y):
                res_set.add((x, y))
    if (sx, sy) in res_set:
        return [0, 0]

    best_t = None
    best_score = -10**18
    for (rx, ry) in res_set:
        sd = cheb(sx, sy, rx, ry)
        od = cheb(ox, oy, rx, ry)
        # Aim for resources we can reach first; if tie, prefer far from opponent.
        # Also add slight preference for closer-to-completion routes.
        claim = 1.0 if sd <= od else 0.0
        score = claim * 1000 - sd * 8 + (od - sd) * 3
        # If opponent is already closer, still allow targets that are "high separation"
        if claim == 0.0:
            score = -sd * 10 + (od - sd) * 6
        if score > best_score:
            best_score = score
            best_t = (rx, ry)

    if best_t is None:
        return [0, 0]

    tx, ty = best_t
    # Choose next step among legal neighbors, maximizing progress while keeping away from obstacles
    moves = (-1, 0, 1)
    best_m = [0, 0]
    best_mv = -10**18
    for dx in moves:
        for dy in moves:
            nx, ny = sx + dx, sy + dy
            if not inb(nx, ny):
                continue
            if (nx, ny) in obs:
                continue
            # Avoid getting into positions where opponent can immediately pick next turn: penalize proximity to opponent.
            nd = cheb(nx, ny, tx, ty)
            oppd = cheb(nx, ny, ox, oy)
            my_to_opp = cheb(nx, ny, ox, oy)
            t_opp = cheb(ox, oy, tx, ty)
            # Reward reducing distance to target and increasing separation
            v = (cheb(sx, sy, tx, ty) - nd) * 20 + (oppd) * 0.8
            # If opponent can reach target sooner, prioritize staying at distance from opponent while moving
            if t_opp < nd:
                v -= 15
            # Mild obstacle pressure: prefer positions that have more free adjacent cells
            free = 0
            for ax in moves:
                for ay in moves:
                    xx, yy = nx + ax, ny + ay
                    if inb(xx, yy) and (xx, yy) not in obs:
                        free += 1
            v += free * 0.1
            if v > best_mv:
                best_mv = v
                best_m = [dx, dy]

    return best_m
