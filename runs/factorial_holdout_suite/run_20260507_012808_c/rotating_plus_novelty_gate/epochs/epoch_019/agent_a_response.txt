def choose_move(observation):

    h = int(observation.get("grid_height", 0) or 0)
    resources = observation.get("resources") or []

    if w <= 0 or h <= 0:
        return [0, 0]
    obstacles = set()

        try:

            obstacles.add((int(x), int(y)))
        except:
            pass

    def ok(x, y):

        r = (int(rx), int(ry))
        myd = d(s, r)
        opd = d(o, r)
        return (myd - opd, myd, r[0], r[1])

    target = min(resources, key=pick_key)
