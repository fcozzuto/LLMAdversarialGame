def choose_move(observation):
    w = observation.get("grid_width", 8)
    h = observation.get("grid_height", 8)
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    resources = observation.get("resources") or []
    obstacles = set(tuple(p) for p in (observation.get("obstacles") or []))
    turns_remaining = observation.get("turns_remaining", 0)

    def steps(a, b, c, d):
        dx = a - c
        if dx < 0: dx = -dx
        dy = b - d
        if dy < 0: dy = -dy
        return dx if dx > dy else dy

    def valid(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obstacles

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    if w <= 0 or h <= 0:
        return [0, 0]

    if not resources:
        tx, ty = (w - 1) // 2, (h - 1) // 2
        best = [0, 0]
        bestv = -10**9
        for dx, dy in moves:
            nx, ny = sx + dx, sy + dy
            if not valid(nx, ny):
                continue
            v = -steps(nx, ny, tx, ty)
            if v > bestv or (v == bestv and (dx, dy) < tuple(best)):
                bestv = v
                best = [dx, dy]
        return best

    best_move = [0, 0]
    best_score = -10**18
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not valid(nx, ny):
            continue

        # evaluate by best resource we could collect soon and safely (vs opponent)
        local_best = 10**18
        for rx, ry in resources:
            if (rx, ry) in obstacles:
                continue
            my_t = steps(nx, ny, rx, ry)
            op_t = steps(ox, oy, rx, ry)
            # prefer resources I can reach earlier; penalize contested ones
            urgency = my_t
            contest = op_t - my_t
            # if opponent arrives no later, discourage
            penalty = 0
            if contest >= 0:
                penalty = 6 + contest
            # slight prefer closer than remaining time
            time_bonus = 0
            if turns_remaining and my_t <= turns_remaining:
                time_bonus = -2
            v = urgency + penalty + time_bonus
            if v < local_best:
                local_best = v

        # tie-breaker: also keep moving toward the currently best (lowest v) direction implicitly
        # add small term for direct improvement to reduce dithering
        cur_best = local_best
        # higher score for smaller local_best
        score = -cur_best

        if score > best_score:
            best_score = score
            best_move = [dx, dy]
        elif score == best_score:
            # deterministic tie-break: prefer smaller step that reduces my distance to closest resource
            # (compute minimal distance from candidate and current)
            # measure to closest resource
            md_curr = 10**18
            md_next = 10**18
            for rx, ry in resources:
                t1 = steps(sx, sy, rx, ry)
                if t1 < md_curr: md_curr = t1
                t2 = steps(nx, ny, rx, ry)
                if t2 < md_next: md_next = t2
            if md_next < md_curr or (md_next == md_curr and (dx, dy) < tuple(best_move)):
                best_move = [dx, dy]

    return best_move
