def choose_move(observation):
    w = observation["grid_width"]
    h = observation["grid_height"]
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    resources = observation.get("resources") or []
    obstacles = observation.get("obstacles") or []
    obs_set = set(map(tuple, obstacles)) if obstacles else set()

    def cheb(x1, y1, x2, y2):
        dx = x1 - x2
        if dx < 0:
            dx = -dx
        dy = y1 - y2
        if dy < 0:
            dy = -dy
        return dx if dx > dy else dy

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    if not resources:
        return [0, 0]

    best = None  # (priority, urgency, tx, ty)
    for rx, ry in resources:
        ds = cheb(sx, sy, rx, ry)
        do = cheb(ox, oy, rx, ry)
        # Deny if opponent can reach no slower; urgency = how much I lag.
        if do <= ds:
            urgency = (ds - do)  # <=0 when I lag or tie
            priority = 0  # highest priority class
        else:
            urgency = (do - ds)  # >0 I am ahead
            priority = 1  # lower priority class
        # Also slightly prefer resources nearer to me to keep pressure.
        key = (priority, urgency, ds, rx, ry)
        if best is None or key < best:
            best = key + (rx, ry)

    _, _, _, tx, ty, = best[:5], best[5], best[6] if len(best) > 5 else (None, None)  # safety
    # Unpack deterministically
    tx = best[-2]
    ty = best[-1]

    # Move greedily toward target while avoiding obstacles.
    best_move = [0, 0]
    best_score = None
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obs_set:
            continue
        d = cheb(nx, ny, tx, ty)
        # Prefer smaller distance; then fewer "turns" from current move shape.
        # Deterministic tie-break: fixed move order already, so just compare score.
        score = (d, abs(dx) + abs(dy), 0 if dx == 0 and dy == 0 else 1)
        if best_score is None or score < best_score:
            best_score = score
            best_move = [dx, dy]

    if best_score is None:
        return [0, 0]
    return best_move

    if not resources or w is None or h is None:

    def cheb(x1, y1, x2, y2):
        dx = x1 - x2
        if dx < 0:
            dx = -dx
        dy = y1 - y2
        if dy < 0:
            dy = -dy
        return dx if dx > dy else dy

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    best_r = None
    best_key = None
    for r in resources:
        rx, ry = r
        ds = cheb(sx, sy, rx, ry)
        do = cheb(ox, oy, rx, ry)
        score = do - ds  # positive => I am closer
        key = (-score, ds, rx, ry)
        if best_key is None or key < best_key:
            best_key = key
            best_r = (rx, ry)

    tx, ty = best_r
    best_step = None
    best_step_key = None

        if nx < 0 or nx >= w or ny < 0 or ny >= h:
