def choose_move(observation):
    w = observation["grid_width"]
    h = observation["grid_height"]
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    resources = observation.get("resources") or []
    obstacles = observation.get("obstacles") or []
    obs_set = set((x, y) for x, y in obstacles)

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def cheb(ax, ay, bx, by):
        dx = ax - bx
        if dx < 0:
            dx = -dx
        dy = ay - by
        if dy < 0:
            dy = -dy
        return dx if dx > dy else dy

    if not resources:
        tx, ty = ox, oy
        best = None
        for dx, dy in moves:
            nx, ny = sx + dx, sy + dy
            if not inb(nx, ny) or (nx, ny) in obs_set:
                continue
            d = cheb(nx, ny, tx, ty)
            if best is None or (d, dx, dy) < best[0]:
                best = (d, [dx, dy])
        return best[1] if best else [0, 0]

    # Opponent's nearest resource distance right now
    opp_nearest = min(cheb(ox, oy, rx, ry) for rx, ry in resources)

    best = None
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obs_set:
            continue

        self_nearest = min(cheb(nx, ny, rx, ry) for rx, ry in resources)

        # Denial: after our move, how much can we increase opponent distance to their nearest resource?
        # (Choose the resource that realizes opponent_nearest; then compute its distance from us.)
        # If multiple resources tie for opponent_nearest, use the one that is closest to us (hardest to deny), then
        # penalize that; otherwise boost denial.
        opp_target_d = None
        deny_best = None
        for rx, ry in resources:
            d_opp = cheb(ox, oy, rx, ry)
            if d_opp != opp_nearest:
                continue
            d_us = cheb(nx, ny, rx, ry)
            # we want this target to be far from opponent after our move indirectly; approximate by making it hard for us to reach it fast
            # but encourage blocking by preferring states where opponent's nearest target is relatively far from us too.
            if deny_best is None or (d_us, rx, ry) < deny_best[0]:
                deny_best = (d_us, rx, ry)
        if deny_best is not None:
            d_us_for_opp_target = deny_best[0]
        else:
            d_us_for_opp_target = self_nearest

        # Primary: maximize lead (opp_nearest - self_nearest).
        # Secondary: minimize self_nearest (more direct collection).
        # Tertiary: maximize d_us_for_opp_target (approximate denial by not committing too hard to their nearest target).
        lead = opp_nearest - self_nearest
        key = (-lead, self_nearest, -d_us_for_opp_target, dx, dy)
        if best is None or key < best[0]:
            best = (key, [dx, dy])

    return best[1] if best else [0, 0]
