def choose_move(observation):
    w = observation.get("grid_width", 8)
    h = observation.get("grid_height", 8)
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    resources = observation.get("resources") or []
    obstacles = set(tuple(p) for p in (observation.get("obstacles") or []))

    def valid(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obstacles

    def cheb(x1, y1, x2, y2):
        dx = x1 - x2
        if dx < 0:
            dx = -dx
        dy = y1 - y2
        if dy < 0:
            dy = -dy
        return dx if dx > dy else dy

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    if not valid(sx, sy):
        sx = 0 if sx < 0 else (w - 1 if sx >= w else sx)
        sy = 0 if sy < 0 else (h - 1 if sy >= h else sy)

    if not resources:
        tx, ty = (w - 1) // 2, (h - 1) // 2
        best = (-10**18, 0, 0)
        for dx, dy in moves:
            nx, ny = sx + dx, sy + dy
            if not valid(nx, ny):
                continue
            d = -cheb(nx, ny, tx, ty)
            if d > best[0]:
                best = (d, dx, dy)
        return [best[1], best[2]]

    scored = set()  # avoid pathological duplicates (same resource points)
    res_list = []
    for r in resources:
        if isinstance(r, (list, tuple)) and len(r) == 2:
            rx, ry = int(r[0]), int(r[1])
            if (rx, ry) not in scored:
                scored.add((rx, ry))
                res_list.append((rx, ry))

    # Prefer moves that create/extend a "capture" advantage over the opponent.
    # Also keep moving toward the best contested resource.
    best_score = -10**18
    best_move = [0, 0]
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not valid(nx, ny):
            continue

        # Compute advantages per resource
        advantages = []
        my_best = 10**18
        for rx, ry in res_list:
            myd = cheb(nx, ny, rx, ry)
            opd = cheb(ox, oy, rx, ry)
            if myd < my_best:
                my_best = myd
            # advantage: positive means we're closer
            adv = opd - myd
            # strong bonus if we can be closer than opponent (capture-ish), mild penalty if far
            advantages.append((adv, myd, opd))

        advantages.sort(key=lambda t: t[0], reverse=True)

        # Take top few contested resources to guide multi-target behavior.
        k = 3 if len(advantages) >= 3 else len(advantages)
        top = advantages[:k]

        # Score: prioritize capture advantage, then closeness, penalize being behind overall.
        capture = 0
        closeness = 0
        behind_pen = 0
        for adv, myd, opd in top:
            if adv > 0:
                capture += adv * 6 + (10 - myd)
                closeness += -myd
            else:
                behind_pen += (-adv) * 3 + myd * 0.5

        # Tie-break toward shorter distance to the best available resource.
        # (Opponent-advantaged moves get higher capture already.)
        my_immediate = -my_best
        total = capture + closeness + my_immediate - behind_pen * 1.0

        if total > best_score:
            best_score = total
            best_move = [dx, dy]

    return best_move
