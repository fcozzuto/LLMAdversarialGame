def choose_move(observation):
    w = observation.get("grid_width", 0)
    h = observation.get("grid_height", 0)
    sx, sy = observation.get("self_position", (0, 0))
    ox, oy = observation.get("opponent_position", (0, 0))
    resources = observation.get("resources") or []
    obstacles = set(tuple(p) for p in (observation.get("obstacles") or []))
    if not resources or w <= 0 or h <= 0:
        dx, dy = sx - ox, sy - oy
        step_x = 0 if dx == 0 else (1 if dx > 0 else -1)
        step_y = 0 if dy == 0 else (1 if dy > 0 else -1)
        for mx, my in ((step_x, step_y), (step_x, 0), (0, step_y), (0, 0)):
            nx, ny = sx + mx, sy + my
            if 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obstacles:
                return [mx, my]
        return [0, 0]

    def nearest_score(nx, ny):
        best = 10**9
        for rx, ry in resources:
            d = abs(nx - rx) + abs(ny - ry)
            if d < best:
                best = d
        return best

    opp_d = abs(sx - ox) + abs(sy - oy)
    candidates = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    best_move = (0, 0)
    best_val = -10**18
    for mx, my in candidates:
        nx, ny = sx + mx, sy + my
        if not (0 <= nx < w and 0 <= ny < h) or (nx, ny) in obstacles:
            continue
        ns = -nearest_score(nx, ny)
        adv = (opp_d - (abs(nx - ox) + abs(ny - oy))) * 0.01 if opp_d <= 6 else 0.0
        val = ns + adv
        if val > best_val:
            best_val = val
            best_move = (mx, my)
    return [best_move[0], best_move[1]]
