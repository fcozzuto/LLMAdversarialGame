def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)
    sx = 0 if sx < 0 else (w - 1 if sx >= w else sx)
    sy = 0 if sy < 0 else (h - 1 if sy >= h else sy)
    ox = 0 if ox < 0 else (w - 1 if ox >= w else ox)
    oy = 0 if oy < 0 else (h - 1 if oy >= h else oy)

    obstacles = set()
    for p in (observation.get("obstacles") or []):
        try:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))
        except:
            pass

    resources = []
    for r in (observation.get("resources") or []):
        try:
            x, y = int(r[0]), int(r[1])
            if 0 <= x < w and 0 <= y < h and (x, y) not in obstacles:
                resources.append((x, y))
        except:
            pass

    def cheb(ax, ay, bx, by):
        dx = ax - bx
        if dx < 0: dx = -dx
        dy = ay - by
        if dy < 0: dy = -dy
        return dx if dx > dy else dy

    def best_for_opponent():
        if not resources:
            return None, None
        best = None
        bestd = None
        for rx, ry in resources:
            d = cheb(ox, oy, rx, ry)
            if best is None or d < bestd or (d == bestd and (rx < best[0] or (rx == best[0] and ry < best[1]))):
                best = (rx, ry)
                bestd = d
        return best, bestd

    def pick_target():
        if not resources:
            return (w - 1, h - 1) if (sx + sy) <= (ox + oy) else (0, 0)
        op_t, op_d = best_for_opponent()
        cand = []
        for rx, ry in resources:
            myd = cheb(sx, sy, rx, ry)
            opd = cheb(ox, oy, rx, ry)
            # Prefer denying opponent's nearest; otherwise take resources we can reach first.
            deny = 1 if op_t == (rx, ry) else 0
            lead = 1 if myd < opd else 0
            tie = 1 if myd == opd else 0
            cand.append((deny * 100000 + lead * 10000 + tie * 1000 + (-opd) * 5 + (-myd), rx, ry))
        cand.sort(reverse=True)
        return (cand[0][1], cand[0][2])

    tx, ty = pick_target()

    dx = 0 if tx == sx else (1 if tx > sx else -1)
    dy = 0 if ty == sy else (1 if ty > sy else -1)

    def step(nx, ny):
        if nx < 0 or ny < 0 or nx >= w or ny >= h:
            return False
        if (nx, ny) in obstacles:
            return False
        return True

    nx, ny = sx + dx, sy + dy
    if step(nx, ny):
        return [dx, dy]

    # Deterministic sidestep order (prioritize keeping direction in the major axis).
    options = []
    if dx != 0 and dy != 0:
        options = [(dx, 0), (0, dy), (dx, -dy), (-dx, dy), (0, 0)]
    else:
        options = [(dx, 0), (0, dy), (dx, dy), (0, 0)]
    for mx, my in options:
        nx, ny = sx + mx, sy + my
        if step(nx, ny):
            return [mx, my]

    return [0, 0]
