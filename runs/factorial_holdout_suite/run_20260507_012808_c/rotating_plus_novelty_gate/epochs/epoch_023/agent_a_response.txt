def choose_move(observation):
    w = observation.get("grid_width", 8)
    h = observation.get("grid_height", 8)
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    resources = observation.get("resources") or []
    obstacles = set(tuple(p) for p in (observation.get("obstacles") or []))

    if w <= 0 or h <= 0:
        return [0, 0]

    def cd(ax, ay, bx, by):
        return max(abs(ax - bx), abs(ay - by))

    def free(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obstacles

    cand_moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    if not resources:
        # drift toward center to reduce pathing bias
        tx, ty = (w - 1) // 2, (h - 1) // 2
        dx = 0 if sx == tx else (1 if tx > sx else -1)
        dy = 0 if sy == ty else (1 if ty > sy else -1)
        return [dx, dy]

    # Predict opponent's immediate priority (sweep_rows tends to keep taking nearby items)
    opp_best = None
    opp_best_d = 10**9
    for rx, ry in resources:
        d = cd(ox, oy, rx, ry)
        if d < opp_best_d:
            opp_best_d = d
            opp_best = (rx, ry)

    # Choose a resource that maximizes advantage, but actively contest opponent's priority
    best_r = None
    best_val = -10**18
    for rx, ry in resources:
        myd = cd(sx, sy, rx, ry)
        opd = cd(ox, oy, rx, ry)
        immediate = 500 if (rx, ry) == (sx, sy) else 0
        adv = opd - myd
        contest = 0
        if opp_best is not None:
            # If this is the opponent's target, we weight it higher to deny collection
            contest = 300 if (rx, ry) == opp_best else 0
            # If we are worse on the opponent target, steal another close-by on the same row/col
            if adv < 0 and (rx == opp_best[0] or ry == opp_best[1]):
                contest += 120
        # Additional pressure: prefer reducing distance to target quickly
        val = adv * 1200 + contest + immediate - myd
        if val > best_val:
            best_val = val
            best_r = (rx, ry)

    tx, ty = best_r

    # Move toward target with deterministic tie-breaks; avoid obstacles
    best_move = [0, 0]
    best_score = -10**18
    for dx, dy in cand_moves:
        nx, ny = sx + dx, sy + dy
        if not free(nx, ny):
            continue
        dcur = cd(sx, sy, tx, ty)
        dnew = cd(nx, ny, tx, ty)
        pick = 1 if (nx, ny) in resources else 0
        # Prefer faster progress, then slight contest with opponent if near their target
        contest = 0
        if opp_best is not None:
            contest = 1 if cd(nx, ny, opp_best[0], opp_best[1]) <= cd(sx, sy, opp_best[0], opp_best[1]) else 0
        score = (dcur - dnew) * 1000 + pick * 800 + contest * 10
        # Deterministic tie-break: smallest dx, then smallest dy after score
        if score > best_score or (score == best_score and (dx, dy) < (best_move[0], best_move[1])):
            best_score = score

        a = ax - bx
        b = ay - by
        if a < 0:
            a = -a
        if b < 0:
            b = -b
        return a if a > b else b

    def valid(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obstacles

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

        if valid(sx + dx, sy + dy):
            return [dx, dy]
        for dx, dy in moves:
            nx, ny = sx + dx, sy + dy
            if valid(nx, ny):
                return [dx, dy]
        return [0, 0]

    best_score = -10**18
