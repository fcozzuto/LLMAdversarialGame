def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position", (0, 0))
    ox, oy = observation.get("opponent_position", (0, 0))
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    obstacles = set()
    for p in observation.get("obstacles", []) or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

    resources = []
    for r in observation.get("resources", []) or []:
        if isinstance(r, (list, tuple)) and len(r) >= 2:
            x, y = int(r[0]), int(r[1])
            if 0 <= x < w and 0 <= y < h and (x, y) not in obstacles:
                resources.append((x, y))

    if not resources:
        return [0, 0]

    def man(a, b):
        ax, ay = a
        bx, by = b
        dx = ax - bx
        if dx < 0: dx = -dx
        dy = ay - by
        if dy < 0: dy = -dy
        return dx if dx > dy else dy  # Chebyshev

    def bounds(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obstacles

    def step_toward(tx, ty, curx, cury):
        dx = 0 if tx == curx else (1 if tx > curx else -1)
        dy = 0 if ty == cury else (1 if ty > cury else -1)
        nx, ny = curx + dx, cury + dy
        if not bounds(nx, ny):
            # try axis-aligned alternatives deterministically
            for adx, ady in ((dx, 0), (0, dy), (0, 0)):
                nx2, ny2 = curx + adx, cury + ady
                if bounds(nx2, ny2):
                    return [adx, ady]
            return [0, 0]
        return [dx, dy]

    # Choose target: normally maximize (opp_dist - self_dist).
    # If we're behind everywhere, switch to denial: chase the resource where
    # opponent is closest (smallest opp_dist), but still head to improve our catch likelihood.
    best = None
    best_key = None
    behind_all = True
    for r in resources:
        sd = man((r[0], r[1]), (sx, sy))
        od = man((r[0], r[1]), (ox, oy))
        if od <= sd:
            pass
        else:
            behind_all = False  # there exists at least one resource we are closer to
        lead = od - sd
        # tie-break prefers closer to self, then smaller x,y
        key = (lead, -sd, -r[0], -r[1])
        if best_key is None or key > best_key:
            best_key = key
            best = r

    if behind_all:
        # denial: target the resource opponent is closest to (min opp_dist),
        # break ties by maximizing our lead (od - sd), then prefer nearer to us.
        target = None
        tkey = None
        for r in resources:
            sd = man((r[0], r[1]), (sx, sy))
            od = man((r[0], r[1]), (ox, oy))
            key = (-od, od - sd, -sd, r[0], r[1])
            if tkey is None or key < tkey:
                tkey = key
                target = r
    else:
        target = best

    tx, ty = int(target[0]), int(target[1])
    dx, dy = step_toward(tx, ty, sx, sy)

    # Safety: if the chosen step is blocked, attempt other legal moves (deterministic).
    if not bounds(sx + dx, sy + dy):
        for adx, ady in ((-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)):
            if bounds(sx + adx, sy + ady):
                return [adx, ady]
        return [0, 0]

    return [dx, dy]
