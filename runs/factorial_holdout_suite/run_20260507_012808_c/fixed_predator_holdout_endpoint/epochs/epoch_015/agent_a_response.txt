def choose_move(observation):
    w = observation["grid_width"]
    h = observation["grid_height"]
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    resources = observation.get("resources", []) or []
    obstacles = set()
    for p in observation.get("obstacles", []) or []:
        if len(p) >= 2:
            obstacles.add((p[0], p[1]))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    moves = []
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            nx, ny = sx + dx, sy + dy
            if inb(nx, ny) and (nx, ny) not in obstacles:
                moves.append((dx, dy))
    if not moves:
        return [0, 0]

    if not resources:
        tx, ty = w // 2, h // 2
        best = (-10**18, 0, 0)
        for dx, dy in moves:
            nx, ny = sx + dx, sy + dy
            sd = abs(nx - tx) + abs(ny - ty)
            od = abs(nx - ox) + abs(ny - oy)
            v = (od - sd, -sd)
            if v > (best[0], best[1]):
                best = (v[0], v[1], 0)
                best_move = (dx, dy)
        return [best_move[0], best_move[1]]

    def man(a, b, c, d):
        return abs(a - c) + abs(b - d)

    best_v = (-10**18, -10**18)
    best_move = (0, 0)
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        best_res_v = (-10**18, -10**18)
        for rx, ry in resources:
            sd = man(nx, ny, rx, ry)
            od = man(nx, ny, ox, oy)  # opponent proximity pressure
            rel = man(nx, ny, ox, oy) - sd  # prefer cutting down our distance vs being close to them
            # primary: maximize (opp-to-resource - our-to-resource) so we arrive first
            primary = (man(ox, oy, rx, ry) - sd)
            # secondary: prefer closer resource and being farther from opponent (denier contest)
            secondary = (-sd, -man(nx, ny, ox, oy))
            v = (primary, secondary[0], secondary[1], rel)
            comp = (v[0], v[1])
            if comp > (best_res_v[0], best_res_v[1]):
                best_res_v = (v[0], v[1])
        # slight tie-break: also prefer moves that reduce distance to the closest resource overall
        closest_sd = min(manh(nx, ny, rx, ry) for rx, ry in resources)
        total_v = (best_res_v[0], -closest_sd)
        if total_v > best_v:
            best_v = total_v
            best_move = (dx, dy)

    return [best_move[0], best_move[1]]
