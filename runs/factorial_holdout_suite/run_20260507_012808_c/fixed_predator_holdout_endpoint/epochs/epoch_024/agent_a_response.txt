def choose_move(observation):
    w = int(observation["grid_width"])
    h = int(observation["grid_height"])
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]

    obstacles = set()
    for p in observation.get("obstacles", []) or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

    resources = []
    for r in observation.get("resources", []) or []:
        if isinstance(r, (list, tuple)) and len(r) >= 2:
            x, y = int(r[0]), int(r[1])
            if 0 <= x < w and 0 <= y < h and (x, y) not in obstacles:
                resources.append((x, y))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def md(x1, y1, x2, y2):
        return abs(x1 - x2) + abs(y1 - y2)

    moves = []
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            nx, ny = sx + dx, sy + dy
            if inb(nx, ny) and (nx, ny) not in obstacles:
                moves.append((dx, dy))
    if not moves:
        return [0, 0]

    if not resources:
        # drift toward the farthest corner to avoid getting stuck in obstacle clusters
        tx, ty = (w - 1, 7) if (sx + sy) < ((w - 1) + 7) else (0, 0)
        best = None
        bestk = None
        for dx, dy in moves:
            nx, ny = sx + dx, sy + dy
            k = md(nx, ny, tx, ty)
            if bestk is None or k > bestk:
                bestk = k
                best = (dx, dy)
        return [best[0], best[1]]

    # Evaluate each move by the best "advantage" it creates among resources.
    # Advantage = (opponent closer time) - (self closer time) under tie-breaks.
    # We also add a small term to prefer not moving away from the single best target.
    top = []
    for rx, ry in resources:
        ds0 = md(sx, sy, rx, ry)
        do0 = md(ox, oy, rx, ry)
        top.append((do0 - ds0, -ds0, rx, ry))
    top.sort(reverse=True)
    candidates = top[:min(6, len(top))]

    best_move = (0, 0)
    best_score = None
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        # choose the strongest resource after this move, but penalize if all are poor
        local_best = None
        local_second = None
        for adv, negds0, rx, ry in candidates:
            ds = md(nx, ny, rx, ry)
            do = md(ox, oy, rx, ry)
            adv2 = do - ds
            key = (adv2, -ds)
            if local_best is None or key > local_best:
                local_second = local_best
                local_best = key
            elif local_second is None or key > local_second:
                local_second = key
        # stable choice: maximize advantage, then minimize self distance to that best resource (encoded in -ds),
        # then maximize opponent disadvantage relative to current (avoid conceding too much).
        if local_best is None:
            score = (-10**9, 0, 0)
        else:
            adv2, nds2 = local_best
            # Slight bias toward moves that reduce distance to the current best target too.
            curr_best = candidates[0][2], candidates[0][3]
            cbx, cby = curr_best
            dist_bias = -md(nx, ny, cbx, cby)
            score = (adv2, nds2, dist_bias)
        if best_score is None or score > best_score:
            best_score = score
            best_move = (dx, dy)

    return [int(best_move[0]), int(best_move[1])]

def doppel_bonus(dopp):
    if dopp <= 1:
        return -10000
    if dopp <= 2:
        return -2000
    return dopp * 3
