def choose_move(observation):
    w = int(observation.get("grid_width") or 8)
    h = int(observation.get("grid_height") or 8)
    sx, sy = observation.get("self_position") or (0, 0)
    ox, oy = observation.get("opponent_position") or (0, 0)
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    obstacles = set()
    for p in observation.get("obstacles") or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

    resources = []
    for r in observation.get("resources") or []:
        if isinstance(r, (list, tuple)) and len(r) >= 2:
            x, y = int(r[0]), int(r[1])
            if 0 <= x < w and 0 <= y < h and (x, y) not in obstacles:
                resources.append((x, y))
    if not resources:
        return [0, 0]

    def md(ax, ay, bx, by):
        dx = ax - bx
        if dx < 0:
            dx = -dx
        dy = ay - by
        if dy < 0:
            dy = -dy
        return dx + dy

    best_t = None
    best_key = None
    for t in resources:
        sd = md(t[0], t[1], sx, sy)
        od = md(t[0], t[1], ox, oy)
        # If we can arrive no later, prioritize denial (maximize od-sd, then closer).
        if od - sd >= 0:
            key = (2, od - sd, -sd, -(t[0] + 31 * t[1]))
        else:
            # Otherwise, contest; minimize opponent advantage, then minimize own distance.
            key = (1, -od, sd, -(t[0] + 31 * t[1]))
        if best_key is None or key > best_key:
            best_key = key
            best_t = t

    tx, ty = best_t

    moves = []
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            nx, ny = sx + dx, sy + dy
            if 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obstacles:
                # Prefer lowering our distance; also account for opponent progress vs same target.
                self_d = md(nx, ny, tx, ty)
                opp_d = md(ox, oy, tx, ty)
                # Deny effect: reduce own distance more than opponent's distance changes is unknown,
                # so use current opp advantage as tie-break.
                key = (-self_d, sd - self_d if 'sd' in locals() else 0, -opp_d, (dx, dy))
                moves.append((key, [dx, dy]))
    if not moves:
        return [0, 0]
    moves.sort(key=lambda x: (x[0][0], x[0][1], x[0][2], -x[0][3][0], -x[0][3][1]), reverse=True)
    return moves[0][1]
