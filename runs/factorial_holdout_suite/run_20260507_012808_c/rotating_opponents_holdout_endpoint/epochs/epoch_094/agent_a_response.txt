def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sp = observation.get("self_position", [0, 0])
    op = observation.get("opponent_position", [0, 0])
    sx, sy = int(sp[0]), int(sp[1])
    ox, oy = int(op[0]), int(op[1])

    obstacles = set()
    for p in observation.get("obstacles") or []:
        if p and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

    resources = []
    for r in observation.get("resources") or []:
        if r and len(r) >= 2:
            x, y = int(r[0]), int(r[1])
            if 0 <= x < w and 0 <= y < h and (x, y) not in obstacles:
                resources.append((x, y))
    if not resources:
        return [0, 0]

    role_denier = (observation.get("opponent_role", "") == "resource_denier")
    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def cheb(x1, y1, x2, y2):
        dx = x1 - x2
        if dx < 0:
            dx = -dx
        dy = y1 - y2
        if dy < 0:
            dy = -dy
        return dx if dx > dy else dy

    best = (10**9, 10**9, -10**9, 0, 0)  # (score, selfDist, oppDist, dx, dy)
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not (0 <= nx < w and 0 <= ny < h) or (nx, ny) in obstacles:
            nx, ny = sx, sy
        min_self = 10**9
        min_opp = 10**9
        best_tgt = None
        for rx, ry in resources:
            sd = cheb(nx, ny, rx, ry)
            od = cheb(ox, oy, rx, ry)
            if sd < min_self or (sd == min_self and od < min_opp):
                min_self, min_opp, best_tgt = sd, od, (rx, ry)

        # If we arrive much later than opponent, heavily penalize; if denier, also avoid targets they can deny.
        lead = min_opp - min_self
        if role_denier:
            score = min_self + (0 if lead >= 0 else (10 + (-lead) * 4))
        else:
            score = min_self + (0 if lead >= 0 else (8 + (-lead) * 3))

        # Slight preference to move closer overall, and deterministically break ties by dx,dy ordering.
        cand = (score, min_self, min_opp, -dx * dx - dy * dy, dx, dy)
        if cand < (best[0], best[1], best[2], best[3], best[4] * 0, best[4], best[4]):
            best = (cand[0], cand[1], cand[2], cand[4], cand[5], cand[6])

    return [int(best[4]), int(best[5])]
