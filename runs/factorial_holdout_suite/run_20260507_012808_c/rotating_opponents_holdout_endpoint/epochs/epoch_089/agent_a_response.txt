def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    obstacles = set()
    for p in observation.get("obstacles") or []:
        if p and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

    resources = []
    for r in observation.get("resources") or []:
        if r and len(r) >= 2:
            x, y = int(r[0]), int(r[1])
            if 0 <= x < w and 0 <= y < h and (x, y) not in obstacles:
                resources.append((x, y))

    moves = [(-1, -1), (-1, 0), (-1, 1),
             (0, -1), (0, 0), (0, 1),
             (1, -1), (1, 0), (1, 1)]

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def man(x1, y1, x2, y2):
        return abs(x1 - x2) + abs(y1 - y2)

    if not resources:
        # Drift toward center while avoiding obstacles
        cx, cy = (w - 1) // 2, (h - 1) // 2
        best = (10**9, 10**9, (0, 0))
        for dx, dy in moves:
            nx, ny = sx + dx, sy + dy
            if inb(nx, ny) and (nx, ny) not in obstacles:
                v = (man(nx, ny, cx, cy), man(nx, ny, ox, oy), nx, ny)
                if v < best[:4]:
                    best = (v[0], v[1], (dx, dy))
        return best[2]

    # Pick a resource where we can beat the opponent (or minimize their advantage).
    # Preference: maximize (opp_d - self_d), then smaller self_d, then deterministic coordinate tie-break.
    best_cell = None
    best_key = None
    for tx, ty in resources:
        sd = man(sx, sy, tx, ty)
        od = man(ox, oy, tx, ty)
        key = (-(od - sd), sd, tx, ty)  # smaller is better
        if best_key is None or key < best_key:
            best_key = key
            best_cell = (tx, ty)

    tx, ty = best_cell

    # Choose the best immediate move that reduces our distance to target while not giving too much to opponent.
    # Also handle one-step obstacle collisions by only considering feasible next cells.
    best_move = (0, 0)
    best_val = (10**9, 10**9, 10**9)
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obstacles:
            continue
        sd2 = man(nx, ny, tx, ty)
        od2 = man(ox, oy, tx, ty)
        # Value: prioritize smaller sd2; break ties by larger lead (sd - sd2) vs static od2; then deterministic.
        lead_gain = man(sx, sy, tx, ty) - sd2
        opp_pressure = man(ox, oy, nx, ny)  # encourage moves that are harder for opponent to contest
        val = (sd2, -lead_gain, opp_pressure, nx, ny)
        if val < (best_val[0], best_val[1], best_val[2], 10**9, 10**9):
            best_val = (val[0], val[1], val[2])
            best_move = (dx, dy)

    return [int(best_move[0]), int(best_move[1])]
