def choose_move(observation):
    w = observation["grid_width"]
    h = observation["grid_height"]
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]

    obstacles = observation.get("obstacles", []) or []
    obs = set()
    for p in obstacles:
        if isinstance(p, dict):
            x = p.get("x", p.get("posx", p.get("position", (None, None))[0]))
            y = p.get("y", p.get("posy", p.get("position", (None, None))[1]))
            if x is not None and y is not None:
                obs.add((int(x), int(y)))
        else:
            try:
                obs.add((int(p[0]), int(p[1])))
            except:
                pass

    def to_xy(r):
        if isinstance(r, dict):
            x = r.get("x", r.get("posx", r.get("position", (None, None))[0]))
            y = r.get("y", r.get("posy", r.get("position", (None, None))[1]))
            if x is None or y is None:
                return None
            x, y = int(x), int(y)
        else:
            try:
                x, y = int(r[0]), int(r[1])
            except:
                return None
        if 0 <= x < w and 0 <= y < h:
            return x, y
        return None

    resources = observation.get("resources", []) or []
    res_xy = []
    for r in resources:
        xy = to_xy(r)
        if xy is not None and xy not in obs:
            res_xy.append(xy)

    def cheb(a, b, c, d):
        dx = a - c
        dy = b - d
        return dx if dx >= 0 else -dx if dy is None else (dy if dy >= 0 else -dy)

    def dist(x1, y1, x2, y2):
        dx = x1 - x2
        if dx < 0: dx = -dx
        dy = y1 - y2
        if dy < 0: dy = -dy
        return dx if dx > dy else dy

    # Target selection: prefer resources we are likely to reach earlier than opponent.
    best = None
    best_key = None
    for rx, ry in res_xy:
        sd = dist(sx, sy, rx, ry)
        od = dist(ox, oy, rx, ry)
        # Lower is better; self advantage dominates.
        key = (sd - 0.85 * od, sd, rx, ry)
        if best_key is None or key < best_key:
            best_key = key
            best = (rx, ry)

    if best is None:
        tx, ty = w // 2, h // 2
    else:
        tx, ty = best

    # Choose move among valid deltas, avoiding obstacles, minimizing next heuristic.
    deltas = [(dx, dy) for dx in (-1, 0, 1) for dy in (-1, 0, 1)]
    best_move = (0, 0)
    best_val = None
    for dx, dy in deltas:
        nx, ny = sx + dx, sy + dy
        if nx < 0 or nx >= w or ny < 0 or ny >= h:
            continue
        if (nx, ny) in obs:
            continue
        sd_next = dist(nx, ny, tx, ty)
        if best is None:
            target_val = sd_next
        else:
            od_now = dist(ox, oy, tx, ty)
            # Encourage continuing to be ahead of opponent on the chosen target.
            target_val = sd_next - 0.85 * od_now
        # Slight preference for reducing direct distance to target; tie-break toward staying/diagonal.
        val = (target_val, sd_next, -abs(dx) - abs(dy), nx, ny)
        if best_val is None or val < best_val:
            best_val = val
            best_move = (dx, dy)

    return [int(best_move[0]), int(best_move[1])]
