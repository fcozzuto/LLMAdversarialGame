def choose_move(observation):
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    sp = observation.get("self_position", [0, 0])
    op = observation.get("opponent_position", [0, 0])
    sx, sy = int(sp[0]), int(sp[1])
    ox, oy = int(op[0]), int(op[1])
    resources = observation.get("resources") or []
    obstacles = observation.get("obstacles") or []
    obst = set()
    for p in obstacles:
        try:
            obst.add((int(p[0]), int(p[1])))
        except Exception:
            pass
    res = []
    for r in resources:
        try:
            x, y = int(r[0]), int(r[1])
            if 0 <= x < w and 0 <= y < h and (x, y) not in obst:
                res.append((x, y))
        except Exception:
            pass

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def bfs_dist(start):
        INF = 10**9
        dist = [[INF] * h for _ in range(w)]
        if not (0 <= start[0] < w and 0 <= start[1] < h) or start in obst:
            return dist
        q = [(start[0], start[1])]
        dist[start[0]][start[1]] = 0
        head = 0
        while head < len(q):
            x, y = q[head]
            head += 1
            d = dist[x][y] + 1
            for dx, dy in dirs:
                nx, ny = x + dx, y + dy
                if 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obst:
                    if d < dist[nx][ny]:
                        dist[nx][ny] = d
                        q.append((nx, ny))
        return dist

    ds = bfs_dist((sx, sy))
    do = bfs_dist((ox, oy))
    if not res:
        return [0, 0]

    best = None
    for rx, ry in res:
        d1 = ds[rx][ry]
        d2 = do[rx][ry]
        if d1 >= 10**9:
            continue
        # Prefer resources we can reach earlier; then maximize opponent delay; then minimize our time.
        key = (-(d2 - d1), d1, rx, ry)
        if best is None or key < best[0]:
            best = (key, (rx, ry), d1)
    if best is None:
        return [0, 0]
    (rx, ry) = best[1]
    if sx == rx and sy == ry:
        return [0, 0]

    target_d = ds[rx][ry]
    best_next = None
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obst and ds[nx][ny] == target_d - 1:
            # tie-break: keep reducing opponent access to the target region
            val = (do[nx][ny] - do[rx][ry], abs(nx - rx) + abs(ny - ry), nx, ny)
            if best_next is None or val < best_next[0]:
                best_next = (val, dx, dy)
    if best_next is None:
        # fallback: step minimizing our distance to target
        best2 = None
        for dx, dy in dirs:
            nx, ny = sx + dx, sy + dy
            if 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obst:
                val = (ds[nx][ny], nx, ny)
                if best2 is None or val < best2[0]:
                    best2 = (val, dx, dy)
        return [int(best2[1]), int(best2[2])]
    return [int(best_next[1]), int(best_next[2])]
