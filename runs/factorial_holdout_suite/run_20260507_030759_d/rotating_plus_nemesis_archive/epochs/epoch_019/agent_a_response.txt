def choose_move(observation):
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    res = observation.get("resources") or []
    if not res:
        return [0, 0]

    obstacles = observation.get("obstacles") or []
    obst = set()
    for p in obstacles:
        try:
            obst.add((int(p[0]), int(p[1])))
        except Exception:
            pass

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    width = int(observation.get("grid_width", 8))
    height = int(observation.get("grid_height", 8))

    def inb(x, y):
        return 0 <= x < width and 0 <= y < height

    def md(x1, y1, x2, y2):
        return abs(x1 - x2) + abs(y1 - y2)

    # Target: prioritize resources we can contest (opp not much closer), then ensure we are the closest.
    best_r = None
    best_key = None
    for r in res:
        tx, ty = int(r[0]), int(r[1])
        ds = md(sx, sy, tx, ty)
        do = md(ox, oy, tx, ty)
        contest = (do - ds)
        # prefer reducing our distance, and resources far from opponent for denier matchup
        key = (contest, -ds, abs(ty - oy), -abs(tx - ox), tx, ty)
        if best_key is None or key > best_key:
            best_key = key
            best_r = (tx, ty)
    tx, ty = best_r

    # Evaluate immediate move: maximize advantage and avoid moving toward positions likely to be denied.
    best_m = (0, 0)
    best_score = None
    res_set = set((int(r[0]), int(r[1])) for r in res)

    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obst:
            continue

        if (nx, ny) in res_set:
            score = 10**9  # immediate collection
        else:
            ds2 = md(nx, ny, tx, ty)
            do2 = md(ox, oy, tx, ty)
            advantage = do2 - ds2

            # Denial avoidance: if we move such that opponent can likely steal next (same tx, reduced do).
            # Approximate by whether opponent is already at equal/closer distance.
            deny_risk = 1 if do2 <= ds2 else 0

            # Encourage movement that increases separation in "useful" coordinate (toward target).
            toward = md(nx, ny, tx, ty) < md(sx, sy, tx, ty)

            score = (advantage, toward, -ds2, -deny_risk, tx, ty)
        if best_score is None or score > best_score:
            best_score = score
            best_m = (dx, dy)

    return [int(best_m[0]), int(best_m[1])]

        score = (do - ds) * 1000 - ds

    tx, ty = best

    # Choose move maximizing improvement toward target and avoiding obstacles.
