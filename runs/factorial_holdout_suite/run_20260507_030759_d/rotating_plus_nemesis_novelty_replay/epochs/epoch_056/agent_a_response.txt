def choose_move(observation):
    sp = observation.get("self_position", [0, 0]) or [0, 0]
    op = observation.get("opponent_position", [0, 0]) or [0, 0]
    sx, sy = int(sp[0]), int(sp[1])
    ox, oy = int(op[0]), int(op[1])
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)

    obstacles = set()
    for o in observation.get("obstacles", []) or []:
        if isinstance(o, (list, tuple)) and len(o) >= 2:
            obstacles.add((int(o[0]), int(o[1])))

    resources = []
    for r in observation.get("resources", []) or []:
        if isinstance(r, (list, tuple)) and len(r) >= 2:
            cell = (int(r[0]), int(r[1]))
            if cell not in obstacles:
                resources.append(cell)

    if not resources:
        return [0, 0]

    def cheb(x1, y1, x2, y2):
        dx = x1 - x2
        if dx < 0:
            dx = -dx
        dy = y1 - y2
        if dy < 0:
            dy = -dy
        return dx if dx > dy else dy

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    valid_cache = {}
    def valid(nx, ny):
        if (nx, ny) in valid_cache:
            return valid_cache[(nx, ny)]
        ok = 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obstacles
        valid_cache[(nx, ny)] = ok
        return ok

    # Opponent greedy target from its current position
    opp_t = None
    opp_d = 10**9
    for rx, ry in resources:
        d = cheb(ox, oy, rx, ry)
        if d < opp_d:
            opp_d, opp_t = d, (rx, ry)

    def opp_step():
        tx, ty = opp_t
        best = (0, 0)
        bestd = 10**9
        for dx, dy in moves:
            nx, ny = ox + dx, oy + dy
            if not valid(nx, ny):
                continue
            d = cheb(nx, ny, tx, ty)
            if d < bestd:
                bestd, best = d, (dx, dy)
        return best, bestd

    (odx, ody), opp_dist_after = opp_step()
    opp_next = (ox + odx, oy + ody)

    best_move = (0, 0)
    best_val = -10**18

    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not valid(nx, ny):
            continue

        # Choose our nearest resource from the next position
        my_t = None
        my_d = 10**9
        for rx, ry in resources:
            d = cheb(nx, ny, rx, ry)
            if d < my_d:
                my_d, my_t = d, (rx, ry)

        # If we can land on a resource now, prioritize strongly
        score = 0
        if my_d == 0:
            score += 10**6
        score += (opp_dist_after - my_d) * 1000

        # Prefer moves that keep advantage if both targets are similar
        if my_t == opp_t:
            score += 200

        # Mildly prefer increasing distance from opponent (deterministic tie-break)
        sep = cheb(nx, ny, opp_next[0], opp_next[1])
        score += sep

        if score > best_val:
            best_val = score
            best_move = (dx, dy)

    return [int(best_move[0]), int(best_move[1])]
