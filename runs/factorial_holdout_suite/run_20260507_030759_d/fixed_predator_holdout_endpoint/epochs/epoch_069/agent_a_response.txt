def choose_move(observation):
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [w - 1, h - 1])
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    resources = observation.get("resources") or []
    obstacles = observation.get("obstacles") or []
    obst = set()
    for p in obstacles:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            obst.add((int(p[0]), int(p[1])))

    def ok(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obst

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def man(ax, ay, bx, by):
        return abs(ax - bx) + abs(ay - by)

    if not resources:
        tx, ty = w // 2, h // 2
        best = (0, 0)
        bestv = -10**9
        for dx, dy in dirs:
            nx, ny = sx + dx, sy + dy
            if ok(nx, ny):
                v = -man(nx, ny, tx, ty)
                if v > bestv:
                    bestv = v
                    best = (dx, dy)
        return [best[0], best[1]]

    best_t = None
    best_gain = -10**9
    for r in resources:
        if not (isinstance(r, (list, tuple)) and len(r) >= 2):
            continue
        tx, ty = int(r[0]), int(r[1])
        if not ok(tx, ty):
            continue
        sd = man(sx, sy, tx, ty)
        od = man(ox, oy, tx, ty)
        gain = (od - sd, -sd)  # prefer being closer; then closer to reduce time
        if gain > (best_gain, -10**9):
            best_gain = gain
            best_t = (tx, ty)

    if best_t is None:
        for dx, dy in dirs:
            nx, ny = sx + dx, sy + dy
            if ok(nx, ny):
                return [dx, dy]
        return [0, 0]

    tx, ty = best_t
    cur_best = (0, 0)
    cur_val = -10**18
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not ok(nx, ny):
            continue
        sd = man(nx, ny, tx, ty)
        od = man(ox, oy, tx, ty)
        # also softly deny by preferring moves that reduce opponent margin if they pursue this cell
        val = (od - sd, -sd, -abs(nx - ox) - abs(ny - oy))
        if val > cur_val:
            cur_val = val
            cur_best = (dx, dy)

    return [int(cur_best[0]), int(cur_best[1])]
