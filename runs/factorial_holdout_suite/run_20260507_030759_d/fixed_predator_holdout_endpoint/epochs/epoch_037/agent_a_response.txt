def choose_move(observation):
    w = observation["grid_width"]
    h = observation["grid_height"]
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    resources = observation.get("resources") or []
    obstacles = set(map(tuple, observation.get("obstacles") or []))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def bfs_from(px, py):
        INF = 10**9
        dist = [[INF] * h for _ in range(w)]
        if not inb(px, py) or (px, py) in obstacles:
            return dist
        dist[px][py] = 0
        q = [(px, py)]
        head = 0
        while head < len(q):
            x, y = q[head]
            head += 1
            nd = dist[x][y] + 1
            for dx, dy in dirs:
                nx, ny = x + dx, y + dy
                if inb(nx, ny) and (nx, ny) not in obstacles and nd < dist[nx][ny]:
                    dist[nx][ny] = nd
                    q.append((nx, ny))
        return dist

    dself = bfs_from(sx, sy)
    dop = bfs_from(ox, oy)

    # If no resources, drift toward center avoiding obstacles
    if not resources:
        tx, ty = w // 2, h // 2
        best = (0, 0)
        bestv = -10**18
        for dx, dy in dirs:
            nx, ny = sx + dx, sy + dy
            if not inb(nx, ny) or (nx, ny) in obstacles:
                continue
            v = -(abs(nx - tx) + abs(ny - ty))
            if v > bestv:
                bestv = v
                best = (dx, dy)
        return [best[0], best[1]]

    best_move = (0, 0)
    best_score = -10**18
    # Main decision: maximize lead over opponent to contested resource using shortest-path BFS
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obstacles:
            continue
        score = -10**18
        for rx, ry in resources:
            if (rx, ry) in obstacles:
                continue
            ds = dself[nx][ny]
            do = dop[rx][ry]
            if ds >= 10**8 or do >= 10**8:
                continue
            # Prefer resources where we can arrive sooner; also break ties by proximity
            lead = do - ds
            # Slightly prefer closer resources once lead is similar (and away from opponent distance)
            v = lead * 100 + (dop[rx][ry] < 10**8) * 0 + (-ds)
            if v > score:
                score = v
        if score > best_score:
            best_score = score
            best_move = (dx, dy)

    # Fallback: move toward nearest reachable resource by BFS distance
    if best_score <= -10**17:
        target = None
        td = 10**9
        for rx, ry in resources:
            if (rx, ry) in obstacles:
                continue
            d = dself[sx][sy]  # not useful, keep deterministic nearest via BFS from self
        for rx, ry in resources:
            if (rx, ry) in obstacles:
                continue
            d = dself[rx][ry]
            if d < td:
                td = d
                target = (rx, ry)
        if target is None:
            return [0, 0]
        tx, ty = target
        # Choose step that decreases BFS distance toward target
        curd = dself[sx][sy]
        best = (0, 0)
        bestd = curd
