def choose_move(observation):
    w = observation["grid_width"]
    h = observation["grid_height"]
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    resources = observation.get("resources") or []
    obstacles = {(p[0], p[1]) for p in (observation.get("obstacles") or [])}

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    valid = []
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if inb(nx, ny) and (nx, ny) not in obstacles:
            valid.append((dx, dy, nx, ny))
    if not valid:
        return [0, 0]

    if not resources:
        tx, ty = w // 2, h // 2
        best = None
        for dx, dy, nx, ny in valid:
            d = (abs(nx - tx) + abs(ny - ty), abs(nx - sx) + abs(ny - sy))
            if best is None or d < best[0]:
                best = (d, dx, dy)
        return [best[1], best[2]]

    # Choose the move that maximizes immediate advantage over opponent for the best resource.
    # Primary: (opp_dist - my_dist) after move; Secondary: my_dist; Tertiary: closest to opponent
    # for pursuit when both are far behind.
    best_move = None  # (primary, secondary, tertiary, dx, dy)
    for dx, dy, nx, ny in valid:
        my_best_primary = None
        my_best_secondary = None
        my_best_tertiary = None
        for rx, ry in resources:
            my_d = abs(nx - rx) + abs(ny - ry)
            opp_d = abs(ox - rx) + abs(oy - ry)
            primary = opp_d - my_d
            secondary = my_d
            tertiary = opp_d
            if (my_best_primary is None or
                primary > my_best_primary or
                (primary == my_best_primary and (secondary < my_best_secondary or
                 (secondary == my_best_secondary and tertiary < my_best_tertiary))))):
                my_best_primary, my_best_secondary, my_best_tertiary = primary, secondary, tertiary
        cand = (my_best_primary, my_best_secondary, my_best_tertiary, dx, dy)
        if best_move is None or cand > best_move:
            best_move = cand

    return [best_move[3], best_move[4]]

def choose_move(observation):
    w = observation["grid_width"]
    h = observation["grid_height"]
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    obstacles = set(map(tuple, observation.get("obstacles") or []))
    resources = observation.get("resources") or []

    dirs = [(-1,-1),(-1,0),(-1,1),(0,-1),(0,0),(0,1),(1,-1),(1,0),(1,1)]

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def d(a, b, c, d2):
        return abs(a - c) + abs(b - d2)

    def nearest_dist(x, y):
        if not resources:
            return None
        best = None
        for rx, ry in resources:
            if (rx, ry) in obstacles:
                continue
            dv = d(x, y, rx, ry)
            if best is None or dv < best:
                best = dv
        return best if best is not None else None

    best = None  # (primary, secondary, tertiary, dx, dy)
    target = (w // 2, h // 2)

    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obstacles:
            continue

        my_nd = nearest_dist(nx, ny)
        opp_nd = nearest_dist(ox, oy)
        if my_nd is None or opp_nd is None:
            my_d = d(nx, ny, target[0], target[1])
            opp_d = d(ox, oy, target[0], target[1])
            primary = opp_d - my_d
            secondary = my_d
            tertiary = d(nx, ny, ox, oy)
