def choose_move(observation):
    w = observation["grid_width"]
    h = observation["grid_height"]
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    resources = observation.get("resources") or []
    obstacles = set(map(tuple, observation.get("obstacles") or []))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def cheb(x1, y1, x2, y2):
        a = x1 - x2
        if a < 0:
            a = -a
        b = y1 - y2
        if b < 0:
            b = -b
        return a if a > b else b

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    if not resources:
        cx, cy = w // 2, h // 2
        dx = 0 if sx == cx else (1 if cx > sx else -1)
        dy = 0 if sy == cy else (1 if cy > sy else -1)
        nx, ny = sx + dx, sy + dy
        if inb(nx, ny) and (nx, ny) not in obstacles:
            return [dx, dy]
        return [0, 0]

    best = None
    for rx, ry in resources:
        if (rx, ry) in obstacles:
            continue
        sd = cheb(sx, sy, rx, ry)
        od = cheb(ox, oy, rx, ry)
        if sd == 0:
            return [0, 0]
        # Favor resources opponent is relatively worse on; tie-break by closer to self.
        adv = od - sd
        # Small deterministic bias: prefer moving toward that vector (helps focus).
        dxr = rx - sx
        dyr = ry - sy
        bias = 0
        if dxr != 0:
            bias += 1 if (dxr > 0) == (od > sd) else 0
        if dyr != 0:
            bias += 1 if (dyr > 0) == (od > sd) else 0
        key = (adv, -sd, bias, -rx, -ry)
        if best is None or key > best[0]:
            best = (key, (rx, ry))

    tx, ty = best[1]
    dx = 0 if sx == tx else (1 if tx > sx else -1)
    dy = 0 if sy == ty else (1 if ty > sy else -1)

    def cell_value(x, y):
        # Evaluate immediate neighbor by best achievable advantage from there.
        best_local = None
        for rx, ry in resources:
            if (rx, ry) in obstacles:
                continue
            sd = cheb(x, y, rx, ry)
            if sd == 0:
                return (10**6, 0)  # instant collect
            od = cheb(ox, oy, rx, ry)
            adv = od - sd
            key = (adv, -sd)
            if best_local is None or key > best_local:
                best_local = key
        return best_local if best_local is not None else (-10**9, 0)

    # Prefer direct step toward chosen target if valid.
    nx, ny = sx + dx, sy + dy
    if inb(nx, ny) and (nx, ny) not in obstacles:
        return [dx, dy]

    # Otherwise, take the best valid move by local evaluation.
    best_move = [0, 0]
    best_key = None
    for mx, my in dirs:
        nx, ny = sx + mx, sy + my
        if not inb(nx, ny) or (nx, ny) in obstacles:
            continue
        k = cell_value(nx, ny)
        if best_key is None or k > best_key:
            best_key = k
            best_move = [mx, my]
    return best_move
