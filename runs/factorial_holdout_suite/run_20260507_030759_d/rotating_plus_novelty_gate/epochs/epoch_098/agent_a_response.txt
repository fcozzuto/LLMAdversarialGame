def choose_move(observation):
    w = observation.get("grid_width", 8) or 8
    h = observation.get("grid_height", 8) or 8
    sx, sy = observation.get("self_position", (0, 0)) or (0, 0)
    ox, oy = observation.get("opponent_position", (0, 0)) or (0, 0)

    resources = observation.get("resources", []) or []
    obstacles = observation.get("obstacles", []) or []
    obs = set()
    for p in obstacles:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            obs.add((p[0], p[1]))

    res = []
    for p in resources:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            r = (p[0], p[1])
            if r not in obs:
                res.append(r)
    if not res:
        return [0, 0]

    def md(x1, y1, x2, y2):
        return abs(x1 - x2) + abs(y1 - y2)

    def bfs_dist(start, target, limit=10):
        if start == target:
            return 0
        tx, ty = target
        x0, y0 = start
        if (x0, y0) in obs or (tx, ty) in obs:
            return None
        q = [(x0, y0, 0)]
        seen = {(x0, y0)}
        while q:
            x, y, d = q.pop(0)
            if d >= limit:
                continue
            for dx in (-1, 0, 1):
                for dy in (-1, 0, 1):
                    if dx == 0 and dy == 0:
                        continue
                    nx, ny = x + dx, y + dy
                    if nx < 0 or ny < 0 or nx >= w or ny >= h:
                        continue
                    if (nx, ny) in obs or (nx, ny) in seen:
                        continue
                    if (nx, ny) == (tx, ty):
                        return d + 1
                    seen.add((nx, ny))
                    q.append((nx, ny, d + 1))
        return None

    deltas = [(dx, dy) for dx in (-1, 0, 1) for dy in (-1, 0, 1)]
    deltas.sort()  # deterministic

    best = None
    best_key = None
    for dx, dy in deltas:
        nx, ny = sx + dx, sy + dy
        if nx < 0 or ny < 0 or nx >= w or ny >= h or (nx, ny) in obs:
            nx, ny = sx, sy
            dx, dy = 0, 0

        # Prefer grabbing next if possible (immediate resource)
        if (nx, ny) in res:
            return [dx, dy]

        score_key = None
        for rx, ry in res:
            our = bfs_dist((nx, ny), (rx, ry), 10)
            if our is None:
                continue
            their = bfs_dist((ox, oy), (rx, ry), 10)
            if their is None:
                their = md(ox, oy, rx, ry)  # fallback heuristic
            their_md = md(ox, oy, rx, ry)
            # Key: maximize lead; then minimize our distance; then minimize opponent distance.
            key = (-(their - our), our, their_md, rx, ry)
            if score_key is None or key < score_key:
                score_key = key

        if score_key is None:
            continue

        # Small tie-break to prefer moving closer to board center (reduces stalling)
        center_bias = md(nx, ny, (w - 1) / 2, (h - 1) / 2)
        final_key = (score_key, center_bias)
        if best_key is None or final_key < best_key:
            best_key = final_key
            best = (dx, dy)

    if best is None:
        return [0, 0]
    return [int(best[0]), int(best[1])]
