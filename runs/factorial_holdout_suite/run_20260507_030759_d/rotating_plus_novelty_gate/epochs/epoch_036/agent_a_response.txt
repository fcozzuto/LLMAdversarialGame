def choose_move(observation):
    w = observation["grid_width"]
    h = observation["grid_height"]
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    resources = observation.get("resources", []) or []
    obstacles_list = observation.get("obstacles", []) or []
    obstacles = set((p[0], p[1]) for p in obstacles_list)

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    res_set = set((r[0], r[1]) for r in resources)
    if (sx, sy) in res_set and (sx, sy) not in obstacles:
        return [0, 0]

    moves = []
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            nx, ny = sx + dx, sy + dy
            if inb(nx, ny) and (nx, ny) not in obstacles:
                moves.append((dx, dy))
    if not moves:
        return [0, 0]
    if not resources:
        tx, ty = (w - 1, h - 1) if (sx + sy) <= (w - 1 - sx + h - 1 - sy) else (0, 0)
        dx = 0 if tx == sx else (1 if tx > sx else -1)
        dy = 0 if ty == sy else (1 if ty > sy else -1)
        if inb(sx + dx, sy + dy) and (sx + dx, sy + dy) not in obstacles:
            return [dx, dy]
        for ddx, ddy in ((dx, dy), (dx, 0), (0, dy), (0, 0)):
            nx, ny = sx + ddx, sy + ddy
            if inb(nx, ny) and (nx, ny) not in obstacles:
                return [ddx, ddy]
        return [0, 0]

    def cell_score(cx, cy):
        # Prefer states where we beat the opponent to the best nearby resource.
        best = 10**9
        tie = 10**9
        for rx, ry in resources:
            sdist = abs(cx - rx) + abs(cy - ry)
            odist = abs(ox - rx) + abs(oy - ry)
            # Higher when sdist is smaller than odist; weight for urgency.
            val = (sdist - odist, sdist)
            if val < best:
                best = val
                tie = sdist
        # Encourage moving toward the "current best" race and away from being trapped.
        # Also slightly prefer cells with more options.
        options = 0
        for mx in (-1, 0, 1):
            for my in (-1, 0, 1):
                nx, ny = cx + mx, cy + my
                if inb(nx, ny) and (nx, ny) not in obstacles:
                    options += 1
        return (best[0], best[1], -options)

    best_move = None
    best_val = None
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        immediate = 0
        if (nx, ny) in res_set:
            immediate = -10**6
        v = cell_score(nx, ny)
        # Lexicographic: minimize (race diff, our distance, negative options), with immediate boost.
        val = (immediate + v[0], immediate + v[1], v[2], abs(nx - ox) + abs(ny - oy))
        if best_val is None or val < best_val:
            best_val = val
            best_move = (dx, dy)

    return [int(best_move[0]), int(best_move[1])]

    res = set((r[0], r[1]) for r in resources)

    if (sx, sy) in res and (sx, sy) not in obstacles:
        return [0, 0]

    moves = []
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            if dx == 0 and dy == 0:
                continue
            nx, ny = sx + dx, sy + dy
            if 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obstacles:
                moves.append((dx, dy))
    if not moves:
        return [0, 0]

    def best_target(x, y):
        if not res:
            return None
        # Deterministic: if multiple at same distance, tie-break by coordinates
        bx, by = None, None
        bd = None
        for rx, ry in res:
            if (rx, ry) in obstacles:
                continue
            d = abs(rx - x) + abs(ry - y)
            if bd is None or d < bd or (d == bd and (rx, ry) < (bx, by)):
                bd = d
                bx, by = rx, ry
        return bx, by

    target = best_target(sx, sy)

    best = None
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if target is None:
            # No resources: head away from opponent slightly, prefer reducing distance to border corner
            d_opp = abs(nx - ox) + abs(ny - oy)
            border_score = (nx, ny)  # deterministic
            val = (-d_opp, -border_score[0], -border_score[1])
        else:
            tx, ty = target
            d_self = abs(tx - nx) + abs(ty - ny)
            d_opp = abs(tx - ox) + abs(ty - oy)
            # Prefer moves that reduce our distance to target; if tie, prefer increasing opponent distance
            val = (d_self, -d_opp, nx, ny)
        if best is None or val < best[0]:
            best = (val, dx, dy)

    return [best[1], best[2]]
