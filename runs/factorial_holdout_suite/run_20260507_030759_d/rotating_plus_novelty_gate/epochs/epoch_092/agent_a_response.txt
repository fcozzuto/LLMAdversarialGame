def choose_move(observation):
    w = observation.get("grid_width", 8) or 8
    h = observation.get("grid_height", 8) or 8
    sx, sy = observation.get("self_position", (0, 0)) or (0, 0)
    ox, oy = observation.get("opponent_position", (0, 0)) or (0, 0)

    resources = observation.get("resources", []) or []
    obstacles = observation.get("obstacles", []) or []

    obs = set()
    for p in obstacles:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            obs.add((p[0], p[1]))

    res = []
    for p in resources:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            res.append((p[0], p[1]))
    if not res:
        return [0, 0]

    legal = []
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            nx, ny = sx + dx, sy + dy
            if 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obs:
                legal.append((dx, dy))
    if not legal:
        return [0, 0]

    if (sx, sy) in res:
        return [0, 0]

    def md(x1, y1, x2, y2):
        return abs(x1 - x2) + abs(y1 - y2)

    # Select target resources we can reach at least not slower than opponent (tie favors us).
    candidates = []
    best_d_opp = 10**9
    for rx, ry in res:
        d_opp = md(ox, oy, rx, ry)
        d_self = md(sx, sy, rx, ry)
        if d_self <= d_opp:
            candidates.append((rx, ry, d_self, d_opp))
        if d_opp < best_d_opp:
            best_d_opp = d_opp

    if not candidates:
        # Fallback: go to closest resource to us.
        rx, ry = min(res, key=lambda p: md(sx, sy, p[0], p[1]))
        candidates = [(rx, ry, md(sx, sy, rx, ry), md(ox, oy, rx, ry))]

    # Deterministic move ordering preference: toward fewer-distance, then toward breaking ties (diagonal ok).
    def score_move(dx, dy):
        nx, ny = sx + dx, sy + dy
        best = -10**18
        for rx, ry, d_self, d_opp in candidates:
            nd_self = md(nx, ny, rx, ry)
            nd_opp = md(ox, oy, rx, ry)
            # Prefer landing on a resource.
            on_res = 1 if (nx, ny) == (rx, ry) else 0
            # Prefer resources we are (still) not slower for.
            lead = (nd_self - nd_opp)  # smaller is better
            # Encourage closeness.
            closeness = -nd_self
            # Small bias to move within grid and avoid being static unless necessary.
            static_pen = 1 if (dx == 0 and dy == 0) else 0
            # If opponent is very close to this resource, make it only secondary unless we can also reach quickly.
            opp_pressure = -nd_opp
            val = (on_res * 1000) + (-(lead) * 50) + (closeness * 5) + (opp_pressure * 0.5) - static_pen
            if val > best:
                best = val
        return best

    # Break ties deterministically by move ordering.
    move_order = []
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            if dx == 0 and dy == 0:
                continue
            move_order.append((dx, dy))
    move_order.append((0, 0))

    best_move = None
    best_val = -10**18
    for dx, dy in move_order:
        if (dx, dy) not in legal:
            continue
        v = score_move(dx, dy)
        if v > best_val:
            best_val = v
            best_move = (dx, dy)

    if best_move is None:
        return [0, 0]
    return [int(best_move[0]), int(best_move[1])]
