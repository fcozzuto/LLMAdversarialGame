def choose_move(observation):
    w = observation.get("grid_width", 8) or 8
    h = observation.get("grid_height", 8) or 8
    sx, sy = observation.get("self_position", (0, 0))
    ox, oy = observation.get("opponent_position", (0, 0))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    obstacles = set(tuple(p) for p in (observation.get("obstacles", []) or []))
    resources = set(tuple(r) for r in (observation.get("resources", []) or []))

    if (sx, sy) in resources:
        return [0, 0]

    dirs = (-1, 0, 1)
    legal = []
    for dx in dirs:
        for dy in dirs:
            nx, ny = sx + dx, sy + dy
            if inb(nx, ny) and (nx, ny) not in obstacles:
                legal.append((dx, dy))
    if not legal:
        return [0, 0]
    if not resources:
        # deterministic fallback: move to increase distance from opponent
        best = None
        for dx, dy in legal:
            nx, ny = sx + dx, sy + dy
            val = (-(abs(nx - ox) + abs(ny - oy)), dx, dy)
            if best is None or val > best:
                best = val
        return [best[1], best[2]]

    def dist(x1, y1, x2, y2):
        return abs(x1 - x2) + abs(y1 - y2)

    best_move = None
    for dx, dy in legal:
        nx, ny = sx + dx, sy + dy
        # pick the resource that gives the best "reach-first" advantage
        best_res = None
        for rx, ry in resources:
            myd = dist(nx, ny, rx, ry)
            opd = dist(nx, ny, ox, oy)
            # maximize: opponent being farther than us, then closer to resource
            cand = (opd - myd, -myd)
            if best_res is None or cand > best_res:
                best_res = cand
        # tie-break deterministically by move coordinates
        move_val = (best_res[0], best_res[1], -dx, -dy)
        if best_move is None or move_val > best_move:
            best_move = (move_val, dx, dy)

    return [best_move[1], best_move[2]]
