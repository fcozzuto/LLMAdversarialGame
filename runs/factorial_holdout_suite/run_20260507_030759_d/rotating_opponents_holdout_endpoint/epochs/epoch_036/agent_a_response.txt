def choose_move(observation):
    w = observation.get("grid_width", 8)
    h = observation.get("grid_height", 8)
    sx, sy = observation.get("self_position", (0, 0))
    ox, oy = observation.get("opponent_position", (0, 0))
    resources = observation.get("resources", []) or []
    obstacles = observation.get("obstacles", []) or []

    obs = set()
    for p in obstacles:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = p[0], p[1]
            if isinstance(x, int) and isinstance(y, int) and 0 <= x < w and 0 <= y < h:
                obs.add((x, y))

    def valid(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obs

    res = []
    for r in resources:
        if isinstance(r, (list, tuple)) and len(r) >= 2:
            x, y = r[0], r[1]
            if isinstance(x, int) and isinstance(y, int) and valid(x, y):
                res.append((x, y))
    if not res:
        return [0, 0]

    def man(a, b, c, d):
        return abs(a - c) + abs(b - d)

    deltas = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    deltas.sort()

    best = None
    for tx, ty in res:
        sd = man(sx, sy, tx, ty)
        od = man(ox, oy, tx, ty)
        contested = 1 if sd <= od else 0
        # Prefer resources we can reach first, then closer, then tie-break by coordinates.
        score = (10 * contested) - sd - 0.2 * od + (-0.01 * (tx + ty))
        cand = (score, -sd, tx, ty)
        if best is None or cand > best:
            best = cand
    _, __, tx, ty = best

    best_move = None
    for dx, dy in deltas:
        nx, ny = sx + dx, sy + dy
        if not valid(nx, ny):
            continue
        dcur = man(nx, ny, tx, ty)
        # Choose moves that reduce distance most; if equal, prefer moving toward opponent-blocking.
        od_next = man(ox, oy, tx, ty)  # opponent position fixed this turn
        step_score = (-dcur, -od_next, nx, ny, dx, dy)
        if best_move is None or step_score > best_move:
            best_move = step_score
    if best_move is None:
        return [0, 0]
    # Decode chosen move from nx,ny
    _, __, nx, ny, dx, dy = best_move
    return [int(nx - sx), int(ny - sy)]
