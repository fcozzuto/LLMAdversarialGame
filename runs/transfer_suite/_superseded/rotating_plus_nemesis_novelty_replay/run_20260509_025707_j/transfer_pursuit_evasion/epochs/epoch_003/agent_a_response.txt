def choose_move(observation):

    ox, oy = observation["opponent_position"]

    opp_role = observation.get("opponent_role", "") or ""
    self_is_pursuer = ("pursuer" in self_role.lower()) or ("pursuit" in self_role.lower())
    # if roles are swapped, infer from opponent role; default to pursue heuristic if ambiguous
    if not self_is_pursuer and (("pursuer" in opp_role.lower()) or ("pursuit" in opp_role.lower())):
        self_is_pursuer = False

    moves = [(dx, dy) for dx in (-1, 0, 1) for dy in (-1, 0, 1)]
    def in_bounds(x, y):
        return 0 <= x < w and 0 <= y < h
    def valid(x, y):
        return in_bounds(x, y) and (x, y) not in obstacles
    def dist2(a, b, c, d):
        dx, dy = a - c, b - d
        return dx * dx + dy * dy
    def free_degree(x, y):
        c = 0
        for dx in (-1, 0, 1):
            for dy in (-1, 0, 1):
                if dx == 0 and dy == 0:
                    continue
                nx, ny = x + dx, y + dy
                if valid(nx, ny):
                    c += 1
        return c

    # Predict opponent next position by assuming it moves like an evader if we are pursuer, else like pursuer.
    def predict_next(px, py, tx, ty, assume_evader):

            score = d if assume_evader else -d
