    h = int(observation.get("grid_height", 8) or 8)
    sp = observation.get("self_position") or [0, 0]
    op = observation.get("opponent_position") or [0, 0]
    sx, sy = int(sp[0]), int(sp[1])
    ox, oy = int(op[0]), int(op[1])

    obstacles = set()
    for t in observation.get("obstacles") or []:
        if isinstance(t, (list, tuple)) and len(t) >= 2:
            x, y = int(t[0]), int(t[1])
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

    def is_evader(role):
        r = (role or "").lower()
        return ("evader" in r) or ("runner" in r) or ("evasion" in r) or ("evasive" in r)

    self_evade = is_evader(observation.get("self_role"))
    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def valid(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obstacles

    def dist_after(ax, ay, bx, by):
        dx = abs(ax - bx)
        dy = abs(ay - by)
        return dx if dx > dy else dy  # Chebyshev

    def opp_best_move_for_assumption(myx, myy):
        # If opponent is evader, assume it maximizes distance from our hypothetical.
        # If opponent is pursuer, assume it minimizes distance to our hypothetical.
        opp_evade = not is_evader(observation.get("opponent_role"))
        # Opponent role string might not be reliable; use our own role: in pursuit_evasion,
        # typically opposite roles. If we are evader, opponent likely pursuer, else opposite.
        if self_evade:
            opp_evade = False
        else:
            opp_evade = True

        best = None
        bestv = None
        for ddx, ddy in dirs:

                continue
            dv = dist_after(myx, myy, nx, ny)
            if best is None:
                best, bestv = (nx, ny), dv
            else:
                if opp_evade:
                    if dv > bestv:
                        best, bestv = (nx, ny), dv
                else:
                    if dv < bestv:
                        best, bestv = (nx, ny), dv
        return best if best is not None else (ox, oy)

    # Block-aware greedy with one-step lookahead.
    best_move = (0, 0)
    best_val = None

        for dx, dy in dirs:

            oxm, oym = opp_best_move_for_assumption(nx, ny)
            v = dist_after(nx, ny, oxm, oym)
            # prefer diagonal motion when safe to disrupt zigzags
            diag_bonus = 0.01 if (dx != 0 and dy != 0) else 0.0
