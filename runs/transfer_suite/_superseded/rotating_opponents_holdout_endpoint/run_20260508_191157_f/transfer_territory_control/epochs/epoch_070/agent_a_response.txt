def choose_move(observation):
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    sx, sy = observation.get("self_position", [0, 0])
    sx, sy = int(sx), int(sy)

    obstacles = set()
    for p in observation.get("obstacles") or []:
        if p and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

    opp_terr = []
    for t in observation.get("opponent_territory") or []:
        if t and len(t) >= 2:
            x, y = int(t[0]), int(t[1])
            if 0 <= x < w and 0 <= y < h:
                opp_terr.append((x, y))
    opp_set = set(opp_terr)

    unclaimed = []
    for c in observation.get("unclaimed_cells") or []:
        if c and len(c) >= 2:
            x, y = int(c[0]), int(c[1])
            if 0 <= x < w and 0 <= y < h:
                unclaimed.append((x, y))

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    best = None

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    # If we can step onto opponent territory, prioritize that to flip.
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obstacles:
            continue
        if (nx, ny) in opp_set:
            rank = (0, abs(nx - sx) + abs(ny - sy), -nx, -ny)  # deterministic tie-break
            best = (rank, [dx, dy])
            break
    if best is not None:
        return best[1]

    targets = unclaimed if unclaimed else opp_terr
    if not targets:
        return [0, 0]

    # Choose move that minimizes distance to nearest target; penalize moving away from denser zones deterministically.
    # Also slight bias to expand into cells closer to center when counts are close.
    self_cnt = int(observation.get("self_territory_count", 0))
    opp_cnt = int(observation.get("opponent_territory_count", 0))
    cx, cy = (w - 1) / 2.0, (h - 1) / 2.0
    center_weight = 0.12 if abs(self_cnt - opp_cnt) <= 2 else 0.0

    # Precompute deterministic center score for targets
    def nearest_target_dist(nx, ny):
        bestd = None
        for tx, ty in targets:
            d = abs(tx - nx) + abs(ty - ny)
            if bestd is None or d < bestd:
                bestd = d
        return bestd if bestd is not None else 999

    # Count a mild preference for targets on the line toward opponent territory centroid if available.
    if opp_terr:
        ox = sum(x for x, _ in opp_terr) / len(opp_terr)
        oy = sum(y for _, y in opp_terr) / len(opp_terr)
    else:
        ox, oy = w - 1, h - 1

    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obstacles:
            continue
        d = nearest_target_dist(nx, ny)

        # Encourage directionality toward opponent side centroid (territory_center_claim style).
        dir_bonus = 0
        cur_dir = abs(sx - ox) + abs(sy - oy)
        new_dir = abs(nx - ox) + abs(ny - oy)
        if new_dir < cur_dir:
            dir_bonus = -0.25 * (cur_dir - new_dir)

        sc = d + center_weight * (abs(nx - cx) + abs(ny - cy)) + dir_bonus
        rank = (round(sc, 6), -nx, -ny, dx, dy)
        if best is None or rank < best[0]:
            best = (rank, [dx, dy])
