def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position") or (0, 0)
    ox, oy = observation.get("opponent_position") or (w - 1, h - 1)
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    blocked = set()
    for p in observation.get("obstacles") or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                blocked.add((x, y))

    self_t = set(map(tuple, observation.get("self_territory") or []))
    opp_t = set(map(tuple, observation.get("opponent_territory") or []))
    unclaimed = list(map(tuple, observation.get("unclaimed_cells") or []))

    def valid(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in blocked

    # Target selection: prioritize nearest unclaimed; if none, attack/contend opponent territory; else head to center.
    moves_dirs = [(dx, dy) for dx in (-1, 0, 1) for dy in (-1, 0, 1)]
    cx, cy = (w - 1) // 2, (h - 1) // 2
    candidates = []
    if unclaimed:
        # Deterministically choose best candidate by a composite score.
        for tx, ty in unclaimed:
            # Encourage reaching cells near center and slightly toward opponent side.
            score = (abs(tx - cx) + abs(ty - cy)) * 2
            score += abs(tx - ox) + abs(ty - oy)  # prefer toward opponent
            # Prefer cells not already ours/mostly free.
            if (tx, ty) in self_t:
                score += 5
            candidates.append((score, tx, ty))
        candidates.sort()
        tx, ty = candidates[0][1], candidates[0][2]
    elif opp_t:
        candidates = []
        for tx, ty in opp_t:
            score = (abs(tx - cx) + abs(ty - cy)) * 2 + (abs(tx - ox) + abs(ty - oy))
            candidates.append((score, tx, ty))
        candidates.sort()
        tx, ty = candidates[0][1], candidates[0][2]
    else:
        # No info: go to center, biased toward opponent.
        tx = cx + (1 if ox > sx else (-1 if ox < sx else 0))
        ty = cy + (1 if oy > sy else (-1 if oy < sy else 0))

    # Evaluate one-step moves; deterministically tie-break.
    best = None
    # Deterministic tie-break: prefer +dx then +dy then stay.
    moves_dirs.sort(key=lambda m: (-m[0], -m[1], 0 if m == (0, 0) else 1))
    for dx, dy in moves_dirs:
        nx, ny = sx + dx, sy + dy
        if not valid(nx, ny):
            continue
        # Base: distance to target.
        dist = abs(nx - tx) + abs(ny - ty)

        # Territory incentive:
        # entering opponent territory is good if flips; but still deterministic by raw heuristic.
        enter_opp = (nx, ny) in opp_t
        enter_unclaimed = (nx, ny) not in self_t and (nx, ny) not in opp_t and (nx, ny) not in blocked
        remain_ours = (nx, ny) in self_t

        bonus = 0
        if enter_opp:
            bonus -= 18
        elif enter_unclaimed:
            bonus -= 8
        elif remain_ours:
            bonus -= 2

        # Obstacle/edge safety: mildly discourage moving away from center when already close.
        edge_pen = (0 if 1 <= nx <= w - 2 else 3) + (0 if 1 <= ny <= h - 2 else 3)

        # Encourage approach in both axes (diagonal allowed).
        axis_pen = 0
        if dx == 0:
            axis_pen += 1
        if dy == 0:
            axis_pen += 1

        total = dist + edge_pen + axis_pen + (-bonus)

        key = (total, -dx, -dy, 0 if (dx, dy) == (0, 0) else 1)
        if best is None or key < best[0]:
            best = (key, dx, dy)

    if best is None:
        return [0, 0]
    return [int(best[1]), int(best[2])]
