def choose_move(observation):
    w = int(observation.get("grid_width") or 8)
    h = int(observation.get("grid_height") or 8)
    sp = observation.get("self_position") or [0, 0]
    op = observation.get("opponent_position") or [0, 0]
    sx, sy = int(sp[0]), int(sp[1])
    ox, oy = int(op[0]), int(op[1])

    blocked = set()
    for p in observation.get("obstacles") or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            bx, by = int(p[0]), int(p[1])
            if 0 <= bx < w and 0 <= by < h:
                blocked.add((bx, by))

    deltas = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def legal(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in blocked

    def sqdist(x1, y1, x2, y2):
        dx, dy = x1 - x2, y1 - y2
        return dx * dx + dy * dy

    def role_is_evader(role):
        r = str(role or "").lower()
        return ("evader" in r) or ("escape" in r) or ("runner" in r) or ("flee" in r) or ("defender" in r)

    self_evader = role_is_evader(observation.get("self_role"))
    opp_evader = role_is_evader(observation.get("opponent_role"))

    def best_response_value(myx, myy):
        # Opponent moves next to extremize distance from us, depending on their role.
        vals = []
        for dx, dy in deltas:
            nx, ny = ox + dx, oy + dy
            if not legal(nx, ny):
                continue
            d = sqdist(myx, myy, nx, ny)
            # Pursuer tries to minimize distance; evader tries to maximize.
            vals.append(d if opp_evader else -d)
        if not vals:
            d = sqdist(myx, myy, ox, oy)
            return d if opp_evader else -d
        return max(vals) if opp_evader else max(vals)  # already signed

    best_move = [0, 0]
    best_score = None

    for dx0, dy0 in deltas:
        nx0, ny0 = sx + dx0, sy + dy0
        if not legal(nx0, ny0):
            continue
        # 1-step lookahead minimax: we optimize our objective, opponent reacts.
        d_after = sqdist(nx0, ny0, ox, oy)
        # If we are pursuer: higher is worse (maximize capture progress -> minimize distance).
        # If we are evader: higher is better (survive -> maximize distance).
        base = -d_after if not self_evader else d_after

        # Convert opponent term back to a consistent "we prefer larger" scale.

        # If opponent is pursuer, they minimize distance => base should reflect actual distance after their move.
        # We approximate by using opp_term as signed distance preference for them and translate:
        # - If opp_evader: opp_term is max distance; use it directly in our base sign.
        # - If opp pursuer: opp_term is max(-distance)= -min distance; recover min distance as -opp_term.
        dist_after = opp_term if opp_evader else -opp_term
        lookahead = (-dist_after if not self_evader else dist_after)
