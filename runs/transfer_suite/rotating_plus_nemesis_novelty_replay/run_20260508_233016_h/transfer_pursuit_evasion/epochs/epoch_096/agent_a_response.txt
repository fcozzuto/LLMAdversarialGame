def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sp = observation.get("self_position", [0, 0]) or [0, 0]
    op = observation.get("opponent_position", [0, 0]) or [0, 0]
    sx, sy = int(sp[0]), int(sp[1])
    ox, oy = int(op[0]), int(op[1])

    blocked = set()
    for b in observation.get("obstacles", []) or []:
        if isinstance(b, (list, tuple)) and len(b) >= 2:
            bx, by = int(b[0]), int(b[1])
            if 0 <= bx < w and 0 <= by < h:
                blocked.add((bx, by))

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def safe(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in blocked

    def manh(x1, y1, x2, y2):
        d = x1 - x2
        if d < 0:
            d = -d
        e = y1 - y2
        if e < 0:
            e = -e
        return d + e

    def mobility(x, y):
        c = 0
        for dx, dy in moves:
            nx, ny = x + dx, y + dy
            if safe(nx, ny):
                c += 1
        return c

    role = str(observation.get("self_role", "") or "").lower()
    is_evader = ("evader" in role) or role.endswith("e")

    # Corner bias: evader tries to keep far while preferring a corner farthest from pursuer.
    corners = [(0, 0), (w - 1, 0), (0, h - 1), (w - 1, h - 1)]
    if is_evader:
        best_corner = None
        best_cd = -1
        for cx, cy in corners:
            cd = manh(cx, cy, ox, oy)
            if cd > best_cd:
                best_cd = cd
                best_corner = (cx, cy)
        tcx, tcy = best_corner
        target = (tcx, tcy)
        desired = 1  # increase distance
    else:
        # Pursuer: chase with slight intercept bias along axis of evader position.
        desired = -1  # decrease distance
        # If opponent is in a corner, try to approach along the dominant axis.
        in_corner = (ox in (0, w - 1)) and (oy in (0, h - 1))
        target = None
        if in_corner:
            # Approach the corner cell itself (greedy), but fall back to direct chase if blocked.
            target = (ox, oy)

    candidates = []
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not safe(nx, ny):
            continue
        d_opp = manh(nx, ny, ox, oy)

        if is_evader:
            # Prefer moving away from pursuer, and also toward chosen corner, while keeping mobility.
            corner_d = manh(nx, ny, target[0], target[1])
            # maximize: (d_opp) - small corner_d + mobility
            score = (desired * (-d_opp)) + (-0.15 * corner_d) + (0.08 * mobility(nx, ny))
        else:
            # minimize: distance; intercept if possible by aiming at target when provided
            if target is not None:
                d_t = manh(nx, ny, target[0], target[1])
                score = (-d_opp) + (-0.08 * d_t) + (0.06 * mobility(nx, ny))
            else:
                score = (-d_opp) + (0.06 * mobility(nx, ny))

        # Deterministic tie-break: lexicographic (dx,dy) preference after score.
        candidates.append((score, dx, dy))

    if not candidates:
        return [0, 0]
