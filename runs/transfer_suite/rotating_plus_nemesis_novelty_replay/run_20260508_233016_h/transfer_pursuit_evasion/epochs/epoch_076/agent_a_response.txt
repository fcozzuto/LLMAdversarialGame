    if isinstance(path, list) and len(path) >= 2:
        p0, p1 = path[-2], path[-1]
        prev = (int(p1[0]) - int(p0[0]), int(p1[1]) - int(p0[1]))

    best_val = None

    for dx, dy in dlist:
