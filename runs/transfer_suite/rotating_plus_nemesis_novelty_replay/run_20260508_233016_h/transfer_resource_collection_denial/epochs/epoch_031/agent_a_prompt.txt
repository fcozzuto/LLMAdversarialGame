    resources = observation.get("resources", [])
    if not resources:
        return [0, 0]
    def score(cell):
        self_d = abs(cell[0] - sx) + abs(cell[1] - sy)
        opp_d = abs(cell[0] - ox) + abs(cell[1] - oy)
        return (self_d - opp_d, self_d)
    target = min(resources, key=score)
