def choose_move(observation):
    sx, sy = observation["self_position"]
    w, h = observation["grid_width"], observation["grid_height"]
    ox, oy = observation["opponent_position"]
    obstacles = set(tuple(p) for p in (observation.get("obstacles") or []))

    unclaimed = set(tuple(p) for p in (observation.get("unclaimed_cells") or []))
    selfT = set(tuple(p) for p in (observation.get("self_territory") or []))
    oppT = set(tuple(p) for p in (observation.get("opponent_territory") or []))

    candidates = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    cx, cy = w // 2, h // 2

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def dist(ax, ay, bx, by):
        return abs(ax - bx) + abs(ay - by)

    # Prefer pushes into unclaimed / opponent, while avoiding dead-ends near obstacles
    def cell_value(nx, ny):
        if (nx, ny) in obstacles:
            return -10**9
        if (nx, ny) in oppT:
            flip_gain = 180
        elif (nx, ny) in unclaimed:
            flip_gain = 90
        elif (nx, ny) in selfT:
            flip_gain = 10
        else:
            flip_gain = 20

        # Progress toward center and away from opponent slightly, except when we are flipping
        dcenter = dist(nx, ny, cx, cy)
        dome = dist(nx, ny, sx, sy)
        doppy = dist(nx, ny, ox, oy)

        center_bonus = 30 - dcenter
        opp_penalty = (-10 if dppy > 0 else 0)  # placeholder replaced below

        # Deterministic obstacle "stickiness": if many neighbors blocked, discourage move
        blocked = 0
        for dx, dy in candidates:
            tx, ty = nx + dx, ny + dy
            if not inb(tx, ty) or (tx, ty) in obstacles:
                blocked += 1
        mobility_penalty = blocked * 6

        # Opponent distance handling
        dppy = doppy
        opp_term = (5 if dppy <= 1 else 0) + (-4 if dppy > 3 else 0)

        return flip_gain + center_bonus + (-3 * dome) + opp_term - mobility_penalty

    best = (0, 0)
    best_v = -10**18
    for dx, dy in candidates:
        nx, ny = sx + dx, sy + dy
        v = cell_value(nx, ny) if inb(nx, ny) else -10**9
        if v > best_v:
            best_v = v
            best = (dx, dy)

    return [int(best[0]), int(best[1])]
