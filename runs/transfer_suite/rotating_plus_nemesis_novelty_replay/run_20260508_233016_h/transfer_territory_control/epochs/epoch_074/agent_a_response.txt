def choose_move(observation):
    sx, sy = observation["self_position"]
    w, h = observation["grid_width"], observation["grid_height"]
    ox, oy = observation["opponent_position"]
    obstacles = set(tuple(p) for p in (observation.get("obstacles") or []))

    selfT = set(tuple(p) for p in (observation.get("self_territory") or []))
    oppT = set(tuple(p) for p in (observation.get("opponent_territory") or []))
    unclaimed = [tuple(c) for c in (observation.get("unclaimed_cells") or [])]

    def man(ax, ay, bx, by):
        return abs(ax - bx) + abs(ay - by)

    def neighbors(x, y):
        for dx in (-1, 0, 1):
            for dy in (-1, 0, 1):
                if dx == 0 and dy == 0:
                    continue
                nx, ny = x + dx, y + dy
                if 0 <= nx < w and 0 <= ny < h:
                    yield nx, ny

    # Pick deterministic target: prefer unclaimed cells near opponent boundary; else near center
    opp_adj = set()
    for (x, y) in oppT:
        for nx, ny in neighbors(x, y):
            if (nx, ny) not in oppT and (nx, ny) not in selfT:
                opp_adj.add((nx, ny))

    candidates_cells = unclaimed[:] if unclaimed else []
    if not candidates_cells:
        for x in range(w):
            for y in range(h):
                if (x, y) not in selfT and (x, y) not in oppT and (x, y) not in obstacles:
                    candidates_cells.append((x, y))
                    if len(candidates_cells) >= 16:
                        break
            if len(candidates_cells) >= 16:
                break

    if candidates_cells:
        def cell_key(c):
            x, y = c
            if (x, y) in obstacles:
                return -10**9
            base = -man(sx, sy, x, y) * 2 - man(ox, oy, x, y)
            edge_bonus = 18 if (x in (0, w - 1) or y in (0, h - 1)) else 0
            boundary_bonus = 30 if (x, y) in opp_adj else 0
            opp_near = 10 if man(ox, oy, x, y) <= 3 else 0
            return base + boundary_bonus + edge_bonus + opp_near

        target = max(candidates_cells, key=cell_key)
    else:
        target = (w // 2, h // 2)

    tx, ty = target
    deltas = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    best_move = (0, 0)
    best_val = -10**18

    for dx, dy in deltas:
        nx, ny = sx + dx, sy + dy
        if not (0 <= nx < w and 0 <= ny < h):
            continue
        if (nx, ny) in obstacles:
            continue

        # Evaluate move by progress to target plus opportunistic flipping
        val = -man(nx, ny, tx, ty)
        if (nx, ny) in oppT:
            val += 60  # immediate flip opportunity
        if (nx, ny) in selfT:
            val += 6   # stable expansion
        # Prefer moves that approach opponent boundary when no unclaimed lead exists
        if unclaimed:
            val += 2 * (1 if (nx, ny) in opp_adj else 0)
        # Slightly avoid getting too close to opponent if we can't flip
        if (nx, ny) not in oppT:
            val -= 0.5 * man(nx, ny, ox, oy)

        # Deterministic tie-break: smaller dx,dy order from deltas list already fixes this
        if val > best_val:
            best_val = val
