def choose_move(observation):
    sx, sy = observation["self_position"]
    w, h = observation["grid_width"], observation["grid_height"]
    ox, oy = observation["opponent_position"]
    selfT = set(tuple(p) for p in (observation.get("self_territory") or []))
    oppT = set(tuple(p) for p in (observation.get("opponent_territory") or []))
    unclaimed = observation.get("unclaimed_cells") or []
    obs = set(tuple(p) for p in (observation.get("obstacles") or []))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def man(ax, ay, bx, by):
        return abs(ax - bx) + abs(ay - by)

    unclaimed_set = set(tuple(c) for c in unclaimed)
    # Best unclaimed distance for global pressure
    # Keep deterministic: pick a few nearest to us and opponent, then use min.
    k = 10
    nearest_to_us = sorted(unclaimed_set, key=lambda c: man(c[0], c[1], sx, sy))[:k]
    nearest_to_opp = sorted(unclaimed_set, key=lambda c: man(c[0], c[1], ox, oy))[:k]
    targets = nearest_to_us + nearest_to_opp
    if not targets:
        targets = [tuple(p) for p in (unclaimed[:1] if unclaimed else [(w // 2, h // 2)])]

    cand = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    best = None
    best_sc = -10**18

    # Strategic mode switch: when ahead, prioritize safe expansion; when behind, prioritize flips.
    self_count = observation.get("self_territory_count", len(selfT))
    opp_count = observation.get("opponent_territory_count", len(oppT))
    behind = 1 if self_count < opp_count else 0

    for dx, dy in cand:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obs:
            continue

        gain = 0
        if (nx, ny) in oppT:
            gain += (3.5 if behind else 2.5)
        elif (nx, ny) in unclaimed_set:
            gain += (2.5 if behind else 3.0)
        elif (nx, ny) in selfT:
            gain += (0.6 if behind else 0.3)

        # Pressure: move toward good unclaimed while slightly avoiding moving into opponent center
        dmin = min(manh(nx, ny, tx, ty) for (tx, ty) in targets) if targets else 0
        # Maintain "frontier" by preferring cells that are closer to unclaimed, not just any
        sc = gain * 10 - dmin

        # If we're close to opponent, prefer moves that can flip immediately or approach their contested border
        dist_opp = man(nx, ny, ox, oy)
        sc += (6 - dist_opp) * (1.2 if behind else 0.6)
        # Mild obstacle/edge preference to keep from getting stuck near blocks
        sc += -0.2 * (dx == 0 and dy == 0)

        if sc > best_sc:
            best_sc = sc
            best = (dx, dy)

    if best is None:
        return [0, 0]
    return [int(best[0]), int(best[1])]
