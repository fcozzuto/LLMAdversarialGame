def choose_move(observation):
    sx, sy = observation["self_position"]
    w, h = observation["grid_width"], observation["grid_height"]
    obstacles = set(tuple(p) for p in (observation.get("obstacles") or []))
    self_terr = set(tuple(p) for p in (observation.get("self_territory") or []))
    opp_terr = set(tuple(p) for p in (observation.get("opponent_territory") or []))
    unclaimed = set(tuple(p) for p in (observation.get("unclaimed_cells") or []))
    opp_pos = tuple(observation.get("opponent_position") or (sx, sy))
    cx, cy = w // 2, h // 2

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    neigh = [(dx, dy) for dx in (-1, 0, 1) for dy in (-1, 0, 1) if not (dx == 0 and dy == 0)]

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def dist(a, b):
        dx, dy = a[0] - b[0], a[1] - b[1]
        return (dx * dx + dy * dy) ** 0.5

    def legal(nx, ny):
        return inb(nx, ny) and (nx, ny) not in obstacles

    def adj_count(cell, S):
        x, y = cell
        c = 0
        for dx, dy in neigh:
            nx, ny = x + dx, y + dy
            if 0 <= nx < w and 0 <= ny < h and (nx, ny) in S:
                c += 1
        return c

    # Prefer contesting around center, but also grab unclaimed adjacent to our expanding boundary.
    best = (0.0, 0, 0)
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not legal(nx, ny):
            nx, ny, dx, dy = sx, sy, 0, 0
        cell = (nx, ny)

        score = 0.0
        if cell in unclaimed:
            score += 5.0
        if cell in self_terr:
            score += 0.4
        if cell in opp_terr:
            score += 2.6  # flipping on entry

        # Move toward center
        score += -0.25 * (dist(cell, (cx, cy)))

        # Boundary pressure: if we can expand from our territory or into opponent area, prioritize.
        if not (dx == 0 and dy == 0):
            score += 0.8 * adj_count(cell, self_terr)
            score += 0.6 * adj_count(cell, opp_terr)

        # Intercept: reduce distance to opponent while heading into their territory/unclaimed nearby.
        score += -0.12 * dist(cell, opp_pos)
        if cell in opp_terr or adj_count(cell, opp_terr) > 0:
            score += 0.35

        # Obstacle proximity penalty to avoid getting stuck near walls
        ox = adj_count(cell, obstacles)
        score -= 0.9 * ox

        if score > best[0] + 1e-12:
            best = (score, dx, dy)
        elif abs(score - best[0]) <= 1e-12:

            def tie_key(ddx, ddy):
                x, y = sx + ddx, sy + ddy
                if not legal(x, y):
                    x, y = sx, sy
                return (dist((x, y), (cx, cy)), 0 if ddx > 0 else 1, 0 if ddy > 0 else 1, 1
