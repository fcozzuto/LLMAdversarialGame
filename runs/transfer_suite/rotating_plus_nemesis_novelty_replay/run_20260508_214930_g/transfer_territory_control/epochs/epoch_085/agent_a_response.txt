def choose_move(observation):
    w = observation.get("grid_width", 8)
    h = observation.get("grid_height", 8)
    sx, sy = observation.get("self_position", (0, 0))
    ox, oy = observation.get("opponent_position", (0, 0))
    obstacles = set(tuple(p) for p in (observation.get("obstacles") or []))
    self_t = set(tuple(p) for p in (observation.get("self_territory") or []))
    opp_t = set(tuple(p) for p in (observation.get("opponent_territory") or []))
    unclaimed = set(tuple(p) for p in (observation.get("unclaimed_cells") or []))

    def ok(nx, ny):
        return 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obstacles

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    man = lambda a, b: abs(a[0] - b[0]) + abs(a[1] - b[1])

    # Deterministic target choice from unclaimed; fallback to opponent territory; fallback to center.
    targets = list(unclaimed)
    if not targets:
        targets = list(opp_t)
    if not targets:
        targets = [(w // 2, h // 2)]

    best_target = targets[0]
    best_val = None
    for t in targets:
        du = man((sx, sy), t)
        do = man((ox, oy), t)
        # Prefer near us, but also try to avoid squares the opponent is closer to.
        val = (du, -do, t[1], t[0])
        if best_val is None or val < best_val:
            best_val = val
            best_target = t

    # Nudge toward our own expansion front: prefer steps that increase distance from opponent
    # unless we're taking opponent territory directly.
    base_to_target = man((sx, sy), best_target)

    best_move = (0, 0)
    best_score = -10**18
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not ok(nx, ny):
            continue

        score = 0
        if (nx, ny) in opp_t:
            score += 800
        if (nx, ny) in unclaimed:
            score += 120
        if (nx, ny) in self_t:
            score += 25

        # Move toward target (greedy)
        dt = man((nx, ny), best_target) - base_to_target
        score += -20 * dt

        # Keep away from opponent unless contesting
        dop = man((nx, ny), (ox, oy))
        curd = man((sx, sy), (ox, oy))

        # Favor increasing our separation slightly
        score += 8 * (dop - curd)

        # Penalize revisiting self territory too much (wastes tempo)
        if (nx, ny) in self_t:
            score -= 2 * man((nx, ny), best_target) // max(1, w + h)

        # Tiny deterministic tie-breaker: prefer staying still last
        score += -0.01 * (abs(dx) + abs(dy))

        if score > best_score:
            best_score = score
            best_move = (dx, dy)

    return [int(best_move[0]), int(best_move[1])]
