def choose_move(observation):
    w = observation.get("grid_width", 8)
    h = observation.get("grid_height", 8)
    sx, sy = observation.get("self_position", (0, 0))
    ox, oy = observation.get("opponent_position", (0, 0))

    obstacles = set(tuple(p) for p in (observation.get("obstacles") or []))
    self_t = set(tuple(p) for p in (observation.get("self_territory") or []))
    opp_t = set(tuple(p) for p in (observation.get("opponent_territory") or []))
    unclaimed = set(tuple(p) for p in (observation.get("unclaimed_cells") or []))

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def ok(nx, ny):
        return 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obstacles

    def nearest_unclaimed_dist(x, y):
        best = 10**9
        for cx, cy in unclaimed:
            d = abs(cx - x) + abs(cy - y)
            if d < best:
                best = d
        return best if unclaimed else 0

    # Bias: opponent tends to center-claim; we prefer expanding outward and denying their approach.
    base_to_opp = abs(sx - ox) + abs(sy - oy)
    toward_far_unclaimed = None
    if unclaimed:
        best_key = None
        for cx, cy in unclaimed:
            if (cx, cy) in obstacles:
                continue
            d_opp = abs(cx - ox) + abs(cy - oy)
            # Prefer farther from opponent, but not too far to waste turns.
            d_self = abs(cx - sx) + abs(cy - sy)
            key = (d_self, -d_opp, cy, cx)
            if best_key is None or key < best_key:
                best_key = key
                toward_far_unclaimed = (cx, cy)

    if toward_far_unclaimed is not None:
        tx, ty = toward_far_unclaimed
    else:
        tx, ty = (0, 0)

    best_move = (0, 0)
    best_score = -10**18

    # Compute a cheap obstacle proximity term from local legality.
    for ddx, ddy in dirs:
        nx, ny = sx + ddx, sy + ddy
        if not ok(nx, ny):
            continue

        score = 0
        if (nx, ny) in unclaimed:
            score += 120
        if (nx, ny) in self_t:
            score += 20
        if (nx, ny) in opp_t:
            score += 260  # flipping on entry is enabled; prioritize forcing captures
        else:
            score -= 10 if (nx, ny) in opp_t else 0

        d_opp = abs(nx - ox) + abs(ny - oy)
        if d_opp > base_to_opp:
            score += 18
        else:
            score -= 6

        # Goal-directed: head to chosen far-unclaimed, otherwise to the local unclaimed frontier.
        if toward_far_unclaimed is not None:
            score += (-(abs(tx - nx) + abs(ty - ny)) + 2 * (abs(tx - sx) + abs(ty - sy)) / 2)
        else:
            score += -nearest_unclaimed_dist(nx, ny)

        # Avoid stepping into immediate "trap" behind opponent territory.
        adj_to_opp = 0
        for ax in (-1, 0, 1):
            for ay in (-1, 0, 1):
                if ax == 0 and ay == 0:
                    continue
                px, py = nx + ax, ny + ay
                if 0 <= px < w and 0 <= py < h and (px, py) in opp_t:
                    adj_to_opp += 1
        score += adj_to_opp * (-8 if (nx, ny) not in opp_t else 5)

        # Deterministic tie-break via dir ordering and stable score.
        if score > best_score:
            best_score = score
