def choose_move(observation):
    w, h = observation["grid_width"], observation["grid_height"]
    sx, sy = observation["self_position"]
    obstacles = set(tuple(p) for p in (observation.get("obstacles") or []))
    self_t = set(tuple(p) for p in (observation.get("self_territory") or []))
    opp_t = set(tuple(p) for p in (observation.get("opponent_territory") or []))
    unclaimed = set(tuple(p) for p in (observation.get("unclaimed_cells") or []))

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def inside(nx, ny):
        return 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obstacles

    def count_adj(nx, ny, sset):
        c = 0
        for ddx in (-1, 0, 1):
            for ddy in (-1, 0, 1):
                if ddx == 0 and ddy == 0:
                    continue
                if (nx + ddx, ny + ddy) in sset:
                    c += 1
        return c

    # Prefer decisive moves: enter opponent territory if possible, else take unclaimed near front.
    opp_adj = []
    for ddx, ddy in dirs:
        nx, ny = sx + ddx, sy + ddy
        if inside(nx, ny) and (nx, ny) in opp_t:
            # capture flip on entry; reward adjacency to our territory for stability
            opp_adj.append(((ddx, ddy), 20 + count_adj(nx, ny, self_t) - count_adj(nx, ny, opp_t)))

    if opp_adj:
        return max(opp_adj, key=lambda t: t[1])[0]

    near_targets = []
    maxd = 3
    for x in range(max(0, sx - maxd), min(w, sx + maxd + 1)):
        for y in range(max(0, sy - maxd), min(h, sy + maxd + 1)):
            if (x, y) == (sx, sy) or (x, y) in obstacles:
                continue
            if (x, y) in unclaimed or (x, y) in opp_t:
                # avoid re-entering our own cell unless it helps keep options open
                base = 3 if (x, y) in unclaimed else 7
                adj_self = count_adj(x, y, self_t)
                adj_opp = count_adj(x, y, opp_t)
                dist = abs(x - sx) + abs(y - sy)
                # Encourage targets that border us and not heavily surrounded by opponent
                sc = base + 2 * adj_self - 1.5 * adj_opp - 0.6 * dist
                near_targets.append(((x, y), sc))
    if near_targets:
        tx, ty = max(near_targets, key=lambda t: t[1])[0]
        dx = 0 if tx == sx else (1 if tx > sx else -1)
        dy = 0 if ty == sy else (1 if ty > sy else -1)
        if inside(sx + dx, sy + dy):
            return [dx, dy]
        # fall back to best immediate move toward that target
        best = (0, 0)
        best_sc = -10**18
        for ddx, ddy in dirs:
            nx, ny = sx + ddx, sy + ddy
            if not inside(nx, ny):
                continue
            # score immediate cell
            cell_sc = 0
            if (nx, ny) in unclaimed:
                cell_sc += 5
            if (nx, ny) in self_t:
                cell_sc -= 0.5
            cell_sc += 2 * count_adj(nx, ny, self_t) - 1.5 * count_adj(nx, ny, opp_t)
            # encourage closing distance
            cell_sc -= 0.7 * (abs(tx - nx) + abs(ty - ny))
            if cell_sc > best_sc:
                best_sc, best = cell_sc, (ddx, ddy)
        return [best[0], best[1]]
