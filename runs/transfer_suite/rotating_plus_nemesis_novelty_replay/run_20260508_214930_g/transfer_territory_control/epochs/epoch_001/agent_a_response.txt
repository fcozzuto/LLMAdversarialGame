def choose_move(observation):
    w, h = observation["grid_width"], observation["grid_height"]
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    obstacles = observation["obstacles"]
    obs_set = {(x, y) for x, y in obstacles}
    self_cells = {(x, y) for x, y in observation["self_territory"]}
    opp_cells = {(x, y) for x, y in observation["opponent_territory"]}
    unclaimed = {(x, y) for x, y in observation["unclaimed_cells"]}

    cx, cy = (w - 1) / 2.0, (h - 1) / 2.0
    options = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    # Deterministic target selection
    if unclaimed:
        tx, ty = min(unclaimed, key=lambda p: (abs(p[0] - sx) + abs(p[1] - sy), abs(p[0] - cx) + abs(p[1] - cy), p[0], p[1]))
    else:
        # If no unclaimed, aim to cut through opponent territory toward them
        candidates = list(opp_cells) if opp_cells else [(ox, oy)]
        tx, ty = min(candidates, key=lambda p: (abs(p[0] - sx) + abs(p[1] - sy), p[0], p[1]))

    best_move = [0, 0]
    best_score = -10**18
    for dx, dy in options:
        nx, ny = sx + dx, sy + dy
        if nx < 0 or nx >= w or ny < 0 or ny >= h or (nx, ny) in obs_set:
            continue
        score = 0
        if (nx, ny) in unclaimed:
            score += 2
        elif (nx, ny) in opp_cells:
            score += 3  # flipping opponent territory on entry
        elif (nx, ny) in self_cells:
            score += 1
        # Distance shaping
        dist_to_target = abs(nx - tx) + abs(ny - ty)
        score += 5 - dist_to_target
        # Encourage moving toward center slightly to expand control
        score += -0.1 * (abs(nx - cx) + abs(ny - cy))
        # If adjacent to opponent, prioritize going onto their territory if possible
        if abs(nx - ox) <= 1 and abs(ny - oy) <= 1:
            score += 1.5 if (nx, ny) in opp_cells else 0.2 * (1 - (abs(nx - ox) + abs(ny - oy)))
        if score > best_score or (score == best_score and (dx, dy) < (best_move[0], best_move[1])):
            best_score = score
            best_move = [dx, dy]
    return best_move

def choose_move(observation):
    w, h = observation["grid_width"], observation["grid_height"]
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    obstacles = observation.get("obstacles", [])
    obs_set = {(x, y) for x, y in obstacles}
    unclaimed = observation.get("unclaimed_cells", []) or []

    options = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def valid(nx, ny):
        return 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obs_set

    if unclaimed:
        targets = unclaimed
        tx, ty = min(targets, key=lambda p: (abs(p[0] - sx) + abs(p[1] - sy), p[0], p[1]))
    else:
        tx, ty = ox, oy

    best = None
    best_key = None
    for dx, dy in options:
        nx, ny = sx + dx, sy + dy
        if not valid(nx, ny):
            continue
        dist = abs(tx - nx) + abs(ty - ny)
        hit_unclaimed = (nx, ny) in obs_set  # always False; keep simple deterministic scoring
        k = (-1 if hit_unclaimed else 0, dist, nx, ny, dx, dy)
        if best is None or k < best_key:
            best, best_key = (dx, dy), k

    if best is None:
        for dx, dy in options:
            nx, ny = sx + dx, sy + dy
            if valid(nx, ny):
                return [dx, dy]
        return [0, 0]
    return [best[0], best[1]]
