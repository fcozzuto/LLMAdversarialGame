    sx, sy = int(sp[0]), int(sp[1])

    obstacles = set()
    for p in observation.get("obstacles") or []:
        if p and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

    unclaimed = set()
    for c in observation.get("unclaimed_cells") or []:
        if c and len(c) >= 2:
            x, y = int(c[0]), int(c[1])
            if 0 <= x < w and 0 <= y < h and (x, y) not in obstacles:
                unclaimed.add((x, y))

    self_t = set()
    for c in observation.get("self_territory") or []:
        if c and len(c) >= 2:
            x, y = int(c[0]), int(c[1])
            if 0 <= x < w and 0 <= y < h:
                self_t.add((x, y))

    opp_t = set()
    for c in observation.get("opponent_territory") or []:
        if c and len(c) >= 2:
            x, y = int(c[0]), int(c[1])
            if 0 <= x < w and 0 <= y < h:
                opp_t.add((x, y))

    neigh8 = [(dx, dy) for dx in (-1, 0, 1) for dy in (-1, 0, 1)]
    neigh8.remove((0, 0))

    def inb(x, y): return 0 <= x < w and 0 <= y < h
    def free(x, y): return inb(x, y) and (x, y) not in obstacles

    def adj(a, bset):
        x, y = a
        for dx, dy in neigh8:
            if (x + dx, y + dy) in bset:
                return True
        return False

    targets = []
    for (x, y) in unclaimed:
        if abs(x - sx) + abs(y - sy) > 6:
            continue
        if (x, y) in self_t:
            continue
        near_self = adj((x, y), self_t)
        near_opp = adj((x, y), opp_t)
        # Prefer denying opponent edges, then expanding our edge, then closer progress.
        score = (10 if near_opp else 0) + (6 if near_self else 0) - (abs(x - sx) + abs(y - sy))
        targets.append((score, x, y))

    if not targets:
        # Fallback: head toward any unclaimed cell within reach.
        for (x, y) in unclaimed:
            if abs(x - sx) + abs(y - sy) <= 6:
                targets.append((-abs(x - sx) - abs(y - sy), x, y))

        return [0, 0]

    _, tx, ty = targets[0]

        if not free(nx, ny):
