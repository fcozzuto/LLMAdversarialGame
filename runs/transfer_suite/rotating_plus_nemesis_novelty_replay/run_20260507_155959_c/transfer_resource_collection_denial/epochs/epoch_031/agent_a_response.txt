def choose_move(observation):
    w = observation.get("grid_width", 8)
    h = observation.get("grid_height", 8)
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    resources = observation.get("resources", []) or []
    obstacles = observation.get("obstacles", []) or []
    turns_remaining = observation.get("turns_remaining", 0)

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    obs = set((p[0], p[1]) for p in obstacles if isinstance(p, (list, tuple)) and len(p) >= 2)

    legal = []
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            nx, ny = sx + dx, sy + dy
            if inb(nx, ny) and (nx, ny) not in obs:
                legal.append((dx, dy))
    if not legal:
        return [0, 0]

    res = []
    for r in resources:
        if isinstance(r, (list, tuple)) and len(r) >= 2:
            rx, ry = r[0], r[1]
            if inb(rx, ry) and (rx, ry) not in obs:
                res.append((rx, ry))
    if not res:
        return [0, 0]

    def md(x1, y1, x2, y2):
        return abs(x1 - x2) + abs(y1 - y2)

    # Choose a target resource where we're relatively better (smaller dist than opponent).
    best_target = None
    best_gain = None
    for rx, ry in res:
        sd = md(sx, sy, rx, ry)
        od = md(ox, oy, rx, ry)
        gain = od - sd  # positive means we can reach sooner than opponent
        # Tie-break: prefer closer overall and then earlier opponent disadvantage
        key = (-(gain), sd, od, rx, ry)
        if best_target is None or key < best_gain:
            best_target = (rx, ry)
            best_gain = key

    tx, ty = best_target

    # If opponent is extremely close to target, try to "steal" nearby alternatives.
    opp_to_t = md(ox, oy, tx, ty)
    if opp_to_t <= 2 and len(res) > 1:
        alt = None
        alt_key = None
        for rx, ry in res:
            if (rx, ry) == (tx, ty):
                continue
            sd = md(sx, sy, rx, ry)
            od = md(ox, oy, rx, ry)
            gain = od - sd
            # prefer immediate captures or big gain
            key = (-(gain), sd, md(ox, oy, rx, ry), rx, ry)
            if alt is None or key < alt_key:
                alt = (rx, ry)
                alt_key = key
        if alt is not None:
            tx, ty = alt

    # Evaluate next moves.
    best_move = None
    best_val = None
    for dx, dy in legal:
        nx, ny = sx + dx, sy + dy
        d_to_target = md(nx, ny, tx, ty)

        # Capture bonus if stepping onto any resource.
        capture_bonus = 0
        for rx, ry in res:
            if rx == nx and ry == ny:
                capture_bonus = 50
                break

        # If opponent is closer to target, add a penalty proportional to remaining advantage.
        od_next = md(ox, oy, tx, ty)
        sd_next = d_to_target
        rel = od_next - sd_next  # positive good
        val = (20 - d_to_target) + capture_bonus + 3 * rel

        # Minor preference to reduce distance to nearest resource overall (avoid local trap).
        if turns_remaining > 0:
            near = None
            near_d = None
            for rx, ry in res:
                dd = md(nx, ny, rx, ry)
                if near_d is None or dd < near_d:
                    near_d = dd
            val -= 0.2 * near_d

        key = (-val, dx, dy)
        if best_val is None or key < best_val:
            best_val = key
            best_move = (dx, dy)

    return [int(best_move[0]), int(best_move[1])]
