def choose_move(observation):
    w = observation.get("grid_width", 8)
    h = observation.get("grid_height", 8)
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    resources = observation.get("resources", []) or []
    obstacles = observation.get("obstacles", []) or []
    obs = set((p[0], p[1]) for p in obstacles)

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def md(x1, y1, x2, y2):
        return abs(x1 - x2) + abs(y1 - y2)

    def greedy_step(px, py, tx, ty):
        # deterministic tie-break: prefer dx toward target, then dy; use diag if helps
        best = (0, 0)
        best_d = 10**9
        for dx in (-1, 0, 1):
            for dy in (-1, 0, 1):
                nx, ny = px + dx, py + dy
                if not inb(nx, ny) or (nx, ny) in obs:
                    continue
                d = md(nx, ny, tx, ty)
                if d < best_d or (d == best_d and (dx, dy) > best):
                    best_d = d
                    best = (dx, dy)
        return best

    def legal_from(px, py):
        moves = []
        for dx in (-1, 0, 1):
            for dy in (-1, 0, 1):
                nx, ny = px + dx, py + dy
                if inb(nx, ny) and (nx, ny) not in obs:
                    moves.append((dx, dy))
        return moves if moves else [(0, 0)]

    res = []
    for r in resources:
        x, y = r[0], r[1]
        if inb(x, y) and (x, y) not in obs:
            res.append((x, y))
    if not res:
        return [0, 0]

    my_legal = legal_from(sx, sy)
    opp_legal = legal_from(ox, oy)

    # opponent likely target: closest resource; deterministic tie: smallest (d, x, y)
    def opp_target():
        best = None
        best_key = None
        for rx, ry in res:
            d = md(ox, oy, rx, ry)
            key = (d, rx, ry)
            if best_key is None or key < best_key:
                best_key = key
                best = (rx, ry)
        return best

    target = opp_target()
    opp_gdx, opp_gdy = greedy_step(ox, oy, target[0], target[1])
    no = (ox + opp_gdx, oy + opp_gdy)

    my_best_score = -10**18
    best_move = (0, 0)

    # one-step lookahead: immediate collection + advantage on next distances
    for dx, dy in my_legal:
        ns = (sx + dx, sy + dy)
        score = 0

        ns_is_resource = ns in set(res)
        no_is_resource = no in set(res)

        if ns_is_resource:
            score += 2  # prioritize collecting now
        if no_is_resource and not ns_is_resource:
            score -= 2  # prevent immediate loss

        # then maximize winning margin for the closest contested resource next
        for rx, ry in res:
            if (rx, ry) == ns and ns_is_resource:
                continue
            d_self = md(ns[0], ns[1], rx, ry)
            d_opp = md(no[0], no[1], rx, ry)
            margin = d_opp - d_self
            # weight margin, but penalize if opponent is already adjacent
            score += margin * 0.1
            if d_opp == 1 and d_self > 1:
                score -= 0.3

        if score > my_best_score or (score == my_best_score and (dx, dy) > best_move):
            my_best_score = score
            best_move = (dx, dy)

    return [int(best_move[0]), int(best_move[1])]
