def choose_move(observation):
    w = observation["grid_width"]; h = observation["grid_height"]
    sx, sy = observation["self_position"]; ox, oy = observation["opponent_position"]
    resources = observation.get("resources") or []
    obstacles = observation.get("obstacles") or []

    obs = set()
    for p in obstacles:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = p[0], p[1]
            if 0 <= x < w and 0 <= y < h:
                obs.add((x, y))
    if (sx, sy) in obs:
        return [0, 0]

    def inb(x, y): return 0 <= x < w and 0 <= y < h
    def md(a, b, c, d): return abs(a - c) + abs(b - d)

    res = []
    for p in resources:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = p[0], p[1]
            if inb(x, y) and (x, y) not in obs:
                res.append((x, y))
    if not res:
        return [0, 0]

    # Opponent likely target: nearest remaining resource.
    opp_t = min(res, key=lambda r: md(ox, oy, r[0], r[1]))
    tx, ty = opp_t

    # Predict their next move toward the target; try to occupy that cell to deny.
    step_x = 0 if ox == tx else (1 if tx > ox else -1)
    step_y = 0 if oy == ty else (1 if ty > oy else -1)
    nxt = (ox + step_x, oy + step_y)
    if not inb(nxt[0], nxt[1]) or nxt in obs:
        # If diagonal/straight cell blocked, use a best neighbor toward target.
        candidates = [(ox + dx, oy + dy) for dx in (-1, 0, 1) for dy in (-1, 0, 1) if not (dx == 0 and dy == 0)]
        candidates = [c for c in candidates if inb(c[0], c[1]) and c not in obs]
        if candidates:
            nxt = min(candidates, key=lambda c: md(c[0], c[1], tx, ty))
        else:
            nxt = (ox, oy)

    # Choose our move among 9 possibilities to minimize our distance to nxt
    # while not being worse at collecting resources (slight tie-break).
    moves = [(dx, dy) for dx in (-1, 0, 1) for dy in (-1, 0, 1)]
    best = None; best_sc = (-10**18, None)
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obs:
            continue
        # Deny score: closer to nxt is better; if we already deny, go toward their target.
        if (nx, ny) == nxt:
            d_to_goal = md(nx, ny, tx, ty)
            deny = 10**6 - d_to_goal
        else:
            deny = -md(nx, ny, nxt)

        # Secondary: steal advantage if we can.
        steal = -min(md(nx, ny, r[0], r[1]) - md(ox, oy, r[0], r[1]) for r in res)
        sc = (deny, steal)
        if best is None or sc > best_sc:
            best_sc = sc
            best = (dx, dy)

    if best is None:
        return [0, 0]
    return [int(best[0]), int(best[1])]
