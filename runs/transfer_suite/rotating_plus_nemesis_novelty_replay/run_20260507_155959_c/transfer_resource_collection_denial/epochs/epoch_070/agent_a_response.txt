def choose_move(observation):
    w = observation.get("grid_width", 8)
    h = observation.get("grid_height", 8)
    sx, sy = observation.get("self_position", (0, 0))
    resources = observation.get("resources") or []
    obstacles = observation.get("obstacles") or []
    opp_pos = observation.get("opponent_position", (0, 0))
    ox, oy = opp_pos[0], opp_pos[1]

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    obs = set()
    for p in obstacles:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = p[0], p[1]
            if 0 <= x < w and 0 <= y < h:
                obs.add((x, y))

    valid = []
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            nx, ny = sx + dx, sy + dy
            if (nx, ny) != (sx, sy) and inb(nx, ny) and (nx, ny) not in obs:
                valid.append((dx, dy))
    valid.append((0, 0))

    res_set = set()
    for p in resources:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            res_set.add((p[0], p[1]))

    if (sx, sy) in res_set:
        return [0, 0]

    # Immediate pickup if possible
    best_adj = None
    for dx, dy in valid:
        nx, ny = sx + dx, sy + dy
        if (nx, ny) in res_set:
            d2 = abs(nx - ox) + abs(ny - oy)
            key = (0, abs(nx - sx) + abs(ny - sy), d2, dx, dy)
            if best_adj is None or key < best_adj[0]:
                best_adj = (key, dx, dy)
    if best_adj is not None:
        return [best_adj[1], best_adj[2]]

    # Strategic target: prefer resources where we have distance advantage; otherwise still move toward a deny target.
    if not resources:
        # drift to center while avoiding obstacles; deterministic tie-break
        tx, ty = w // 2, h // 2
        best = None
        for dx, dy in valid:
            nx, ny = sx + dx, sy + dy
            key = (abs(nx - tx) + abs(ny - ty), abs(dx) + abs(dy), dx, dy)
            if best is None or key < best:
                best = (key, dx, dy)
        return [best[1], best[2]]

    target = None
    for p in resources:
        if not (isinstance(p, (list, tuple)) and len(p) >= 2):
            continue
        rx, ry = p[0], p[1]
        ds = abs(rx - sx) + abs(ry - sy)
        do = abs(rx - ox) + abs(ry - oy)
        # Primary: our advantage (positive means closer to us). Secondary: our closeness.
        adv = do - ds
        key = (-adv, ds, do, rx, ry)
        if target is None or key < target[0]:
            target = (key, rx, ry)
    _, tx, ty = target

    # Choose move that reduces distance to target; slight bias to limit opponent distance to target resource.
    best = None
    for dx, dy in valid:
        nx, ny = sx + dx, sy + dy
        ds_next = abs(nx - tx) + abs(ny - ty)
        do_next = abs(nx - ox) + abs(ny - oy)
        key = (ds_next, do_next, abs(dx) + abs(dy), dx, dy)
        if best is None or key < best:
            best = (key, dx, dy)
    return [best[1], best[2]]
