def choose_move(observation):
    x, y = observation.get("self_position", [0, 0])
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    turn = int(observation.get("turn_index", 0))
    obstacles = observation.get("obstacles", []) or []
    obs = set((p[0], p[1]) for p in obstacles)

    selft = set((p[0], p[1]) for p in (observation.get("self_territory", []) or []))
    opp = set((p[0], p[1]) for p in (observation.get("opponent_territory", []) or []))
    unclaimed = set((p[0], p[1]) for p in (observation.get("unclaimed_cells", []) or []))

    def inb(nx, ny):
        return 0 <= nx < w and 0 <= ny < h

    # quick helpers
    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def adj_to_self(nx, ny):
        for dx in (-1, 0, 1):
            for dy in (-1, 0, 1):
                if dx == 0 and dy == 0:
                    continue
                if (nx + dx, ny + dy) in selft:
                    return True
        return False

    def manh(a, b):
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

    phase_even = (turn % 2 == 0)

    # choose a strategic target class deterministically
    if phase_even and opp:
        # push toward opponent territory, but only when we can "touch" it from our frontier
        targets = list(opp)
        center = (w // 2, h // 2)
        # deterministic selection: closest, then closer to center
        target = min(targets, key=lambda p: (manh((x, y), p), manh(p, center)))
        target_type = "opp"
    else:
        # expand into unclaimed; if none, fall back to opponent
        if unclaimed:
            targets = list(unclaimed)
            center = (w // 2, h // 2)
            # deterministic: closest, then prefer those closer to center to avoid edge-stalling
            target = min(targets, key=lambda p: (manh((x, y), p), -manh(p, center)))
            target_type = "unclaimed"
        elif opp:
            targets = list(opp)
            center = (w // 2, h // 2)
            target = min(targets, key=lambda p: (manh((x, y), p), manh(p, center)))
            target_type = "opp"
        else:
            return [0, 0]

    best = (-10**18, None)

    # evaluate immediate move
    for dx, dy in dirs:
        nx, ny = x + dx, y + dy
        if not inb(nx, ny) or (nx, ny) in obs:
            continue

        score = 0

        # base: reduce distance to chosen target
        score += 2.0 * (-manh((nx, ny), target))

        # prefer claiming unclaimed
        if (nx, ny) in unclaimed:
            score += 12.0
            score += 1.0 * (1 if adj_to_self(nx, ny) else 0)

        # handle stepping into opponent territory
        if (nx, ny) in opp:
            # only commit if we are on/near our frontier; otherwise avoid "sweeper" traps
            if adj_to_self(nx, ny):
                score += 9.0
            else:
                score -= 8.0

                score += 3.0

def choose_move(observation):

    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    obstacles = observation.get("obstacles", []) or []
    obs = set((p[0], p[1]) for p in obstacles if p and len(p) >= 2)
    ox, oy = observation.get("opponent_position", [x, y])
    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    best = None
    best_sc = -10**9
    for dx, dy in dirs:
        nx, ny = x + dx, y + dy
        if nx < 0 or nx >= w or ny < 0 or ny >= h:
            continue
        if (nx, ny) in obs:
            continue
        d = abs(nx - ox) + abs(ny - oy)
        score = -d
        if dx == 0 and dy == 0:
            score -= 1
        un = set((p[0], p[1]) for p in ((observation.get("unclaimed_cells", []) or [])) if p and len(p) >= 2)
        if (nx, ny) in un:
            score += 2
        rt = set((p[0], p[1]) for p in ((observation.get("resources", []) or [])) if p and len(p) >= 2)
        if (nx, ny) in rt:
            score += 3
        if score > best_sc or (score == best_sc and (dx, dy) < best):
            best_sc = score
            best = (dx, dy)
    if best is None:
        return [0, 0]
