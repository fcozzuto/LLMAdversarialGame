def choose_move(observation):
    w = observation["grid_width"]
    h = observation["grid_height"]
    x, y = observation["self_position"]
    ox, oy = observation["opponent_position"]
    obstacles = set((a[0], a[1]) for a in observation["obstacles"])
    self_cells = set((a[0], a[1]) for a in observation["self_territory"])
    opp_cells = set((a[0], a[1]) for a in observation["opponent_territory"])
    unclaimed = set((a[0], a[1]) for a in observation["unclaimed_cells"])

    dirs = [(-1,-1),(-1,0),(-1,1),(0,-1),(0,0),(0,1),(1,-1),(1,0),(1,1)]
    t = observation["turn_index"]

    def dist_cheb(a, b):
        dx = a[0] - b[0]
        dy = a[1] - b[1]
        if dx < 0: dx = -dx
        if dy < 0: dy = -dy
        return dx if dx > dy else dy

    # Prioritize: take opponent cells if possible, otherwise unclaimed, otherwise expand/hold.
    nearest_opp = None
    if len(opp_cells) > 0:
        nearest_opp = min(opp_cells, key=lambda c: dist_cheb((x, y), c))
    nearest_un = None
    if len(unclaimed) > 0:
        nearest_un = min(unclaimed, key=lambda c: dist_cheb((x, y), c))

    best = None
    best_score = None
    for dx, dy in dirs:
        nx, ny = x + dx, y + dy
        if nx < 0 or nx >= w or ny < 0 or ny >= h:
            continue
        if (nx, ny) in obstacles:
            score = -10**9
        else:
            score = 0
            if (nx, ny) in opp_cells:
                score += 50
            elif (nx, ny) in unclaimed:
                score += 30
            elif (nx, ny) in self_cells:
                score += 0
            else:
                score += 5

            # Attraction to targets
            if nearest_un is not None:
                score += max(0, 12 - dist_cheb((nx, ny), nearest_un))
            if nearest_opp is not None:
                # If we can steal nearby, value increases
                score += 2 * max(0, 10 - dist_cheb((nx, ny), nearest_opp))

            # Avoid moving into positions that are immediately adjacent to many obstacles (simple safety)
            adj_obs = 0
            for adx, ady in dirs:
                mx, my = nx + adx, ny + ady
                if 0 <= mx < w and 0 <= my < h and (mx, my) in obstacles:
                    adj_obs += 1
            score -= adj_obs

            # Deterministic tie-break: prefer smaller dx, then smaller dy, then later dir order via score compare only
        if best_score is None or score > best_score or (score == best_score and (dx, dy) < best):
            best_score = score
            best = (dx, dy)

    if best is None:
        return [0, 0]
    return [best[0], best[1]]
