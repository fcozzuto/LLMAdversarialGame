def choose_move(observation):
    x, y = observation.get("self_position", [0, 0])
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    obs_cells = observation.get("obstacles", []) or []
    obs = set((p[0], p[1]) for p in obs_cells)
    opp = set((p[0], p[1]) for p in (observation.get("opponent_territory", []) or []))
    unclaimed = set((p[0], p[1]) for p in (observation.get("unclaimed_cells", []) or []))
    selft = set((p[0], p[1]) for p in (observation.get("self_territory", []) or []))

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    def inb(nx, ny):
        return 0 <= nx < w and 0 <= ny < h
    def neigh8(nx, ny):
        for dx in (-1, 0, 1):
            for dy in (-1, 0, 1):
                if dx == 0 and dy == 0:
                    continue
                ax, ay = nx + dx, ny + dy
                if inb(ax, ay):
                    yield ax, ay

    def dist(a, b):
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

    # Choose a good target to claim next (unclaimed preferred; prefer ones near opponent territory)
    best_t = None
    best_score = -10**18
    for tx, ty in unclaimed:
        if (tx, ty) in obs:
            continue
        d = dist((x, y), (tx, ty))
        frontier = 0
        for nx, ny in neigh8(tx, ty):
            if (nx, ny) in opp:
                frontier += 1
        # Prefer closeness to us + being a "frontier capture"; small preference to being near our current territory
        near_self = 0
        if selft:
            near_self = min(abs(px - tx) + abs(py - ty) for px, py in selft)
        sc = (-d) + (0.9 * frontier) + (0.15 * (-near_self if selft else 0))
        if sc > best_score or (sc == best_score and (tx, ty) < (best_t or (10**9, 10**9))):
            best_score = sc
            best_t = (tx, ty)

    if best_t is None:
        # Fallback: move toward opponent territory
        if opp:
            best_t = min(opp, key=lambda p: (dist((x, y), p), p[0], p[1]))
        else:
            return [0, 0]

    tx, ty = best_t

    # If target is adjacent (or behind obstacle), evaluate legal moves by improving distance to target, with a capture bias.
    def move_score(dx, dy):
        nx, ny = x + dx, y + dy
        if not inb(nx, ny) or (nx, ny) in obs:
            return -10**18
        dcur = abs(x - tx) + abs(y - ty)
        dnext = abs(nx - tx) + abs(ny - ty)
        cap_bias = 0.0
        if (nx, ny) in opp:
            cap_bias += 6.0
        if (nx, ny) in unclaimed:
            cap_bias += 2.5
        # Slightly discourage stepping too deep into opponent if there's no unclaimed frontier nearby
        sc = (dcur - dnext) + cap_bias
        if not unclaimed:
            sc -= 0.5 if (nx, ny) in opp else 0.0
        return sc
