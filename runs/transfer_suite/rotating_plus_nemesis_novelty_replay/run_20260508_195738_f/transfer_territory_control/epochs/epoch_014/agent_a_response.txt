def choose_move(observation):
    x, y = observation.get("self_position", [0, 0])
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    obstacles = observation.get("obstacles", []) or []
    obs = set((int(p[0]), int(p[1])) for p in obstacles)

    opp_terr = observation.get("opponent_territory", []) or []
    opp = set((int(p[0]), int(p[1])) for p in opp_terr)
    unclaimed_list = observation.get("unclaimed_cells", []) or []
    unclaimed = set((int(p[0]), int(p[1])) for p in unclaimed_list)
    self_terr_list = observation.get("self_territory", []) or []
    selft = set((int(p[0]), int(p[1])) for p in self_terr_list)

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def inb(nx, ny):
        return 0 <= nx < w and 0 <= ny < h

    def score_move(nx, ny):
        if (nx, ny) in obs or not inb(nx, ny):
            return -10**9

        s = 0
        # Direct value of the cell we land on
        if (nx, ny) in unclaimed:
            s += 3
        if (nx, ny) in opp:
            s += 6
        if (nx, ny) in selft:
            s += 1

        # Frontier pressure: reward adjacency to unclaimed/opponent, penalize being boxed by obstacles/opponent
        near_unclaimed = 0
        near_opp = 0
        near_self = 0
        near_obs = 0
        for dx in (-1, 0, 1):
            for dy in (-1, 0, 1):
                if dx == 0 and dy == 0:
                    continue
                ax, ay = nx + dx, ny + dy
                if not inb(ax, ay):
                    continue
                if (ax, ay) in unclaimed:
                    near_unclaimed += 1
                if (ax, ay) in opp:
                    near_opp += 1
                if (ax, ay) in selft:
                    near_self += 1
                if (ax, ay) in obs:
                    near_obs += 1

        s += 1.5 * near_unclaimed
        s += 1.0 * near_opp
        s += 0.3 * near_self
        s -= 0.8 * near_obs

        # Prefer staying in/near our territory band to avoid reckless flips
        if (nx, ny) in opp and near_opp < 2:
            s -= 1.0

        # Tie-breaker: prefer not moving away from our territory center
        # (deterministic, cheap): compute nearest self cell distance approx by min Manhattan to self_territory
        if selft:
            md = 10**9
            for px, py in selft:
                d = abs(px - nx) + abs(py - ny)
                if d < md:
                    md = d
            s -= 0.02 * md

        return s

    best_s = -10**18
    best = (0, 0)
    # Deterministic tie-break: smallest dx,dy lexicographically after sorting dirs
    for dx, dy in sorted(dirs):
        nx, ny = x + dx, y + dy
        s = score_move(nx, ny)
        if s > best_s:
            best_s = s
            best = (dx, dy)

    return [int(best[0]), int(best[1]]]

        return dx + dy
