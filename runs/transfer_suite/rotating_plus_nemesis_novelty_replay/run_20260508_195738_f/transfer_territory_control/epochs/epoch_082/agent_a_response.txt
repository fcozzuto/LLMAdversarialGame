def choose_move(observation):
    sx, sy = observation.get("self_position", [0, 0])
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    obstacles = observation.get("obstacles", []) or []
    obs = set((p[0], p[1]) for p in obstacles)
    unclaimed = observation.get("unclaimed_cells", []) or []
    oppt = observation.get("opponent_territory", []) or []
    opp_cx = sum(p[0] for p in oppt) / len(oppt) if oppt else (w - 1) / 2
    opp_cy = sum(p[1] for p in oppt) / len(oppt) if oppt else (h - 1) / 2

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def inb(x, y): return 0 <= x < w and 0 <= y < h
    opp_set = set((p[0], p[1]) for p in oppt)

    def man(a, b): return abs(a[0] - b[0]) + abs(a[1] - b[1])

    # Pick a deterministic "best" frontier target: far from opponent center, but reachable
    if unclaimed:
        best_t = None
        best_v = -10**18
        for ux, uy in unclaimed:
            v = (abs(ux - opp_cx) + abs(uy - opp_cy)) * 4 - man((sx, sy), (ux, uy))
            if v > best_v:
                best_v = v
                best_t = (ux, uy)
        tx, ty = best_t
    else:
        tx, ty = observation.get("opponent_position", [w - 1, h - 1])

    best_move = (0, 0)
    best_score = -10**18

    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obs:
            # invalid moves yield staying in place; model that deterministically
            nx, ny = sx, sy

        # Prefer stepping toward target, and prefer unclaimed (and opponent cells if it also advances)
        d_t = abs(nx - tx) + abs(ny - ty)
        d_un = 0
        if unclaimed:
            # approximate: closeness to nearest unclaimed cell to this neighbor
            # (bounded by small constant to keep deterministic/cheap)
            local_best = 10**9
            for k in range(len(unclaimed)):
                ux, uy = unclaimed[k]
                dist = abs(ux - nx) + abs(uy - ny)
                if dist < local_best:
                    local_best = dist
                if local_best == 0:
                    break
            d_un = local_best

        v = 0
        v += -d_t * 3
        v += -d_un * 1
        if (nx, ny) in opp_set:
            v += 2  # flipping on entry: sometimes worth it
        if (nx, ny) not in opp_set and [nx, ny] in unclaimed:
            v += 5
        # discourage oscillating near opponent center
        v += -(abs(nx - opp_cx) + abs(ny - opp_cy)) * 0.2

        # tie-break deterministically: small dx then small dy then non-stay preferred
        if v > best_score or (v == best_score and (abs(dx), abs(dy), dx, dy, (dx, dy) == (0, 0)) <
                                 (abs(best_move[0]), abs(best_move[1]), best_move[0], best_move[1], best_move == (0, 0))):
            best_score = v
            best_move = (dx, dy)

def choose_move(observation):
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [w - 1, h - 1])

    obstacles = observation.get("obstacles", []) or []
    obs = set((p[0], p[1]) for p in obstacles if isinstance(p, (list, tuple)) and len(p) >= 2)

    unclaimed = observation.get("unclaimed_cells", []) or []
    if unclaimed:
        tx, ty = min(unclaimed, key=lambda p: (p[0] - ox) ** 2 + (p[1] - oy) ** 2)[0:2]
    else:
        tx, ty = w - 1, h - 1

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    dirs = [(-1, 0), (1, 0), (0, -1), (0, 1), (0, 0)]
    best = (0, 0)
    bestv = -10**18

    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obs:
            continue
        d_me = abs(nx - tx) + abs(ny - ty)
        d_opp = abs(nx - ox) + abs(ny - oy)
        v = 3 * d_opp - d_me
        if v > bestv or (v == bestv and (dx, dy) < best):
            bestv = v
            best = (dx, dy)

    return [int(best[0]), int(best[1])]
