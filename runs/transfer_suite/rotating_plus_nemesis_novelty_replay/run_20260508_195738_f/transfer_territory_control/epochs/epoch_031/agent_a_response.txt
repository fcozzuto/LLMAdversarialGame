def choose_move(observation):
    x, y = observation.get("self_position", [0, 0])
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    obstacles = observation.get("obstacles", []) or []
    obs = set((p[0], p[1]) for p in obstacles)
    opp = set((p[0], p[1]) for p in (observation.get("opponent_territory", []) or []))
    unclaimed = set((p[0], p[1]) for p in (observation.get("unclaimed_cells", []) or []))
    selft = set((p[0], p[1]) for p in (observation.get("self_territory", []) or []))
    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def inb(nx, ny):
        return 0 <= nx < w and 0 <= ny < h

    opp_list = list(opp)
    un_list = list(unclaimed)
    if not opp_list:
        opp_list = [observation.get("opponent_position", [w - 1, h - 1])]

    def min_manh(nx, ny, pts, default=10**9):
        if not pts:
            return default
        md = 10**9
        for px, py in pts:
            d = abs(px - nx) + abs(py - ny)
            if d < md:
                md = d
        return md

    # If losing, be more aggressive: prioritize opponent cells and frontier.
    self_cnt = int(observation.get("self_territory_count", len(selft)))
    opp_cnt = int(observation.get("opponent_territory_count", len(opp)))
    aggressive = 1.0 if self_cnt < opp_cnt else 0.6

    best = (-10**18, 0, 0)
    for dx, dy in dirs:
        nx, ny = x + dx, y + dy
        if not inb(nx, ny) or (nx, ny) in obs:
            continue

        # Immediate outcome shaping
        if (nx, ny) in opp:
            immediate = 6.0 * aggressive
        elif (nx, ny) in unclaimed:
            immediate = 3.2
        elif (nx, ny) in selft:
            immediate = 1.0
        else:
            immediate = 0.4

        # Frontier pressure: move toward opponent territory, but only if it also moves toward expansion.
        d_opp = min_manh(nx, ny, opp_list, default=10**9)
        d_un = min_manh(nx, ny, un_list, default=10**9)

        frontier = (10.0 / (1.0 + d_opp)) * (0.9 + 0.8 * aggressive)
        expansion = (7.0 / (1.0 + d_un)) * (1.0 + 0.4 * (self_cnt < opp_cnt))

        # Small preference to avoid obstacles-adjacent stagnation
        around_obs = 0
        for adx in (-1, 0, 1):
            for ady in (-1, 0, 1):
                if adx == 0 and ady == 0:
                    continue
                ax, ay = nx + adx, ny + ady
                if 0 <= ax < w and 0 <= ay < h and (ax, ay) in obs:
                    around_obs += 1
        risk = 0.35 * around_obs

        val = immediate + frontier + expansion - risk

def choose_move(observation):
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    x, y = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [w - 1, h - 1])

    obs = observation.get("obstacles", []) or []
    blocked = set()
    for p in obs:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            blocked.add((int(p[0]), int(p[1])))

    dirs = [(1, 0), (0, 1), (-1, 0), (0, -1), (0, 0)]
    best_d = 10**18
    best_move = (0, 0)

    for dx, dy in dirs:
        nx, ny = x + dx, y + dy
        if 0 <= nx < w and 0 <= ny < h and (nx, ny) not in blocked:
            d = abs(ox - nx) + abs(oy - ny)
            if d < best_d:
                best_d = d
                best_move = (dx, dy)

    return [int(best_move[0]), int(best_move[1])]
