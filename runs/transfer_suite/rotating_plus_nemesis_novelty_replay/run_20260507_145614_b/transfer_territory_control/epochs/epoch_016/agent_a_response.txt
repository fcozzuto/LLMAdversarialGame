def choose_move(observation):
    w = observation["grid_width"]
    h = observation["grid_height"]
    sx, sy = observation["self_position"]
    obstacles = set()
    for p in (observation.get("obstacles") or []):
        if p and len(p) >= 2:
            obstacles.add((p[0], p[1]))

    self_t = set(tuple(p) for p in (observation.get("self_territory") or []) if p and len(p) >= 2)
    opp_t = set(tuple(p) for p in (observation.get("opponent_territory") or []) if p and len(p) >= 2)
    unclaimed = set(tuple(p) for p in (observation.get("unclaimed_cells") or []) if p and len(p) >= 2)

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def adj8_cell(a, b):
        return max(abs(a[0] - b[0]), abs(a[1] - b[1])) == 1

    # Precompute some local frontiers for heuristic targets
    frontier = []
    for (ux, uy) in unclaimed:
        if any(adj8_cell((ux, uy), t) for t in self_t):
            frontier.append((ux, uy))
    opp_adj_unclaimed = []
    for (ux, uy) in unclaimed:
        if any(adj8_cell((ux, uy), o) for o in opp_t):
            opp_adj_unclaimed.append((ux, uy))

    # Choose whether to push (expand) or invade (harass) deterministically
    invade = False
    if opp_t:
        # If opponent leads by a lot or we are small, prioritize invasion
        if observation.get("opponent_territory_count", len(opp_t)) > observation.get("self_territory_count", len(self_t)) + 6:
            invade = True
        # If there exists reachable adjacent opp territory, invade
        else:
            for dx, dy in dirs:
                nx, ny = sx + dx, sy + dy
                if inb(nx, ny) and (nx, ny) in opp_t:
                    invade = True
                    break

    best = None
    best_val = -10**30

    # Helper to get value components
    def value_for_cell(nx, ny):
        if not inb(nx, ny) or (nx, ny) in obstacles:
            return -10**20
        v = 0.0

        if (nx, ny) in opp_t:
            v += 260.0  # strong immediate flip gain on entry
        elif (nx, ny) in unclaimed:
            v += 45.0
        elif (nx, ny) in self_t:
            v += 6.0

        # Being adjacent to opp can enable next-step flips
        if opp_t and any(adj8_cell((nx, ny), o) for o in opp_t):
            v += 14.0 if invade else 8.0

        # Prefer moving toward our frontier to keep growth steady
        if frontier:
            # Use a cheap nearest check with early stopping
            md = 10**9
            for tx, ty in frontier:
                d = abs(tx - nx) + abs(ty - ny)
                if d < md:
                    md = d
                    if md == 0:
                        break
            v += (22.0 / (1.0 + md)) if not invade else (10.0 / (1.0 + md))

        # If pushing back against opponent, prefer cutting unclaimed cells adjacent to them
        if opp_adj_unclaimed and not invade:
            md2 = 10**9
            for ux, uy in opp_adj_unclaimed:
                d = abs(ux - nx) + abs(uy - ny)
                if d < md2:
                    md2 = d
                    if md2 == 0:
                        break
            v += 18.0 / (1.0 + md2)

        # Mild anti-stall: encourage moves that change position unless no options
        if (nx, ny) == (sx, sy):
            v -= 2.0

        return v

    # Deterministic tie-break by fixed dir order; higher value wins
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        v = value_for_cell(nx, ny)
        if v > best_val:
            best_val = v
            best = (dx, dy)

    if best is None:
        return [0, 0]
    return [best[0], best[1]]

    h = int(observation.get("grid_height", 0) or 0)
    sx, sy = observation.get("self_position", (0, 0))
    ox, oy = observation.get("opponent_position", (0, 0))

    for p in (observation.get("obstacles") or []):
        if p and len(p) >= 2:
            obstacles.add((int(p[0]), int(p[1])))

    resources = []
    for p in (observation.get("resources") or []):
        if p and len(p) >= 2:
            resources.append((int(p[0]), int(p[1])))

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    best = None
    best_val = -10**18

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def dist2(a, b):
        dx = a[0] - b[0]
        dy = a[1] - b[1]
        return dx * dx + dy * dy

    target = None
    if resources:
        bestd = None
        for r in resources:
            if inb(r[0], r[1]) and r not in obstacles:
                d = dist2((sx, sy), r)
                if bestd is None or d < bestd:
                    bestd = d
                    target = r
    elif w and h:
        target = (ox, oy)  # fallback: posture vs opponent

    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obstacles:
            continue

        v = 0
        if target is not None and resources:
            v += -dist2((nx, ny), target) * 10  # chase nearest resource
        # keep some distance from opponent, but don't freeze
        v += dist2((nx, ny), (ox, oy)) * 2

        # deterministic nudge toward the board center to avoid loops in empty maps
        if w and h:
            cx, cy = (w - 1) / 2.0, (h - 1) / 2.0
            v += -((nx - cx) * (nx - cx) + (ny - cy) * (ny - cy)) * 0.01

        if v > best_val:
            best_val = v
            best = [dx, dy]

    if best is None:
        return [0, 0]
