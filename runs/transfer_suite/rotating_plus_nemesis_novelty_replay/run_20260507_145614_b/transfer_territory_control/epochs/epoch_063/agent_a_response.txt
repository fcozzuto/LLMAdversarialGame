        if p and len(p) >= 2:
            self_t.add((p[0], p[1]))

    for p in observation.get("opponent_territory") or []:
        if p and len(p) >= 2:
            opp_t.add((p[0], p[1]))

    neigh8 = [(dx, dy) for dx, dy in dirs if not (dx == 0 and dy == 0)]

    def inside(x, y):
        return 0 <= x < w and 0 <= y < h

    # Frontier expansion: choose an unclaimed cell adjacent to our territory that is not blocked.
    candidates = []
