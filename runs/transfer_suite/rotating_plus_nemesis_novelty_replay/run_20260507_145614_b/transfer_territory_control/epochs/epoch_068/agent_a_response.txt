def choose_move(observation):
    w = int(observation["grid_width"])
    h = int(observation["grid_height"])
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]

    obstacles = set()
    for p in (observation.get("obstacles") or []):
        if p and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

    unclaimed = []
    for p in (observation.get("unclaimed_cells") or []):
        if p and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h and (x, y) not in obstacles:
                unclaimed.append((x, y))

    def man(x1, y1, x2, y2):
        return abs(x1 - x2) + abs(y1 - y2)

    if not unclaimed:
        steps = [(0, 0)]
        for dx in (-1, 0, 1):
            for dy in (-1, 0, 1):
                if dx or dy:
                    nx, ny = sx + dx, sy + dy
                    if 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obstacles:
                        steps.append((dx, dy))
        best = [0, 0]
        bestv = -10**9
        for dx, dy in steps:
            nx, ny = sx + dx, sy + dy
            v = man(nx, ny, ox, oy)  # maximize distance from opponent
            if v > bestv:
                bestv = v
                best = [dx, dy]
        return best

    best_t = unclaimed[0]
    best_u = -10**18
    for x, y in unclaimed[:]:
        edge = 1.0 if (x == 0 or x == w - 1 or y == 0 or y == h - 1) else 0.0
        dself = man(sx, sy, x, y)
        dopp = man(x, y, ox, oy)
        # Prefer expanding toward far edge away from opponent; keep it close to us.
        u = 2.0 * edge + 0.35 * dopp - 0.75 * dself
        if u > best_u:
            best_u = u
            best_t = (x, y)

    tx, ty = best_t
    # Greedy 8-direction step (deterministic tie-break by dx,dy order)
    candidates = []
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            nx, ny = sx + dx, sy + dy
            if 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obstacles:
                if dx == 0 and dy == 0:
                    candidates.append((0, 0, man(sx, sy, tx, ty)))
                else:
                    candidates.append((dx, dy, man(nx, ny, tx, ty)))
    if not candidates:
        return [0, 0]

    # Choose minimal distance to target; if tie, maximize distance from opponent; then deterministic dx/dy.
    best = None
    best_key = None
    for dx, dy, d_to = candidates:
        nx, ny = sx + dx, sy + dy
        key = (d_to, -man(nx, ny, ox, oy), dx, dy)
        if best_key is None or key < best_key:
            best_key = key
            best = [dx, dy]
    return best
