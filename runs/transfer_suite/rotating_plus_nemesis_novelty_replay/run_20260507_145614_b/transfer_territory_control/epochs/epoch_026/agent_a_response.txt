def choose_move(observation):
    w = observation["grid_width"]
    h = observation["grid_height"]
    sx, sy = observation["self_position"]
    opp_t_list = observation.get("opponent_territory") or []
    self_t_list = observation.get("self_territory") or []
    unclaimed_list = observation.get("unclaimed_cells") or []
    obstacles_list = observation.get("obstacles") or []
    opp_pos = observation.get("opponent_position") or [w - 1, h - 1]

    opp_t = {tuple(p) for p in opp_t_list if p and len(p) >= 2}
    self_t = {tuple(p) for p in self_t_list if p and len(p) >= 2}
    unclaimed = {tuple(p) for p in unclaimed_list if p and len(p) >= 2}
    obstacles = {tuple(p) for p in obstacles_list if p and len(p) >= 2}

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    cx, cy = (w - 1) / 2.0, (h - 1) / 2.0

    def is_adj_to_opp(cell):
        x, y = cell
        for dx in (-1, 0, 1):
            for dy in (-1, 0, 1):
                if dx == 0 and dy == 0:
                    continue
                if (x + dx, y + dy) in opp_t:
                    return True
        return False

    # Build frontier preference: unclaimed adjacent to opponent, else unclaimed farthest from opponent to avoid being hunted.
    frontier = []
    if opp_t:
        for c in unclaimed:
            if is_adj_to_opp(c):
                frontier.append(c)
    candidates_cells = frontier if frontier else (list(unclaimed) if unclaimed else [])

    if candidates_cells:
        # Deterministic best target: maximize (unclaimed) and prefer cutting frontier; otherwise go toward closest target.
        # If both lists exist, frontier already picked.
        tx, ty = min(candidates_cells, key=lambda c: (abs(c[0] - sx) + abs(c[1] - sy), abs(c[0] - opp_pos[0]) + abs(c[1] - opp_pos[1])))
    else:
        # No unclaimed: pressure opponent territory if possible, else drift to center.
        if opp_t:
            ox = sum(x for x, _ in opp_t) / len(opp_t)
            oy = sum(y for _, y in opp_t) / len(opp_t)
            tx, ty = ox, oy
        else:
            tx, ty = cx, cy

    best = [0, 0]
    bestv = -10**9
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not (0 <= nx < w and 0 <= ny < h):
            continue
        cell = (nx, ny)
        v = 0.0

        if cell in obstacles:
            v -= 100.0
        if cell in unclaimed:
            v += 8.0
        if cell in opp_t:
            v += 6.0  # flipping on entry
        if cell in self_t:
            v += 0.5  # allow re-entry, but not over-expand
        if opp_t and is_adj_to_opp(cell):
            v += 3.5  # likely to convert nearby

        # Reachability to target
        dist = abs(nx - tx) + abs(ny - ty)
        v -= 0.9 * dist

        # Mild center bias early; reduced near end.
        turns_remaining = observation.get("turns_remaining", 0)
        progress = 1.0
        if isinstance(turns_remaining, int) and turns_remaining > 0:
            progress = 0.5 + 0.5 * (min(64, turns_remaining) / 64.0)
        v += 0.3 * progress * (-(abs(nx - cx) + abs(ny - cy)))

        # Tie-break deterministically: prefer not-staying if equal, then lexicographically.
        if v > bestv + 1e-9 or (abs(v - bestv) <= 1e-9 and (dx, dy) != tuple(best) and (dx, dy) < (best[0], best[1])):
            bestv = v

def choose_move(observation):
    w = int(observation.get("grid_width", 0) or 0)
    h = int(observation.get("grid_height", 0) or 0)
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [w - 1, h - 1])
    try:
        sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)
    except:
        sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    obstacles = observation.get("obstacles") or []
    obs = set()
    for p in obstacles:
        if p and len(p) >= 2:
            try:
                obs.add((int(p[0]), int(p[1])))
            except:
                pass

    resources = observation.get("resources") or []
    res = []
    for p in resources:
        if p and len(p) >= 2:
            try:
                x, y = int(p[0]), int(p[1])
                res.append((x, y))
            except:
                pass

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    cand = []
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if w and h:
            if nx < 0 or ny < 0 or nx >= w or ny >= h:
                continue
        if (nx, ny) in obs:
            continue

        if res:
            tx, ty = min(res, key=lambda t: (t[0] - nx) * (t[0] - nx) + (t[1] - ny) * (t[1] - ny))
            score = -((tx - nx) * (tx - nx) + (ty - ny) * (ty - ny))
        else:
            score = (nx - ox) * (nx - ox) + (ny - oy) * (ny - oy)  # flee when no resources known

        # deterministic tie-break: prefer smaller dx, then dy, then higher score
        cand.append((score, dx, dy))

    if not cand:
        return [0, 0]

    cand.sort(key=lambda t: (-(t[0]), t[1], t[2]))
    _, dx, dy = cand[0]
    return [int(dx), int(dy)]
