def choose_move(observation):
    w = int(observation["grid_width"])
    h = int(observation["grid_height"])
    sx, sy = observation["self_position"]
    obstacles = set()
    for p in (observation.get("obstacles") or []):
        if p and len(p) >= 2:
            x, y = p[0], p[1]
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

    opp_terr = observation.get("opponent_territory") or []
    opp_set = set()
    for p in opp_terr:
        if p and len(p) >= 2:
            opp_set.add((p[0], p[1]))

    unclaimed = observation.get("unclaimed_cells") or []
    targets = []
    for p in unclaimed:
        if not p or len(p) < 2:
            continue
        x, y = p[0], p[1]
        if not (0 <= x < w and 0 <= y < h) or (x, y) in obstacles:
            continue
        if (x == 0 or x == w - 1 or y == 0 or y == h - 1):
            edge = 1
        else:
            edge = 0
        adj = 0
        for dx in (-1, 0, 1):
            for dy in (-1, 0, 1):
                if dx == 0 and dy == 0:
                    continue
                if (x + dx, y + dy) in opp_set:
                    adj = 1
                    break
            if adj:
                break
        if adj or edge:
            targets.append((x, y, adj, edge))
    if not targets:
        return [0, 0]

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    # Prefer moves that (1) claim near opponent/edge, (2) avoid obstacles, (3) advance.
    best = None
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            nx, ny = sx + dx, sy + dy
            if not inb(nx, ny):
                continue
            if (nx, ny) in obstacles:
                score = -10**9
            else:
                # Choose best target from this move position.
                # Deterministic scan with fixed ordering.
                maxv = -10**9
                for (tx, ty, adj, edge) in targets:
                    dist = abs(tx - nx) + abs(ty - ny)
                    # Higher for adj-edge, lower for distance. Small tie-breakers encourage diagonals/forward.
                    v = 8 * adj + 2 * edge - dist
                    # If moving onto/near opponent territory is risky, slightly penalize being adjacent to opponent
                    # unless we are targeting adjacency (keeps pressure without suicidal flips).
                    near_opp = 0
                    for ddx in (-1, 0, 1):
                        for ddy in (-1, 0, 1):
                            if ddx == 0 and ddy == 0:
                                continue
                            if (nx + ddx, ny + ddy) in opp_set:
                                near_opp = 1
                                break
                        if near_opp:
                            break
                    if near_opp and not adj:
                        v -= 1
                    # Tie-break: prefer increasing x then y deterministically to reduce jitter.
                    v += 0.001 * (nx * 10 + ny)
                    if v > maxv:
                        maxv = v
                # Obstacle proximity penalty: count adjacent obstacles.
                adj_obs = 0
                for ddx in (-1, 0, 1):
                    for ddy in (-1, 0, 1):
                        if ddx == 0 and ddy == 0:
                            continue
                        if (nx + ddx, ny + ddy) in obstacles:
                            adj_obs += 1
                score = maxv - 0.3 * adj_obs
            cand = (score, dx, dy)
            if best is None or cand > best:
                best = cand
    return [int(best[1]), int(best[2])]
