def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)

    resources = observation.get("resources", []) or []
    res = []
    for r in resources:
        if isinstance(r, (list, tuple)) and len(r) >= 2:
            x, y = int(r[0]), int(r[1])
            if 0 <= x < w and 0 <= y < h and (x, y) not in obstacles:
                val = r[2] if len(r) > 2 and isinstance(r[2], (int, float)) else 1
                res.append((x, y, float(val)))
        elif isinstance(r, dict):
            x = int(r.get("x", r.get("X", 0)))
            y = int(r.get("y", r.get("Y", 0)))
            if 0 <= x < w and 0 <= y < h and (x, y) not in obstacles:
                val = r.get("value", 1)
                if not isinstance(val, (int, float)):
                    val = 1
                res.append((x, y, float(val)))

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def legal(nx, ny):
        return 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obstacles

    def manh(x1, y1, x2, y2):
        return abs(x1 - x2) + abs(y1 - y2)

    best = (None, -10**18)

    if res:
        target = min(res, key=lambda t: (manh(sx, sy, t[0], t[1]) - (0.5 if evade else -0.5) * t[2], -t[2], t[0], t[1]))
        tx, ty, tv = target
