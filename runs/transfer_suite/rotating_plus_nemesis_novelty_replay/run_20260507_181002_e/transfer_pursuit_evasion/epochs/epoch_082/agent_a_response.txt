def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sp = observation.get("self_position", [0, 0]) or [0, 0]
    op = observation.get("opponent_position", [0, 0]) or [0, 0]
    sx, sy = int(sp[0]), int(sp[1])
    ox, oy = int(op[0]), int(op[1])

    obstacles = set()
    for p in observation.get("obstacles", []) or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            obstacles.add((int(p[0]), int(p[1])))

    def legal(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obstacles

    def boundary_dist(x, y):
        return min(x, y, w - 1 - x, h - 1 - y)

    def adj_obst(x, y):
        c = 0
        for ex in (-1, 0, 1):
            for ey in (-1, 0, 1):
                if ex == 0 and ey == 0:
                    continue
                if (x + ex, y + ey) in obstacles:
                    c += 1
        return c

    def corner_bias(x, y):
        # Prefer moving toward the same general boundary area as the opponent
        ob = boundary_dist(ox, oy)
        sb = boundary_dist(x, y)
        # If opponent is close to a wall, follow its "walliness"
        if ob <= 1:
            return 0.08 * (1 - sb)  # larger when sb small
        # Otherwise avoid sticking to walls too much
        return -0.04 * sb

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    best_dx, best_dy = 0, 0
    best = -10**9

    # Small deterministic tie-breaker toward direct pursuit
    want_dx = 0 if ox == sx else (1 if ox > sx else -1)
    want_dy = 0 if oy == sy else (1 if oy > sy else -1)

    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not legal(nx, ny):
            continue

        man = abs(nx - ox) + abs(ny - oy)
        cheb = max(abs(nx - ox), abs(ny - oy))
        # Two-step lookahead to avoid stepping into immediate obstacle pinch
        blocked_next = adj_obst(nx, ny)
        pin_pen = 0.10 * blocked_next

        score = (-1.2 * cheb) + (-0.02 * man) + corner_bias(nx, ny) - pin_pen
