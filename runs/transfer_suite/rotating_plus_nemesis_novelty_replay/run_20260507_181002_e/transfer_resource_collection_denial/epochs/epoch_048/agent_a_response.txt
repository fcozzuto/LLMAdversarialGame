def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)

    def gp(v, default=(0, 0)):
        if isinstance(v, (list, tuple)) and len(v) >= 2:
            return (int(v[0]), int(v[1]))
        if isinstance(v, dict):
            p = v.get("position", v.get("pos", None))
            if isinstance(p, (list, tuple)) and len(p) >= 2:
                return (int(p[0]), int(p[1]))
        return default

    sx, sy = gp(observation.get("self_position", None), (0, 0))
    ox, oy = gp(observation.get("opponent_position", None), (sx, sy))

    obstacles = set()
    for o in observation.get("obstacles", []) or []:
        x, y = gp(o, None)
        if x is None:
            continue
        if 0 <= x < w and 0 <= y < h:
            obstacles.add((x, y))

    resources = []
    for r in observation.get("resources", []) or []:
        x, y = gp(r, None)
        if x is None:
            continue
        if 0 <= x < w and 0 <= y < h and (x, y) not in obstacles:
            resources.append((x, y))

    def bfs(start):
        INF = 10**9
        dist = [[INF] * h for _ in range(w)]
        sx0, sy0 = start
        if not (0 <= sx0 < w and 0 <= sy0 < h) or (sx0, sy0) in obstacles:
            return dist
        dist[sx0][sy0] = 0
        q = [(sx0, sy0)]
        head = 0
        while head < len(q):
            x, y = q[head]
            head += 1
            nd = dist[x][y] + 1
            for dx in (-1, 0, 1):
                nx = x + dx
                if nx < 0 or nx >= w:
                    continue
                for dy in (-1, 0, 1):
                    if dx == 0 and dy == 0:
                        continue
                    ny = y + dy
                    if ny < 0 or ny >= h:
                        continue
                    if (nx, ny) in obstacles:
                        continue
                    if nd < dist[nx][ny]:
                        dist[nx][ny] = nd
                        q.append((nx, ny))
        return dist

    sd = bfs((sx, sy))
    od = bfs((ox, oy))

    if not resources:
        tx, ty = ox, oy
    else:
        # Prefer resources where we can arrive no later than opponent; otherwise contest closest gap.

def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)

    def pos_of(v):
        if isinstance(v, (list, tuple)) and len(v) >= 2:
            return int(v[0]), int(v[1])
        if isinstance(v, dict):
            p = v.get("position", v.get("pos", v.get("location", None)))
            if isinstance(p, (list, tuple)) and len(p) >= 2:
                return int(p[0]), int(p[1])
        return None

    def in_bounds(x, y):
        return 0 <= x < w and 0 <= y < h

    obstacles = set()
    for o in observation.get("obstacles", []) or []:
        p = pos_of(o)
        if p:
            x, y = p
            if in_bounds(x, y):
                obstacles.add((x, y))

    s = pos_of(observation.get("self_position", None))
    o = pos_of(observation.get("opponent_position", None))
    if not s:
        s = (0, 0)
    if not o:
        o = s

    resources = []
    for r in observation.get("resources", []) or []:
        p = pos_of(r)
        if p:
            x, y = p
            if in_bounds(x, y) and (x, y) not in obstacles:
                resources.append((x, y))

    sx, sy = s
    dirs = [(0, 0), (1, 0), (-1, 0), (0, 1), (0, -1)]
    best_move = (0, 0)
    best_val = -10**18

    if not resources:
        for dx, dy in dirs:
            nx, ny = sx + dx, sy + dy
            if in_bounds(nx, ny) and (nx, ny) not in obstacles:
                return [dx, dy]
        return [0, 0]

    def mindist(px, py):
        md = 10**9
        for rx, ry in resources:
            d = abs(px - rx) + abs(py - ry)
            if d < md:
                md = d
        return md

    opp_best = mindist(o[0], o[1])

    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not in_bounds(nx, ny) or (nx, ny) in obstacles:
            continue
        my_best = mindist(nx, ny)
        value = (opp_best - my_best) * 100 - my_best
        if nx == o[0] and ny == o[1]:
            value += 10**6
        if value > best_val:
            best_val = value
            best_move = (dx, dy)

    return [int(best_move[0]), int(best_move[1])]
