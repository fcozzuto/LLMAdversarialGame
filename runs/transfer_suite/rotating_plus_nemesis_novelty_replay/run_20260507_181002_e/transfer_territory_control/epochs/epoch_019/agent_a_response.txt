def choose_move(observation):
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    sp = observation.get("self_position") or [0, 0]
    op = observation.get("opponent_position") or [w - 1, h - 1]
    sx, sy = int(sp[0]), int(sp[1])
    ox, oy = int(op[0]), int(op[1])

    def toset(v):
        s = set()
        for p in (v or []):
            if isinstance(p, (list, tuple)) and len(p) >= 2:
                x, y = int(p[0]), int(p[1])
                if 0 <= x < w and 0 <= y < h:
                    s.add((x, y))
        return s

    obstacles = toset(observation.get("obstacles"))
    unclaimed = toset(observation.get("unclaimed_cells"))
    resources = toset(observation.get("resources"))
    selfT = toset(observation.get("self_territory"))
    oppT = toset(observation.get("opponent_territory"))
    if not unclaimed:
        unclaimed = set(resources)

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def best_unclaimed_score(tx, ty):
        if (tx, ty) in obstacles:
            return -10**9
        dso = abs(tx - ox) + abs(ty - oy)
        dself = abs(tx - sx) + abs(ty - sy)
        # Prefer squares near opponent but still unclaimed (edge-claimer behavior)
        return (-dself) + (0.6 * dso)

    def opp_boundary_boost(nx, ny):
        if (nx, ny) in oppT:
            return 200
        # If near opponent territory, slightly encourage to keep pressure at the frontier.
        for dx in (-1, 0, 1):
            for dy in (-1, 0, 1):
                if dx or dy:
                    if (nx + dx, ny + dy) in oppT:
                        return 30
        return 0

    best = None
    bestv = -10**18
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not (0 <= nx < w and 0 <= ny < h):
            continue
        if (nx, ny) in obstacles:
            continue

        v = 0
        # Hard prioritize capturing opponent territory on entry.
        v += opp_boundary_boost(nx, ny)
        if (nx, ny) in selfT:
            v += 5
        if (nx, ny) in unclaimed:
            v += 60

        # If unclaimed exists, bias toward a good frontier cell.
        if unclaimed:
            # find deterministically nearest "frontier-ish" unclaimed without scanning whole board
            # by sampling up to 9 candidates closest to opponent.
            # (Still deterministic: fixed iteration order from sorted set.)
            # We'll approximate by iterating a small prefix of sorted unclaimed by (dist to opp).
            cand = None
            cnt = 0
            for (ux, uy) in sorted(unclaimed, key=lambda p: (abs(p[0] - ox) + abs(p[1] - oy), abs(p[0] - sx) + abs(p[1] - sy))):
                cand = (ux, uy)
                # only a few to keep deterministic and fast
                cnt += 1
                if cnt >= 12:
                    break
            if cand:
                tx, ty = cand
                v += 0.7 * (-(abs(nx - tx) + abs(ny - ty)))
        else:
            # No unclaimed: pressure opponent directly.
            v += -0.35 * (abs(nx - ox) + abs(ny - oy))

        # Tie-breakers: prefer diagonal progress and avoid staying when capture/closer move exists.
        if dx != 0 and dy != 0:
            v += 1
        if (dx, dy) == (0, 0):
            v -= 2

        if v > bestv:
            bestv = v
            best = [dx, dy]

    if best is None:
        return [0, 0]
    return [int(best[0]), int(best[1])]
