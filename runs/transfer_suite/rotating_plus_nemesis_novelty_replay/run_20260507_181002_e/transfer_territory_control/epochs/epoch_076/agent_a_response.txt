def choose_move(observation):
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    sp = observation.get("self_position") or [0, 0]
    op = observation.get("opponent_position") or [w - 1, h - 1]
    sx, sy = int(sp[0]), int(sp[1])
    ox, oy = int(op[0]), int(op[1])

    def toset(v):
        s = set()
        for p in (v or []):
            if isinstance(p, (list, tuple)) and len(p) >= 2:
                x, y = int(p[0]), int(p[1])
                if 0 <= x < w and 0 <= y < h:
                    s.add((x, y))
        return s

    obstacles = toset(observation.get("obstacles"))
    unclaimed = toset(observation.get("unclaimed_cells"))
    selfT = toset(observation.get("self_territory"))
    oppT = toset(observation.get("opponent_territory"))
    resources = toset(observation.get("resources"))
    if resources and not unclaimed:
        unclaimed = set(resources)
    if not unclaimed:
        return [0, 0]

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def neighbors8(x, y):
        out = []
        for dx in (-1, 0, 1):
            for dy in (-1, 0, 1):
                if dx == 0 and dy == 0:
                    continue
                nx, ny = x + dx, y + dy
                if 0 <= nx < w and 0 <= ny < h:
                    out.append((nx, ny))
        return out

    frontier = []
    if selfT:
        for x, y in list(selfT):
            for nx, ny in neighbors8(x, y):
                if (nx, ny) in unclaimed and (nx, ny) not in obstacles:
                    frontier.append((nx, ny))
    if not frontier:
        frontier = list(unclaimed)

    def dist(ax, ay, bx, by):
        return abs(ax - bx) + abs(ay - by)

    # Target score: get close to frontier, avoid moving into opponent pressure, prefer cells that are not adjacent to opponent territory.
    best = None
    best_score = -10**18
    opp_zone = oppT | set()
    for tx, ty in sorted(set(frontier)):
        if (tx, ty) in obstacles:
            continue
        ds = dist(sx, sy, tx, ty)
        do = dist(ox, oy, tx, ty)
        adj_opp = 0
        for nx, ny in neighbors8(tx, ty):
            if (nx, ny) in opp_zone:
                adj_opp = 1
                break
        center_bias = -abs(tx - (w - 1) / 2.0) - abs(ty - (h - 1) / 2.0)
        # Higher is better
        score = (0.6 * do) - (1.0 * ds) - (3.0 * adj_opp) + (0.15 * center_bias)
        # small tie-break to keep determinism: lexicographic already via sorted iteration
        if score > best_score:
            best_score = score
            best = (tx, ty)

    tx, ty = best
    # Greedy step toward target with obstacle avoidance; prefer moves that reduce Manhattan distance.
    best_step = (0, 0)
    best_step_score = -10**18
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not (0 <= nx < w and 0 <= ny < h):
            continue
        if (nx, ny) in obstacles:
            continue
        nd = dist(nx, ny, tx, ty)
        step_score = -nd
        # If adjacent to opponent territory, slightly discourage (territory could flip on entry).
        if any((px, py) in oppT for px, py in neighbors8(nx, ny)):
            step_score -= 0.5
        if step_score > best_step_score:
            best_step_score = step_score
            best_step = (dx, dy)

    return [int(best_step[0]), int(best_step[1])]
