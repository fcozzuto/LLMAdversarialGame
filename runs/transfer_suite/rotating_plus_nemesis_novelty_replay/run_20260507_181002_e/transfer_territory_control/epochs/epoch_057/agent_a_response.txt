def choose_move(observation):
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    sp = observation.get("self_position") or [0, 0]
    op = observation.get("opponent_position") or [w - 1, h - 1]
    sx, sy = int(sp[0]), int(sp[1])
    ox, oy = int(op[0]), int(op[1])

    def toset(v):
        s = set()
        for p in (v or []):
            if isinstance(p, (list, tuple)) and len(p) >= 2:
                s.add((int(p[0]), int(p[1])))
        return s

    obstacles = toset(observation.get("obstacles"))
    unclaimed = toset(observation.get("unclaimed_cells"))
    selfT = toset(observation.get("self_territory"))
    resources = toset(observation.get("resources"))
    if not unclaimed and resources:
        unclaimed = set(resources)

    if not unclaimed:
        return [0, 0]

    cx, cy = (w - 1) / 2.0, (h - 1) / 2.0

    def md(a, b, c, d):
        return abs(a - c) + abs(b - d)

    # Determine "frontline": prefer cells adjacent to our territory (locking) early,
    # otherwise go for cells we can reach sooner than the opponent (denial).
    # This is tuned to resist a center-claimer by denying their near-center reach.
    frontier = set()
    if selfT:
        for x, y in selfT:
            for dx in (-1, 0, 1):
                for dy in (-1, 0, 1):
                    if dx == 0 and dy == 0:
                        continue
                    nx, ny = x + dx, y + dy
                    if 0 <= nx < w and 0 <= ny < h:
                        if (nx, ny) in unclaimed:
                            frontier.add((nx, ny))

    use_frontier = (len(selfT) > 0 and len(frontier) > 0 and len(unclaimed) > 12)
    candidates = frontier if use_frontier else unclaimed

    best = None
    best_val = -10**18
    self_len = len(selfT)

    for tx, ty in candidates:
        if (tx, ty) in obstacles:
            continue
        ds = md(sx, sy, tx, ty)
        do = md(ox, oy, tx, ty)
        if ds == 0:
            continue

        # Advantage term: prefer cells we can reach earlier than the opponent.
        adv = (do - ds)

        # Center pressure: if opponent is closer to center, we bias outward lanes;
        # if we're behind, bias toward center to contest.
        dist_center = abs(tx - cx) + abs(ty - cy)
        opp_center_bias = md(ox, oy, cx, cy) - md(sx, sy, cx, cy)
        center_term = (-0.25 * dist_center) + (0.08 * opp_center_bias * (dist_center / (w + h)))

        # Frontier lock bonus: slight preference for closer-to-front expansion.
        lock = 0.0
        if use_frontier:
            lock = 1.2 / (1 + ds)

        # Early game: avoid overreaching into far corners if opponent is already strong.
        strength_ratio = self_len / (self_len + len(toset(observation.get("opponent_territory")) or []))
        overreach_pen = 0.0
        if not use_frontier and ds > 6 and strength_ratio < 0.55:
            overreach_pen = 0.9

        val = 3.0 * adv + center_term + lock - overreach_pen

        # Small deterministic tiebreaker: prefer upper-left on ties
        if val > best_val or (val == best_val and (best is None or (tx, ty) < best)):
            best_val = val
            best = (tx, ty)

    if best is None:
        best = min(unclaimed, key=lambda p: (md(sx, sy, p[0], p[1]), p[0], p[1]))

    tx, ty = best
    dx = 0 if tx == sx else (1 if tx > sx else -1)
    dy = 0 if ty == sy else (1 if ty > sy else -1)

    nx, ny = sx + dx, sy + dy
