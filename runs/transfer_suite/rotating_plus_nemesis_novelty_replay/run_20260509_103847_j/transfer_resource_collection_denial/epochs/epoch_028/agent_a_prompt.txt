    resources = observation.get("resources", [])
    if not resources:
        return [0, 0]
    def score(cell):

        opp_d = abs(cell[0] - ox) + abs(cell[1] - oy)
        return (opp_d - self_d, self_d)
    target = max(resources, key=score)
    dx = 0 if target[0] == sx else (1 if target[0] > sx else -1)
