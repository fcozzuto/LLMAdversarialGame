def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [w - 1, h - 1])
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    obs = set()
    for p in (observation.get("obstacles") or []):
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            obs.add((int(p[0]), int(p[1])))

    un = [(int(x), int(y)) for x, y in (observation.get("unclaimed_cells") or [])]
    if not un:
        un = [(x, y) for x in range(w) for y in range(h) if (x, y) not in obs]

    dirs = [(-1, 0), (0, -1), (0, 0), (1, 0), (0, 1)]
    def valid(nx, ny):
        return 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obs

    def dist(a, b, c, d):
        return abs(a - c) + abs(b - d)

    # pick a target deterministically: closest unclaimed to our position, tie-break far from opponent
    best_t = None
    best_td = None
    for tx, ty in un:
        td = dist(sx, sy, tx, ty)
        if best_t is None or td < best_td or (td == best_td and dist(ox, oy, tx, ty) > dist(ox, oy, best_t[0], best_t[1])):
            best_t = (tx, ty)
            best_td = td
    tx, ty = best_t

    # choose move that minimizes distance to target and maximizes distance from opponent
    best_m = (0, 0)
    best_val = None
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not valid(nx, ny):
            continue
        v1 = dist(nx, ny, tx, ty)
        v2 = dist(nx, ny, ox, oy)
        # deterministic combine: prioritize target, then opponent distance, then smaller dx,dy magnitude
        val = (v1, -v2, abs(dx) + abs(dy))
        if best_val is None or val < best_val:
            best_val = val
            best_m = (dx, dy)

    # if all moves blocked, stay put (must still be valid in-bounds and not obstacle)
    if valid(sx, sy):
        return [best_m[0], best_m[1]]
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if valid(nx, ny):

    return [0, 0]
