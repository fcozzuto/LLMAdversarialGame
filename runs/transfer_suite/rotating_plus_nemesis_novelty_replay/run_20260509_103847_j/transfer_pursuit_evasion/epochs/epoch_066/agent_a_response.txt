def choose_move(observation):
    w = int(observation.get("grid_width") or 8)
    h = int(observation.get("grid_height") or 8)
    sp = observation.get("self_position") or [0, 0]
    op = observation.get("opponent_position") or [0, 0]
    sx, sy = int(sp[0]), int(sp[1])
    ox, oy = int(op[0]), int(op[1])

    obstacles = set()
    for p in observation.get("obstacles") or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    def inside(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obstacles

    role = (observation.get("self_role") or "").lower()
    is_evader = ("evader" in role) or ("escape" in role) or ("evade" in role) or ("runner" in role)

    if (sx, sy) == (ox, oy):
        return [0, 0]

    # BFS distances on demand from target (opponent) to all cells
    from collections import deque
    INF = 10**9
    dist = [[INF] * h for _ in range(w)]
    if inside(ox, oy) or (ox, oy) == (sx, sy):
        dq = deque()
        dist[ox][oy] = 0
        dq.append((ox, oy))
        while dq:
            x, y = dq.popleft()
            d = dist[x][y] + 1
            for dx, dy in moves:
                nx, ny = x + dx, y + dy
                if inside(nx, ny) and d < dist[nx][ny]:
                    dist[nx][ny] = d
                    dq.append((nx, ny))

    def degree(x, y):
        c = 0
        for dx, dy in moves:
            nx, ny = x + dx, y + dy
            if inside(nx, ny):
                c += 1
        return c

    best = None
    best_val = -10**18
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not inside(nx, ny):
            continue
        d = dist[nx][ny]
        if d >= INF:
            # fallback: greedy manhattan if BFS unreachable
            md = abs(nx - ox) + abs(ny - oy)
            d = md
        # Prefer lower distance as pursuer, higher as evader; break ties with mobility
        if is_evader:
            val = d * 1000 + degree(nx, ny)
        else:
            val = (-d) * 1000 + degree(nx, ny)
        if best is None or val > best_val:
            best_val = val
            best = (dx, dy)

    return [int(best[0]) if best else 0, int(best[1]) if best else 0]
