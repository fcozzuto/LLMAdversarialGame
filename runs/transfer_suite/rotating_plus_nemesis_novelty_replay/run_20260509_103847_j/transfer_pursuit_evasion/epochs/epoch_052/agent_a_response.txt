def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sp = observation.get("self_position") or [0, 0]
    op = observation.get("opponent_position") or [0, 0]
    sx, sy = int(sp[0]), int(sp[1])
    ox, oy = int(op[0]), int(op[1])

    obs = observation.get("obstacles") or []
    obs_set = set()
    for p in obs:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obs_set.add((x, y))

    def inside(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obs_set

    sr = (observation.get("self_role") or "").lower()
    pr = (observation.get("opponent_role") or "").lower()
    self_purs = any(k in sr for k in ("purs", "catch", "hunt", "chaser", "pursuer"))
    opp_purs = any(k in pr for k in ("purs", "catch", "hunt", "chaser", "pursuer"))
    i_am_pursuer = self_purs and not (opp_purs and not self_purs)
    if not self_purs and opp_purs:
        i_am_pursuer = False  # evader

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def bfs_dist(ax, ay, bx, by, limit=24):
        if (ax, ay) == (bx, by):
            return 0
        if not inside(ax, ay) or not inside(bx, by):
            return 10**9
        q = [(ax, ay)]
        dist = {(ax, ay): 0}
        head = 0
        while head < len(q):
            x, y = q[head]
            head += 1
            d = dist[(x, y)]
            if d >= limit:
                continue
            for dx, dy in ((-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)):
                nx, ny = x + dx, y + dy
                if not inside(nx, ny):
                    continue
                if (nx, ny) not in dist:
                    nd = d + 1
                    dist[(nx, ny)] = nd
                    if (nx, ny) == (bx, by):
                        return nd
                    q.append((nx, ny))
        return dist.get((bx, by), 10**9)

    def dist2(x, y):
        dx = x - ox
        dy = y - oy
        return dx * dx + dy * dy

    best = None
    best_val = None
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not inside(nx, ny):
            continue
        dpath = bfs_dist(nx, ny, ox, oy, limit=24)
        d2 = dist2(nx, ny)
        if i_am_pursuer:
            # minimize path distance; tie-breaker by smaller direct distance
            val = (dpath, d2)
            if best_val is None or val < best_val:
                best_val = val
                best = (dx, dy)
        else:
            # maximize path distance; tie-breaker by larger direct distance
            val = (-dpath, -d2)
            if best_val is None or val > best_val:
                best_val = val
