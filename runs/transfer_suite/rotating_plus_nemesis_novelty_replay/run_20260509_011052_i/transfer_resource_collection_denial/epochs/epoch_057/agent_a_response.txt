def choose_move(observation):
    w = int(observation.get("grid_width") or 8)
    h = int(observation.get("grid_height") or 8)
    sp = observation.get("self_position") or (0, 0)
    op = observation.get("opponent_position") or (0, 0)
    sx, sy = int(sp[0]), int(sp[1])
    ox, oy = int(op[0]), int(op[1])

    blocked = set()
    for p in observation.get("obstacles") or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                blocked.add((x, y))

    def cheb(ax, ay, bx, by):
        dx = ax - bx
        if dx < 0:
            dx = -dx
        dy = ay - by
        if dy < 0:
            dy = -dy
        return dx if dx > dy else dy

    resources = []
    for p in observation.get("resources") or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h and (x, y) not in blocked:
                resources.append((x, y))

    if not resources:
        tx, ty = ox, oy
        if sx > tx:
            dx = -1
        elif sx < tx:
            dx = 1
        else:
            dx = 0
        if sy > ty:
            dy = -1
        elif sy < ty:
            dy = 1
        else:
            dy = 0
        return [dx, dy]

    best = None
    best_key = (-10**18, -10**18, 10**18, 10**18)
    for x, y in resources:
        sd = cheb(sx, sy, x, y)
        od = cheb(ox, oy, x, y)
        race = od - sd  # higher is better (opponent farther or you closer)
        # Prefer reachable quickly; slight preference for resources closer to center to reduce opponent sweeps
        cx, cy = (w - 1) / 2.0, (h - 1) / 2.0
        dist_center = abs(x - cx) + abs(y - cy)
        # Deterministic tie-break using coordinates
        key = (race, -sd, -dist_center, -(x + y))
        if key > best_key:
            best_key = key
            best = (x, y)

    tx, ty = best
    dx = 0
    if sx < tx:
        dx = 1
    elif sx > tx:
        dx = -1
    dy = 0
    if sy < ty:
        dy = 1
    elif sy > ty:
        dy = -1

    # Avoid moving into obstacle if possible; otherwise accept (engine keeps in place)
    nx, ny = sx + dx, sy + dy
    if (nx, ny) in blocked:
        # Try axis-aligned alternatives deterministically
        candidates = []
        if dx != 0 and (sx + dx, sy) not in blocked:
            candidates.append((dx, 0))
        if dy != 0 and (sx, sy + dy) not in blocked:
            candidates.append((0, dy))
        if dx != 0 and dy != 0 and (sx + dx, sy) not in blocked and (sx, sy + dy) not in blocked:
            candidates.append((dx, 0))
            candidates.append((0, dy))
        if candidates:
            # choose the candidate that reduces cheb distance more
            best_move = None
            best_d = 10**18
            for mdx, mdy in candidates:
                nd = cheb(sx + mdx, sy + mdy, tx, ty)
                if nd < best_d:
                    best_d = nd
                    best_move = (mdx, mdy)
            return [int(best_move[0]), int(best_move[1])]
    return [int(dx), int(dy)]
