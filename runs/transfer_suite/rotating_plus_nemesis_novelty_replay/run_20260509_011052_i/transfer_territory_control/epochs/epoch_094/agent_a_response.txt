def choose_move(observation):
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    sx, sy = observation.get("self_position", (0, 0))
    ox, oy = observation.get("opponent_position", (w - 1, h - 1))
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    obs_cells = set()
    for p in (observation.get("obstacles", []) or []):
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            obs_cells.add((int(p[0]), int(p[1])))

    opp_terr = set()
    for p in (observation.get("opponent_territory", []) or []):
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            opp_terr.add((int(p[0]), int(p[1])))

    unclaimed = []
    for p in (observation.get("unclaimed_cells", []) or []):
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            unclaimed.append((int(p[0]), int(p[1])))

    resources = []
    for p in (observation.get("resources", []) or []):
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            resources.append((int(p[0]), int(p[1])))

    cand_steps = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def dist(a, b):
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

    targets = resources if resources else unclaimed
    if not targets:
        targets = [(ox, oy)]

    # Prefer contested frontier: unclaimed cells adjacent to opponent territory.
    opp_neighbors = set()
    for (tx, ty) in opp_terr:
        for dx in (-1, 0, 1):
            for dy in (-1, 0, 1):
                if dx == 0 and dy == 0:
                    continue
                nx, ny = tx + dx, ty + dy
                if inb(nx, ny) and (nx, ny) not in opp_terr:
                    opp_neighbors.add((nx, ny))

    frontier_targets = [t for t in targets if t in opp_neighbors]
    if frontier_targets:
        targets = frontier_targets

    # Deterministic tie-breaking: scan in given order, pick best.
    if targets:
        best_t = None
        best_td = 10**9
        best_type = 10**9
        for t in targets:
            td = dist((sx, sy), t)
            ttype = 0 if t in opp_neighbors else 1
            if ttype < best_type or (ttype == best_type and td < best_td):
                best_type, best_td, best_t = ttype, td, t
        target = best_t
    else:
        target = (ox, oy)

    best_move = [0, 0]
    best_val = -10**18
    for dx, dy in cand_steps:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny):
            continue
        if (nx, ny) in obs_cells:
            continue

        val = 0
        nxt = (nx, ny)

        if nxt in opp_terr:
            val += 1200  # immediate flip pressure
        # Move toward target (frontier / resources)
        val += 80 - 10 * dist(nxt, target)

        # If moving off frontier, discourage; if on/near opponent, encourage.
        if nxt in opp_neighbors:
            val += 180
        val += 0 if not resources else (200 if nxt in resources else 0)

        # Defensive nudge: if too close to opponent, prefer not to approach recklessly
        val += -0.5 * dist(nxt, (ox, oy))

        # Prefer stepping into unclaimed (claim extension) over staying in own territory
        # We infer via observation lists: if not in opp_terr, then if not in self_territory it's likely unclaimed.
        self_terr = set((tuple(p) for p in (observation.get("self_territory", []) or [])))
        if nxt not in opp_terr and nxt not in self_terr:
            val += 25

        if val > best_val:
            best_val = val
            best_move = [dx, dy]

    return best_move
