        if isinstance(p, (list, tuple)) and len(p) >= 2:

    targets = resources or []
    if not targets:

            if isinstance(p, (list, tuple)) and len(p) >= 2:
                targets.append((int(p[0]), int(p[1])))

            if isinstance(p, (list, tuple)) and len(p) >= 2:

    if not targets:
        targets = [(w // 2, h // 2)]

    cx, cy = w // 2, h // 2
    def dist2(x1, y1, x2, y2):
        dx = x1 - x2
        dy = y1 - y2
        return dx * dx + dy * dy

    best = None
    best_score = None
