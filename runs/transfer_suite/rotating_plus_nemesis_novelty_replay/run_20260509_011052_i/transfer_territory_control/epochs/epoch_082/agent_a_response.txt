    else:
        best = None

        for rx, ry in resources:
            dx = rx - sx
            dy = ry - sy
            d = dx * dx + dy * dy
            if bestd is None or d < bestd:
                bestd = d
                best = (rx, ry)
        tx, ty = best

    opp_dx = ox - sx
