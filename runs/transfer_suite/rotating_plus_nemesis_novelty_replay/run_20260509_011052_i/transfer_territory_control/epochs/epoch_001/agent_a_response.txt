def choose_move(observation):
    w = observation["grid_width"]
    h = observation["grid_height"]
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]

    obstacles = set(tuple(p) for p in observation["obstacles"])
    self_terr = set(tuple(p) for p in observation["self_territory"])
    opp_terr = set(tuple(p) for p in observation["opponent_territory"])
    unclaimed = set(tuple(p) for p in observation["unclaimed_cells"])

    deltas = [(-1,-1),(-1,0),(-1,1),(0,-1),(0,0),(0,1),(1,-1),(1,0),(1,1)]
    inf = 10**9

    def clamp_in(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obstacles

    unclaimed_list = list(unclaimed)
    closest_un = inf
    for ux, uy in unclaimed_list:
        d = abs(ux - sx) + abs(uy - sy)
        if d < closest_un:
            closest_un = d
    if not unclaimed_list:
        closest_un = inf

    best = None
    best_score = -inf
    sc_self = observation.get("self_territory_count", 0)
    sc_opp = observation.get("opponent_territory_count", 0)
    score_bias = 1.0 if sc_self < sc_opp else 0.0

    for dx, dy in deltas:
        nx, ny = sx + dx, sy + dy
        if not clamp_in(nx, ny):
            continue

        tgt = (nx, ny)
        d_opp = abs(nx - ox) + abs(ny - oy)
        d_un = inf
        for ux, uy in unclaimed_list[:25]:
            dd = abs(ux - nx) + abs(uy - ny)
            if dd < d_un:
                d_un = dd
        if not unclaimed_list:
            d_un = inf

        if tgt in self_terr:
            base = 1.0
        elif tgt in unclaimed:
            base = 3.0
        elif tgt in opp_terr:
            base = 4.0 + 2.0 * score_bias
        else:
            base = 0.5

        # Prefer approaching unclaimed, and avoid getting too close to opponent unless flipping.
        dist_term = 0.0
        if d_un != inf:
            dist_term = -0.25 * d_un
        opp_term = 0.12 * d_opp
        if tgt in opp_terr:
            opp_term = -0.06 * d_opp

        # Slight preference for moving away from edges less often (keeps mobility).
        edge_pen = 0.0
        if nx == 0 or nx == w - 1 or ny == 0 or ny == h - 1:
            edge_pen = -0.05

        score = base + dist_term + opp_term + edge_pen
        if score > best_score or (score == best_score and (best is None or (dx, dy) < best)):
            best_score = score
            best = (dx, dy)

    if best is None:
        return [0, 0]
    return [int(best[0]), int(best[1])]
