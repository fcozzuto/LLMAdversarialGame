def choose_move(observation):
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    sx, sy = observation.get("self_position", (0, 0))
    sx, sy = int(sx), int(sy)

    def norm_list(key):
        out = set()
        for p in (observation.get(key) or []):
            if isinstance(p, (list, tuple)) and len(p) >= 2:
                out.add((int(p[0]), int(p[1])))
        return out

    obstacles = norm_list("obstacles")
    self_terr = norm_list("self_territory")
    opp_terr = norm_list("opponent_territory")
    unclaimed = norm_list("unclaimed_cells")
    resources = norm_list("resources")

    cxands = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    cand = []
    for dx, dy in cxands:
        nx, ny = sx + dx, sy + dy
        if 0 <= nx < w and 0 <= ny < h:
            cand.append((dx, dy, nx, ny))
        else:
            cand.append((dx, dy, sx, sy))  # invalid -> engine keeps in place

    def obstacle_near(x, y):
        for tx in (x - 1, x, x + 1):
            for ty in (y - 1, y, y + 1):
                if (tx, ty) in obstacles:
                    return 1
        return 0

    def dist_to_set(x, y, s):
        if not s:
            return 10**9
        best = 10**9
        for tx, ty in s:
            d = abs(tx - x) + abs(ty - y)
            if d < best:
                best = d
                if best == 0:
                    break
        return best

    # Compute a deterministic "influence" target: prefer capturing opponent territory edges.
    best_move = (0, 0)
    best_val = -10**18
    ox, oy = observation.get("opponent_position", (w - 1, h - 1))
    ox, oy = int(ox), int(oy)

    # Build a small derived set of likely edge cells (opponent-occupied cells that neighbor non-opponent).
    edge = set()
    for px, py in opp_terr:
        for tx, ty in ((px - 1, py), (px + 1, py), (px, py - 1), (px, py + 1), (px - 1, py - 1), (px + 1, py - 1), (px - 1, py + 1), (px + 1, py + 1)):
            if (tx, ty) != (px, py) and 0 <= tx < w and 0 <= ty < h and (tx, ty) not in opp_terr and (tx, ty) not in obstacles:
                edge.add((px, py))
                break

    for dx, dy, nx, ny in cand:
        if (nx, ny) in obstacles:
            nx, ny = sx, sy  # treat as invalid -> stay

        val = 0

        # Primary objective: capture opponent territory directly.
        if (nx, ny) in opp_terr:
            val += 1200

        # Secondary: move onto unclaimed that is adjacent to opponent territory to build an edge assault.
        if (nx, ny) in unclaimed:
            adj_opp = 0
            for tx in (nx - 1, nx, nx + 1):
                for ty in (ny - 1, ny, ny + 1):
                    if (tx, ty) in opp_terr:
                        adj_opp = 1
                        break
                if adj_opp:
                    break
            val += 200 if adj_opp else 40

        # Resources if present (tertiary).
        if (nx, ny) in resources:
            val += 120

        # Avoid stepping deep into our own area if we can progress; prefer moving away from our center but
        # still keep some safety by not clustering near obstacles.
        if (nx, ny) in self_terr:
            val -= 25
            val -= 8 * obstacle_near(nx, ny)
        else:
            val -= 2

def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position", (0, 0))
    sx, sy = int(sx), int(sy)
    ox, oy = observation.get("opponent_position", (sx, sy))
    ox, oy = int(ox), int(oy)

    def to_set(key):
        out = set()
        for p in observation.get(key) or []:
            if isinstance(p, (list, tuple)) and len(p) >= 2:
                out.add((int(p[0]), int(p[1])))
        return out

    obstacles = to_set("obstacles")
    resources = to_set("resources")

    cand = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    best = None
    best_score = None

    def inside(x, y):
        return 0 <= x < w and 0 <= y < h

    def blocked(x, y):
        return (x, y) in obstacles

    def dman(x1, y1, x2, y2):
        return abs(x1 - x2) + abs(y1 - y2)

    nearest_res = None
    nearest_dist = 10**9
    for rx, ry in resources:
        d = dman(sx, sy, rx, ry)
        if d < nearest_dist:
            nearest_dist = d
            nearest_res = (rx, ry)

    for dx, dy in cand:
        nx, ny = sx + dx, sy + dy
        if not inside(nx, ny):
            nx, ny = sx, sy
        if blocked(nx, ny):
            continue

        # Prefer moving closer to nearest resource; if no resources, move toward opponent's vicinity.
        if nearest_res is not None:
            goalx, goaly = nearest_res
            prog = dman(nx, ny, goalx, goaly)
            opp = dman(nx, ny, ox, oy)
            score = (prog, -opp)
        else:
            prog = dman(nx, ny, ox, oy)
            score = (prog, 0)

        # Tie-break deterministically: smaller (dx,dy) preferred.
        tie = (dx, dy)
        key = (score, tie)
        if best is None or key < best_score:
            best = (dx, dy)
            best_score = key

    if best is None:
        return [0, 0]
    return [int(best[0]), int(best[1])]
