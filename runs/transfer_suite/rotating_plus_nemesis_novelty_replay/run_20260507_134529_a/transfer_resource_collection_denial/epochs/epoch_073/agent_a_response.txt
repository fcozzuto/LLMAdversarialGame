def choose_move(observation):
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    w, h = observation["grid_width"], observation["grid_height"]
    resources = observation.get("resources") or []
    obstacles = observation.get("obstacles") or []

    obs = set()
    for p in obstacles:
        if p and len(p) >= 2:
            obs.add((p[0], p[1]))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def free(x, y):
        return inb(x, y) and (x, y) not in obs

    dirs = [(-1,-1),(-1,0),(-1,1),(0,-1),(0,0),(0,1),(1,-1),(1,0),(1,1)]

    def cheb(x1, y1, x2, y2):
        dx = x1 - x2
        if dx < 0: dx = -dx
        dy = y1 - y2
        if dy < 0: dy = -dy
        return dx if dx > dy else dy

    def best_res_dist(x, y):
        if not resources:
            return 10**9
        bd = 10**9
        for rx, ry in resources:
            d = cheb(x, y, rx, ry)
            if d < bd:
                bd = d
        return bd

    def greedy_step(px, py, tx, ty):
        # choose move minimizing cheb distance to target; deterministic tie-break by (dx,dy)
        bestd = 10**9
        bestm = (0, 0)
        for dx, dy in dirs:
            nx, ny = px + dx, py + dy
            if not free(nx, ny):
                nx, ny = px, py
            d = cheb(nx, ny, tx, ty)
            if d < bestd or (d == bestd and (dx, dy) < bestm):
                bestd = d
                bestm = (dx, dy)
        return bestm

    if not resources:
        # drift toward opponent to contest the middle after sweeping rows
        tx, ty = (w//2, h//2)
        dx, dy = greedy_step(sx, sy, tx, ty)
        return [dx, dy]

    # choose our candidate move by simulating opponent's greedy response to their nearest resource
    my_best = 10**18
    my_move = (0, 0)

    # determine opponent's current greedy target deterministically (nearest by cheb, tie by lex)
    opp_target = resources[0]
    best_od = 10**9
    for r in resources:
        d = cheb(ox, oy, r[0], r[1])
        if d < best_od or (d == best_od and (r[0], r[1]) < (opp_target[0], opp_target[1])):
            best_od = d
            opp_target = r
    tx, ty = opp_target[0], opp_target[1]

    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not free(nx, ny):
            nx, ny = sx, sy

        # opponent step assuming they keep going for their greedy target
        odx, ody = greedy_step(ox, oy, tx, ty)
        onx, ony = ox + odx, oy + ody
        if not free(onx, ony):
            onx, ony = ox, oy

        # evaluate: increase our closeness to nearest resource and decrease theirs
        my_d = best_res_dist(nx, ny)
        opp_d = best_res_dist(onx, ony)

        # additional pressure: if we can reach a resource strictly sooner than opponent, prioritize
        # (cheb distance proxy; deterministic tie-break)
        reach_bias = opp_d - my_d

        # objective: maximize (reach_bias), then maximize (opp_d - my_d) implicitly, then minimize my_d
        obj = reach_bias * 1000 - my_d

        if obj > my_best or (obj == my_best and (dx, dy) < my_move):
            my_best = obj
            my_move = (dx, dy)

    return [my_move[0], my_move[1]]
