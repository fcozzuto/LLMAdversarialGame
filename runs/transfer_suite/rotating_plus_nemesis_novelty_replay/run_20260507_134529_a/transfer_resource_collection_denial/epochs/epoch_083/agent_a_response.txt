def choose_move(observation):
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    w, h = observation["grid_width"], observation["grid_height"]
    blocked = {(p[0], p[1]) for p in (observation.get("obstacles") or [])}
    resources = observation.get("resources") or []
    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in blocked

    def best_resource_score(x, y):
        if not resources:
            return 0, None
        best = None
        for r in resources:
            rx, ry = r[0], r[1]
            d_self = abs(x - rx) + abs(y - ry)
            d_opp = abs(ox - rx) + abs(oy - ry)
            val = (d_self, d_self - d_opp)
            if best is None or val < best[0]:
                best = (val, (rx, ry))
        return best[0], best[1]

    best_move = (0, 0)
    best_val = None
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny):
            continue
        (d_self, deny_gap), target = best_resource_score(nx, ny)
        if target is not None:
            rx, ry = target
            on_target = 0 if (nx == rx and ny == ry) else 1
            score = (d_self * 10 + on_target,
                     deny_gap,
                     abs(nx - ox) + abs(ny - oy),
                     abs(nx - rx) + abs(ny - ry))
        else:
            # No resources: head toward opponent (likely sweep) but prefer blocking via distance tie-break
            score = (abs(nx - ox) + abs(ny - oy), -(abs(nx - sx) + abs(ny - sy)))
        if best_val is None or score < best_val:
            best_val = score
            best_move = (dx, dy)

    dx, dy = best_move
    return [int(dx), int(dy)]
