    obs = set((p[0], p[1]) for p in obstacles)

    def inb(x, y):

        return inb(x, y) and (x, y) not in obs

            dx = -dx
        dy = y1 - y2
        if dy < 0:
            dy = -dy
        return dx if dx > dy else dy
