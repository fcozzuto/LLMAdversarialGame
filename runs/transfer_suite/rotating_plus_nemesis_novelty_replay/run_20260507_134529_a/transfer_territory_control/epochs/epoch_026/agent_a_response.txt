def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position", (0, 0))
    ox, oy = observation.get("opponent_position", (w - 1, h - 1))
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    def to_set(key):
        out = set()
        for p in observation.get(key) or []:
            if isinstance(p, (list, tuple)) and len(p) >= 2:
                x, y = int(p[0]), int(p[1])
                if 0 <= x < w and 0 <= y < h:
                    out.add((x, y))
        return out

    obstacles = to_set("obstacles")
    self_terr = to_set("self_territory")
    opp_terr = to_set("opponent_territory")
    unclaimed = to_set("unclaimed_cells")

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def valid(nx, ny):
        return 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obstacles

    def neigh4(x, y):
        return [(x, y - 1), (x, y + 1), (x - 1, y), (x + 1, y)]

    # Strategy shift: instead of expanding from our territory, cut opponent by stepping onto/adjacent to their territory.
    opp_targets = set()
    if opp_terr:
        for x, y in opp_terr:
            for nx, ny in neigh4(x, y):
                if 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obstacles:
                    opp_targets.add((nx, ny))

    # If no opponent territory info, fall back to unclaimed.
    if not opp_targets:
        opp_targets = set(unclaimed) if unclaimed else {(ox, oy)}

    best = None
    best_score = -10**18
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not valid(nx, ny):
            continue

        # Higher is better: prefer moving onto opponent territory (flip), otherwise adjacent to it.
        is_opp = (nx, ny) in opp_terr
        adj_opp = sum((px, py) in opp_terr for px, py in neigh4(nx, ny))
        dist_to_opp = min(abs(nx - tx) + abs(ny - ty) for tx, ty in opp_targets) if opp_targets else 0

        # Light penalty for moving into our already-controlled cells (to avoid wasting moves).
        waste = 1 if (nx, ny) in self_terr else 0

        score = (200 if is_opp else 0) + 20 * adj_opp - 2 * dist_to_opp - 5 * waste

        # Deterministic tie-breaker: lexicographic by (dx, dy)
        cand = (score, -dx, -dy)
        if cand > best:
            best = cand
            best_move = (dx, dy)

    if best is None:
        return [0, 0]
    return [int(best_move[0]), int(best_move[1])]

        return 0 <= x < w and 0 <= y < h and (x, y) not in obst

    if not candidates:
        return [0, 0]

    resources = []

            x, y = int(p[0]), int(p[1])

    def d2(x, y, tx, ty):
        dx = x - tx
        dy = y - ty
        return dx * dx + dy * dy

    best = None
    for dx, dy, nx, ny in candidates:
