def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position", (0, 0))
    ox, oy = observation.get("opponent_position", (w - 1, h - 1))
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    def to_set(key):
        out = set()
        for p in observation.get(key) or []:
            if isinstance(p, (list, tuple)) and len(p) >= 2:
                x, y = int(p[0]), int(p[1])
                if 0 <= x < w and 0 <= y < h:
                    out.add((x, y))
        return out

    obs = to_set("obstacles")
    unclaimed = to_set("unclaimed_cells")
    opp_terr = to_set("opponent_territory")
    self_terr = to_set("self_territory")

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    best = None
    best_score = -10**18

    # Prefer deterministic tie-break: iterate moves in fixed order.
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not (0 <= nx < w and 0 <= ny < h):
            continue
        if (nx, ny) in obs:
            continue

        # Heuristic: territory counterclaim => reward stepping into opponent-controlled cells.
        if (nx, ny) in opp_terr:
            base = 800.0
        elif (nx, ny) in unclaimed:
            base = 120.0
        elif (nx, ny) in self_terr:
            base = 30.0
        else:
            base = 10.0

        # Discourage getting pinned near opponent unless it flips them.
        d_opp = abs(nx - ox) + abs(ny - oy)
        pin_pen = 0.0 if (nx, ny) in opp_terr else (80.0 if d_opp <= 1 else 30.0 if d_opp <= 2 else 0.0)

        # Encourage moving toward the biggest "frontier" away from obstacles by minimizing distance
        # to unclaimed, but with an opponent-distance safeguard.
        if unclaimed:
            # simple local target: nearest unclaimed to the candidate
            near_un = min(abs(nx - ux) + abs(ny - uy) for ux, uy in unclaimed)
            expand = 60.0 / (1.0 + near_un)
        else:
            expand = 0.0

        score = base + expand - pin_pen + (d_opp * (5.0 if (nx, ny) not in opp_terr else -2.0))

        if score > best_score:
            best_score = score
            best = (dx, dy)

    if best is None:
        return [0, 0]
    return [int(best[0]), int(best[1]]]
