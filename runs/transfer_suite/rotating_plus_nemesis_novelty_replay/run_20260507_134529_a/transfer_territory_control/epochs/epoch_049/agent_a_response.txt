def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position", (0, 0))
    ox, oy = observation.get("opponent_position", (w - 1, h - 1))
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    def to_set(key):
        out = set()
        for p in observation.get(key) or []:
            if isinstance(p, (list, tuple)) and len(p) >= 2:
                x, y = int(p[0]), int(p[1])
                if 0 <= x < w and 0 <= y < h:
                    out.add((x, y))
        return out

    obstacles = to_set("obstacles")
    unclaimed = to_set("unclaimed_cells")
    self_terr = to_set("self_territory")
    opp_terr = to_set("opponent_territory")

    cx, cy = (w - 1) / 2.0, (h - 1) / 2.0
    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def dist(a, b):
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

    def valid(nx, ny):
        return 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obstacles

    best = None
    best_sc = -10**18

    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not valid(nx, ny):
            continue
        dopp = dist((nx, ny), (ox, oy))
        to_center = -dist((nx, ny), (cx, cy))
        in_unclaimed = (nx, ny) in unclaimed
        in_self = (nx, ny) in self_terr
        in_opp = (nx, ny) in opp_terr

        # Prefer capturing new/unclaimed; if contested, prefer moves that increase separation.
        cap = 0.0
        if in_unclaimed:
            cap += 6.0
        elif in_self:
            cap += 1.0
        if in_opp:
            cap += 2.0  # flipping can help, but separated by dmin term

        # Opponent edge-claiming: keep advancing along the axis that reduces their ability to swing into center.
        vx, vy = (nx - sx), (ny - sy)
        opp_vec = (sx - ox, sy - oy)
        align = (vx * opp_vec[0] + vy * opp_vec[1])  # positive => moving away from opponent
        sep_bonus = 0.0
        if doup := dists := dopp:
            sep_bonus = min(8.0, dopp / 1.5)

        sc = cap + 0.7 * to_center + 1.2 * sep_bonus + 0.25 * align + 0.02 * dopp
        if sc > best_sc:
            best_sc = sc
            best = [dx, dy]

    return best if best is not None else [0, 0]
