def choose_move(observation):
    w = int(observation.get("grid_width", 0) or 0)
    h = int(observation.get("grid_height", 0) or 0)
    sx, sy = observation.get("self_position", (0, 0))
    ox, oy = observation.get("opponent_position", (0, 0))
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    obs = set()
    for p in (observation.get("obstacles") or []):
        if p is not None and len(p) >= 2:
            obs.add((int(p[0]), int(p[1])))

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    def inb(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obs

    def cheb(x1, y1, x2, y2):
        dx = x1 - x2
        if dx < 0: dx = -dx
        dy = y1 - y2
        if dy < 0: dy = -dy
        return dx if dx > dy else dy

    def free_neighbors(x, y):
        c = 0
        for ax in (-1, 0, 1):
            for ay in (-1, 0, 1):
                if ax == 0 and ay == 0:
                    continue
                if inb(x + ax, y + ay):
                    c += 1
        return c

    curd = cheb(sx, sy, ox, oy)
    best = None
    best_key = None

    # Deterministic tie-break order: closer-first, then more freedom, then preferring movement that continues pursuit.
    order = {(-1, -1): 0, (-1, 0): 1, (-1, 1): 2, (0, -1): 3, (0, 0): 4, (0, 1): 5, (1, -1): 6, (1, 0): 7, (1, 1): 8}
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny):
            continue
        d = cheb(nx, ny, ox, oy)
        step_progress = (abs(nx - ox) + abs(ny - oy)) - (abs(sx - ox) + abs(sy - oy))
        # Penalize moving away in either metric, heavily avoid dead-ends, and break ties deterministically.
        key = (
            d,                       # minimize distance in Chebyshev
            -free_neighbors(nx, ny),# maximize mobility
            step_progress,           # prefer non-increasing Manhattan distance
            abs(nx - mx) + abs(ny - my) if (w and h) else 0,  # mild preference toward centerline interception
            order[(dx, dy)]
        )
        if best_key is None or key < best_key:
            best_key = key
            best = [dx, dy]

    if best is None:
        return [0, 0]

    # If no move improves distance, force an "intercept stride" toward opponent using sign steps (still deterministic).
    if best_key[0] >= curd:
        sdx = 0 if ox == sx else (1 if ox > sx else -1)
        sdy = 0 if oy == sy else (1 if oy > sy else -1)
        cand = (sdx, sdy)
        if cand in moves and inb(sx + cand[0], sy + cand[1]):
            return [cand[0], cand[1]]
        # fallback: best among valid axis directions
        for dx, dy in moves:
            if dx == 0 and dy == 0:
                continue
            nx, ny = sx + dx, sy + dy
            if inb(nx, ny) and cheb(nx, ny, ox, oy) <= curd:
                return [dx, dy]
        return best

    return best
