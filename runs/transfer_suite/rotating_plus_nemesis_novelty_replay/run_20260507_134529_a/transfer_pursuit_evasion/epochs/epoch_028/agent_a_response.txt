    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    obstacles = set()
    for p in observation.get("obstacles") or []:
        if p is not None and len(p) >= 2:
            obstacles.add((int(p[0]), int(p[1])))

    role = (observation.get("self_role") or "").lower()
    pursuer = ("pursuer" in role) or ("catch" in role) or ("hunter" in role)

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obstacles

    def dist2(x1, y1, x2, y2):
        dx, dy = x1 - x2, y1 - y2
        return dx * dx + dy * dy

    def clamp(x, lo, hi):
        return lo if x < lo else hi if x > hi else x

    # Predict opponent's next step (zigzag evasion: usually move away from pursuer)
    away_x = 1 if ox > sx else (-1 if ox < sx else 0)
    away_y = 1 if oy > sy else (-1 if oy < sy else 0)
    px = clamp(ox + away_x, 0, w - 1)
    py = clamp(oy + away_y, 0, h - 1)

            if (ox, ty) not in obstacles:
