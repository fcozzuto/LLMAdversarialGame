def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position", (0, 0)) or (0, 0)
    ox, oy = observation.get("opponent_position", (0, 0)) or (0, 0)
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    obstacles = set()
    for p in observation.get("obstacles") or []:
        if p is not None and len(p) >= 2:
            obstacles.add((int(p[0]), int(p[1])))

    self_role = (observation.get("self_role") or "").lower()
    opponent_role = (observation.get("opponent_role") or "").lower()
    self_is_evader = ("evader" in self_role) or ("pursuer" not in self_role and "evader" in opponent_role)

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def ok(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obstacles

    def dist2(x, y):
        dx, dy = x - ox, y - oy
        return dx * dx + dy * dy

    def mobility(x, y):
        m = 0
        for dx, dy in moves:
            nx, ny = x + dx, y + dy
            if ok(nx, ny):
                m += 1
        return m

    candidates = []
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not ok(nx, ny):
            continue
        if self_is_evader:
            # Evade: maximize distance, prefer moves that reduce pursuer's immediate mobility if we were pursuer.
            # Here use our own mobility after move as a proxy for not getting boxed in.
            val = dist2(nx, ny) + 0.15 * mobility(nx, ny)
            tie = (-(dx == 0 and dy == 0),)
        else:
            # Pursue: minimize distance; if close, also reduce opponent escape mobility by moving to choke zones.
            # Approximate by: choose moves that would minimize opponent mobility if opponent were at its current spot.
            # (mobility doesn't change, so instead bias toward decreasing our mobility while staying near opponent)
            val = dist2(nx, ny) - 0.10 * mobility(nx, ny)
            tie = (dx == 0 and dy == 0,)
        candidates.append((val, tie, (dx, dy), nx, ny))

    if not candidates:
        return [0, 0]

    # Deterministic tie-break: sort by value, then by a parity-based preference to avoid repeating identical policy.
    parity = int(observation.get("turn_index", 0) or 0) & 1
    # Prefer diagonal when parity=0, axial when parity=1 (deterministic, helps vs zigzag patterns).
    def preference(move):
        dx, dy = move
        diag = (dx != 0 and dy != 0)
        return 0 if (diag == (parity == 0)) else 1

    best = None
    best_key = None
    for val, tie, mv, nx, ny in candidates:
        key = (val, preference(mv), mv[0], mv[1])
        if best_key is None or key > best_key if self_is_evader else key < best_key:
            best_key = key
            best = mv

    if best is None:
        return [0, 0]
    return [int(best[0]), int(best[1])]
