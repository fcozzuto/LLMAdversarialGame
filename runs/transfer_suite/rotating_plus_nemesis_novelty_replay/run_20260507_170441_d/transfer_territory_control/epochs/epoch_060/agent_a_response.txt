def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = map(int, observation.get("self_position", (0, 0)))
    ox, oy = map(int, observation.get("opponent_position", (0, 0)))
    obstacles = observation.get("obstacles") or []
    blocked = set()
    for p in obstacles:
        try:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                blocked.add((x, y))
        except:
            pass

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    self_t = set((int(x), int(y)) for x, y in (observation.get("self_territory") or []))
    opp_t = set((int(x), int(y)) for x, y in (observation.get("opponent_territory") or []))
    unclaimed = set((int(x), int(y)) for x, y in (observation.get("unclaimed_cells") or []))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in blocked

    def neighbors8(p):
        x, y = p
        for dx, dy in dirs:
            if dx == 0 and dy == 0:
                continue
            nx, ny = x + dx, y + dy
            if inb(nx, ny):
                yield (nx, ny)

    # Primary: claim cells adjacent to opponent territory (deny their expansion)
    cand = set()
    if opp_t and unclaimed:
        for ux, uy in unclaimed:
            for nx, ny in neighbors8((ux, uy)):
                if (nx, ny) in opp_t:
                    cand.add((ux, uy))
                    break

    # Secondary: claim cells adjacent to our territory
    if not cand and self_t and unclaimed:
        for ux, uy in unclaimed:
            for nx, ny in neighbors8((ux, uy)):
                if (nx, ny) in self_t:
                    cand.add((ux, uy))
                    break

    # Tertiary: move toward nearest unclaimed cell
    if not cand and unclaimed:
        target = min(unclaimed, key=lambda p: abs(p[0] - sx) + abs(p[1] - sy) + 0.3 * (abs(p[0] - ox) + abs(p[1] - oy)))
        cand = {target}

    # If nothing, drift toward opponent to keep pressure
    if not cand:
        tx, ty = ox, oy
        target_set = {(tx, ty)}
    else:
        # Choose deterministically: closest, then lexicographic
        target = min(cand, key=lambda p: (abs(p[0] - sx) + abs(p[1] - sy), p[0], p[1]))
        target_set = {target}

    tx, ty = next(iter(target_set))

    # Pick best immediate step by greedy score: closer to target, avoid obstacles, slight bias away from opponent
    best = [0, 0]
    best_key = None
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny):
            continue
        # Prefer stepping into unclaimed or opponent territory adjacent to our aim
        step_to_target = abs(nx - tx) + abs(ny - ty)
        into_unclaimed = 1 if (nx, ny) in unclaimed else 0
        into_opp = 1 if (nx, ny) in opp_t else 0
        dist_from_opp = abs(nx - ox) + abs(ny - oy)
        key = (step_to_target, -3 * into_unclaimed - 1.5 * into_opp, -dist_from_opp, dx, dy)
        if best_key is None or key < best_key:
            best_key = key
            best = [dx, dy]
    return best

                res.append((x, y))
        except:
            pass

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    cx, cy = w // 2, h // 2
    best = None
    best_score = None

    for dx, dy in dirs:
