def choose_move(observation):
    w = observation.get("grid_width", 8)
    h = observation.get("grid_height", 8)
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [w - 1, h - 1])
    resources = observation.get("resources", []) or []
    obstacles = set(tuple(p) for p in (observation.get("obstacles", []) or []))

    dirs8 = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def bfs_dist(x0, y0):
        INF = 10**9
        dist = [[INF] * h for _ in range(w)]
        if not inb(x0, y0) or (x0, y0) in obstacles:
            return dist
        dist[x0][y0] = 0
        qx, qy = [x0], [y0]
        qi = 0
        steps = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
        while qi < len(qx):
            x, y = qx[qi], qy[qi]
            qi += 1
            d = dist[x][y] + 1
            for dx, dy in steps:
                nx, ny = x + dx, y + dy
                if inb(nx, ny) and (nx, ny) not in obstacles and d < dist[nx][ny]:
                    dist[nx][ny] = d
                    qx.append(nx); qy.append(ny)
        return dist

    if not resources:
        return [0, 0]

    INF = 10**9
    d_op = bfs_dist(ox, oy)
    best_move = [0, 0]
    best_val = -10**30

    # Deterministic ordering: prefer closer-forward directions by Manhattan decrease, then lexicographic.
    def pref_key(dx, dy):
        return (-(abs((sx + dx) - ox) + abs((sy + dy) - oy)), dx, dy)

    moves = sorted(dirs8, key=pref_key)

    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obstacles:
            continue
        if (nx, ny) in obstacles:
            continue
        d_me = bfs_dist(nx, ny)
        # Evaluate best target resource from this state.
        # Advantage: we want to be earlier than opponent; also capture now if possible.
        cell_best = -10**30
        for rx, ry in resources:
            dm = d_me[rx][ry]
            do = d_op[rx][ry]
            if dm >= INF or do >= INF:
                continue
            if dm == 0:
                val = 10**9
            else:
                # Win earlier strongly; otherwise prefer targets where opponent is late.
                val = (do - dm) * 1000 - dm
            if val > cell_best:
                cell_best = val
        # If no reachable resources exist (rare), neutral.
        if cell_best < -10**20:
            cell_best = -10**20
        if cell_best > best_val:
            best_val = cell_best
            best_move = [dx, dy]

    return [int(best_move[0]), int(best_move[1])]
