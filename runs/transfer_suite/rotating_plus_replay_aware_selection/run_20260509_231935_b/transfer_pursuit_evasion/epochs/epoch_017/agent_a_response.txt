    obs_list = observation.get("obstacles") or []
    obs = set()

    pursue = True

        pursue = False
    if ("evad" in orr) and ("purs" in sr):

        pursue = False

    def in_bounds(x, y):
        return 0 <= x < w and 0 <= y < h

    def dist(x, y):
        return abs(x - ox) + abs(y - oy)

    best = None
