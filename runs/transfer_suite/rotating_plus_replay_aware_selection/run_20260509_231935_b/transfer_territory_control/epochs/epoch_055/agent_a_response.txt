    self_terr = set(tuple(p) for p in (observation.get("self_territory") or []))
    opp_terr = set(tuple(p) for p in (observation.get("opponent_territory") or []))
    unclaimed = list(observation.get("unclaimed_cells") or [])

        return [0, 0]

    def inside(x, y): return 0 <= x < w and 0 <= y < h
    def man(x1, y1, x2, y2): return abs(x1 - x2) + abs(y1 - y2)

    cx, cy = (w - 1) / 2.0, (h - 1) / 2.0

    candidates = []
    t = int(observation.get("turn_index", 0))
    if unclaimed:
        # Prefer cells that we can reach before opponent, also near center.
        for x, y in unclaimed:
            dS = man(sx, sy, x, y)
            dO = man(ox, oy, x, y)
            center_bias = abs(x - cx) + abs(y - cy)
            # Alternate between greedy unclaimed and center-biased to avoid stagnation.
            wO = 1.0 if (t % 2 == 0) else 0.75
            score = (dS - wO * dO) + 0.02 * center_bias + 0.001 * (x + y)
            candidates.append((score, x, y))
        candidates.sort()
        target = (candidates[0][1], candidates[0][2])
