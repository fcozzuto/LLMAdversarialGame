def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    obstacles = set()
    for p in observation.get("obstacles") or []:
        if p and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

    role = (observation.get("self_role", "") or "").lower()
    pursuer = any(k in role for k in ("pursuer", "hunter", "chaser", "pursuit"))

    dxs = (-1, 0, 1)
    dys = (-1, 0, 1)

    def clamp(nx, ny):
        if nx < 0: nx = 0
        elif nx >= w: nx = w - 1
        if ny < 0: ny = 0
        elif ny >= h: ny = h - 1
        return nx, ny

    def obst_cost(nx, ny):
        if not obstacles:
            return 0.0
        best = 10**9
        for x, y in obstacles:
            d = abs(nx - x) + abs(ny - y)
            if d < best:
                best = d
                if best == 0:
                    break
        if best == 0:
            return 1e6
        if best == 1:
            return 30.0
        if best == 2:
            return 10.0
        return float(best) * 0.15

    def best_evade_tie(nx, ny):
        # Deterministic tie-break: prefer increasing "freedom" (more non-obstacle neighbors)
        cnt = 0
        for a in dxs:
            for b in dys:
                if a == 0 and b == 0:
                    continue
                px, py = clamp(nx + a, ny + b)
                if (px, py) not in obstacles:
                    cnt += 1
        return cnt

    best_move = (0, 0)
    if pursuer:
        best_val = -10**18
        for dx in dxs:
            for dy in dys:
                nx, ny = clamp(sx + dx, sy + dy)
                if (nx, ny) in obstacles:
                    continue
                dist = abs(nx - ox) + abs(ny - oy)
                val = -dist - obst_cost(nx, ny)
                if nx == ox and ny == oy:
                    val += 100000.0
                # If tied, prefer closer by distance then fewer obstacle costs then determinism by dx,dy order
                if val > best_val:
                    best_val = val
                    best_move = (dx, dy)
        return [int(best_move[0]), int(best_move[1])]
    else:
        best_val = -10**18
        best_freedom = -10**18
        for dx in dxs:
            for dy in dys:
                nx, ny = clamp(sx + dx, sy + dy)
                if (nx, ny) in obstacles:
                    continue
                dist = abs(nx - ox) + abs(ny - oy)
                val = dist - obst_cost(nx, ny)
                freedom = best_evade_tie(nx, ny)
                if val > best_val or (val == best_val and freedom > best_freedom):
                    best_val = val
                    best_freedom = freedom
                    best_move = (dx, dy)
        return [int(best_move[0]), int(best_move[1])]
