    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position", (0, 0))
    ox, oy = observation.get("opponent_position", (0, 0))
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    obs_list = observation.get("obstacles") or []
    obstacles = set()
    for p in obs_list:
        if p and len(p) >= 2:
            obstacles.add((int(p[0]), int(p[1])))

    self_role = (observation.get("self_role", "") or "").lower()
    is_evader = ("evader" in self_role) or ("runner" in self_role) or ("evasion" in self_role)

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def blocked(x, y):
        return (x, y) in obstacles

    def manh(x1, y1, x2, y2):
        return abs(x1 - x2) + abs(y1 - y2)

    # Obstacle field: prefer cells closer to free space rather than hugging obstacles.
    obst_points = list(obstacles)
    def obstacle_score(x, y):
        if not obst_points:
            return 0.0
        dmin = 10**9
        for ax, ay in obst_points:
            d = abs(x - ax) + abs(y - ay)
            if d < dmin:
                dmin = d
        if dmin <= 0:
            return -1e6
        # Mildly penalize being adjacent; avoid hard block only if actually blocked.
        if dmin == 1:
            return -120.0
        if dmin == 2:
            return -35.0
        return 6.0 / (dmin + 1)

    # Predict evader/pursuer preference using current relative position (no history available).
    # pursuer tries to reduce distance; evader tries to increase it, especially along diagonals toward far corner.
    corners = [(0, 0), (0, h - 1), (w - 1, 0), (w - 1, h - 1)]
    far_corner = max(corners, key=lambda c: manh(c[0], c[1], ox, oy)) if is_evader else None

    best = None
    best_val = -10**18 if is_evader else 10**18

    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or blocked(nx, ny):
            continue
        d = manh(nx, ny, ox, oy)

        # Capture ends game; if evader, avoid; if pursuer, seek.
        if d == 0:
            val = -1e9 if is_evader else 1e9
            if is_evader:
                if val > best_val:
                    best_val, best = val, [dx, dy]
            else:
                if val < best_val:
                    best_val, best = val, [dx, dy]
            continue

            target_bonus = 0.0

                before = manh(sx, sy, far_corner[0], far_corner[1])
