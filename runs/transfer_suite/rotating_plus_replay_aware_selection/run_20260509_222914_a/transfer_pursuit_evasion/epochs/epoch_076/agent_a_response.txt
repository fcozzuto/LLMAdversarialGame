def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    obstacles = set()
    for p in observation.get("obstacles") or []:
        if p and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

    role = (observation.get("self_role", "") or "").lower()
    pursuer = any(k in role for k in ("pursuer", "hunter", "chaser", "pursuit"))

    corners = [(0, 0), (0, h - 1), (w - 1, 0), (w - 1, h - 1)]
    far_corner = max(corners, key=lambda c: abs(c[0] - ox) + abs(c[1] - oy))

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def clamp(nx, ny):
        if nx < 0: nx = 0
        elif nx >= w: nx = w - 1
        if ny < 0: ny = 0
        elif ny >= h: ny = h - 1
        return nx, ny

    def obst_pen(nx, ny):
        if (nx, ny) in obstacles:
            return 1e9
        if not obstacles:
            return 0.0
        md = 10**9
        for x, y in obstacles:
            d = abs(nx - x) + abs(ny - y)
            if d < md: md = d
            if md == 1: break
        return 10.0 / (md + 1.0)

    best = None
    best_score = -1e18 if pursuer else 1e18
    for dx, dy in moves:
        nx, ny = clamp(sx + dx, sy + dy)
        if (nx, ny) in obstacles:
            continue
        dist_op = abs(nx - ox) + abs(ny - oy)
        dist_corner = abs(nx - far_corner[0]) + abs(ny - far_corner[1])
        pen = obst_pen(nx, ny)
        score = (-dist_op - 0.2 * dist_corner - 2.0 * pen) if pursuer else (dist_op - 0.2 * dist_corner - 2.0 * pen)
        if best is None or (score > best_score if pursuer else score < best_score):
            best_score, best = score, (dx, dy)

    if best is None:
        return [0, 0]
    return [int(best[0]), int(best[1]]]

    res_list = []
    for p in resources:
        if p and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                res_list.append((x, y))

    role = (observation.get("self_role") or "").lower()
    pursue = any(k in role for k in ("pursuer", "hunter", "chaser", "pursuit"))

    moves = [(-1,-1),(-1,0),(-1,1),(0,-1),(0,0),(0,1),(1,-1),(1,0),(1,1)]
    best_dxdy = [0, 0]
    best_score = -10**18

    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if nx < 0 or nx >= w or ny < 0 or ny >= h:
            continue
        if (nx, ny) in obstacles:
            score = -10**15
        else:
            md_res = 10**9
            for rx, ry in res_list:
                d = abs(nx - rx) + abs(ny - ry)
                if d < md_res:
                    md_res = d
            res_score = 0.0 if not res_list else (50.0 / (md_res + 1.0))
            if pursue:
                opp_score = - (abs(nx - ox) + abs(ny - oy))

                opp_score = (abs(nx - ox) + abs(ny - oy)) * 0.4
            edge_pen = 0

                edge_pen = -0.5
            score = res_score + opp_score + edge_pen
        if score > best_score:
            best_score = score
            best_dxdy = [dx, dy]
