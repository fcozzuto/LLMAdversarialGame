def choose_move(observation):
    w = int(observation.get("grid_width", 0) or 0)
    h = int(observation.get("grid_height", 0) or 0)
    if w <= 0 or h <= 0:
        return [0, 0]

    sx, sy = observation.get("self_position", (0, 0))
    ox, oy = observation.get("opponent_position", (0, 0))
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    obstacles = observation.get("obstacles") or []
    obs = set()
    for p in obstacles:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            ax, ay = int(p[0]), int(p[1])
            if 0 <= ax < w and 0 <= ay < h:
                obs.add((ax, ay))

    resources = []
    for p in (observation.get("resources") or []):
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            rx, ry = int(p[0]), int(p[1])
            if 0 <= rx < w and 0 <= ry < h and (rx, ry) not in obs:
                resources.append((rx, ry))

    if not resources:
        return [0, 0]

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def dist(x1, y1, x2, y2):
        dx = x1 - x2
        if dx < 0: dx = -dx
        dy = y1 - y2
        if dy < 0: dy = -dy
        return dx + dy

    opp_row = oy
    best = (-10**9, None)

    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not (0 <= nx < w and 0 <= ny < h) or (nx, ny) in obs:
            continue

        # Score move by best resource we can contest soonest.
        best_res = (-10**9, 10**9, 10**9, 10**9)
        for rx, ry in resources:
            sd = dist(nx, ny, rx, ry)
            od = dist(ox, oy, rx, ry)
            # Prefer cells we can reach no later than opponent, and reach sooner.
            contest = od - sd  # higher is better
            # Small anti-sweep behavior: bias toward resources nearer opponent's current row.
            row_bias = -abs(ry - opp_row)  # closer to opp row is better
            tie = -sd
            cand = (contest, row_bias, tie)
            if cand > best_res[0:3]:
                best_res = (cand[0], 0, 0, cand[1])  # pack first component + row_bias

        # Combine: maximize contest; then maximize row_bias; then minimize distance to closest resource from nx,ny.
        contest = -10**9
        row_bias = -10**9
        min_sd = 10**9
        for rx, ry in resources:
            sd = dist(nx, ny, rx, ry)
            od = dist(ox, oy, rx, ry)
            c = od - sd
            if c > contest:
                contest = c
            if -abs(ry - opp_row) > row_bias:
                row_bias = -abs(ry - opp_row)
            if sd < min_sd:
                min_sd = sd

        # If we can't contest, still pick move that reduces our distance most.
        # Deterministic final tie-break: lexicographic preference on (dx,dy) after score.
        score = (contest, row_bias, -min_sd, -abs(nx - sx), -abs(ny - sy), dx, dy)
        if score > best[0]:
            best = (score, (dx, dy))

    if best[1] is None:
        return [0, 0]
    return [int(best[1][0]), int(best[1][1])]
