def choose_move(observation):
    w = int(observation.get("grid_width", 0) or 0)
    h = int(observation.get("grid_height", 0) or 0)
    if w <= 0 or h <= 0:
        return [0, 0]

    sx, sy = observation.get("self_position", (0, 0))
    ox, oy = observation.get("opponent_position", (0, 0))
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    obs = set()
    for p in (observation.get("obstacles") or []):
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            ax, ay = int(p[0]), int(p[1])
            if 0 <= ax < w and 0 <= ay < h:
                obs.add((ax, ay))

    resources = []
    for p in (observation.get("resources") or []):
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            rx, ry = int(p[0]), int(p[1])
            if 0 <= rx < w and 0 <= ry < h and (rx, ry) not in obs:
                resources.append((rx, ry))
    if not resources:
        return [0, 0]

    def cheb(x1, y1, x2, y2):
        dx = x1 - x2
        if dx < 0: dx = -dx
        dy = y1 - y2
        if dy < 0: dy = -dy
        return dx if dx > dy else dy

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    best = (None, None, None)  # (score, opp_lead, tie_dist)
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not (0 <= nx < w and 0 <= ny < h) or (nx, ny) in obs:
            continue

        best_cell = None
        for rx, ry in resources:
            self_t = cheb(nx, ny, rx, ry)
            opp_t = cheb(ox, oy, rx, ry)
            # Strongly prefer resources we can reach earlier, otherwise still move toward good races.
            lead = opp_t - self_t  # positive is good
            score = lead * 100 - self_t
            # Secondary: prefer less total urgency/contestedness when scores tie
            opp_lead = lead
            tie_dist = self_t
            key = (score, opp_lead, -tie_dist, rx, ry)
            if best_cell is None or key > best_cell:
                best_cell = key

        if best_cell is None:
            continue
        if best[0] is None or (best_cell[0], best_cell[1], best_cell[2]) > (best[0], best[1], best[2]):
            best = (best_cell[0], best_cell[1], best_cell[2])
            best_move = [dx, dy]
        elif best_cell is not None:
            cur = (best_cell[0], best_cell[1], best_cell[2])
            if cur == (best[0], best[1], best[2]):
                # Deterministic tiebreaker: prefer smaller dx, then smaller dy
                if [dx, dy] < best_move:
                    best_move = [dx, dy]

    return best_move if 'best_move' in locals() else [0, 0]
