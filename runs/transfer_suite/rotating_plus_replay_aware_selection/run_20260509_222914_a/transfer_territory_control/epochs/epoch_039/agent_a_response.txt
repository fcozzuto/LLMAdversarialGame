def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position") or (0, 0)
    ox, oy = observation.get("opponent_position") or (sx, sy)
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    def inside(x, y):
        return 0 <= x < w and 0 <= y < h

    obstacles = set()
    for p in observation.get("obstacles") or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if inside(x, y):
                obstacles.add((x, y))

    self_t = set()
    for p in observation.get("self_territory") or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if inside(x, y):
                self_t.add((x, y))

    opp_t = set()
    for p in observation.get("opponent_territory") or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if inside(x, y):
                opp_t.add((x, y))

    unclaimed = []
    for p in observation.get("unclaimed_cells") or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if inside(x, y) and (x, y) not in obstacles and (x, y) not in self_t and (x, y) not in opp_t:
                unclaimed.append((x, y))

    def dist(a, b):
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

    def best_target():
        if not unclaimed:
            return None
        best = None
        bestv = -10**9
        for x, y in unclaimed:
            dme = dist((sx, sy), (x, y))
            doe = dist((ox, oy), (x, y))
            adj_ours = 0
            for dx in (-1, 0, 1):
                for dy in (-1, 0, 1):
                    if dx == 0 and dy == 0:
                        continue
                    if (x + dx, y + dy) in self_t:
                        adj_ours = 1
            v = (10 - dme) + (doe - dme) + 3 * adj_ours
            if v > bestv:
                bestv = v
                best = (x, y)
        return best

    target = best_target()
    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def cell_value(nx, ny):
        if (nx, ny) in obstacles:
            return -10**9
        if (nx, ny) in self_t:
            return 1.0
        if (nx, ny) in opp_t:
            return 6.0  # likely to flip
        # unclaimed or unknown: prefer expansion
        if not unclaimed or (nx, ny) not in opp_t:
            return 3.0
        return 0.5

    # If no unclaimed targets, try to move into opponent territory; otherwise toward center-like
    if target is None:
        target = (w // 2, h // 2)

    best_move = (0, 0)
    best_score = -10**9
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not inside(nx, ny) or (nx, ny) in obstacles:
            continue
        d_to_target = dist((nx, ny), target)
        d_from_opp = dist((ox, oy), (nx, ny))
        gain = cell_value(nx, ny)
        # heuristic: maximize immediate control gain, then improve target approach, then widen opponent gap
        sc = gain * 10 + (40 - d_to_target) + (d_from_opp - d_to_target) * 2
        if sc > best_score:
            best_score = sc
            best_move = (dx, dy)

    return [int(best_move[0]), int(best_move[1])]
