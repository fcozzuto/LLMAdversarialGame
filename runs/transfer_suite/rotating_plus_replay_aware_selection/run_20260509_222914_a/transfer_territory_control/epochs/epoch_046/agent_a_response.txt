    sx, sy = int(sp[0]), int(sp[1])
    ox, oy = int(op[0]), int(op[1])

        if isinstance(p, (list, tuple)) and len(p) >= 2:
            try:
                x, y = int(p[0]), int(p[1])
                if 0 <= x < w and 0 <= y < h:
                    obstacles.add((x, y))
            except:
                pass

    def inside(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obstacles

    my_ter = set()
    for p in observation.get("self_territory") or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            try:
                my_ter.add((int(p[0]), int(p[1])))
            except:
                pass

        if isinstance(p, (list, tuple)) and len(p) >= 2:
