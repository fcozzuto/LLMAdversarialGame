def choose_move(observation):
    w = observation["grid_width"]
    h = observation["grid_height"]
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    obstacles = set((p[0], p[1]) for p in observation.get("obstacles", []) if p is not None and len(p) >= 2)

    candidates = observation.get("unclaimed_cells") or []
    if not candidates:
        candidates = observation.get("self_territory", []) or []
    if not candidates:
        return [0, 0]

    # Prefer targets that are somewhat reachable, but maximally far from opponent (edge/corner bias)
    best_t = None
    best_key = None
    for x, y in candidates:
        if x is None or y is None:
            continue
        if (x, y) in obstacles:
            continue
        du = abs(x - sx) + abs(y - sy)
        do = abs(x - ox) + abs(y - oy)
        # Key: minimize du while maximizing do; deterministic via tuple ordering
        key = (du, -do, x, y)
        if best_key is None or key < best_key:
            best_key = key
            best_t = (x, y)

    tx, ty = best_t
    dx = 0 if tx == sx else (1 if tx > sx else -1)
    dy = 0 if ty == sy else (1 if ty > sy else -1)

    # Evaluate immediate moves with obstacle avoidance and slight preference toward target
    moves = [(dx, dy), (dx, 0), (0, dy), (0, 0), (dx, -dy), (-dx, dy), (-dx, 0), (0, -dy), (-dx, -dy)]
    seen = set()
    for mx, my in moves:
        if (mx, my) in seen:
            continue
        seen.add((mx, my))
        nx, ny = sx + mx, sy + my
        if 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obstacles:
            # Greedy: minimize distance to target, then stay not-oscillating
            dtar = abs(nx - tx) + abs(ny - ty)
            key = (dtar, abs(mx) + abs(my), mx, my)
            if best_key is None or key < best_key:
                best_key = key
                best_move = (mx, my)

    return [best_move[0], best_move[1]] if "best_move" in locals() else [0, 0]
