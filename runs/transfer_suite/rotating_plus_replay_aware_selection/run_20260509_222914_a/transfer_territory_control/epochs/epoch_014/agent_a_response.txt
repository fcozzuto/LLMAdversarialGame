                targets.append((x, y))
    if not targets:
        targets = [(w // 2, h // 2)]

    tx, ty = min(targets, key=lambda t: abs(t[0] - sx) + abs(t[1] - sy))
    options = [(0, 0), (1, 0), (0, 1), (-1, 0), (0, -1)]

    best_val = None
    for dx, dy in options:
