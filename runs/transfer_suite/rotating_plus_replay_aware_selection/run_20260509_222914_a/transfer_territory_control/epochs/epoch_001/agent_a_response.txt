def choose_move(observation):
    w = observation["grid_width"]
    h = observation["grid_height"]
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    obstacles = set((p[0], p[1]) for p in observation["obstacles"])
    self_t = set((p[0], p[1]) for p in observation["self_territory"])
    opp_t = set((p[0], p[1]) for p in observation["opponent_territory"])
    unclaimed = set((p[0], p[1]) for p in observation["unclaimed_cells"])
    cur_self_count = observation.get("self_territory_count", len(self_t))
    cur_opp_count = observation.get("opponent_territory_count", len(opp_t))
    moves = [(dx, dy) for dx in (-1, 0, 1) for dy in (-1, 0, 1)]
    dirs = sorted(moves, key=lambda t: (t[0] != 0 or t[1] != 0, t[0], t[1]))
    def inside(x, y):
        return 0 <= x < w and 0 <= y < h
    def dist(a, b):
        return abs(a[0]-b[0]) + abs(a[1]-b[1])
    opp_cells = list(opp_t)
    un_cells = list(unclaimed)
    best_t = None
    if un_cells:
        best_t = min(un_cells, key=lambda p: dist((sx, sy), p))
    elif opp_cells:
        best_t = min(opp_cells, key=lambda p: dist((sx, sy), p))
    best_score = -10**9
    best_move = (0, 0)
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not inside(nx, ny) or (nx, ny) in obstacles:
            continue
        immediate = 0
        if (nx, ny) in unclaimed:
            immediate += 2
        if (nx, ny) in opp_t:
            immediate += 5
        if (nx, ny) in self_t:
            immediate -= 1
        # "Frontier" pressure: move toward nearest opponent territory and nearest unclaimed.
        if opp_cells:
            d_opp = min(dist((nx, ny), p) for p in opp_cells)
        else:
            d_opp = 999
        if un_cells:
            d_un = min(dist((nx, ny), p) for p in un_cells)
        else:
            d_un = 999
        # Blend: prioritize stealing when adjacent; otherwise expand to unclaimed.
        target_bias = (-d_opp if immediate >= 1 else -d_un)
        # Center preference to reduce cornering.
        cx, cy = (w - 1) / 2.0, (h - 1) / 2.0
        center_pref = -(abs(nx - cx) + abs(ny - cy)) * 0.02
        # Deterministic tie-breaker: prefer staying if scores equal and we have no gain.
        score = immediate * 10 + target_bias * 1.0 + center_pref
        if score > best_score or (score == best_score and (dx, dy) == (0, 0)):
            best_score = score
            best_move = (dx, dy)
    return [int(best_move[0]), int(best_move[1])]
