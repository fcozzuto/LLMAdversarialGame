def choose_move(observation):

    ox, oy = observation.get("opponent_position", [0, 0])
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    resources = observation.get("resources", []) or []
    obstacles = observation.get("obstacles", []) or []
    obs = set((int(x), int(y)) for x, y in obstacles)

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def cheb(x1, y1, x2, y2):
        dx = x1 - x2
        if dx < 0:
            dx = -dx
        dy = y1 - y2
        if dy < 0:
            dy = -dy
        return dx if dx > dy else dy

    def manh(x1, y1, x2, y2):
        dx = x1 - x2
        if dx < 0:
            dx = -dx
        dy = y1 - y2
        if dy < 0:
            dy = -dy
        return dx + dy

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    candidates = []
    for rx, ry in resources:
        rx = int(rx)
        ry = int(ry)
        if not inb(rx, ry) or (rx, ry) in obs:
            continue
        sd = cheb(sx, sy, rx, ry)
        od = cheb(ox, oy, rx, ry)
        # Prefer resources we can reach much earlier than opponent; break ties by closeness to us.
        lead = od - sd
        if lead < -2:
            continue
        near_benefit = 0
        # Small bonus for resources that are "open" around them (fewer obstacle neighbors).
        for nx in (rx - 1, rx, rx + 1):
            for ny in (ry - 1, ry, ry + 1):
                if (nx, ny) in obs:
                    near_benefit -= 1
        # Deterministic scaling to favor consistent selection.
        val = lead * 1000 + (20 - sd) * 10 + near_benefit - manh(rx, ry, 0, 0) * 0.01
        candidates.append((val, sd, od, rx, ry))
    candidates.sort(reverse=True)
    if not candidates:
        # No available resources: head to nearest corner (deterministic).
        corners = [(0, 0), (0, h - 1), (w - 1, 0), (w - 1, h - 1)]
        cx, cy = min(corners, key=lambda p: cheb(sx, sy, p[0], p[1]))
        dx = 0 if cx == sx else (1 if cx > sx else -1)
        dy = 0 if cy == sy else (1 if cy > sy else -1)
        if inb(sx + dx, sy + dy) and (sx + dx, sy + dy) not in obs:
            return [dx, dy]
