def choose_move(observation):
    sx, sy = map(int, observation.get("self_position", [0, 0]))
    ox, oy = map(int, observation.get("opponent_position", [0, 0]))
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    resources = observation.get("resources", []) or []
    obstacles = observation.get("obstacles", []) or []
    tr = int(observation.get("turns_remaining", 0))
    obs = set((int(x), int(y)) for x, y in obstacles)
    inb = lambda x, y: 0 <= x < w and 0 <= y < h

    res = []
    for x, y in resources:
        x, y = int(x), int(y)
        if inb(x, y) and (x, y) not in obs:
            res.append((x, y))
    if not res:
        return [0, 0]

    nbrs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def bfs_dist(start):
        inf = 10**9
        dist = [[inf] * h for _ in range(w)]
        x0, y0 = start
        if not inb(x0, y0) or (x0, y0) in obs:
            return dist
        dist[x0][y0] = 0
        qx, qy, qi = [x0], [y0], 0
        while qi < len(qx):
            x, y = qx[qi], qy[qi]
            qi += 1
            nd = dist[x][y] + 1
            for dx, dy in nbrs:
                nx, ny = x + dx, y + dy
                if inb(nx, ny) and (nx, ny) not in obs and nd < dist[nx][ny]:
                    dist[nx][ny] = nd
                    qx.append(nx)
                    qy.append(ny)
        return dist

    op_dist = bfs_dist((ox, oy))

    deltas = nbrs
    best = None
    for dx, dy in deltas:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obs:
            nx, ny = sx, sy
            dx, dy = 0, 0
        my_dist = bfs_dist((nx, ny))
        best_res = None
        for rx, ry in res:
            ds = my_dist[rx][ry]
            do = op_dist[rx][ry]
            if ds >= 10**9:
                continue
            if tr > 0 and ds > tr:
                continue
            # maximize advantage; tie-break toward earlier collection and farther denial
            score = (do - ds, -(ds), do, rx, ry)
            if best_res is None or score > best_res[0]:
                best_res = (score, (rx, ry))
        if best_res is None:
            cand = ((-10**9, 10**9, -10**9, 0, 0), (nx, ny))
            if best is None or cand[0] > best[0]:
                best = (dx, dy, cand[0])
        else:
            cand_score = best_res[0]
            if best is None or cand_score > best[0]:
                best = (dx, dy, cand_score)

    return [int(best[0]), int(best[1])]
