            x, y = int(p[0]), int(p[1])

                resources.append((x, y))

    dirs = [(-1, 0), (1, 0), (0, -1), (0, 1)]
    candidates = []
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if 0 <= nx < w and 0 <= ny < h and (nx, ny) not in blocked:
            candidates.append((dx, dy))
