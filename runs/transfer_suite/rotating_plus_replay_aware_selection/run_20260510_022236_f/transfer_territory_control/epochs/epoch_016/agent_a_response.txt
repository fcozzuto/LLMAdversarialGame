        if ok(x, y):
            targets.append((x, y))
        if len(targets) >= 20:
            break

    def man(a, b, c, d):
        return abs(a - c) + abs(b - d)

    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
