    sx, sy = observation.get("self_position") or (0, 0)
    ox, oy = observation.get("opponent_position") or (w - 1, h - 1)
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    def to_set(lst):
        s = set()
        if not lst:
            return s
        for p in lst:
            if isinstance(p, (list, tuple)) and len(p) >= 2:
                x, y = int(p[0]), int(p[1])
                if 0 <= x < w and 0 <= y < h:
                    s.add((x, y))
        return s

    selfT = to_set(observation.get("self_territory"))
    oppT = to_set(observation.get("opponent_territory"))
    unclaimed = to_set(observation.get("unclaimed_cells") or observation.get("unclaimed"))
    obstacles = to_set(observation.get("obstacles"))
    resources = to_set(observation.get("resources"))

    my_count = int(observation.get("self_territory_count") or 0)
    opp_count = int(observation.get("opponent_territory_count") or 0)
    aggressive = (my_count <= opp_count)

    # Directional bias: head towards opponent, but avoid moving away from resources if present.
    rtx = rty = None
    if resources:
        rtx, rty = min(resources, key=lambda p: (abs(sx - p[0]) + abs(sy - p[1]), p[0], p[1]))
    def dist(a, b):
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

    opp_dx, opp_dy = ox - sx, oy - sy
    pref_forward = (1 if opp_dx > 0 else -1 if opp_dx < 0 else 0, 1 if opp_dy > 0 else -1 if opp_dy < 0 else 0)

    cand = []
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            nx, ny = sx + dx, sy + dy
            if not (0 <= nx < w and 0 <= ny < h):

            cand.append((dx, dy, nx, ny))
    cand.sort(key=lambda t: (t[0], t[1]))  # deterministic tie-break

                score += 50 if aggressive else 80
            elif (nx, ny) in oppT:

            elif (nx, ny) in unclaimed:
                score += 120 if aggressive else 90
            else:
                score += 40  # edge/empty squares still controllable by entry

    except:
        sx, sy, ox, oy = 0, 0, w - 1, h - 1

    obstacles = observation.get("obstacles") or []
    res = []

        if isinstance(p, (list, tuple)) and len(p) >= 2:

            if 0 <= x < w and 0 <= y < h:
                obs.add((x, y))

    target = None

        target = min(res, key=lambda p: (abs(p[0] - sx) + abs(p[1] - sy), p[0], p[1]))
