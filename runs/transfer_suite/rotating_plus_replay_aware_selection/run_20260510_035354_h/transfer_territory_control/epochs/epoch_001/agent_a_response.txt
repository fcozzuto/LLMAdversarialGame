def choose_move(observation):
    tw = observation["turn_index"]
    w = observation["grid_width"]
    h = observation["grid_height"]
    self_pos = observation["self_position"]
    opp_pos = observation["opponent_position"]
    self_terr = observation.get("self_territory", [])
    opp_terr = observation.get("opponent_territory", [])
    unclaimed = observation.get("unclaimed_cells", [])
    obstacles = observation.get("obstacles", [])

    ox = set()
    for x, y in obstacles:
        ox.add((x, y))
    st = set()
    for x, y in self_terr:
        st.add((x, y))
    ot = set()
    for x, y in opp_terr:
        ot.add((x, y))
    uc = unclaimed
    uc_set = set((x, y) for x, y in uc)

    ax, ay = self_pos
    bx, by = opp_pos

    moves = [(-1,-1),(-1,0),(-1,1),(0,-1),(0,0),(0,1),(1,-1),(1,0),(1,1)]
    best = (0, 0)
    best_score = -10**9

    attack_mode = (tw % 4 == 2) or (tw % 4 == 3)

    # Pick a target: expand toward unclaimed unless in attack mode
    target = None
    if (not attack_mode) and uc:
        bestd = 10**9
        for x, y in uc:
            if (x, y) in ox:
                continue
            d = abs(x - ax) + abs(y - ay)
            if d < bestd:
                bestd = d
                target = (x, y)
    if target is None:
        # If no unclaimed or attack mode, aim near opponent territory / position
        if ot:
            # choose nearest opponent territory cell
            bestd = 10**9
            for x, y in ot:
                d = abs(x - ax) + abs(y - ay)
                if d < bestd:
                    bestd = d
                    target = (x, y)
        else:
            target = (bx, by)

    tx, ty = target

    for dx, dy in moves:
        nx, ny = ax + dx, ay + dy
        if nx < 0 or nx >= w or ny < 0 or ny >= h:
            continue
        if (nx, ny) in ox:
            continue

        in_self = (nx, ny) in st
        in_opp = (nx, ny) in ot
        in_un = (nx, ny) in uc_set
        # Greedy deterministic evaluation
        dtarget = abs(nx - tx) + abs(ny - ty)
        doopp = abs(nx - bx) + abs(ny - by)

        score = 0
        if in_opp:
            score += 30
        if in_un:
            score += 12
        if in_self:
            score += 3
        if (nx, ny) == (bx, by):
            score += 20
        if (nx, ny) in ot and attack_mode:
            score += 10
        # Distance shaping
        score += max(0, 20 - dtarget)
        # Mild preference to not drift too far from opponent when attacking
        if attack_mode:
            score += max(0, 12 - doopp)

        # Slight deterministic tie-break: favor moves with smaller dx, then dy
        score = score * 100 + (0 if dx == 0 else (1 if dx > 0 else 2)) + (0 if dy == 0 else (1 if dy > 0 else 2))

        if score > best_score:
            best_score = score
            best = (dx, dy)

    return [int(best[0]), int(best[1])]
