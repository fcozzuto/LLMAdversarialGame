def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    obs = set()
    for p in (observation.get("obstacles", []) or []):
        if p is not None and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obs.add((x, y))

    r = (observation.get("self_role", "") or "").lower()
    pursuer = ("purs" in r) or ("chaser" in r) or ("hunter" in r) or ("catcher" in r)

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    def inside(x, y): return 0 <= x < w and 0 <= y < h
    def dist2(x, y):
        dx, dy = x - ox, y - oy
        return dx * dx + dy * dy
    def obstacle_pen(x, y):
        if (x, y) in obs:
            return 1000000
        return 0

    best = None
    best_key = None

    # Tie-break encourages moving toward center if distance tie.
    cx, cy = (w - 1) / 2.0, (h - 1) / 2.0

    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not inside(nx, ny):
            continue
        if obstacle_pen(nx, ny):
            continue

        dcur = dist2(sx, sy)
        dnew = dist2(nx, ny)
        step_goal = (dcur - dnew) if pursuer else (dnew - dcur)  # improve for pursuer, for evader invert
        # Extra shaping: prefer not to "hug" walls if evading (wall-runner), but pursue should keep pressure.
        wall_dist = min(nx, ny, (w - 1) - nx, (h - 1) - ny)
        center_dist = (nx - cx) * (nx - cx) + (ny - cy) * (ny - cy)

        # If very close, pursuer prioritizes maximum progress; evader prioritizes maximum separation.
        closeness = dist2(nx, ny)
        tight_bonus = 1000 if closeness <= 4 else (200 if closeness <= 9 else 0)

        # Deterministic scoring; higher is better.
        score = 0
        score += 1000000 * (1 if tight_bonus else 0) + tight_bonus
        score += (step_goal * 10)
        score += (wall_dist if evading else -wall_dist)  # evader wants larger wall_dist, pursuer slightly avoids corners only
        score += (-center_dist * (1 if evader else -0.1))

        # Prefer staying still only if tied well.
        if dx == 0 and dy == 0:
            score -= (5 if pursuer else 2)

        # Secondary deterministic tie-break: lexicographic move order by index in moves list already fixed.
        key = (score, -abs(dx), -abs(dy), )  # stable-ish
        if best_key is None or key > best_key:
            best_key = key
            best = [dx, dy]

    if best is None:
        # Fallback: deterministic legal step toward/away without obstacles.
        dx, dy = moves[4]
        nx, ny = sx + dx, sy + dy
        if inside(nx, ny) and (nx, ny) not in obs:
            return [0, 0]
        for dx, dy in moves:
            nx, ny = sx + dx, sy + dy
            if inside(nx, ny) and (nx, ny) not in obs:
                return [dx, dy]
        return [0, 0]
    return best
