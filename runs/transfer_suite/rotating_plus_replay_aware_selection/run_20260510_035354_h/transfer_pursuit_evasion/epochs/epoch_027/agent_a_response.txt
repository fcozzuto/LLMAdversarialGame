def choose_move(observation):
    w = int(observation["grid_width"]); h = int(observation["grid_height"])
    sx, sy = observation["self_position"]; ox, oy = observation["opponent_position"]
    obstacles = observation.get("obstacles") or []
    obs = set((int(p[0]), int(p[1])) for p in obstacles)
    role = (observation.get("self_role") or "").lower()
    opp_role = (observation.get("opponent_role") or "").lower()
    i_pursuer = ("purs" in role) or (("evad" not in role) and ("evad" in opp_role))
    from collections import deque
    def bfs_dist(start, goal):
        if start == goal:
            return 0
        if not (0 <= start[0] < w and 0 <= start[1] < h) or start in obs:
            return 10**9
        q = deque([start])
        dist = {start: 0}
        while q:
            x, y = q.popleft()
            d = dist[(x, y)] + 1
            for dx in (-1, 0, 1):
                for dy in (-1, 0, 1):
                    if dx == 0 and dy == 0:
                        continue
                    nx, ny = x + dx, y + dy
                    if 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obs and (nx, ny) not in dist:
                        if (nx, ny) == goal:
                            return d
                        dist[(nx, ny)] = d
                        q.append((nx, ny))
        return 10**9

    best = None
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            nx, ny = sx + dx, sy + dy
            if not (0 <= nx < w and 0 <= ny < h) or (nx, ny) in obs:
                continue
            d = bfs_dist((nx, ny), (ox, oy))
            # Tie-breaker: prefer reducing actual geometric distance to avoid stalls.
            g = (nx - ox) * (nx - ox) + (ny - oy) * (ny - oy)
            key = (d, g) if i_pursuer else (-d, -g)
            if best is None or key < best[0] or (key == best[0] and (dx, dy) < best[1]):
                best = (key, (dx, dy))
    return [best[1][0], best[1][1]] if best is not None else [0, 0]

def choose_move(observation):
    w = int(observation["grid_width"])
    h = int(observation["grid_height"])
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]

    obstacles = observation.get("obstacles") or []
    obs = set()
    for p in obstacles:
        try:
            x, y = int(p[0]), int(p[1])
            obs.add((x, y))
        except Exception:
            pass

    role = (observation.get("self_role") or "").lower()
    opp_role = (observation.get("opponent_role") or "").lower()
    evade = ("evad" in role) or ("purs" not in role and "evad" in opp_role)

    resources = observation.get("resources") or []
    res = []
    for p in resources:
        try:
            x, y = int(p[0]), int(p[1])
            res.append((x, y))
        except Exception:
            pass

    best = None
    best_val = None

    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            nx, ny = sx + dx, sy + dy
            if not (0 <= nx < w and 0 <= ny < h):
                continue
            if (nx, ny) in obs:
                continue

            d_op = abs(nx - ox) + abs(ny - oy)
            if res:
                d_res = min(abs(nx - rx) + abs(ny - ry) for rx, ry in res)
            else:
                d_res = 0

            val = d_res - (d_op if evade else -d_op)
            if best_val is None or (val > best_val) or (val == best_val and (dx, dy) < best):
                best_val = val
                best = (dx, dy)

    if best is None:
        return [0, 0]
    return [int(best[0]), int(best[1])]
