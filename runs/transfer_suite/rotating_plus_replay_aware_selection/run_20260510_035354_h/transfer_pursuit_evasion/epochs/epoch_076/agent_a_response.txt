        if p is not None and len(p) >= 2:

            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

        return 0 <= x < w and 0 <= y < h

        return abs(ax - bx) + abs(ay - by)

    corners = [(0, 0), (0, h - 1), (w - 1, 0), (w - 1, h - 1)]

        sign = 1.0  # maximize closeness
