def choose_move(observation):
    w = observation["grid_width"]; h = observation["grid_height"]
    sx, sy = observation["self_position"]; ox, oy = observation["opponent_position"]
    obstacles = set()
    for p in observation.get("obstacles", []):
        if p and len(p) == 2:
            obstacles.add((p[0], p[1]))
    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    def valid(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obstacles
    def dist2(x, y, px, py):
        dx = x - px; dy = y - py
        return dx * dx + dy * dy
    role = observation.get("self_role", "pursuer")
    best = (0, 0)
    best_score = None
    if role == "pursuer":
        # Try to reduce opponent distance; anticipate evader response by looking 2-ply worst-case.
        for dx, dy in moves:
            nx, ny = sx + dx, sy + dy
            if not valid(nx, ny):
                continue
            worst_d = -1
            for odx, ody in moves:
                ex, ey = ox + odx, oy + ody
                if not valid(ex, ey):
                    continue
                d = dist2(nx, ny, ex, ey)
                if d > worst_d:
                    worst_d = d
            score = -worst_d
            tie = (-abs(dx) - abs(dy), -dist2(nx, ny, ox, oy))  # prefer larger move? invert via negative; prefer closer too
            if best_score is None or (score, tie) > (best_score, None):
                best_score = score
                best = (dx, dy)
            elif score == best_score:
                # deterministic tie-break: choose move with minimum current dist to opponent, then lexicographic
                cur_d = dist2(nx, ny, ox, oy)
                best_nx, best_ny = sx + best[0], sy + best[1]
                best_d = dist2(best_nx, best_ny, ox, oy)
                if cur_d < best_d or (cur_d == best_d and (dx, dy) < best):
                    best = (dx, dy)
    else:
        # Evader: maximize distance from pursuer; 2-ply assumes pursuer minimizes distance.
        for dx, dy in moves:
            nx, ny = sx + dx, sy + dy
            if not valid(nx, ny):
                continue
            best_p_d = 10**18
            for pdx, pdy in moves:
                px, py = ox + pdx, oy + pdy
                if not valid(px, py):
                    continue
                d = dist2(px, py, nx, ny)
                if d < best_p_d:
                    best_p_d = d
            score = best_p_d
            tie = (-abs(dx) - abs(dy), dist2(nx, ny, ox, oy))
            if best_score is None or (score, tie) > (best_score, None):
                best_score = score
                best = (dx, dy)
            elif score == best_score:
                cur_d = dist2(nx, ny, ox, oy)
                best_nx, best_ny = sx + best[0], sy + best[1]
                best_d = dist2(best_nx, best_ny, ox, oy)
                if cur_d > best_d or (cur_d == best_d and (dx, dy) < best):
                    best = (dx, dy)
    return [int(best[0]), int(best[1])]
