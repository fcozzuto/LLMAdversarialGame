def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position", [0, 0]) or [0, 0]
    try:
        sx, sy = int(sx), int(sy)
    except:
        sx, sy = 0, 0

    obstacles = set()
    for p in (observation.get("obstacles", []) or []):
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            try:
                obstacles.add((int(p[0]), int(p[1])))
            except:
                pass

    unclaimed = set()
    for c in (observation.get("unclaimed_cells", []) or []):
        if isinstance(c, (list, tuple)) and len(c) >= 2:
            try:
                unclaimed.add((int(c[0]), int(c[1])))
            except:
                pass

    self_territory = set()
    for c in (observation.get("self_territory", []) or []):
        if isinstance(c, (list, tuple)) and len(c) >= 2:
            try:
                self_territory.add((int(c[0]), int(c[1])))
            except:
                pass

    opp_territory = set()
    for c in (observation.get("opponent_territory", []) or []):
        if isinstance(c, (list, tuple)) and len(c) >= 2:
            try:
                opp_territory.add((int(c[0]), int(c[1])))
            except:
                pass

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    neigh8 = dirs[:]
    opp_pos = observation.get("opponent_position", [None, None]) or [None, None]
    try:
        ox, oy = int(opp_pos[0]), int(opp_pos[1])
    except:
        ox, oy = None, None

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def score_cell(x, y):
        if not inb(x, y) or (x, y) in obstacles:
            return -10**9
        if (x, y) in unclaimed:
            base = 1200
        elif (x, y) in opp_territory:
            base = 420  # flip on entry is enabled
        elif (x, y) in self_territory:
            base = 25
        else:
            base = 0

        # Frontier pressure: move toward squares adjacent to unclaimed (claiming next turn).
        frontier = 0
        for dx, dy in neigh8:
            nx, ny = x + dx, y + dy
            if (nx, ny) in unclaimed:
                frontier += 1
            if (nx, ny) in opp_territory:
                frontier += 0.5
        base += 55 * frontier

        # Avoid getting trapped: penalize low-mobility cells next to obstacles.
        mob_pen = 0
        for dx, dy in neigh8:
            nx, ny = x + dx, y + dy
            if not inb(nx, ny) or (nx, ny) in obstacles:
                mob_pen += 1
        base -= 3.0 * mob_pen

        # Subtle chase if no good unclaimed: prefer nearer opponent.
        if ox is not None and oy is not None:
            base += -0.8 * (abs(x - ox) + abs(y - oy)) * (1 if (x, y) not in unclaimed else 0)
        return base

    best = -10**18
    best_move = (0, 0)
    # Deterministic tie-break: prefer moves with smaller dx, then dy, then staying still last.
    for dx, dy in dirs:
        x, y = sx + dx, sy + dy
        sc = score_cell(x, y)
        if sc > best or (sc == best and (dx, dy) < best_move):
            best, best_move = sc, (dx, dy)

    return [int(best_move[0]), int(best_move[1])]

def choose_move(observation):
    w = int(observation.get("grid_width", 0) or 0)
    h = int(observation.get("grid_height", 0) or 0)
    if w <= 0:
        w = 8
    if h <= 0:
        h = 8

    sp = observation.get("self_position", [0, 0]) or [0, 0]
    op = observation.get("opponent_position", [0, 0]) or [0, 0]
    try:
        sx, sy = int(sp[0]), int(sp[1])
    except:
        sx, sy = 0, 0
    try:
        ox, oy = int(op[0]), int(op[1])
    except:
        ox, oy = 0, 0

    obs = set()
    for p in (observation.get("obstacles", []) or []):
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            try:
                obs.add((int(p[0]), int(p[1])))
            except:
                pass

    resources = []
    for r in (observation.get("resources", []) or []):
        if isinstance(r, (list, tuple)) and len(r) >= 2:
            try:
                resources.append((int(r[0]), int(r[1])))
            except:
                pass

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def dist(a, b):
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    target = (ox, oy)
    if resources:
        best = None
        bd = None
        for t in resources:
            if not inb(t[0], t[1]) or t in obs:
                continue
            d = dist((sx, sy), t)
            if bd is None or d < bd:
                bd = d
                best = t
        if best is not None:
            target = best

    best_move = (0, 0)
    best_score = None
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny):
            continue
        if (nx, ny) in obs:
            continue
        score = dist((nx, ny), target)
        if best_score is None or score < best_score:
            best_score = score
            best_move = (dx, dy)
        elif score == best_score:
            if (dx, dy) < best_move:
                best_move = (dx, dy)

    return [int(best_move[0]), int(best_move[1])]
