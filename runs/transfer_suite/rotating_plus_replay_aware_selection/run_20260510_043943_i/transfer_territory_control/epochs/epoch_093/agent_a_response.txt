        cx, cy = ox, oy
        # Prefer cells that are closer to us; tie-break to reduce distance to opponent's cell (more likely contested).
        targets.sort(key=lambda t: (abs(t[0] - sx) + abs(t[1] - sy), abs(t[0] - cx) + abs(t[1] - cy), t[0], t[1]))
