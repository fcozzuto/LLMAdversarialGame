def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position", [0, 0]) or [0, 0]
    sx, sy = int(sx), int(sy)

    obstacles = set()
    for p in observation.get("obstacles", []) or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            try:
                obstacles.add((int(p[0]), int(p[1])))
            except:
                pass

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    our = set()
    for c in observation.get("self_territory", []) or []:
        if isinstance(c, (list, tuple)) and len(c) >= 2:
            try:
                our.add((int(c[0]), int(c[1])))
            except:
                pass

    opp = set()
    for c in observation.get("opponent_territory", []) or []:
        if isinstance(c, (list, tuple)) and len(c) >= 2:
            try:
                opp.add((int(c[0]), int(c[1])))
            except:
                pass

    unclaimed = set()
    for c in observation.get("unclaimed_cells", []) or []:
        if isinstance(c, (list, tuple)) and len(c) >= 2:
            try:
                unclaimed.add((int(c[0]), int(c[1])))
            except:
                pass

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    ox, oy = (observation.get("opponent_position", [None, None]) or [None, None])
    try:
        ox, oy = int(ox), int(oy)
    except:
        ox, oy = None, None

    my_adj_unclaimed = set()
    if our:
        for (x, y) in our:
            for dx, dy in dirs:
                nx, ny = x + dx, y + dy
                if inb(nx, ny) and (nx, ny) in unclaimed:
                    my_adj_unclaimed.add((nx, ny))

    def neighbor_count(x, y, S):
        cnt = 0
        for dx, dy in dirs:
            if (x + dx, y + dy) in S:
                cnt += 1
        return cnt

    if my_adj_unclaimed:
        targets = list(my_adj_unclaimed)
    else:
        targets = list(unclaimed) if unclaimed else []

    def score_cell(x, y):
        if not inb(x, y) or (x, y) in obstacles:
            return -10**9
        s = 0
        if (x, y) in unclaimed:
            s += 1200
            if our and (x, y) in my_adj_unclaimed:
                s += 300 + 20 * neighbor_count(x, y, our)
            s += 2 * neighbor_count(x, y, opp) * (-1)  # avoid feeding opponent by proximity
            if ox is not None and oy is not None:
                s += -0.5 * (abs(x - ox) + abs(y - oy))
        elif (x, y) in opp:
            # flipping enabled; prefer stealing when no unclaimed
            s += 600 + 10 * neighbor_count(x, y, opp) * (-1) + 5 * neighbor_count(x, y, our)
            if ox is not None and oy is not None:
                s += -0.2 * (abs(x - ox) + abs(y - oy))
        else:
            # staying on our territory is fine; otherwise minor penalty
            s += 50 if (x, y) in our else 0
            s += 5 * neighbor_count(x, y, our)
        return s

    best = (score_cell(sx, sy), 0, 0)
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        sc = score_cell(nx, ny)
        if sc > best[0]:
            best = (sc, dx, dy)
    if best[0] < -10**8:
        return [0, 0]
    return [int(best[1]), int(best[2])]
