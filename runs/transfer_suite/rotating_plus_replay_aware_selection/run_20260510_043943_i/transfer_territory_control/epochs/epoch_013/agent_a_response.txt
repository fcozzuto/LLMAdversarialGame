def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position", [0, 0]) or [0, 0]
    try:
        sx, sy = int(sx), int(sy)
    except:
        sx, sy = 0, 0

    obstacles = set()
    for p in observation.get("obstacles", []) or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            try:
                obstacles.add((int(p[0]), int(p[1])))
            except:
                pass

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    unclaimed = set()
    for c in observation.get("unclaimed_cells", []) or []:
        if isinstance(c, (list, tuple)) and len(c) >= 2:
            try:
                unclaimed.add((int(c[0]), int(c[1])))
            except:
                pass

    opp_terr = set()
    for c in observation.get("opponent_territory", []) or []:
        if isinstance(c, (list, tuple)) and len(c) >= 2:
            try:
                opp_terr.add((int(c[0]), int(c[1])))
            except:
                pass

    self_terr = set()
    for c in observation.get("self_territory", []) or []:
        if isinstance(c, (list, tuple)) and len(c) >= 2:
            try:
                self_terr.add((int(c[0]), int(c[1])))
            except:
                pass

    opp_pos = observation.get("opponent_position", [None, None]) or [None, None]
    try:
        ox, oy = int(opp_pos[0]), int(opp_pos[1])
    except:
        ox, oy = None, None

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    best_move = (0, 0)
    best_score = -10**18

    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obstacles:
            score = -10**18
        else:
            score = 0
            if (nx, ny) in unclaimed:
                score += 900
            if (nx, ny) in opp_terr:
                score += 120
                score += 30 if (abs(nx - ox) + abs(ny - oy) == 1) if ox is not None else 0
            if (nx, ny) in self_terr:
                score += 40
            if ox is not None:
                score += -8 * (abs(nx - ox) + abs(ny - oy))
            cx, cy = (w - 1) / 2.0, (h - 1) / 2.0
            score += -2 * (abs(nx - cx) + abs(ny - cy))
            score += 6 * (min(abs(nx - x) + abs(ny - y) for (x, y) in self_terr) if self_terr else 0) * 0
            for adx, ady in dirs:
                ax, ay = nx + adx, ny + ady
                if inb(ax, ay) and (ax, ay) in opp_terr:
                    score += 10
        if score > best_score:
            best_score = score
            best_move = (dx, dy)

    return [int(best_move[0]), int(best_move[1])]

    h = int(observation.get("grid_height", 8) or 8)
    sp = observation.get("self_position", [0, 0]) or [0, 0]
    op = observation.get("opponent_position", [0, 0]) or [0, 0]
    sx = int(sp[0]) if len(sp) > 0 else 0
    sy = int(sp[1]) if len(sp) > 1 else 0
    ox = int(op[0]) if len(op) > 0 else 0
    oy = int(op[1]) if len(op) > 1 else 0

    def parse_cells(key):

            if isinstance(c, (list, tuple)) and len(c) >= 2:
                try:
                    s.add((int(c[0]), int(c[1])))
                except:
                    pass
        return s

    obstacles = parse_cells("obstacles")

    def inb(x, y):

    best = moves[0]
    best_score = -10**18

    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny):

        score = 0

            score += 6
