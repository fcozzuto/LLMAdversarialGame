            targets.append((int(p[0]), int(p[1])))
    if not targets:
        targets = [(w // 2, h // 2)]
    else:
        # Prefer nearby targets deterministically
        targets.sort(key=lambda t: abs(t[0] - sx) + abs(t[1] - sy))
