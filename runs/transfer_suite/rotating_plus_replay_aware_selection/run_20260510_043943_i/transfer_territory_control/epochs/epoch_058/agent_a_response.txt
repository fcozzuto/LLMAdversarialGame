    sx, sy = int(sp[0]), int(sp[1])
    ox, oy = int(op[0]), int(op[1])

        if p and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obstacles

    for p in observation.get("self_territory", []) or []:

            x, y = int(p[0]), int(p[1])

                self_terr.add((x, y))
