def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = map(int, observation.get("self_position") or (0, 0))

    obstacles = set()
    for p in observation.get("obstacles") or []:
        if p and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

    def ok(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obstacles

    self_terr = observation.get("self_territory") or []
    opp_terr = observation.get("opponent_territory") or []
    resources = [tuple(map(int, p[:2])) for p in (observation.get("resources") or []) if p and len(p) >= 2 and ok(int(p[0]), int(p[1]))]
    unclaimed = [tuple(map(int, p[:2])) for p in (observation.get("unclaimed_cells") or []) if p and len(p) >= 2 and ok(int(p[0]), int(p[1]))]
    opp_set = set((int(x), int(y)) for x, y in opp_terr if ok(int(x), int(y)))

    def d2(ax, ay, bx, by):
        dx, dy = ax - bx, ay - by
        return dx * dx + dy * dy

    targets = []
    if resources:
        targets = resources
    elif unclaimed:
        bases = [(int(p[0]), int(p[1])) for p in self_terr if p and len(p) >= 2 and ok(int(p[0]), int(p[1]))]
        if not bases:
            bases = [(sx, sy)]
        for tx, ty in unclaimed:
            best_base = None
            for bx, by in bases:
                dd = d2(bx, by, tx, ty)
                if best_base is None or dd < best_base:
                    best_base = dd
            min_opp = None
            if opp_set:
                # Approx: distance to nearest opponent cell if listed; otherwise ignore
                for ox, oy in opp_set:
                    dd = d2(ox, oy, tx, ty)
                    if min_opp is None or dd < min_opp:
                        min_opp = dd
            # Prefer expansion outward from our territory, and slight preference away from immediate opponent reach
            score = best_base + (0 if min_opp is None else (-(min_opp * 0.03)))
            targets.append((score, tx, ty))

        targets.sort(key=lambda t: (t[0], t[1], t[2]))
        tx, ty = targets[0][1], targets[0][2]
    else:
        # Fallback: drift toward center
        tx, ty = w // 2, h // 2

    # Greedy one-step toward (tx, ty) while staying valid
    best_move = (0, 0)
    best_dist = None
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            nx, ny = sx + dx, sy + dy
            if not ok(nx, ny):
                continue
            dist = d2(nx, ny, tx, ty)
            if best_dist is None or dist < best_dist or (dist == best_dist and (dx, dy) < best_move):
                best_dist = dist
                best_move = (dx, dy)
    return [int(best_move[0]), int(best_move[1])]

    moves = [(1, 0), (-1, 0), (0, 1), (0, -1)]
