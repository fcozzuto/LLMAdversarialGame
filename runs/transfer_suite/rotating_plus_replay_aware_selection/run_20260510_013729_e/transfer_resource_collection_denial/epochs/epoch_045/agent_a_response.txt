def choose_move(observation):
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    w, h = observation["grid_width"], observation["grid_height"]
    resources = observation.get("resources") or []
    obstacles = observation.get("obstacles") or []
    if not resources:
        return [0, 0]

    obst = {(x, y) for x, y in obstacles}

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def cheb(x1, y1, x2, y2):
        dx = x2 - x1
        if dx < 0:
            dx = -dx
        dy = y2 - y1
        if dy < 0:
            dy = -dy
        return dx if dx > dy else dy

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    cur_moves = []
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny):
            continue
        if (nx, ny) in obst:
            continue
        cur_moves.append((nx, ny, dx, dy))
    if not cur_moves:
        return [0, 0]

    # Pick opponent's current best target for "denial" pressure.
    best_opp_t = None
    best_opp_d = None
    for rx, ry in resources:
        d = cheb(ox, oy, rx, ry)
        if best_opp_d is None or d < best_opp_d or (d == best_opp_d and (rx, ry) < best_opp_t):
            best_opp_d = d
            best_opp_t = (rx, ry)
    opp_rx, opp_ry = best_opp_t

    best = None
    for nx, ny, dx, dy in cur_moves:
        # Evaluate against all resources; prioritize captures and lead.
        total = -10**9
        for rx, ry in resources:
            ds = cheb(nx, ny, rx, ry)
            do = cheb(ox, oy, rx, ry)
            capture = 12 if (nx, ny) == (rx, ry) else 0
            lead = do - ds  # positive means we are closer than opponent
            denom = cheb(nx, ny, opp_rx, opp_ry) - best_opp_d  # closer to their target is good
            score = capture + 2.0 * lead + 0.6 * (-denom)
            # Deterministic tie-break: earlier resource in sorted order
            tr = rx * 8 + ry
            key = (score, -ds, -lead, -tr)
            if key > total:
                total = key

        # Also slightly avoid moving away from the set of resources overall
        avg_d = 0
        for rx, ry in resources:
            avg_d += cheb(nx, ny, rx, ry)
        avg_d = avg_d / len(resources)

        final_key = (total[0], -total[1], total[2], -avg_d, dx, dy)
        if best is None or final_key > best[0]:
            best = (final_key, [dx, dy])

    return best[1] if best else [0, 0]
