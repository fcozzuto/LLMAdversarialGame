def choose_move(observation):
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    w, h = observation["grid_width"], observation["grid_height"]
    resources = observation.get("resources") or []
    obstacles = observation.get("obstacles") or []
    if not resources:
        return [0, 0]

    obst = {(x, y) for x, y in obstacles}
    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def man(ax, ay, bx, by):
        dx = bx - ax
        if dx < 0:
            dx = -dx
        dy = by - ay
        if dy < 0:
            dy = -dy
        return dx + dy

    def score_move(nx, ny):
        if (nx, ny) in obst:
            return -10**9
        # Choose best target to race for
        best_adv = -10**30
        best_res = None
        best_sd = 10**9
        best_od = 10**9
        worst_opp_adv = -10**30  # for denial: reduce opponent's best lead
        for rx, ry in resources:
            sd = man(nx, ny, rx, ry)
            od = man(ox, oy, rx, ry)
            adv = od - sd  # positive: we are closer
            if adv > best_adv or (adv == best_adv and sd < best_sd):
                best_adv = adv
                best_res = (rx, ry)
                best_sd, best_od = sd, od
            opp_lead = sd - od  # positive means opponent is closer
            if opp_lead > worst_opp_adv:
                worst_opp_adv = opp_lead
        # Decision heuristic:
        # - Primary: maximize our racing advantage to best target
        # - Secondary: if we are already behind, prioritize reducing opponent's lead
        # - Tertiary: prefer closer to our chosen target (lower sd) and slightly stay safe
        if observation["turns_remaining"] <= 6:
            # Endgame: go for nearest resource
            chosen_sd = best_sd
            return -chosen_sd - (0 if (nx, ny) in obst else 0)
        behind = 1 if best_adv < 0 else 0
        # Prefer moves that increase best_adv and shrink distance to best_res
        bx, by = best_res if best_res else (sx, sy)
        dist_to_target = man(nx, ny, bx, by)
        return (best_adv * 10) + (-dist_to_target) + (behind * (-worst_opp_adv))

    best = None
    bestv = -10**30
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny):
            continue
        v = score_move(nx, ny)
        if v > bestv or (v == bestv and (dx == 0 and dy == 0) and best is not None):
            bestv = v
            best = (dx, dy)

    if best is None:
        return [0, 0]
    return [best[0], best[1]]
