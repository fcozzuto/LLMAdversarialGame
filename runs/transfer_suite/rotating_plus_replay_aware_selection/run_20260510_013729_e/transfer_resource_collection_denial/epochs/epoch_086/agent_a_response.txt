def choose_move(observation):
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    w, h = observation["grid_width"], observation["grid_height"]
    resources = observation.get("resources") or []
    obstacles = observation.get("obstacles") or []
    if not resources:
        return [0, 0]

    obst = {(x, y) for x, y in obstacles}
    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def legal(x, y):
        return inb(x, y) and (x, y) not in obst

    def man(ax, ay, bx, by):
        dx = ax - bx
        if dx < 0:
            dx = -dx
        dy = ay - by
        if dy < 0:
            dy = -dy
        return dx + dy

    def step_toward(ax, ay, tx, ty):
        bestx, besty, bestd = ax, ay, 10**9
        for mdx, mdy in moves:
            nx, ny = ax + mdx, ay + mdy
            if not legal(nx, ny):
                continue
            d = man(nx, ny, tx, ty)
            if d < bestd:
                bestd = d
                bestx, besty = nx, ny
        return bestx, besty

    def rollout_steps(ax, ay, tx, ty, k):
        x, y = ax, ay
        if (x, y) == (tx, ty):
            return 0
        for t in range(1, k + 1):
            nx, ny = step_toward(x, y, tx, ty)
            x, y = nx, ny
            if (x, y) == (tx, ty):
                return t
        return k + 1

    # Predict opponent target as their nearest resource (Manhattan) and their greedy travel to it
    opp_target = resources[0]
    bestm = man(ox, oy, opp_target[0], opp_target[1])
    for r in resources[1:]:
        m = man(ox, oy, r[0], r[1])
        if m < bestm:
            bestm = m
            opp_target = r

    k = 6
    opp_time = rollout_steps(ox, oy, opp_target[0], opp_target[1], k)

    best_res = None
    best_key = None
    for rx, ry in resources:
        if (rx, ry) == tuple(opp_target):
            # Still allow; scoring will handle collision-like scenarios
            pass
        st = rollout_steps(sx, sy, rx, ry, k)
        # Prefer taking a resource we can reach strictly earlier than opponent's predicted collection,
        # but if not, delay them by choosing a resource whose collection is much later for us.
        # Key: (we_score_now, tie_dist) higher is better for we_score_now.
        # we_score_now positive means advantage, negative means behind.
        adv = opp_time - st
        key = (adv, -st, man(sx, sy, rx, ry))
        if best_key is None or key > best_key:
            best_key = key
            best_res = (rx, ry)

    tx, ty = best_res
    nx, ny = step_toward(sx, sy, tx, ty)
    dx = nx - sx
    dy = ny - sy
    if dx < -1:
        dx = -1
    if dx > 1:
        dx = 1
    if dy < -1:
        dy = -1
    if dy > 1:
        dy = 1
    return [int(dx), int(dy)]
