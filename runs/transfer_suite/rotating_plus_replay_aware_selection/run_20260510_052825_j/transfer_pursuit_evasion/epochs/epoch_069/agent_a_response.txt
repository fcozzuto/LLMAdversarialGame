def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)
    self_role = str(observation.get("self_role", "")).lower()
    obstacles = observation.get("obstacles") or []
    obs = set()
    for p in obstacles:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obs.add((x, y))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obs

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    corners = [(0, 0), (0, h - 1), (w - 1, 0), (w - 1, h - 1)]

    def free_degree(x, y):
        c = 0
        for dx, dy in moves:
            nx, ny = x + dx, y + dy
            if inb(nx, ny):
                c += 1
        return c

    def dist2(x1, y1, x2, y2):
        dx, dy = x1 - x2, y1 - y2
        return dx * dx + dy * dy

    pursuer = ("purs" in self_role) or ("pursuer" in self_role)
    best = (10**18, 0, 0)

    if pursuer:
        # Chase opponent; prefer safer positions and avoid being boxed near corners.
        for dx, dy in moves:
            nx, ny = sx + dx, sy + dy
            if not inb(nx, ny):
                continue
            d_opp = dist2(nx, ny, ox, oy)
            deg = free_degree(nx, ny)
            # Penalize moving into extreme corner-adjacency when it reduces mobility.
            nearest_corner = min(corners, key=lambda c: dist2(nx, ny, c[0], c[1]))
            corner_pen = dist2(nx, ny, nearest_corner)
            corner_pen = 0 if corner_pen > 1 else 200  # discourage hugging corners too tightly
            # Also slightly prefer reducing distance to the closest corner that matches opponent's side.
            side_corner = min(corners, key=lambda c: dist2(ox, oy, c[0], c[1]))
            side_pref = dist2(nx, ny, side_corner)
            score = d_opp - 0.8 * deg + 0.01 * side_pref + corner_pen
            if score < best[0]:
                best = (score, dx, dy)
    else:
        # Evader: increase distance from pursuer; head toward the farthest corner that is not blocked.
        for dx, dy in moves:
            nx, ny = sx + dx, sy + dy
            if not inb(nx, ny):
                continue
            d_opp = dist2(nx, ny, ox, oy)
            deg = free_degree(nx, ny)
            # Prefer corners that are far from opponent, but keep mobility.
            far_corner = max(corners, key=lambda c: dist2(c[0], c[1], ox, oy))
            corner_go = dist2(nx, ny, far_corner)
            # Avoid moving into tight areas (low free degree).
            tight_pen = 30 if deg <= 2 else 0
            score = -d_opp - 0.5 * deg + 0.01 * corner_go + tight_pen
            if score < best[0]:
                best = (score, dx, dy)

    _, bx, by = best
    return [int(bx), int(by)]
