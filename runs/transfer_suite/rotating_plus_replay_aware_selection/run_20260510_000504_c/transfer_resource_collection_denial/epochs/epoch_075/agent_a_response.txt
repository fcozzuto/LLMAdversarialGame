    w = int(observation.get("grid_width") or 8)
    h = int(observation.get("grid_height") or 8)
    resources = observation.get("resources", []) or []
    obstacles_raw = observation.get("obstacles", []) or []

    obstacles = set()
    for p in obstacles_raw:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            obstacles.add((int(p[0]), int(p[1])))
        elif isinstance(p, dict):
            q = p.get("position", None)
            if isinstance(q, (list, tuple)) and len(q) >= 2:
                obstacles.add((int(q[0]), int(q[1])))
            elif "x" in p and "y" in p:
                obstacles.add((int(p["x"]), int(p["y"])))

    res = []
    for r in resources:
        if isinstance(r, (list, tuple)) and len(r) >= 2:
            res.append((int(r[0]), int(r[1])))

                res.append((int(q[0]), int(q[1])))
            elif "x" in r and "y" in r:
                res.append((int(r["x"]), int(r["y"])))

    def cd(x1, y1, x2, y2):
        dx = abs(x1 - x2)
        dy = abs(y1 - y2)
        return dx if dx > dy else dy  # Chebyshev

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def step_toward(tx, ty):

        for dx in (-1, 0, 1):

                if not inb(nx, ny) or (nx, ny) in obstacles:
                    continue
                d = cd(nx, ny, tx, ty)
                # Prefer closer, then closer to corners away from opponent bias
                corner_bias = (nx + ny) - (abs(nx - (w - 1)) + abs(ny - (h - 1)))
                score = (-d, corner_bias)
                if best is None or score > best[0]:
                    best = (score, [dx, dy])
