    h = int(observation["grid_height"])
    sx, sy = observation["self_position"]
    obstacles = set(map(tuple, observation.get("obstacles") or []))
    unclaimed = set(map(tuple, observation.get("unclaimed_cells") or []))
    self_terr = set(map(tuple, observation.get("self_territory") or []))
    opp_terr = set(map(tuple, observation.get("opponent_territory") or []))
    resources = observation.get("resources") or []

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    cx, cy = (w - 1) / 2.0, (h - 1) / 2.0

    def clamp_step(a, b):
        if a == b:
            return 0
        return 1 if b > a else -1

    def mdist(x1, y1, x2, y2):
        d = x1 - x2
        if d < 0:
            d = -d
        e = y1 - y2
        if e < 0:
            e = -e
        return d + e

    # Pick target deterministically: resources -> frontier unclaimed -> nearest unclaimed -> nearest opponent territory
    target = None
    if resources:
        target = tuple(resources[0])
    else:
        frontier = []
        for (x, y) in self_terr:
            for dx in (-1, 0, 1):
                for dy in (-1, 0, 1):
                    if dx == 0 and dy == 0:
                        continue
                    nx, ny = x + dx, y + dy
                    if inb(nx, ny) and (nx, ny) in unclaimed and (nx, ny) not in obstacles:
                        frontier.append((nx, ny))
        if frontier:
            bx, by = sx, sy
            # prioritize closeness to our position, then closeness to center
            target = min(frontier, key=lambda p: (mdist(p[0], p[1], bx, by), abs(p[0] - cx) + abs(p[1] - cy)))
        else:
            if unclaimed:
                target = min(unclaimed, key=lambda p: (mdist(p[0], p[1], sx, sy), abs(p[0] - cx) + abs(p[1] - cy)))
            elif opp_terr:
                target = min(opp_terr, key=lambda p: (mdist(p[0], p[1], sx, sy), -(p[0] - cx) * (p[0] - cx) - (p[1] - cy) * (p[1] - cy)))
