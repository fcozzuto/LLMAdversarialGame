def choose_move(observation):
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    sx, sy = observation.get("self_position", [0, 0])
    obstacles = set(map(tuple, observation.get("obstacles") or []))
    unclaimed = list(map(tuple, observation.get("unclaimed_cells") or []))
    resources = list(map(tuple, observation.get("resources") or []))
    opp_terr = set(map(tuple, observation.get("opponent_territory") or []))
    self_terr = set(map(tuple, observation.get("self_territory") or []))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def manhattan(x1, y1, x2, y2):
        dx = x1 - x2
        if dx < 0:
            dx = -dx
        dy = y1 - y2
        if dy < 0:
            dy = -dy
        return dx + dy

    # Pick target deterministically; prioritize immediate conquest/territory growth over long chases.
    if resources:
        targets = resources
    else:
        targets = unclaimed if unclaimed else (list(opp_terr) if opp_terr else [(w - 1, h - 1)])

    # Prefer targets that are either unclaimed, or adjacent to our territory (easy expansion).
    def score_target(t):
        tx, ty = t
        d = manhattan(sx, sy, tx, ty)
        adj_self = 0
        for ax in (-1, 0, 1):
            for ay in (-1, 0, 1):
                if ax == 0 and ay == 0:
                    continue
                if (tx + ax, ty + ay) in self_terr:
                    adj_self = 1
                    break
            if adj_self:
                break
        adj_opp = 0
        for ax in (-1, 0, 1):
            for ay in (-1, 0, 1):
                if ax == 0 and ay == 0:
                    continue
                if (tx + ax, ty + ay) in opp_terr:
                    adj_opp = 1
                    break
            if adj_opp:
                break
        return (d, -adj_self, -adj_opp, ty, tx)

    target = min(targets, key=score_target)
    tx, ty = target

    # Choose best neighboring move that reduces distance to target.
    moves = []
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            if dx == 0 and dy == 0:
                continue
            nx, ny = sx + dx, sy + dy
            if not inb(nx, ny) or (nx, ny) in obstacles:
                continue
            d = manhattan(nx, ny, tx, ty)
            flip_bias = -1 if (nx, ny) in opp_terr else 0  # deterministic incentive to attack.
            own_bias = -1 if (nx, ny) in self_terr else 0  # prefer holding/expanding near us.
            next_unclaimed = 1 if (nx, ny) in unclaimed else 0
            moves.append((d + flip_bias + own_bias - next_unclaimed * 0.1, dx, dy))

    if not moves:
        # Fallback: deterministic valid step toward board center if possible, else stay.
        center = (w // 2, h // 2)
        best = (manhattan(sx, sy, center[0], center[1]), 0, 0)
        for dx in (-1, 0, 1):
            for dy in (-1, 0, 1):
                nx, ny = sx + dx, sy + dy
                if not inb(nx, ny) or (nx, ny) in obstacles:
                    continue
                d = manhattan(nx, ny, center[0], center[1])
                cand = (d, dx, dy)
                if cand < best:
                    best = cand
