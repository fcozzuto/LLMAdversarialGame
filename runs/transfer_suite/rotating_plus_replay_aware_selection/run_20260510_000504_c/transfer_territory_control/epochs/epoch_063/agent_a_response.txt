    self_terr = set(map(tuple, observation.get("self_territory") or []))
    opp_terr = set(map(tuple, observation.get("opponent_territory") or []))
    resources = observation.get("resources") or []
    if resources:
        tx, ty = resources[0]
        best_t = (tx, ty)
    else:
        # Prefer unclaimed cells adjacent to our territory; else any unclaimed; else fall back.
        frontier = []
        for (x, y) in self_terr:
            for dx in (-1, 0, 1):
                for dy in (-1, 0, 1):
                    if dx == 0 and dy == 0:
                        continue
                    nx, ny = x + dx, y + dy
                    if 0 <= nx < w and 0 <= ny < h and (nx, ny) in unclaimed and (nx, ny) not in obstacles:
                        frontier.append((nx, ny))
        targets = frontier if frontier else list(unclaimed) if unclaimed else list(opp_terr) if opp_terr else [(sx, sy)]
        cx, cy = (w - 1) / 2.0, (h - 1) / 2.0

        def md(a, b, c, d):
            dx = a - c
            if dx < 0:
                dx = -dx
            dy = b - d
            if dy < 0:
                dy = -dy
            return dx + dy

        # Deterministic tie-breaker: closest to frontier goal, then closer to our position, then towards center.
        best_t = None
        best_k = None
        for t in targets:
            tx, ty = t
            k = (md(tx, ty, cx, cy), md(tx, ty, sx, sy), tx, ty)
            if best_k is None or k < best_k:
                best_k = k
                best_t = t

    tx, ty = best_t

    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            nx, ny = sx + dx, sy + dy

            dx = -dx

        if dy < 0:
            dy = -dy
        return dx + dy

    best = None
    best_score = None
    for dx, dy, nx, ny in moves:
        on_unclaimed = (nx, ny) in unclaimed
        on_own = (nx, ny) in self_terr
        on_opp = (nx, ny) in opp_terr
        base = 0
        if on_unclaimed:
            base += 100
        elif on_opp:
            base += 40
        elif on_own:
            base += 10
        else:
            base += 0
        d_to_target = md(nx, ny, tx, ty)
        # If opponent is adjacent, slightly prefer moving onto them to flip.
        opp_adj = 0
        for ax in (-1, 0, 1):
            for ay in (-1, 0, 1):
                if ax == 0 and ay == 0:
                    continue

                if 0 <= ax2 < w and 0 <= ay2 < h and (ax2, ay2) in opp_terr:
                    opp_adj = 1
        score = (-base, d_to_target, -opp_adj, dx, dy)
        if best_score is None or score < best_score:
            best_score = score
            best = (dx, dy)
    return [int(best[0]), int(best[1])]
