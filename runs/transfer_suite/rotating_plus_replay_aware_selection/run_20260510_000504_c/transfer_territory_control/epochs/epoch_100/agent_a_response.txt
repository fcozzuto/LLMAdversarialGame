def choose_move(observation):
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    sx, sy = observation.get("self_position", (0, 0))
    ox, oy = observation.get("opponent_position", (0, 0))
    obstacles = set(map(tuple, observation.get("obstacles") or []))
    unclaimed = set(map(tuple, observation.get("unclaimed_cells") or []))
    self_terr = set(map(tuple, observation.get("self_territory") or []))
    opp_terr = set(map(tuple, observation.get("opponent_territory") or []))
    scores = observation.get("scores") or {}
    self_name = observation.get("self_name", "agent_a")
    opp_name = observation.get("opponent_name", "agent_b")
    my_score = float(scores.get(self_name, 0.0) or 0.0)
    op_score = float(scores.get(opp_name, 0.0) or 0.0)

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    inb = lambda x, y: 0 <= x < w and 0 <= y < h

    def legal(nx, ny):
        return inb(nx, ny) and (nx, ny) not in obstacles

    def manh(x1, y1, x2, y2):
        return abs(x1 - x2) + abs(y1 - y2)

    targets = unclaimed if unclaimed else [(w - 1, h - 1)]
    closest_un = None
    closest_d = 10**9
    for tx, ty in targets:
        d = manh(sx, sy, tx, ty)
        if d < closest_d:
            closest_d = d
            closest_un = (tx, ty)
    if closest_un is None:
        closest_un = (w - 1, h - 1)

    aggressive = my_score + 1.0 < op_score  # if behind, push contests

    best = None
    best_val = -10**18
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not legal(nx, ny):
            continue
        val = 0.0

        if (nx, ny) in unclaimed:
            val += 6.0
        if (nx, ny) in self_terr:
            val += 1.5
        if (nx, ny) in opp_terr:
            val += 5.0 if aggressive else 3.0

        # Expand toward nearest unclaimed cell
        dn = manh(nx, ny, closest_un[0], closest_un[1])
        val += (closest_d - dn) * 1.2

        # Prefer moving generally away from obstacles / not being trapped
        open_neighbors = 0
        for ddx, ddy in dirs:
            ax, ay = nx + ddx, ny + ddy
            if legal(ax, ay):
                open_neighbors += 1
        val += open_neighbors * 0.15

        # Soft pressure on opponent: when aggressive, move toward their territory frontier
        if aggressive and (nx, ny) not in self_terr:
            val += (manh(nx, ny, ox, oy) - manh(sx, sy, ox, oy)) * -0.4

        # Deterministic tie-breaker: fixed direction order via comparison on (val, idx)
        idx = dirs.index((dx, dy))
        key = (val, -idx)
        if key > (best_val, best):
            best_val = val
            best = (dx, dy)

    if best is None:
        return [0, 0]
    return [int(best[0]), int(best[1])]
