def choose_move(observation):
    w = observation["grid_width"]
    h = observation["grid_height"]
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    obstacles = observation.get("obstacles", [])
    blocked = {(p[0], p[1]) for p in obstacles}
    role = (observation.get("self_role") or "").lower()
    is_evader = ("evad" in role) or ("runner" in role) or ("escape" in role)
    moves = [(-1,-1),(-1,0),(-1,1),(0,-1),(0,0),(0,1),(1,-1),(1,0),(1,1)]

    def in_bounds(x, y):
        return 0 <= x < w and 0 <= y < h

    def passable(x, y):
        return in_bounds(x, y) and (x, y) not in blocked

    def dist2(x1, y1, x2, y2):
        dx = x1 - x2
        dy = y1 - y2
        return dx * dx + dy * dy

    def bfs_dist(start, goal):
        if start == goal:
            return 0
        if not passable(*start) or not passable(*goal):
            return None
        q = [start]
        head = 0
        seen = {start: 0}
        while head < len(q):
            x, y = q[head]
            d = seen[(x, y)]
            head += 1
            for dx, dy in moves:
                nx, ny = x + dx, y + dy
                if not passable(nx, ny) or (nx, ny) in seen:
                    continue
                nd = d + 1
                if (nx, ny) == goal:
                    return nd
                seen[(nx, ny)] = nd
                q.append((nx, ny))
        return None

    corners = [(0, 0), (0, h - 1), (w - 1, 0), (w - 1, h - 1)]
    best_move = (0, 0)
    best_score = None

    if is_evader:
        # Choose moves that maximize distance to pursuer, while (deterministically) biasing toward the farthest reachable corner.
        best_corner = None
        best_corner_score = None
        for cx, cy in corners:
            d = bfs_dist((sx, sy), (cx, cy))
            if d is None:
                continue
            s = dist2(sx, sy, ox, oy) + 0.001 * (-d)
            if best_corner_score is None or s > best_corner_score:
                best_corner_score = s
                best_corner = (cx, cy)
        if best_corner is None:
            best_corner = max(corners, key=lambda c: dist2(c[0], c[1], ox, oy))

        tcx, tcy = best_corner
        for dx, dy in moves:
            nx, ny = sx + dx, sy + dy
            if not passable(nx, ny):
                continue
            far = dist2(nx, ny, ox, oy)
            d_corner = bfs_dist((nx, ny), (tcx, tcy))
            # Prefer staying far; if corner unreachable, penalize.
            score = far * 1000 + (-999999 if d_corner is None else -d_corner)
            if best_score is None or score > best_score:
                best_score = score
                best_move = (dx, dy)

def choose_move(observation):
    w = int(observation.get("grid_width", 0) or 0)
    h = int(observation.get("grid_height", 0) or 0)
    sx, sy = observation.get("self_position", (0, 0))
    ox, oy = observation.get("opponent_position", (0, 0))
    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    obstacles = observation.get("obstacles", []) or []
    blocked = {(p[0], p[1]) for p in obstacles if isinstance(p, (list, tuple)) and len(p) >= 2}

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def passable(x, y):
        return inb(x, y) and (x, y) not in blocked

    def man(x1, y1, x2, y2):
        return abs(x1 - x2) + abs(y1 - y2)

    def get_scores():

        if isinstance(s, dict):

            b = s.get("opponent", s.get("op", s.get("enemy", 0)))
            return a if isinstance(a, (int, float)) else 0, b if isinstance(b, (int, float)) else 0
