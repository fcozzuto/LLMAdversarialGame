def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position", [0, 0]) or [0, 0]
    ox, oy = observation.get("opponent_position", [0, 0]) or [0, 0]
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)
    role = str(observation.get("self_role", "")).lower()
    evader = ("evader" in role) or ("runner" in role) or ("evade" in role)

    blocked = set()
    for p in (observation.get("obstacles", []) or []):
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            blocked.add((int(p[0]), int(p[1])))

    def in_bounds(x, y):
        return 0 <= x < w and 0 <= y < h

    def dist2(x1, y1, x2, y2):
        dx = x1 - x2
        dy = y1 - y2
        return dx * dx + dy * dy

    resources = observation.get("resources", []) or []
    has_res = bool(resources)

    corners = [(0, 0), (0, h - 1), (w - 1, 0), (w - 1, h - 1)]
    moves = [(dx, dy) for dx in (-1, 0, 1) for dy in (-1, 0, 1)]

    # Targeting: pursue nearest visible resource for pursuer; evader tries to stay far from pursuer and corners.
    if (not evader) and has_res:
        best = None
        for r in resources:
            if isinstance(r, (list, tuple)) and len(r) >= 2:
                rx, ry = int(r[0]), int(r[1])
                if (rx, ry) in blocked:
                    continue
                d = abs(sx - rx) + abs(sy - ry)
                if best is None or d < best[0] or (d == best[0] and (rx, ry) < best[1]):
                    best = (d, (rx, ry))
        target = best[1] if best is not None else (ox, oy)
    else:
        target = None

    # Heuristic evaluation among candidate moves
    best_move = (0, 0)
    best_score = None
    # Tie-break deterministically by preferring smaller (dx,dy) lexicographically
    for dx, dy in sorted(moves):
        nx, ny = sx + dx, sy + dy
        if not in_bounds(nx, ny) or (nx, ny) in blocked:
            continue

        d_opp = dist2(nx, ny, ox, oy)

        if evader:
            # Prefer: maximize distance from pursuer; also prefer moving toward the farthest corner direction.
            far_corner = None
            far_d = None
            for cx, cy in corners:
                dd = dist2(nx, ny, cx, cy)
                if far_d is None or dd > far_d or (dd == far_d and (cx, cy) < far_corner):
                    far_d = dd
                    far_corner = (cx, cy)
            # Minor preference to align toward that corner
            step_to_corner = -abs(nx - far_corner[0]) - abs(ny - far_corner[1])
            score = (d_opp, step_to_corner)
            # Maximize lexicographically for (d_opp, step_to_corner)
            if best_score is None or score > best_score:
                best_score = score
                best_move = (dx, dy)
        else:
            # Pursuer: minimize distance to opponent; if using resources, also minimize to target.
            d_main = d_opp
            if target is not None:
                d_t = dist2(nx, ny, target[0], target[1])
            else:
                d_t = 0
            # Primary: reduce d_main. Secondary: reduce d_t. Tertiary: stay away from obstacles via small penalty.
            obs_pen = 0
            # Penalize proximity to obstacles (simple)
            for bx, by in blocked:
                dd = abs(nx - bx) + abs(ny - by)
                if dd == 0:
                    obs_pen += 1000
                elif dd == 1:
                    obs_pen += 3
                elif dd == 2:
                    obs_pen += 1
            score = (-d_main, -(d_t), -obs_pen)
            if best_score is None or score > best_score:
                best_score = score
                best_move = (dx, dy)

    return [int(best_move[0]), int(best_move[1])]
