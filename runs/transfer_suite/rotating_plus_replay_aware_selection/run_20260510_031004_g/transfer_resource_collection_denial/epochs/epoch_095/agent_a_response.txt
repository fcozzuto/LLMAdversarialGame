def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    obstacles = set()
    for p in observation.get("obstacles") or []:
        if p and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

    resources = []
    for r in observation.get("resources") or []:
        if r and len(r) >= 2:
            x, y = int(r[0]), int(r[1])
            if 0 <= x < w and 0 <= y < h and (x, y) not in obstacles:
                resources.append((x, y))
    if not resources:
        return [0, 0]

    def cheb(x1, y1, x2, y2):
        dx = x1 - x2
        if dx < 0:
            dx = -dx
        dy = y1 - y2
        if dy < 0:
            dy = -dy
        return dx if dx >= dy else dy

    # Pick target that maximizes "opponent race gap" then closeness.
    best_r = resources[0]
    best_key = None
    for rx, ry in resources:
        my_d = cheb(sx, sy, rx, ry)
        opp_d = cheb(ox, oy, rx, ry)
        key = (opp_d - my_d, -my_d, -(abs(sx - rx) + abs(sy - ry)), rx, ry)
        if best_key is None or key > best_key:
            best_key = key
            best_r = (rx, ry)
    tx, ty = best_r

    # Evaluate candidate next steps with deterministic scoring + obstacle penalty.
    moves = []
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            if dx == 0 and dy == 0:
                moves.append((0, 0))
            else:
                moves.append((dx, dy))
    seen = set()
    moves2 = []
    for m in moves:
        if m not in seen:
            seen.add(m)
            moves2.append(m)

    best_m = (0, 0)
    best_s = None
    for dx, dy in moves2:
        nx, ny = sx + dx, sy + dy
        if not (0 <= nx < w and 0 <= ny < h):
            continue
        if (nx, ny) in obstacles:
            s = -10**9
        else:
            my_next = cheb(nx, ny, tx, ty)
            my_now = cheb(sx, sy, tx, ty)
            opp_now = cheb(ox, oy, tx, ty)
            opp_tight = cheb(ox, oy, nx, ny) <= 1
            # Encourage getting closer than opponent and reduce wandering.
            s = (-(my_next - my_now), (opp_now - my_next), -my_next, -(abs(tx - nx) + abs(ty - ny)))
            if opp_tight:
                s += 1
            # Small preference to advance generally toward target.
            if dx == 0 and dy == 0:
                s -= 0.1
            else:
                s += 0.01 * ((1 if (tx - nx) * (tx - sx) > 0 else -1) + (1 if (ty - ny) * (ty - sy) > 0 else -1))
        if best_s is None or s > best_s:
            best_s = s
            best_m = (dx, dy)

    return [int(best_m[0]), int(best_m[1])]
