    def cheb(ax, ay, bx, by):

        if dx < 0: dx = -dx
        dy = ay - by
        if dy < 0: dy = -dy
        return dx if dx >= dy else dy

    def inside(x, y):
        return 0 <= x < w and 0 <= y < h

    def free(x, y):
        return inside(x, y) and (x, y) not in obstacles
