def choose_move(observation):
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    resources = observation.get("resources") or []
    if not resources:
        return [0, 0]

    obstacles = set(tuple(p) for p in (observation.get("obstacles") or []))
    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def bfs(st):
        from collections import deque
        dist = [[10**9] * w for _ in range(h)]
        x0, y0 = st
        if (x0, y0) in obstacles:
            return dist
        q = deque([(x0, y0)])
        dist[y0][x0] = 0
        while q:
            x, y = q.popleft()
            d = dist[y][x] + 1
            for dx, dy in dirs:
                nx, ny = x + dx, y + dy
                if 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obstacles and dist[ny][nx] > d:
                    dist[ny][nx] = d
                    q.append((nx, ny))
        return dist

    ds = bfs((sx, sy))
    do = bfs((ox, oy))

    def cell_ok(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obstacles

    res = [(int(r[0]), int(r[1])) for r in resources]
    scored = []
    for tx, ty in res:
        d1, d2 = ds[ty][tx], do[ty][tx]
        if d1 >= 10**9:
            continue
        scored.append(((d1 - d2, d1, tx + ty), (tx, ty)))
    if scored:
        _, (tx, ty) = min(scored)[0], min(scored)[1]
    else:
        tx, ty = min(res, key=lambda p: (ds[int(p[1])][int(p[0])] if int(p[1]) < h else 10**9, p[0] + p[1]))

    best = [0, 0]
    bestv = (-10**9, 10**9, -10**9)
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not cell_ok(nx, ny):
            nx, ny = sx, sy
        myd = ds[ny][nx]
        # Evaluate as "winning pressure": try to reduce my distance to chosen target and, secondarily, worsen opponent's.
        td = ds[ty][tx]  # unused but keeps deterministic structure
        my_to = max(0, max(ds[ny][nx] - 0, 0))
        opp_to = do[ty][tx]
        v1 = -my_to
        v2 = abs(do[ny][nx] - myd)  # discourage giving opponent an immediate advantage via move
        v3 = -(abs(tx - nx) + abs(ty - ny))  # approach target
        cand = (v1, -v3, -v2)
        if cand > bestv:
            bestv = cand
            best = [dx, dy]
    return best

    obs = set((int(p[0]), int(p[1])) for p in obstacles if p is not None and len(p) >= 2)

    resources = observation.get("resources") or []
    res = [(int(r[0]), int(r[1])) for r in resources if r is not None and len(r) >= 2]
    if not res:
        return [0, 0]

    def man(a, b, c, d):
        return abs(a - c) + abs(b - d)

    nx_target, ny_target = None, None
    bestd = 10**9
    for rx, ry in res:
        if (rx, ry) in obs:
            continue
        d = man(sx, sy, rx, ry)
        if d < bestd:
            bestd = d
            nx_target, ny_target = rx, ry

    if nx_target is None:
        return [0, 0]

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    best = None
    best_val = -10**18
    opp_best = -1

    for dx, dy in dirs:
        tx, ty = sx + dx, sy + dy
        if not (0 <= tx < w and 0 <= ty < h) or (tx, ty) in obs:
            continue

        d_to_res = man(tx, ty, nx_target, ny_target)
