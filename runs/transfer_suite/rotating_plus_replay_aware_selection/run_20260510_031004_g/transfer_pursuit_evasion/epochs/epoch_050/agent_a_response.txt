def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sp = observation.get("self_position", [0, 0]) or [0, 0]
    op = observation.get("opponent_position", [0, 0]) or [0, 0]
    sx, sy = int(sp[0]), int(sp[1])
    ox, oy = int(op[0]), int(op[1])

    obs = set()
    for p in observation.get("obstacles", []) or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obs.add((x, y))

    self_role = (observation.get("self_role", "") or "").lower()
    opponent_role = (observation.get("opponent_role", "") or "").lower()
    pursuer = ("purs" in self_role) or ("hunter" in self_role) or (self_role in ("pursuer", "pursuer_agent"))
    if not self_role and opponent_role:
        pursuer = not ("evad" in opponent_role)
    if not self_role and not opponent_role:
        pursuer = True

    corners = [(0, 0), (0, h - 1), (w - 1, 0), (w - 1, h - 1)]
    if pursuer:
        tx, ty = max(corners, key=lambda c: abs(c[0] - sx) + abs(c[1] - sy))
    else:
        tx, ty = max(corners, key=lambda c: abs(c[0] - ox) + abs(c[1] - oy))

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    candidates = []
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obs:
            candidates.append((dx, dy, nx, ny))
    if not candidates:
        return [0, 0]

    def md(a, b, c, d):
        return abs(a - c) + abs(b - d)

    eps = 1e-6
    best = None
    best_score = None

    for dx, dy, nx, ny in sorted(candidates, key=lambda t: (t[0], t[1])):
        d_opp = md(nx, ny, ox, oy)
        d_self_to_target = md(nx, ny, tx, ty)
        move_to_corner = -d_self_to_target if pursuer else d_self_to_target
        if pursuer:
            score = (-d_opp) * 10.0 + move_to_corner * 2.0
            # Encourage not moving away from opponent when opponent is near a wall/corner
            opp_wall = (ox == 0 or ox == w - 1 or oy == 0 or oy == h - 1)
            if opp_wall:
                score += (-d_opp)
        else:
            score = d_opp * 10.0 + move_to_corner * 2.0
            opp_wall = (ox == 0 or ox == w - 1 or oy == 0 or oy == h - 1)
            if opp_wall:
                score += d_opp

        # Small deterministic bias to avoid oscillation: prefer reducing distance to target corner direction
        score += -(eps * d_self_to_target if pursuer else -eps * d_self_to_target)

        if best_score is None or score > best_score:
            best_score = score
            best = [dx, dy]
