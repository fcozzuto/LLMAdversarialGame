def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sp = observation.get("self_position", [0, 0]) or [0, 0]
    op = observation.get("opponent_position", [0, 0]) or [0, 0]
    sx, sy = int(sp[0]), int(sp[1])
    ox, oy = int(op[0]), int(op[1])

    obs = set()
    for p in observation.get("obstacles", []) or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obs.add((x, y))

    self_role = (observation.get("self_role", "") or "").lower()
    opponent_role = (observation.get("opponent_role", "") or "").lower()
    pursuer = ("purs" in self_role) or ("hunter" in self_role) or (self_role == "pursuer") or ("pursuer" in self_role)
    if not self_role and opponent_role:
        pursuer = not ("evad" in opponent_role)
    if not self_role and not opponent_role:
        pursuer = True

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    best = None
    best_move = [0, 0]
    tie = 0
    sdx, sdy = ox - sx, oy - sy
    prefer = (1 if sdx > 0 else (-1 if sdx < 0 else 0), 1 if sdy > 0 else (-1 if sdy < 0 else 0))

    def valid(nx, ny):
        return 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obs

    def manhattan(x, y):
        return abs(x - ox) + abs(y - oy)

    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not valid(nx, ny):
            continue
        if nx == ox and ny == oy:
            return [dx, dy]  # immediate capture (radius 0)
        dm = manhattan(nx, ny)
        mx = abs(dx - prefer[0]) + abs(dy - prefer[1])
        center = (nx - (w // 2)) ** 2 + (ny - (h // 2)) ** 2
        # small obstacle proximity pressure to avoid running into walls
        adj_obs = 0
        for ax in (-1, 0, 1):
            for ay in (-1, 0, 1):
                if ax == 0 and ay == 0:
                    continue
                tx, ty = nx + ax, ny + ay
                if (tx, ty) in obs:
                    adj_obs += 1
        if pursuer:
            sc = dm * 100 + mx * 3 + adj_obs * 7 + center * 0.01
            # discourage waiting unless it helps
            if dx == 0 and dy == 0:
                sc += 20
            better = (best is None) or (sc < best) or (sc == best and tie > mx)
        else:
            sc = -dm * 100 + mx * 3 + adj_obs * 7 + center * 0.01
            # discourage suicidal staying
            if dx == 0 and dy == 0:
                sc -= 10
            better = (best is None) or (sc > best) or (sc == best and tie > mx)
        if better:
            best = sc
            best_move = [dx, dy]
