    op = observation.get("opponent_position", [0, 0]) or [0, 0]
    sx, sy = int(sp[0]), int(sp[1])
    ox, oy = int(op[0]), int(op[1])

    role = str(observation.get("self_role", "") or "").lower()
    evader = ("evader" in role) or ("runner" in role) or ("escape" in role) or ("pursuer" not in role and "chaser" not in role)

    avoid = set()
    for p in observation.get("obstacles", []) or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                avoid.add((x, y))

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def dist2(ax, ay, bx, by):
        dx, dy = ax - bx, ay - by
        return dx * dx + dy * dy

    best = (None, None, None, None)  # (score, nd, ntie1, ntie2)
    best_dxdy = (0, 0)

    if evader:

            d_to_t = dist2(nx, ny, tcx, tcy)
            score = d_to_p * 3 - d_to_t  # primary: maximize distance from pursuer
            nd = -d_to_p

        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                avoid.add((x, y))

    resources = observation.get("resources", []) or []
    res = []
    for p in resources:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h and (x, y) not in avoid:
                res.append((x, y))

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def dist2(ax, ay, bx, by):
        dx, dy = ax - bx, ay - by
        return dx * dx + dy * dy

    remaining = observation.get("remaining_resource_count", None)
    has_res = True
    try:
        has_res = (remaining is None) or (int(remaining) > 0) or (len(res) > 0)
    except:
        has_res = len(res) > 0

    best = None
    best_score = None
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not (0 <= nx < w and 0 <= ny < h) or (nx, ny) in avoid:
            continue
        d_op = dist2(nx, ny, ox, oy)
        score = 0
        if has_res and res:
            d_r = min(dist2(nx, ny, rx, ry) for rx, ry in res)
            score = -d_r * 100 + -d_op
