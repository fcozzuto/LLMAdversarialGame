def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sp = observation.get("self_position", [0, 0]) or [0, 0]
    op = observation.get("opponent_position", [0, 0]) or [0, 0]
    sx, sy = int(sp[0]), int(sp[1])
    ox, oy = int(op[0]), int(op[1])

    self_role = (observation.get("self_role", "") or "").lower()
    opp_role = (observation.get("opponent_role", "") or "").lower()
    is_pursuer = None
    if "purs" in self_role:
        is_pursuer = True
    if "evad" in self_role:
        is_pursuer = False
    if is_pursuer is None:
        if "evad" in opp_role:
            is_pursuer = False
        elif "purs" in opp_role:
            is_pursuer = True
    if is_pursuer is None:
        is_pursuer = True

    obs = set()
    for p in observation.get("obstacles", []) or []:
        try:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obs.add((x, y))
        except Exception:
            pass

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def md(x1, y1, x2, y2):
        return abs(x1 - x2) + abs(y1 - y2)

    corners = [(0, 0), (0, h - 1), (w - 1, 0), (w - 1, h - 1)]
    far_corner = max(corners, key=lambda c: md(c[0], c[1], ox, oy))
    near_corner = min(corners, key=lambda c: md(c[0], c[1], ox, oy))

    # If evading, bias toward far corner; if pursuing, bias toward near corner of pursuer relative to opponent.
    if is_pursuer:
        corner_target = near_corner
    else:
        corner_target = far_corner

    # If opponent is very close, corner bias increases.
    d0 = md(sx, sy, ox, oy)
    corner_weight = 3 if d0 <= 2 else 1

    best = None
    best_val = None
    best_corner_sgn = None

    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not (0 <= nx < w and 0 <= ny < h) or (nx, ny) in obs:
            nx, ny = sx, sy  # engine keeps in place on invalid moves
            dx, dy = 0, 0

        d_op = md(nx, ny, ox, oy)
        d_corner = md(nx, ny, corner_target[0], corner_target[1])

        # Primary objective: pursue/evade distance; secondary: corner bias.
        if is_pursuer:
            # minimize distance to opponent; then minimize distance to corner_target
            val = d_op * 10 + corner_weight * d_corner
        else:
            # maximize distance to opponent; then maximize distance from corner_target (equivalently minimize d_corner for corner_target? invert)
            val = (-d_op) * 10 + corner_weight * d_corner

        # Deterministic tie-breakers
        corner_sgn = (nx - corner_target[0], ny - corner_target[1])
        if best is None or val < best_val or (val == best_val and (corner_sgn, dx, dy) < best_corner_sgn):
            best = (dx, dy)
            best_val = val
            best_corner_sgn = corner_sgn

    return [int(best[0]), int(best[1])]

    resources = observation.get("resources", []) or []
    res_pos = []
    for r in resources:
        try:
            x, y = int(r[0]), int(r[1])
            if 0 <= x < w and 0 <= y < h:
                res_pos.append((x, y))
        except Exception:
            pass
    if not res_pos:
        res_pos = None

    def dist(x1, y1, x2, y2):
        d = abs(x1 - x2) + abs(y1 - y2)
        return d if d else 1

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    best_move = (0, 0)
    best_val = -10**18

    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if nx < 0 or nx >= w or ny < 0 or ny >= h:
            continue
        if (nx, ny) in obs:
            continue
        val = 0
        if res_pos is not None:
            d = min(dist(nx, ny, rx, ry) for rx, ry in res_pos)
            val += (1000 // d) if is_pursuer else -(1000 // d)
        dpo = dist(nx, ny, ox, oy)
        val += (200 // dpo) if is_pursuer else -(200 // dpo)
