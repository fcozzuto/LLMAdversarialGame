def choose_move(observation):

    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position", [0, 0]) or [0, 0]
    ox, oy = observation.get("opponent_position", [w - 1, h - 1]) or [w - 1, h - 1]
    obstacles = set(map(tuple, observation.get("obstacles", []) or []))
    unclaimed = observation.get("unclaimed_cells", []) or []
    self_terr = set(map(tuple, observation.get("self_territory", []) or []))
    opp_terr = set(map(tuple, observation.get("opponent_territory", []) or []))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def blocked(x, y):
        return (x, y) in obstacles

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def md(a, b):
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

    # If opponent is edge-claiming, take "interior expansion": prefer cells adjacent to our territory and
    # not adjacent to opponent territory, while still progressing toward unclaimed cells.
    opp_list = list(opp_terr) if opp_terr else []
    self_list = list(self_terr) if self_terr else []

    def adj_count(cell, terr_set):
        x, y = cell
        c = 0
        for dx in (-1, 0, 1):
            for dy in (-1, 0, 1):
                if dx == 0 and dy == 0:
                    continue
                nx, ny = x + dx, y + dy
                if inb(nx, ny) and (nx, ny) in terr_set:
                    c += 1
        return c

    def score_cell(cell):
        x, y = cell
        if blocked(x, y) or not inb(x, y):
            return -10**9
        dself = md((sx, sy), (x, y))
        # Distance from opponent territory (approx; higher is better)
        dop = 0
        if opp_list:
            dop = min(abs(px - x) + abs(py - y) for px, py in opp_list)
        # Interior-lean: prefer away from borders unless we are desperate (unclaimed far away)
        border = min(x, y, (w - 1) - x, (h - 1) - y)
        # Expansion: prefer cells adjacent to our territory
        a_self = adj_count((x, y), self_terr) if self_terr else 0
        a_opp = adj_count((x, y), opp_terr) if opp_terr else 0
        # Big penalty if cell is right next to opponent territory (avoid immediate contest flip loss)
        return (a_self * 3.5) + (border * 0.35) + (dop * 0.18) - (a_opp * 2.2) - (dself * 0.75)

    # Choose best target among unclaimed, else among opponent territory (to flip).

def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position", [0, 0]) or [0, 0]
    ox, oy = observation.get("opponent_position", [w - 1, h - 1]) or [w - 1, h - 1]
    obstacles = set(map(tuple, observation.get("obstacles", []) or []))
    unclaimed = observation.get("unclaimed_cells", []) or []
    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def ok(x, y):
        return inb(x, y) and (x, y) not in obstacles

    def md(x1, y1, x2, y2):
        return abs(x1 - x2) + abs(y1 - y2)

    target = None
    bestd = None
    for c in unclaimed:
        if not c:
            continue
        x, y = c
        if not ok(x, y):
            continue
        d = md(sx, sy, x, y)
        if bestd is None or d < bestd or (d == bestd and (x, y) < target):
            bestd = d
            target = (x, y)

    if target is None:
        target = (ox, oy)

    best_move = (0, 0)
    best_score = None
    tx, ty = target
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not ok(nx, ny):
            continue
        # Prefer smaller distance to target; break ties deterministically by direction order.
        score = md(nx, ny, tx, ty)
        if best_score is None or score < best_score:
            best_score = score
            best_move = (dx, dy)
    return [best_move[0], best_move[1]]
