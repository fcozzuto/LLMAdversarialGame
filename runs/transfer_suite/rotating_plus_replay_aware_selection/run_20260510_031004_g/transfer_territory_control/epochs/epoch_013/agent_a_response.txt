        return 0 <= x < w and 0 <= y < h

    def best_resource_dist(x, y):

            return 10**9

                dmin = dd
        return dmin if dmin is not None else 10**9

    candidates = []

        nx, ny = sx + dx, sy + dy
        if inb(nx, ny) and (nx, ny) not in obstacles:
            candidates.append((dx, dy, nx, ny))

        return [0, 0]
