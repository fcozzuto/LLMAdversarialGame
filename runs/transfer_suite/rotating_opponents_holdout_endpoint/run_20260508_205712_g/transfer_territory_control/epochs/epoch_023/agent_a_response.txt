def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [w - 1, h - 1])
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    obstacles = observation.get("obstacles", []) or []
    obs = set((int(x), int(y)) for x, y in obstacles)
    unclaimed = observation.get("unclaimed_cells", []) or []
    uc = set((int(x), int(y)) for x, y in unclaimed)
    oppT = set((int(x), int(y)) for x, y in (observation.get("opponent_territory", []) or []))
    selfT = set((int(x), int(y)) for x, y in (observation.get("self_territory", []) or []))

    neigh = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def man(a, b, c, d):
        return abs(a - c) + abs(b - d)

    # Determine primary targets
    if uc:
        # Deterministic: choose nearest unclaimed from current position, tie by coordinates
        target = min(uc, key=lambda p: (man(sx, sy, p[0], p[1]), p[0], p[1]))
        tx, ty = target
        chase_mode = True
    else:
        tx, ty = ox, oy
        chase_mode = False

    best = [0, 0]
    best_score = -10**18

    for dx, dy in neigh:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obs:
            continue

        on_u = (nx, ny) in uc
        on_self = (nx, ny) in selfT
        on_opp = (nx, ny) in oppT

        if chase_mode:
            dist = man(nx, ny, tx, ty)
            # Prefer stepping into unclaimed, otherwise reduce distance.
            score = -dist
            if on_u:
                score += 9

                # Flipping is enabled; still slightly prefer contest only if it helps reach frontier.

                score += 1
        else:
            dist = man(nx, ny, ox, oy)
