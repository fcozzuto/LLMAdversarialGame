def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position") or [0, 0]
    sx, sy = int(sx), int(sy)
    ox, oy = observation.get("opponent_position") or [w - 1, h - 1]
    turn = int(observation.get("turn_index", 0) or 0)

    obstacles = set()
    for p in observation.get("obstacles") or []:
        if p and len(p) >= 2:
            try:
                obstacles.add((int(p[0]), int(p[1])))
            except:
                pass

    self_set = set((int(x), int(y)) for x, y in (observation.get("self_territory") or []))
    opp_set = set((int(x), int(y)) for x, y in (observation.get("opponent_territory") or []))
    un_set = set((int(x), int(y)) for x, y in (observation.get("unclaimed_cells") or []))
    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def manh(a, b, c, d):
        return abs(a - c) + abs(b - d)

    # Heuristic: grab unclaimed; prefer cells that reduce distance to unclaimed frontier;
    # avoid stepping into opponent territory unless close to it (to flip/contend).
    best = -10**9
    best_move = [0, 0]
    prefer_flip = turn > 8

    # Precompute nearest unclaimed distance from candidate neighbors by scanning local ring only
    # (deterministic and cheap).
    cand_ns = []
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if inb(nx, ny) and (nx, ny) not in obstacles:
            cand_ns.append((nx, ny))
    un_list = list(un_set) if un_set else []
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obstacles:
            continue

        base = 0
        if (nx, ny) in self_set:
            base += 1
        elif (nx, ny) in un_set:
            base += 8
        elif (nx, ny) in opp_set:
            base -= 2
            if prefer_flip and manh(nx, ny, ox, oy) <= 3:
                base += 7  # flip on entry is enabled

        d_opp = manh(nx, ny, ox, oy)
        base += 0.4 * d_opp  # keep distance from opponent generally
        if nx <= ox:
            base += 0.1 if sx <= ox else 0  # tiny deterministic tie-breaker bias
        d_un = 999
        if un_list:
            # scan up to 12 unclaimed deterministically
            k = 0
            for ux, uy in un_list:
                d = manh(nx, ny, ux, uy)
                if d < d_un:
                    d_un = d
                k += 1
                if k >= 12:
                    break
            base += 4.0 / (1 + d_un)

        # Edge/center bias: move toward center after early turns if no nearby unclaimed.
        cx, cy = (w - 1) // 2, (h - 1) // 2
        if (not un_list) or (turn > 12 and len(un_set) < 20):
            base += 0.25 * (8 - manh(nx, ny, cx, cy))

        if base > best:
            best = base
            best_move = [dx, dy]
        elif base == best:
            # deterministic tie-break: prefer staying still less, then toward increasing x, then y
            if [dx, dy] == [0, 0] and best_move != [0, 0]:
                pass
            elif best_move == [0, 0]:
                best_move = [dx, dy]
            elif (dx, dy) > (best_move[0], best_move[1]):
                best_move = [dx, dy]

    # If no valid move (shouldn't happen), stay.
    return [int(best_move[0]), int(best_move[1])]
