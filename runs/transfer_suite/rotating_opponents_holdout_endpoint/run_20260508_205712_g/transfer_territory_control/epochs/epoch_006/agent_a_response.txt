def choose_move(observation):
    w = observation.get("grid_width", 8)
    h = observation.get("grid_height", 8)
    sx, sy = observation.get("self_position", (0, 0))
    ox, oy = observation.get("opponent_position", (0, 0))
    obstacles = set(tuple(p) for p in (observation.get("obstacles", []) or []))
    unclaimed = [tuple(p) for p in (observation.get("unclaimed_cells", []) or [])]
    opp_terr = set(tuple(p) for p in (observation.get("opponent_territory", []) or []))
    self_terr = set(tuple(p) for p in (observation.get("self_territory", []) or []))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def adj_opp(x, y):
        for dx in (-1, 0, 1):
            for dy in (-1, 0, 1):
                if dx == 0 and dy == 0:
                    continue
                if (x + dx, y + dy) in opp_terr:
                    return True
        return False

    # If possible, take an unclaimed cell that threatens/feeds expansion near opponent.
    if unclaimed:
        candidates = []
        for x, y in unclaimed:
            if not inb(x, y) or (x, y) in obstacles:
                continue
            # Prefer cells adjacent to opponent territory; otherwise prefer moving toward opponent.
            d_to_opp = abs(x - ox) + abs(y - oy)
            # Prefer cells "on the frontier" of our current territory (expansion) to avoid getting stuck.
            adj_to_self = False
            for dx in (-1, 0, 1):
                for dy in (-1, 0, 1):
                    if dx == 0 and dy == 0:
                        continue
                    if (x + dx, y + dy) in self_terr:
                        adj_to_self = True
                        break
                if adj_to_self:
                    break
            # Deterministic tie-breakers: closeness to center then lexicographic.
            d_center = abs(x - (w - 1) / 2.0) + abs(y - (h - 1) / 2.0)
            key = (
                0 if adj_opp(x, y) else 1,
                d_to_opp,
                0 if adj_to_self else 1,
                d_center,
                y,
                x,
            )
            candidates.append((key, x, y))
        if candidates:
            _, tx, ty = min(candidates)[0], min(candidates)[1], min(candidates)[2]
            # Move one step toward target (including diagonals).
            opts = []
            for dx in (-1, 0, 1):
                for dy in (-1, 0, 1):
                    if dx == 0 and dy == 0:
                        continue
                    nx, ny = sx + dx, sy + dy
                    if not inb(nx, ny) or (nx, ny) in obstacles:
                        continue
                    dist = abs(nx - tx) + abs(ny - ty)
                    # Prefer frontier risk-reward: slightly prefer entering opponent-owned cells.
                    enter_opp = 0 if (nx, ny) in opp_terr else 1
                    opts.append((dist, enter_opp, ny, nx, dx, dy))
            if opts:
                _, _, _, _, mdx, mdy = min(opts)
                return [mdx, mdy]
            return [0, 0]

    # Fallback: head toward opponent, but avoid obstacles; if blocked, go to center.
    cx, cy = (w - 1) // 2, (h - 1) // 2
    target = (ox, oy) if unclaimed == [] else (cx, cy)
    tx, ty = target

    best = None
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            if dx == 0 and dy == 0:
                continue
            nx, ny = sx + dx, sy + dy
            if not inb(nx, ny) or (nx, ny) in obstacles:
                continue
            score = abs(nx - tx) + abs(ny - ty)
            # If opponent territory is next, prioritize it deterministically.
            score2 = 0 if (nx, ny) in opp_terr else 1
            # Slightly prefer staying near our territory to keep control.
            score3 = 0 if (nx, ny) in self_terr else 1
            key = (score2, score, score3, ny, nx, dx, dy)
            if best is None or key < best[0]:
                best = (key, dx, dy)
    if best:
        return [best[1], best[2]]
    return [0, 0]
