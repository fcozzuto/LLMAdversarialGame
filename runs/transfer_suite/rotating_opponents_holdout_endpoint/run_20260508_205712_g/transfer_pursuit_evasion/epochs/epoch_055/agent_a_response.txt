def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sp = observation.get("self_position", [0, 0]) or [0, 0]
    op = observation.get("opponent_position", [0, 0]) or [0, 0]
    sx, sy = int(sp[0]), int(sp[1])
    ox, oy = int(op[0]), int(op[1])

    obstacles = set()
    for a in observation.get("obstacles", []) or []:
        if isinstance(a, (list, tuple)) and len(a) >= 2:
            x, y = int(a[0]), int(a[1])
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

    role = (observation.get("self_role", "pursuer") or "pursuer").lower()
    pursuer = role != "evader"

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def valid(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obstacles

    # BFS distance from opponent (for pursuer) / from opponent (for evader too, via maximizing)
    from collections import deque
    INF = 10**9
    dist = [[INF] * h for _ in range(w)]
    if valid(ox, oy):
        q = deque([(ox, oy)])
        dist[ox][oy] = 0
        while q:
            x, y = q.popleft()
            d0 = dist[x][y] + 1
            for dx, dy in moves:
                nx, ny = x + dx, y + dy
                if valid(nx, ny) and dist[nx][ny] > d0:
                    dist[nx][ny] = d0
                    q.append((nx, ny))

    best = None
    best_move = (0, 0)
    # Deterministic tie-break: iterate moves in given order
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not valid(nx, ny):
            continue
        d = dist[nx][ny]
        if d >= INF:
            d = INF // 2 + (abs(nx - ox) + abs(ny - oy))  # still deterministic preference
        score = -d if pursuer else d
        if best is None or score > best:
            best = score
            best_move = (dx, dy)

    # If no valid move (unlikely), stay
    dx, dy = best_move
    return [int(dx), int(dy)]

                obstacles.add((x, y))

    evader = role == "evader"

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    resources = observation.get("resources", None)
    if not resources:
        resources = []
    res_list = []
    for r in resources:
        if isinstance(r, (list, tuple)) and len(r) >= 2:
            x, y = int(r[0]), int(r[1])
            if 0 <= x < w and 0 <= y < h and (x, y) not in obstacles:
                res_list.append((x, y))
        elif isinstance(r, dict) and "position" in r and r["position"]:
            p = r["position"]
            if isinstance(p, (list, tuple)) and len(p) >= 2:
                x, y = int(p[0]), int(p[1])
                if 0 <= x < w and 0 <= y < h and (x, y) not in obstacles:
                    res_list.append((x, y))

    def valid(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obstacles

    def manhattan(x1, y1, x2, y2):
        dx = x1 - x2
        if dx < 0:
            dx = -dx
        dy = y1 - y2
        if dy < 0:
            dy = -dy
        return dx + dy

    def nearest_res_dist(x, y):
        if not res_list:
            return manhattan(x, y, ox, oy)
        best = 10**9
        for rx, ry in res_list:
            d = manhattan(x, y, rx, ry)
            if d < best:
                best = d
        return best

    best_move = (0, 0)
    best_score = -10**18 if not evader else 10**18
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not valid(nx, ny):
            continue
        dres = nearest_res_dist(nx, ny)
        dop = manhattan(nx, ny, ox, oy)
        if evader:
            score = dop * 10 - dres
            if score > best_score or (score == best_score and (dx, dy) < best_move):
                best_score = score
                best_move = (dx, dy)
        else:
            score = -dres * 10 + dop
            if score > best_score or (score == best_score and (dx, dy) < best_move):
                best_score = score
                best_move = (dx, dy)

    return [int(best_move[0]), int(best_move[1])]
