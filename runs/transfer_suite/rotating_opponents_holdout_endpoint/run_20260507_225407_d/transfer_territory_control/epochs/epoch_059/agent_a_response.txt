def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position", [0, 0])
    sx, sy = int(sx), int(sy)
    ox, oy = observation.get("opponent_position", [w - 1, h - 1])
    ox, oy = int(ox), int(oy)

    obstacles = set()
    for p in observation.get("obstacles") or []:
        if p and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

    self_cells = set(map(tuple, observation.get("self_territory") or []))
    opp_cells = set(map(tuple, observation.get("opponent_territory") or []))
    unclaimed = set(map(tuple, observation.get("unclaimed_cells") or []))

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    pref = dirs  # deterministic order

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def neighbors(x, y):
        res = []
        for dx, dy in dirs:
            if dx == 0 and dy == 0:
                continue
            nx, ny = x + dx, y + dy
            if inb(nx, ny):
                res.append((nx, ny))
        return res

    def nearest_unclaimed_dist(x, y):
        if not unclaimed:
            return 0
        md = 10**9
        for ux, uy in unclaimed:
            d = abs(ux - x) + abs(uy - y)
            if d < md:
                md = d
        return md if md != 10**9 else 0

    # Main heuristic: expand around unclaimed, and counterclaim opponent territory; avoid obstacles/edges of own safety
    best = None
    best_val = -10**18
    for dx, dy in pref:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obstacles:
            continue

        val = 0

        # Prefer moving into unclaimed or opponent-owned cells
        if (nx, ny) in unclaimed:
            val += 6
        if (nx, ny) in opp_cells:
            val += 12

        # Expansion: count adjacent unclaimed (more if we just entered unclaimed/opponent)
        adj_u = 0
        for (px, py) in neighbors(nx, ny):
            if (px, py) in unclaimed:
                adj_u += 1
        val += adj_u * 2

        # Territorial pressure: prefer getting closer to opponent and to frontier cells (adjacent to their territory)
        val += max(0, 6 - (abs(nx - ox) + abs(ny - oy))) * 0.25

        opp_front = 0
        for (px, py) in neighbors(nx, ny):
            if (px, py) in opp_cells:
                opp_front += 1
        val += opp_front * 1.5

        # Slightly avoid getting too deep into our current territory if it isolates from unclaimed
        if (nx, ny) in self_cells:
            val -= 1.0

        # Mild deterrent near obstacles
        adj_obs = 0
        for (px, py) in neighbors(nx, ny):
            if (px, py) in obstacles:
                adj_obs += 1
        val -= adj_obs * 1.0

        # Encourage reaching the nearest unclaimed if it exists
        d_un = nearest_unclaimed_dist(nx, ny)
        if d_un:
            val += (1.0 / (1 + d_un)) * 4.0

        # Deterministic tie-break: favor staying if equal, else earlier in pref
        if val > best_val:
            best_val = val
            best = (dx, dy)
        elif val == best_val and best is not None:
            if (dx, dy) == (0, 0):
                best = (dx, dy)

    if best is None:
        return [0, 0]
    return [int(best[0]), int(best[1])]
