    ox, oy = observation.get("opponent_position", [w - 1, h - 1])
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    def to_set(key):
        s = set()
        for p in observation.get(key) or []:
            if p and len(p) >= 2:
                x, y = int(p[0]), int(p[1])
                if 0 <= x < w and 0 <= y < h:
                    s.add((x, y))
        return s

    un = to_set("unclaimed_cells")
    self_t = to_set("self_territory")
    dirs = [(dx, dy) for dx in (-1, 0, 1) for dy in (-1, 0, 1) if not (dx == 0 and dy == 0)]

    def bfs(start):
        INF = 10**9
        dist = [[INF] * h for _ in range(w)]
        if start in obstacles:
            return dist
        x0, y0 = start
        dist[x0][y0] = 0
        q = [(x0, y0)]
        qi = 0
        while qi < len(q):
            x, y = q[qi]
            qi += 1
            d = dist[x][y] + 1
            for dx in (-1, 0, 1):
                for dy in (-1, 0, 1):
                    if dx == 0 and dy == 0:
                        continue
                    nx, ny = x + dx, y + dy
                    if 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obstacles and d < dist[nx][ny]:
                        dist[nx][ny] = d
                        q.append((nx, ny))
        return dist

    dist_us = bfs((sx, sy))
    dist_opp = bfs((ox, oy))

    best = None
    for (x, y) in un:
        du = dist_us[x][y]
        do = dist_opp[x][y]
        if du >= 10**9:
            continue
        val = du - 0.65 * do + 0.001 * (x * 8 + y)
        if best is None or val < best[0] or (val == best[0] and (x + y) < (best[1] + best[2])):
            best = (val, x, y)
