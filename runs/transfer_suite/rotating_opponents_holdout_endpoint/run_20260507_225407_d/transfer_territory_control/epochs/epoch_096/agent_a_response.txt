def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = map(int, observation.get("self_position") or (0, 0))
    self_territory = observation.get("self_territory") or []
    opp_territory = observation.get("opponent_territory") or []
    unclaimed = observation.get("unclaimed_cells") or []
    obstacles = set()
    for p in observation.get("obstacles") or []:
        if p and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

    self_set = set((int(p[0]), int(p[1])) for p in self_territory if p and len(p) >= 2)
    opp_set = set((int(p[0]), int(p[1])) for p in opp_territory if p and len(p) >= 2)
    un_set = set((int(p[0]), int(p[1])) for p in unclaimed if p and len(p) >= 2)

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    # Frontier pressure: encourage moves that get closer to opponent territory.
    opp_cells = list(opp_set)
    best = (0, 0, -10**9, 10**9, 10**9)  # (dx,dy,score,distOpp,distCorner)
    best_score = -10**9

    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not (0 <= nx < w and 0 <= ny < h) or (nx, ny) in obstacles:
            continue

        gain = 0
        if (nx, ny) in opp_set:
            gain += 2
            if (nx, ny) not in self_set:
                gain += 1
        elif (nx, ny) in un_set:
            gain += 1
        elif (nx, ny) in self_set:
            gain += 0
        else:
            gain += 0

        if opp_cells:
            dopp = min(max(abs(nx - ox), abs(ny - oy)) for ox, oy in opp_cells)
        else:
            dmax = max(abs((w - 1) - nx), abs((h - 1) - ny))
            dopp = dmax

        # Encourage sweeping toward the center early, but pressure opponent when close.
        cx, cy = (w - 1) / 2.0, (h - 1) / 2.0
        dc = abs(nx - cx) + abs(ny - cy)

        # Deterministic scoring with clear priorities.
        score = gain * 10 + (12 - dopp) + (0.2 * (-dc))
        dist_corner = abs(nx - 0) + abs(ny - 0)

        if score > best_score or (score == best_score and (dopp, dc, dist_corner) < (best[3], best[4], best[5])):
            best = (dx, dy, score, dopp, dc)
            best_score = score

    return [int(best[0]), int(best[1])]

def choose_move(observation):
    w = int(observation.get("grid_width") or 8)
    h = int(observation.get("grid_height") or 8)
    sx, sy = observation.get("self_position") or (0, 0)
    ox, oy = observation.get("opponent_position") or (0, 0)
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    obstacles = set()
    for p in observation.get("obstacles") or []:
        if p is None:
            continue
        try:
            x, y = p[0], p[1]
        except Exception:
            continue
        x, y = int(x), int(y)
        if 0 <= x < w and 0 <= y < h:
            obstacles.add((x, y))

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    best = None
    best_val = None

    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not (0 <= nx < w and 0 <= ny < h):
            continue
        if (nx, ny) in obstacles:
            continue
        dist = abs(nx - ox) + abs(ny - oy)
        val = -dist
        if best is None or val > best_val or (val == best_val and (dx, dy) < best):
            best = (dx, dy)
            best_val = val

    if best is None:
        for dx, dy in dirs:
            nx, ny = sx + dx, sy + dy
            if 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obstacles:
                return [dx, dy]
        return [0, 0]
    return [best[0], best[1]]
