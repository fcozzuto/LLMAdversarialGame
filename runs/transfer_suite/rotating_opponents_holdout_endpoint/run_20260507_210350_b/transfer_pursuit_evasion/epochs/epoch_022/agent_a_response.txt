def choose_move(observation):
    w = observation["grid_width"]
    h = observation["grid_height"]
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    obstacles = set(tuple(p) for p in (observation.get("obstacles", []) or []))
    role = str(observation.get("self_role", "")).lower()
    pursuer = any(k in role for k in ("pursuer", "hunter", "chaser", "pursuit"))
    deltas = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def cheb(x1, y1, x2, y2):
        dx = x1 - x2
        if dx < 0: dx = -dx
        dy = y1 - y2
        if dy < 0: dy = -dy
        return dx if dx > dy else dy

    def valid(nx, ny):
        return 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obstacles

    def obs_min_dist(nx, ny):
        md = 99
        if not obstacles:
            return md
        for ax, ay in obstacles:
            d = cheb(nx, ny, ax, ay)
            if d < md:
                md = d
                if md == 0:
                    return 0
        return md

    corners = ((0, 0), (w - 1, 0), (0, h - 1), (w - 1, h - 1))
    if pursuer:
        best_score = -10**18
        best_move = (0, 0)
        # Prefer shrinking distance; also drift toward the opposite corner of the opponent to trap with diagonals.
        for dx, dy in deltas:
            nx, ny = sx + dx, sy + dy
            if not valid(nx, ny):
                continue
            d = cheb(nx, ny, ox, oy)
            od = obs_min_dist(nx, ny)
            # Corner bias: closest corner to opponent is "farther" for pursuer; we want the opposite.
            opp_corner_d = 999
            for cx, cy in corners:
                dd = cheb(cx, cy, ox, oy)
                if dd < opp_corner_d:
                    opp_corner_d = dd
            # Approx "opposite corner" incentive: maximize corner distance to opponent.
            my_best_corner = 0
            for cx, cy in corners:
                cc = cheb(nx, ny, cx, cy)
                oo = cheb(cx, cy, ox, oy)
                val = cc - (oo * 0.2)
                if val > my_best_corner:
                    my_best_corner = val
            score = (-d * 1000) + (od * 8) + (my_best_corner * 2) - (0.5 * cheb(nx, ny, sx, sy))
            if score > best_score:
                best_score = score
                best_move = (dx, dy)
        return [int(best_move[0]), int(best_move[1])]
    else:
        best_score = -10**18
        best_move = (0, 0)
        # Evader: maximize distance from pursuer while maintaining safe clearance from obstacles and tending to farthest corner.
        for dx, dy in deltas:
            nx, ny = sx + dx, sy + dy
            if not valid(nx, ny):
                continue
            d = cheb(nx, ny, ox, oy)
            od = obs_min_dist(nx, ny)
            far_corner_d = 0
            for cx, cy in corners:
                cd = cheb(cx, cy, ox, oy)
                if cd > far_corner_d:
                    far_corner_d = cd
            corner_here = 0
            for cx, cy in corners:
                cd = cheb(nx, ny, cx, cy)
                if cd > corner_here:
                    corner_here = cd
            score = (d * 1000) + (od * 10) + (far_corner_d * 0.5) + (corner_here * 2) - (0.5 * cheb(nx, ny, sx, sy))
            if score > best_score:
                best_score = score
                best_move = (dx, dy)
        return [int(best_move[0]), int(best_move[1])]
