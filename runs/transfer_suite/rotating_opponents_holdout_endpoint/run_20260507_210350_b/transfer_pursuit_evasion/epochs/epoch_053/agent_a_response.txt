def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    obstacles = set(tuple(p) for p in (observation.get("obstacles", []) or []))
    rolestr = (observation.get("self_role", "") or "").lower()
    is_evader = any(k in rolestr for k in ("evader", "escape", "flee", "runner"))
    t = int(observation.get("turn_index", 0) or 0)

    moves = [(-1, -1), (-1, 0), (-1, 1),
             (0, -1), (0, 0), (0, 1),
             (1, -1), (1, 0), (1, 1)]

    def inb(x, y): return 0 <= x < w and 0 <= y < h

    def cheb(x1, y1, x2, y2):
        dx = x2 - x1
        if dx < 0: dx = -dx
        dy = y2 - y1
        if dy < 0: dy = -dy
        return dx if dx > dy else dy

    def obs_penalty(x, y):
        if not obstacles:
            return 0
        bestd = 999
        for ax, ay in obstacles:
            d = cheb(x, y, ax, ay)
            if d < bestd:
                bestd = d
            if bestd <= 0:
                return 10000
        # discourage moving onto/adjacent to obstacles
        if bestd <= 1:
            return 500
        if bestd == 2:
            return 50
        if bestd == 3:
            return 10
        return 0

    # deterministic tie-break: rotate move ordering by parity
    shift = t % len(moves)
    ordered = moves[shift:] + moves[:shift]

    best = None
    best_score = None

    # for evader, bias toward opposite corner from pursuer
    corners = [(0, 0), (w - 1, 0), (0, h - 1), (w - 1, h - 1)]
    if is_evader:
        # choose corner farthest from pursuer as an attractor
        far_corner = None
        far_d = -1
        for cx, cy in corners:
            d = cheb(cx, cy, ox, oy)
            if d > far_d:
                far_d = d
                far_corner = (cx, cy)

    for dx, dy in ordered:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obstacles:
            continue

        d_to_opp = cheb(nx, ny, ox, oy)
        d_opp_to_me = d_to_opp  # capture radius 0 implies immediate capture only if same cell; this is enough

        score = 0
        if is_evader:
            # maximize distance from pursuer, also keep moving toward chosen far corner
            score = d_to_opp * 100 - cheb(nx, ny, far_corner[0], far_corner[1]) * 3 if far_corner else d_to_opp * 100
            # small penalty for "stalling"
            if dx == 0 and dy == 0:
                score -= 2
            score -= obs_penalty(nx, ny)
        else:
            # pursuer: minimize distance to evader
            score = -d_to_opp * 100 - cheb(nx, ny, ox, oy) * 0 + 0
            # small penalty for stalling
            if dx == 0 and dy == 0:
                score -= 1
            score -= obs_penalty(nx, ny)

        if best_score is None or (score > best_score if is_evader else score > best_score):
            best_score = score
            best = (dx, dy)

    if best is None:
        # last resort: any valid move or stay
        for dx, dy in ordered:
            nx, ny = sx + dx, sy + dy
            if inb(nx, ny) and (nx, ny) not in obstacles:
                return [dx, dy]
        return [0, 0]
    return [best[0], best[1]]
