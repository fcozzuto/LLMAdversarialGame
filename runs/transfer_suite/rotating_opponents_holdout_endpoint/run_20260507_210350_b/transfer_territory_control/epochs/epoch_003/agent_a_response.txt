def choose_move(observation):
    w = observation.get("grid_width", 8)
    h = observation.get("grid_height", 8)
    sx, sy = observation.get("self_position", (0, 0))
    ox, oy = observation.get("opponent_position", (0, 0))
    obstacles = observation.get("obstacles", []) or []
    blocked = set()
    for p in obstacles:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            blocked.add((p[0], p[1]))

    def in_bounds(x, y):
        return 0 <= x < w and 0 <= y < h

    self_terr = observation.get("self_territory") or []
    opp_terr = observation.get("opponent_territory") or []
    self_set = set((p[0], p[1]) for p in self_terr if isinstance(p, (list, tuple)) and len(p) >= 2)
    opp_set = set((p[0], p[1]) for p in opp_terr if isinstance(p, (list, tuple)) and len(p) >= 2)

    unclaimed = observation.get("unclaimed_cells") or []
    unclaimed_set = set((p[0], p[1]) for p in unclaimed if isinstance(p, (list, tuple)) and len(p) >= 2)

    # Build targets: unclaimed first; otherwise cells that let us press into opponent territory.
    targets = []
    if unclaimed_set:
        targets = list(unclaimed_set)
    else:
        # Prefer opponent territory cells adjacent to our territory; else closest opponent territory cell.
        frontier = set()
        for (x, y) in self_set:
            for dx in (-1, 0, 1):
                for dy in (-1, 0, 1):
                    if dx == 0 and dy == 0:
                        continue
                    nx, ny = x + dx, y + dy
                    if in_bounds(nx, ny) and (nx, ny) in opp_set:
                        frontier.add((nx, ny))
        targets = list(frontier) if frontier else list(opp_set) if opp_set else [(ox, oy)]

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    best = None
    best_move = [0, 0]

    # Evaluation: moving into opponent cell is best; unclaimed is next; distance drives.
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not in_bounds(nx, ny) or (nx, ny) in blocked:
            continue

        in_opp = (nx, ny) in opp_set
        in_unc = (nx, ny) in unclaimed_set
        in_self = (nx, ny) in self_set

        # Choose nearest target distance from the prospective position.
        # Small, bounded targets list preference: if many targets, cap deterministically.
        if len(targets) > 64:
            # Take deterministically by sorting
            tlist = sorted(targets)[:64]
        else:
            tlist = targets

        # Squared distance
        md = None
        for tx, ty in tlist:
            ddx = nx - tx
            ddy = ny - ty
            d = ddx * ddx + ddy * ddy
            if md is None or d < md:
                md = d

        # Bonus terms to prefer advancing and avoid stalling.
        # If we can enter opponent territory, prioritize strongly.
        score = 0
        if in_opp:
            score += 100000
        if in_unc:
            score += 5000
        if in_self:
            score -= 20  # avoid unnecessary staying in already-owned cells if others exist

        # Encourage moving away from opponent position slightly when no good targets (reduces being countered).
        odx = nx - ox
        ody = ny - oy
        opp_dist2 = odx * odx + ody * ody

        score += (200 - md)  # closer target => larger
        score += (opp_dist2 // 8)  # prefer staying a bit away

        # Deterministic tie-break: prefer smaller (dx,dy) lexicographically when score equal.
        key = (-(score), dx, dy)
        if best is None or key < best:
            best = key
            best_move = [dx, dy]

    # If all moves invalid (shouldn't happen), stay.
    return [int(best_move[0]), int(best_move[1])]
