def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position", (0, 0))
    sx, sy = int(sx), int(sy)

    def to_set(v):
        s = set()
        for p in v or []:
            if isinstance(p, (list, tuple)) and len(p) >= 2:
                s.add((int(p[0]), int(p[1])))
        return s

    obstacles = to_set(observation.get("obstacles"))
    self_t = to_set(observation.get("self_territory"))
    opp_t = to_set(observation.get("opponent_territory"))
    unclaimed = to_set(observation.get("unclaimed_cells"))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def neigh8(x, y):
        for ddx in (-1, 0, 1):
            for ddy in (-1, 0, 1):
                if ddx == 0 and ddy == 0:
                    continue
                nx, ny = x + ddx, y + ddy
                if inb(nx, ny):
                    yield nx, ny

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    candidates = []
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obstacles:
            continue

        base = 0
        if (nx, ny) in unclaimed:
            base += 10
        if (nx, ny) in opp_t:
            base += 7  # flipping enabled on entry
        if (nx, ny) in self_t:
            base += 2

        # Local expansion potential
        neigh_unclaimed = 0
        neigh_opp = 0
        neigh_self = 0
        for ax, ay in neigh8(nx, ny):
            if (ax, ay) in unclaimed:
                neigh_unclaimed += 1
            elif (ax, ay) in opp_t:
                neigh_opp += 1
            elif (ax, ay) in self_t:
                neigh_self += 1

        # Macro target: prefer moving toward nearest unclaimed cell adjacent to opponent,
        # otherwise toward nearest unclaimed cell adjacent to our territory.
        frontier_opp = None
        frontier_ours = None
        if unclaimed:
            bestd = 10**9
            for ox, oy in opp_t if opp_t else []:
                for ax, ay in neigh8(ox, oy):
                    if (ax, ay) in unclaimed and (ax, ay) not in obstacles:
                        d = abs(ax - nx) + abs(ay - ny)
                        if d < bestd:
                            bestd = d
                            frontier_opp = (ax, ay)
            bestd = 10**9
            if frontier_opp is None:
                for px, py in self_t if self_t else []:
                    for ax, ay in neigh8(px, py):
                        if (ax, ay) in unclaimed and (ax, ay) not in obstacles:
                            d = abs(ax - nx) + abs(ay - ny)
                            if d < bestd:
                                bestd = d
                                frontier_ours = (ax, ay)

        dist_term = 0
        if frontier_opp is not None:
            dist_term = - (abs(frontier_opp[0] - nx) + abs(frontier_opp[1] - ny))
        elif frontier_ours is not None:
            dist_term = - (abs(frontier_ours[0] - nx) + abs(frontier_ours[1] - ny))

        score = base + 2 * neigh_unclaimed + 1 * neigh_self + 0.5 * neigh_opp + dist_term
        # Deterministic tie-break: prefer lower (dx,dy) lex to keep stable
        candidates.append((score, dx, dy))

    if not candidates:
        return [0, 0]
    candidates.sort(key=lambda t: (-t[0], t[1], t[2]))
    _, dx, dy = candidates[0]

    def norm_points(v):
        s = set()
        if not v:
            return s
        for p in v:

                    x, y = int(x), int(y)

    obstacles = norm_points(observation.get("obstacles"))
