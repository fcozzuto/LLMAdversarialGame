def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sp = observation.get("self_position", [0, 0]) or [0, 0]
    sx, sy = int(sp[0]), int(sp[1])
    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def to_set(key):
        s = set()
        for p in observation.get(key, []) or []:
            if isinstance(p, (list, tuple)) and len(p) >= 2:
                x, y = int(p[0]), int(p[1])
                if 0 <= x < w and 0 <= y < h:
                    s.add((x, y))
        return s

    obstacles = to_set("obstacles")
    unclaimed = to_set("unclaimed_cells")
    self_terr = to_set("self_territory")
    opp_terr = to_set("opponent_territory")
    resources = to_set("resources")

    self_cnt = int(observation.get("self_territory_count", 0) or 0)
    opp_cnt = int(observation.get("opponent_territory_count", 0) or 0)
    behind = self_cnt < opp_cnt

    # Nearest unclaimed cell for direction preference (local, cheap)
    target = None
    bestd = 10**9
    for (x, y) in unclaimed:
        d = abs(x - sx) + abs(y - sy)
        if d < bestd:
            bestd = d
            target = (x, y)

    ox, oy = None, None
    if target is not None:
        ox, oy = target

    def adj_to(sset, x, y):
        for ax in (-1, 0, 1):
            for ay in (-1, 0, 1):
                if ax == 0 and ay == 0:
                    continue
                nx, ny = x + ax, y + ay
                if (nx, ny) in sset:
                    return True
        return False

    best_move = [0, 0]
    best_score = -10**18

    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not (0 <= nx < w and 0 <= ny < h):
            continue
        if (nx, ny) in obstacles:
            continue

        score = 0
        if (nx, ny) in unclaimed:
            score += 260 if behind else 240
        if (nx, ny) in opp_terr:
            score += 220 if behind else 200  # stealing is valuable
        if (nx, ny) in self_terr:
            score += 25
        if (nx, ny) in resources:
            score += 90 if behind else 70

        # Prefer expanding from our frontier, and contesting the opponent frontier
        if adj_to(self_terr, nx, ny):
            score += 35
        if adj_to(opp_terr, nx, ny):
            score += 18 if behind else 28
        # Discourage pointless wandering: move toward nearest unclaimed if known
        if ox is not None:
            score += (bestd - (abs(nx - ox) + abs(ny - oy))) * (6 if behind else 4)

        # Avoid getting stuck adjacent to obstacles cluster: small penalty
        obs_adj = 0
        for ax in (-1, 0, 1):
            for ay in (-1, 0, 1):
                if ax == 0 and ay == 0:
                    continue
                tx, ty = nx + ax, ny + ay
                if (tx, ty) in obstacles:
                    obs_adj += 1
        score -= obs_adj * 3

        if score > best_score:
            best_score = score
            best_move = [dx, dy]

    return best_move
