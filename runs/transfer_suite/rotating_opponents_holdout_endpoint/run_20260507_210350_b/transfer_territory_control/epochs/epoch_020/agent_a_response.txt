def choose_move(observation):
    w = int(observation.get("grid_width") or 8)
    h = int(observation.get("grid_height") or 8)
    sp = observation.get("self_position") or [0, 0]
    op = observation.get("opponent_position") or [0, 0]
    sx, sy = int(sp[0]), int(sp[1])
    ox, oy = int(op[0]), int(op[1])

    def to_set(key):
        s = set()
        for p in (observation.get(key) or []):
            if isinstance(p, (list, tuple)) and len(p) >= 2:
                x, y = int(p[0]), int(p[1])
                if 0 <= x < w and 0 <= y < h:
                    s.add((x, y))
        return s

    obstacles = to_set("obstacles")
    self_t = to_set("self_territory")
    opp_t = to_set("opponent_territory")
    unclaimed = to_set("unclaimed_cells")

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def adj_cells(x, y):
        for dx in (-1, 0, 1):
            for dy in (-1, 0, 1):
                if dx == 0 and dy == 0:
                    continue
                nx, ny = x + dx, y + dy
                if inb(nx, ny):
                    yield nx, ny

    # Opponent-aware: if they control far more, switch to capture their edge.
    opp_adv = (observation.get("opponent_territory_count") or 0) - (observation.get("self_territory_count") or 0)
    edge_mode = opp_adv >= 8  # deterministic threshold

    best = None
    best_score = -10**18
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obstacles:
            nx, ny = sx, sy  # engine rejects invalid move by keeping in place
        cell = (nx, ny)

        if cell in obstacles:
            continue

        man_to_opp = abs(nx - ox) + abs(ny - oy)
        man_to_self = abs(nx - sx) + abs(ny - sy)  # will be 0/1/2-ish, but keeps structure deterministic

        # Base value: taking unclaimed near our cluster
        near_self = any(p in self_t for p in adj_cells(nx, ny))
        near_opp = any(p in opp_t for p in adj_cells(nx, ny))

        # Flip on entry to opponent-owned cell is enabled; reward that heavily.
        if cell in opp_t:
            score = 2000 - 5 * man_to_opp
        else:
            score = 0
            if cell in unclaimed:
                score += 60
                if near_self:
                    score += 25
                if edge_mode and near_opp:
                    score += 40  # capture/contaminate their edge
            else:
                # Stepping on our own territory: keep momentum, but don't drift aimlessly into their proximity.
                if cell in self_t:
                    score += 8
                    if near_opp:
                        score -= 25
                else:
                    score -= 10  # other non-territory cells (if any)
            # Proximity control: prefer moving toward our safest expansion unless in edge mode.
            if edge_mode:
                score += 15 * (8 - man_to_opp) if man_to_opp <= 8 else -10
            else:
                score += 2 * (8 - man_to_opp) if man_to_opp <= 8 else -2

            # Avoid getting too close to opponent unless we can flip next.
            if near_opp and cell not in opp_t:
                score -= 18

            # Small deterministic tie-breaker: prefer staying closer to center-ish to reduce zigzag.
        cx, cy = (w - 1) / 2.0, (h - 1) / 2.0
        score -= 0.1 * (abs(nx - cx) + abs(ny - cy))
