def choose_move(observation):
    w = observation["grid_width"]
    h = observation["grid_height"]
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    obstacles = set(tuple(p) for p in observation["obstacles"])
    resources = [tuple(p) for p in observation["resources"] if tuple(p) not in obstacles]
    if not resources:
        return [0, 0]

    def man(a, b):
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

    def step_towards(tx, ty):
        dx = 0 if tx == sx else (1 if tx > sx else -1)
        dy = 0 if ty == sy else (1 if ty > sy else -1)
        nx, ny = sx + dx, sy + dy
        if 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obstacles:
            return [dx, dy]
        # Fallback to axis step deterministically
        for ddx, ddy in ((dx, 0), (0, dy), (dx, -dy), (-dx, dy), (-dx, 0), (0, -dy), (0, 0)):
            nx, ny = sx + ddx, sy + ddy
            if 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obstacles:
                return [ddx, ddy]
        return [0, 0]

    my_close = min(resources, key=lambda r: man((sx, sy), r))
    # Contend: pick a resource where opponent is closer or equal, but we can still get there reasonably.
    opp_adv = sorted(resources, key=lambda r: (man((ox, oy), r) - man((sx, sy), r), man((sx, sy), r)))
    best = None
    for r in opp_adv[:6]:
        d_my = man((sx, sy), r)
        d_op = man((ox, oy), r)
        if d_my <= d_op + 2:  # contest window; material change vs always racing ahead
            best = r
            break
    if best is None:
        # If no contestable resource, push center-ish resource (tie-break) to avoid being too predictable.
        cx, cy = (w - 1) / 2.0, (h - 1) / 2.0
        best = min(resources, key=lambda r: (man((sx, sy), r), abs(r[0] - cx) + abs(r[1] - cy)))
    if best == (sx, sy):
        return [0, 0]
    return step_towards(best)
