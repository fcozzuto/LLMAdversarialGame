def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sp = observation.get("self_position", [0, 0]) or [0, 0]
    op = observation.get("opponent_position", [0, 0]) or [0, 0]
    sx, sy = int(sp[0]), int(sp[1])
    ox, oy = int(op[0]), int(op[1])

    obstacles = set()
    for p in observation.get("obstacles", []) or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

    resources = []
    for r in observation.get("resources", []) or []:
        if isinstance(r, (list, tuple)) and len(r) >= 2:
            x, y = int(r[0]), int(r[1])
            if 0 <= x < w and 0 <= y < h and (x, y) not in obstacles:
                resources.append((x, y))
    if not resources:
        return [0, 0]

    def king(a, b, c, d):
        dx = abs(a - c)
        dy = abs(b - d)
        return dx if dx >= dy else dy

    # Deterministic ordering and target selection:
    # 1) Prefer resources where we are faster than opponent (negative my-opp).
    # 2) If tied, prefer shorter our distance.
    # 3) If still tied, prefer resources more "open" (not adjacent to obstacles).
    resources.sort()
    best = None
    best_key = None
    for t in resources:
        tx, ty = t
        myd = king(sx, sy, tx, ty)
        opd = king(ox, oy, tx, ty)
        faster_margin = myd - opd
        adj_block = 0
        for nx in (tx - 1, tx, tx + 1):
            for ny in (ty - 1, ty, ty + 1):
                if (nx, ny) in obstacles:
                    adj_block += 1
        key = (faster_margin, myd, adj_block, tx, ty)
        if best_key is None or key < best_key:
            best_key = key
            best = t

    tx, ty = best
    if sx == tx and sy == ty:
        return [0, 0]

    def moves_ordered():
        # Prefer diagonal if it reduces both; otherwise axis; deterministic.
        for dx in (-1, 0, 1):
            for dy in (-1, 0, 1):
                if dx == 0 and dy == 0:
                    continue
                yield (dx, dy)

    # Candidate step directions toward target (with obstacle avoidance fallback).
    step_dx = 0
    if tx > sx:
        step_dx = 1
    elif tx < sx:
        step_dx = -1
    step_dy = 0
    if ty > sy:
        step_dy = 1
    elif ty < sy:
        step_dy = -1

    preferred = (step_dx, step_dy)
    if preferred != (0, 0):
        nx, ny = sx + preferred[0], sy + preferred[1]
        if 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obstacles:
            return [preferred[0], preferred[1]]

    # Try alternative moves deterministically: closest to target, obstacle-safe.
    candidates = []
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):

            if 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obstacles:
                candidates.append((king(nx, ny, tx, ty), abs(dx), abs(dy), dx, dy))
    candidates.sort()
    if candidates:
        return [int(candidates[0][3]), int(candidates[0][4])]
    return [0, 0]
