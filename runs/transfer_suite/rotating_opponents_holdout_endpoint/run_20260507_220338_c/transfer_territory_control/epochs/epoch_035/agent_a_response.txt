def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position") or [0, 0]
    ox, oy = observation.get("opponent_position") or [0, 0]
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    obstacles = set((int(p[0]), int(p[1])) for p in (observation.get("obstacles") or []))
    unclaimed = set((int(p[0]), int(p[1])) for p in (observation.get("unclaimed_cells") or []))
    my_terr = set((int(p[0]), int(p[1])) for p in (observation.get("self_territory") or []))
    opp_terr = set((int(p[0]), int(p[1])) for p in (observation.get("opponent_territory") or []))
    resources = set((int(p[0]), int(p[1])) for p in (observation.get("resources") or []))

    my_count = int(observation.get("self_territory_count", 0) or 0)
    opp_count = int(observation.get("opponent_territory_count", 0) or 0)
    behind = my_count <= opp_count

    def inb(x, y): return 0 <= x < w and 0 <= y < h
    def md(a, b): return abs(a[0] - b[0]) + abs(a[1] - b[1])

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    cx, cy = w // 2, h // 2

    frontier = []
    if opp_terr:
        for (x, y) in opp_terr:
            for dx, dy in dirs:
                nx, ny = x + dx, y + dy
                if inb(nx, ny) and (nx, ny) in unclaimed and (nx, ny) not in obstacles:
                    frontier.append((nx, ny))
    frontier = list(dict.fromkeys(frontier))

    targets = []
    if resources:
        targets = list(resources)
    elif frontier:
        targets = frontier
    elif unclaimed:
        targets = list(unclaimed)
    elif behind and opp_terr:
        targets = list(opp_terr)
    else:
        targets = [(cx, cy)] if inb(cx, cy) else [(sx, sy)]

    cur = (sx, sy)
    target = min(targets, key=lambda t: (md(cur, t), md((ox, oy), t), t[0], t[1]))

    best = (0, 0)
    best_score = -10**18
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obstacles:
            if (dx, dy) != (0, 0):
                continue
            nx, ny = sx, sy
        pos = (nx, ny)
        cell_type = 0
        if pos in resources:
            cell_type = 5
        elif pos in unclaimed:
            cell_type = 3
        elif pos in opp_terr:
            cell_type = 2 if behind else -3
        elif pos in my_terr:
            cell_type = 0
        else:
            cell_type = 0

        dist = md(pos, target)
        score = cell_type * 10 - dist
        if behind and pos in opp_terr:
            score += 2
        if not behind and pos in opp_terr:
            score -= 2
        score += -min(md(pos, t) for t in targets[:5]) * 0.25 if len(targets) > 1 else 0

        if score > best_score:
            best_score = score
            best = (dx, dy)
