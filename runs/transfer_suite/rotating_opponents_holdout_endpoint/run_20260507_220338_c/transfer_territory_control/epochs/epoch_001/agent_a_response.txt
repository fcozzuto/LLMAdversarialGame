def choose_move(observation):
    w = observation["grid_width"]
    h = observation["grid_height"]
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]

    self_cells = set((x, y) for x, y in observation["self_territory"])
    opp_cells = set((x, y) for x, y in observation["opponent_territory"])
    unclaimed = set((x, y) for x, y in observation["unclaimed_cells"])
    obstacles = set((x, y) for x, y in observation["obstacles"])

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def cheb(a, b):
        ax, ay = a
        bx, by = b
        dx = ax - bx
        dy = ay - by
        if dx < 0: dx = -dx
        if dy < 0: dy = -dy
        return dx if dx > dy else dy

    # Build candidate frontier targets (unclaimed adjacent to opponent territory)
    opp_neighbors = set()
    for (x, y) in opp_cells:
        for dx in (-1, 0, 1):
            for dy in (-1, 0, 1):
                nx, ny = x + dx, y + dy
                if inb(nx, ny) and (nx, ny) not in opp_cells and (nx, ny) not in self_cells:
                    opp_neighbors.add((nx, ny))
    frontier = [c for c in unclaimed if c in opp_neighbors]
    if not frontier:
        frontier = list(unclaimed)

    if not frontier:
        # Fallback: move to reduce distance to opponent while avoiding obstacles
        best = (0, None)
        for dx, dy in dirs:
            nx, ny = sx + dx, sy + dy
            if not inb(nx, ny) or (nx, ny) in obstacles:
                continue
            status = 2 if (nx, ny) in opp_cells else 1 if (nx, ny) in self_cells else 0
            val = 100 * status - cheb((nx, ny), (ox, oy))
            if val > best[0]:
                best = (val, [dx, dy])
        return best[1] if best[1] is not None else [0, 0]

    # Choose move that best approaches frontier; prioritize flipping opponent territory immediately
    best_val = -10**9
    best_move = [0, 0]
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obstacles:
            continue
        if (nx, ny) in opp_cells:
            val = 10**6 - cheb((nx, ny), (ox, oy))
        else:
            # distance to closest target + preference for unclaimed
            dist = 10**3
            for t in frontier:
                d = cheb((nx, ny), t)
                if d < dist:
                    dist = d
            val = -dist + (200 if (nx, ny) in unclaimed else 0) + (20 if (nx, ny) in self_cells else 0)
        if val > best_val:
            best_val = val
            best_move = [dx, dy]
    return best_move
