def choose_move(observation):
    w = int(observation.get("grid_width") or 8)
    h = int(observation.get("grid_height") or 8)
    sx, sy = observation.get("self_position") or (0, 0)
    sx, sy = int(sx), int(sy)
    opp_pos = observation.get("opponent_position") or (w - 1, h - 1)
    ox, oy = int(opp_pos[0]), int(opp_pos[1])

    def to_set(lst):
        s = set()
        for p in lst or []:
            if isinstance(p, (list, tuple)) and len(p) == 2:
                s.add((int(p[0]), int(p[1])))
        return s

    obstacles = to_set(observation.get("obstacles"))
    selfT = to_set(observation.get("self_territory"))
    oppT = to_set(observation.get("opponent_territory"))
    unclaimed = to_set(observation.get("unclaimed_cells"))

    moves = [(-1, -1), (-1, 0), (-1, 1),
             (0, -1), (0, 0), (0, 1),
             (1, -1), (1, 0), (1, 1)]

    def in_bounds(x, y): return 0 <= x < w and 0 <= y < h
    def mindist_to_set(px, py, S):
        best = 10**9
        for (x, y) in S:
            d = abs(px - x) + abs(py - y)
            if d < best:
                best = d
        return best if S else best

    def adjacent_cells(x, y):
        for dx in (-1, 0, 1):
            for dy in (-1, 0, 1):
                if dx == 0 and dy == 0:
                    continue
                nx, ny = x + dx, y + dy
                if in_bounds(nx, ny):
                    yield (nx, ny)

    # Prefer capturing unclaimed that are adjacent to opponent territory (steal/expand against edge-claimers)
    targets = set()
    for (x, y) in oppT:
        for nx, ny in adjacent_cells(x, y):
            if (nx, ny) in unclaimed and (nx, ny) not in obstacles:
                targets.add((nx, ny))
    if not targets:
        # Otherwise, take unclaimed adjacent to self territory to secure expansion
        for (x, y) in selfT:
            for nx, ny in adjacent_cells(x, y):
                if (nx, ny) in unclaimed and (nx, ny) not in obstacles:
                    targets.add((nx, ny))
    if not targets:
        # Fallback: move toward opponent position while avoiding obstacles
        targets = {(ox, oy)}

    # Extra: discourage walking into opponent territory unless it improves distance to their "edge"
    opp_cells = set(oppT)

    best_move = (0, 0)
    best_val = -10**18
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not in_bounds(nx, ny) or (nx, ny) in obstacles:
            continue

        # Base value by cell ownership
        if (nx, ny) in unclaimed:
            val = 40
        elif (nx, ny) in opp_cells:
            # flipping on entry can help, but only if it approaches targets
            val = 18
        elif (nx, ny) in selfT:
            val = 10
        else:
            val = 6

        # Main heuristic: minimize distance to chosen targets
        d_to_targets = mindist_to_set(nx, ny, targets)
        val += 120 - 10 * d_to_targets

        # Small penalty if moving away from opponent too much (edge contest)
        d_opp_now = abs(sx - ox) + abs(sy - oy)
        d_opp_new = abs(nx - ox) + abs(ny - oy)
        val += 6 * (d_opp_now - d_opp_new)

        # If moving into opponent territory, ensure it doesn't drastically increase distance to targets
        if (nx, ny) in opp_cells:
            d_self_to_targets = mindist_to_set(sx, sy, targets)
            if d_to_targets > d_self_to_targets + 2:
                val -= 25

        if val > best_val:
            best_val = val
            best_move = (dx, dy)

    return [int(best_move[0]), int(best_move[1])]
