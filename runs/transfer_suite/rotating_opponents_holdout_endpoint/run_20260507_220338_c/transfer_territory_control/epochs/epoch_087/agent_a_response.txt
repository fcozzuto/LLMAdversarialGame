def choose_move(observation):

    h = int(observation.get("grid_height") or 8)
    sp = observation.get("self_position") or [0, 0]
    sx, sy = int(sp[0]), int(sp[1])

    blocked = set()
    for p in (observation.get("obstacles") or []):
        if p is not None and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                blocked.add((x, y))

    selfT = set((int(t[0]), int(t[1])) for t in (observation.get("self_territory") or []))
    oppT = set((int(t[0]), int(t[1])) for t in (observation.get("opponent_territory") or []))
    unclaimed = set((int(t[0]), int(t[1])) for t in (observation.get("unclaimed_cells") or []))

    def inside(x, y):
        return 0 <= x < w and 0 <= y < h

    def neigh4(x, y):
        for dx, dy in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            nx, ny = x + dx, y + dy
            if inside(nx, ny):
                yield nx, ny

    candidates = []
    for dx, dy in ((-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)):
        nx, ny = sx + dx, sy + dy
        if not inside(nx, ny):
            continue
        if (nx, ny) in blocked:
            continue
        candidates.append((dx, dy, nx, ny))
    if not candidates:
        return [0, 0]

    # Deterministic tie-breaker: prefer lexicographically smaller move deltas.
    candidates.sort(key=lambda t: (t[0], t[1]))

    best_move = (0, 0, sx, sy)
    best_score = -10**18

    # Heuristic weights:
    # - Strongly prefer capturing opponent territory (counterclaim).
    # - Prefer unclaimed cells especially adjacent to our territory (frontier expansion).
    # - Slightly avoid giving opponent easy adjacent capture by staying away from their territory unless countering.
    for dx, dy, nx, ny in candidates:
        score = 0

        in_self = (nx, ny) in selfT
        in_opp = (nx, ny) in oppT
        in_un = (nx, ny) in unclaimed
        if in_self:
            score -= 3  # don't waste time moving within already-owned area
        if in_opp:
            score += 120
        if in_un:
            score += 35

        adj_self = 0
        adj_opp = 0
        for ax, ay in neigh4(nx, ny):
            if (ax, ay) in selfT:
                adj_self += 1
            if (ax, ay) in oppT:
                adj_opp += 1

        score += 12 * adj_self
        if in_un:
            score += 6 * adj_self
            score -= 5 * adj_opp  # avoid stepping into contested zones unless directly countering

        # If we move next to opponent, slight positive only when we also touch our frontier.
        if adj_opp > 0:
            score += (2 if adj_self > 0 else -4) * min(2, adj_opp)

def choose_move(observation):

    h = int(observation.get("grid_height") or 8)
    sp = observation.get("self_position") or [0, 0]
    oxp = observation.get("opponent_position") or [0, 0]
    sx, sy = int(sp[0]), int(sp[1])
    ox, oy = int(oxp[0]), int(oxp[1])

    def inside(x, y):
        return 0 <= x < w and 0 <= y < h

    blocked = set()
    for p in observation.get("obstacles") or []:
        if p is not None and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                blocked.add((x, y))

    unclaimed = set()
    for t in observation.get("unclaimed_cells") or []:
        if t is not None and len(t) >= 2:
            unclaimed.add((int(t[0]), int(t[1])))

    oppT = set()
    for t in observation.get("opponent_territory") or []:
        if t is not None and len(t) >= 2:
            oppT.add((int(t[0]), int(t[1])))

    resources = set()
    for r in observation.get("resources") or []:
        if r is not None and len(r) >= 2:
            resources.add((int(r[0]), int(r[1])))

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    best = (0, 0, -10**9)
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not inside(nx, ny) or (nx, ny) in blocked:
            continue

        md_opp = abs(nx - ox) + abs(ny - oy)
        score = 0
        if (nx, ny) in resources:
            score += 6
        if (nx, ny) in unclaimed:
            score += 3
        if (nx, ny) in oppT:
            score -= 5
        if md_opp <= 2:
            score -= (3 - md_opp) * 2

        if resources:
            mind = 10**9
            for rx, ry in resources:
                d = abs(nx - rx) + abs(ny - ry)
                if d < mind:
                    mind = d
            score += max(0, 4 - mind)

        if (score, -dx, -dy) > (best[2], -best[0], -best[1]):
            best = (dx, dy, score)
        elif score == best[2] and (dx, dy) < (best[0], best[1]):
            best = (dx, dy, score)

    return [int(best[0]), int(best[1])]
