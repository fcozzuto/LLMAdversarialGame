def choose_move(observation):
    w = observation["grid_width"]
    h = observation["grid_height"]
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    obstacles = set(observation.get("obstacles", []))
    unclaimed = set(observation.get("unclaimed_cells", []))
    resources = set(observation.get("resources", []))
    opp_cells = set(observation.get("opponent_territory", []))
    self_cells = set(observation.get("self_territory", []))

    cx, cy = (w - 1) / 2.0, (h - 1) / 2.0

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def cheb(ax, ay, bx, by):
        dx = ax - bx
        if dx < 0:
            dx = -dx
        dy = ay - by
        if dy < 0:
            dy = -dy
        return dx if dx > dy else dy

    # Build a small deterministic preference list: unclaimed adjacent to opponent territory (frontier).
    frontier = set()
    for (x, y) in opp_cells:
        for dx, dy in dirs:
            nx, ny = x + dx, y + dy
            if inb(nx, ny) and (nx, ny) in unclaimed:
                frontier.add((nx, ny))

    targets = []
    if frontier:
        targets = list(frontier)
    else:
        targets = list(unclaimed) if unclaimed else []

    def target_score(t):
        tx, ty = t
        # Prefer closer to center, and closer to us.
        d_center = abs(tx - cx) + abs(ty - cy)
        d_us = abs(tx - sx) + abs(ty - sy)
        # Deterministic tie-break: favor lower y then x.
        return (d_center * 10 + d_us, ty, tx)

    target = min(targets, key=target_score) if targets else None

    best = None
    best_move = (0, 0)
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obstacles:
            continue

        step = 0.0
        if (nx, ny) in resources:
            step += 3.0
        if (nx, ny) in unclaimed:
            step += 2.2
        if (nx, ny) in opp_cells:
            # Avoid feeding opponent territory too easily, but don't suicide into it.
            step -= 1.2
        if (nx, ny) in self_cells:
            step += 0.2

        d_center = abs(nx - cx) + abs(ny - cy)
        step -= d_center * 0.08

        if target is not None:
            step -= cheb(nx, ny, target[0], target[1]) * 0.9

        # If opponent is very close, try to avoid staying adjacent unless we are also moving toward the frontier.
        d_opp = cheb(nx, ny, ox, oy)
        if d_opp <= 1 and (target is None or cheb(nx, ny, target[0], target[1]) > 1):
            step -= 1.0
        if d_opp == 0:
            step -= 2.0

        # Deterministic tie-break
        key = (-(step), abs(dx), abs(dy), dy, dx)
        if best is None or key < best:
            best = key
            best_move = (dx, dy)

    return [int(best_move[0]), int(best_move[1])]

    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    obstacles_raw = observation.get("obstacles", [])
    obstacles = set()
    for p in obstacles_raw:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            obstacles.add((int(p[0]), int(p[1])))

    resources_raw = observation.get("resources", [])
    resources = []
    for p in resources_raw:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h and (x, y) not in obstacles:
                resources.append((x, y))

    dirs = [(-1, 0), (0, -1), (0, 0), (1, 0), (0, 1)]
    best = None

    def man(a, b, c, d):
        dx = a - c
        if dx < 0: dx = -dx
        dy = b - d
        if dy < 0: dy = -dy
        return dx + dy

    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not (0 <= nx < w and 0 <= ny < h):
            continue
        if (nx, ny) in obstacles:
            continue

        opp_d = man(nx, ny, ox, oy)
        if resources:
            rd = 10**9
            for rx, ry in resources:
                d = man(nx, ny, rx, ry)
                if d < rd:
                    rd = d
        else:
            rd = 0

        # Prefer getting closer to resources; also keep away from opponent; slight bias to stay.
        score = -rd - 2 * (0 if opp_d == 0 else opp_d) + (1 if dx == 0 and dy == 0 else 0)
        if best is None or score > best[0] or (score == best[0] and (dx, dy) < best[1]):
            best = (score, (dx, dy))

    if best is None:
        return [0, 0]
    return [best[1][0], best[1][1]]
