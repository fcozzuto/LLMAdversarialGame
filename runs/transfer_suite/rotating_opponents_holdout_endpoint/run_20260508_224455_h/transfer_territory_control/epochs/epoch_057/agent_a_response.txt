def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sp = observation.get("self_position") or [0, 0]
    sx, sy = int(sp[0]), int(sp[1])

    opp_pos = observation.get("opponent_position")
    if opp_pos is None:
        ox, oy = w - 1, h - 1
    else:
        ox, oy = int(opp_pos[0]), int(opp_pos[1])

    obs_set = set()
    for c in (observation.get("obstacles") or []):
        if c is not None and len(c) >= 2:
            try:
                obs_set.add((int(c[0]), int(c[1])))
            except:
                pass

    resources = observation.get("resources") or []

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    best_move = [0, 0]
    best_val = -10**18

    target = None
    if res:
        target = min(res, key=lambda p: (p[0] - sx) * (p[0] - sx) + (p[1] - sy) * (p[1] - sy))

        target = (ox, oy)

    tx, ty = target

    for dx, dy in dirs:

        val = -d_to_target + 0.05 * d_to_opp + (-0.001) * (abs(dx) + abs(dy))
