def choose_move(observation):
    sx, sy = observation.get("self_position", (0, 0))
    ox, oy = observation.get("opponent_position", (0, 0))
    w = int(observation.get("grid_width", 8)) or 8
    h = int(observation.get("grid_height", 8)) or 8

    obstacles = set(map(tuple, observation.get("obstacles", []) or []))
    unclaimed = set(map(tuple, observation.get("unclaimed_cells", []) or []))
    self_terr = set(map(tuple, observation.get("self_territory", []) or []))
    opp_terr = set(map(tuple, observation.get("opponent_territory", []) or []))

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    def inb(x, y):
        return 0 <= x < w and 0 <= y < h
    def adj8(x, y):
        for dx, dy in dirs[:-1]:
            nx, ny = x + dx, y + dy
            if inb(nx, ny):
                yield nx, ny
    def man(x1, y1, x2, y2):
        return abs(x1 - x2) + abs(y1 - y2)

    # Compute opponent frontier (unclaimed or self cells adjacent to opponent territory)
    opp_frontier = set()
    for (px, py) in opp_terr:
        for nx, ny in adj8(px, py):
            if (nx, ny) in obstacles:
                continue
            if (nx, ny) not in opp_terr:
                opp_frontier.add((nx, ny))
    if not opp_frontier:
        opp_frontier = set([(ox, oy)])

    best_move = [0, 0]
    best_val = -10**18

    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obstacles:
            nx, ny = sx, sy
            dx, dy = 0, 0

        is_opp = (nx, ny) in opp_terr
        is_un = (nx, ny) in unclaimed
        is_self = (nx, ny) in self_terr

        # Prefer immediate flips/claiming near opponent frontier, and avoid giving space to opponent
        frontier_dist = min(manh(nx, ny, fx, fy) for (fx, fy) in opp_frontier) if opp_frontier else 0
        opp_press = min(manh(ox, oy, fx, fy) for (fx, fy) in opp_frontier) if opp_frontier else 0

        val = 0
        if is_opp:
            val += 60
        if is_un:
            val += 20
        if is_self:
            val += 5
        # Move toward opponent frontier; and keep relative pressure
        val += 8 * (opp_press - frontier_dist)
        # Tie-break: reduce distance to opponent position when neutral
        val -= 0.2 * man(nx, ny, ox, oy)
        # Tiny deterministic preference to progress right/down if equal
        val += 0.001 * (dx + 2 * (dy == 1))

        if val > best_val:
            best_val = val
            best_move = [dx, dy]

    return [int(best_move[0]), int(best_move[1])]

    if res_cells:
