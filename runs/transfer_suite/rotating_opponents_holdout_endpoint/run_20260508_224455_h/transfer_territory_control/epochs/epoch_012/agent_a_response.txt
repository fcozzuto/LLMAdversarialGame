def choose_move(observation):
    sx, sy = observation.get("self_position", [0, 0])
    w = observation.get("grid_width", 8)
    h = observation.get("grid_height", 8)
    self_t = set(map(tuple, observation.get("self_territory", [])))
    opp_t = set(map(tuple, observation.get("opponent_territory", [])))
    unclaimed = list(map(tuple, observation.get("unclaimed_cells", [])))
    obstacles = set(map(tuple, observation.get("obstacles", [])))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    unclaimed_set = set(unclaimed)

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    best_move = [0, 0]
    best_score = -10**18

    cx, cy = (w - 1) / 2.0, (h - 1) / 2.0

    oxp, oyp = observation.get("opponent_position", [w - 1, h - 1])

    # Pick a disruption target: unclaimed cell adjacent to opponent territory, else closest unclaimed.
    target = None
    for ux, uy in unclaimed:
        for ddx in (-1, 0, 1):
            for ddy in (-1, 0, 1):
                if ddx == 0 and ddy == 0:
                    continue
                nx, ny = ux + ddx, uy + ddy
                if (nx, ny) in opp_t:
                    d = abs(ux - sx) + abs(uy - sy)
                    # prefer near us, but also slightly prefer toward opponent's side
                    d2 = (d + 0.25 * (abs(ux - (w - 1)) + abs(uy - (h - 1)))))
                    if target is None or d2 < target[0]:
                        target = (d2, (ux, uy))
                    break
            if target is not None and target[1][0] == ux and target[1][1] == uy:
                break
    if target is None and unclaimed:
        ux, uy = min(unclaimed, key=lambda p: abs(p[0] - sx) + abs(p[1] - sy))
        target = (abs(ux - sx) + abs(uy - sy), (ux, uy))
    t = target[1] if target else None

    # Immediate gain heuristic + positioning toward target and center, avoid obstacles already handled by move validity.
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obstacles:
            continue

        if (nx, ny) in opp_t:
            score = 6.5  # flipping on entry
        elif (nx, ny) in unclaimed_set:
            score = 4.0  # claim new territory
        elif (nx, ny) in self_t:
            score = 2.0  # maintain control
        else:
            score = 1.0  # should be unclaimed/edge, but keep small

        # Move toward disruption target (break edge-claims)
        if t is not None:
            score += 1.4 * (-(abs(nx - t[0]) + abs(ny - t[1])))

        # Mild pull toward center for long-term reach
        score += 0.15 * (-(abs(nx - cx) + abs(ny - cy)))

        # If close to opponent, prefer taking their territory rather than just claiming neutral cells
        score += 0.05 * (-(abs(nx - oxp) + abs(ny - oyp)))

        # Penalize moving into isolated corner if target exists (keeps mobility)
        score -= 0.2 * (1 if (nx in (0, w - 1) or ny in (0, h - 1)) and t is not None and (abs(nx - t[0]) + abs(ny - t[1]) > abs(sx - t[0]) + abs(sy - t[1])) else 0)

        if score > best_score:
            best_score = score
            best_move = [dx, dy]

    return best_move
