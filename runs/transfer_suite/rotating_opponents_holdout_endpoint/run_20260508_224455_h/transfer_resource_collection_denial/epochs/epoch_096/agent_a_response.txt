def choose_move(observation):
    gw = int(observation["grid_width"])
    gh = int(observation["grid_height"])
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    sx = int(sx); sy = int(sy); ox = int(ox); oy = int(oy)

    resources = observation.get("resources") or []
    obstacles = observation.get("obstacles") or []
    if not resources:
        return [0, 0]

    obs = {(int(p[0]), int(p[1])) for p in obstacles}
    res = [(int(r[0]), int(r[1])) for r in resources]

    def inb(x, y):
        return 0 <= x < gw and 0 <= y < gh

    def dist8(ax, ay, bx, by):
        dx = ax - bx
        dy = ay - by
        if dx < 0: dx = -dx
        if dy < 0: dy = -dy
        return dx if dx > dy else dy

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def opp_best_step(tx, ty, cx, cy):
        best = None
        bestd = 10**9
        bestx = cx; besty = cy
        # deterministic tie: smaller (d, dx, dy) lexicographically using ordered moves
        for dx, dy in moves:
            nx, ny = cx + dx, cy + dy
            if not inb(nx, ny) or (nx, ny) in obs:
                continue
            d = dist8(nx, ny, tx, ty)
            if d < bestd:
                bestd = d
                best = (dx, dy)
                bestx, besty = nx, ny
        return best if best is not None else (0, 0), (bestx, besty)

    # Candidate targets: closest resources first with deterministic ordering
    candidates = sorted(res, key=lambda r: (dist8(sx, sy, r[0], r[1]), r[0], r[1]))

    # Evaluate a target by how likely opponent can reach/contest it immediately
    def score_target(tx, ty):
        # Prefer closer to us
        d_us = dist8(sx, sy, tx, ty)
        # Opponent one-step best move toward target
        (odx, ody), (nxo, nyo) = opp_best_step(tx, ty, ox, oy)
        d_opp_after = dist8(nxo, nyo, tx, ty)
        # If opponent can get onto target now or in one step, penalize heavily
        opp_reach_now = (nxo == tx and nyo == ty)
        opp_adj = (d_opp_after <= 1)
        # Also penalize if we will both head into same neighborhood (resource_denier)
        # Our best move toward the target
        best_my = (0, 0); bestd = 10**9
        for dx, dy in moves:
            nx, ny = sx + dx, sy + dy
            if not inb(nx, ny) or (nx, ny) in obs:
                continue
            d = dist8(nx, ny, tx, ty)
            if d < bestd:
                bestd = d; best_my = (dx, dy)
        # Higher is better
        sc = -d_us
        if opp_reach_now:
            sc -= 50
        elif opp_adj:
            sc -= 20
        # Encourage moving into same cell only if beneficial
        if best_my == (0, 0) and d_us > 0:
            sc -= 3
        return sc

    best_tx, best_ty = None, None
    best_sc = -10**18
    for tx, ty in candidates[:8]:
        sc = score_target(tx, ty)
        if sc > best_sc:
            best_sc = sc
            best_tx, best_ty = tx, ty

    tx, ty = best_tx, best_ty
    # Move toward chosen target, but never into obstacles/out of bounds
    best_move = (0, 0)
    bestd = 10**9
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obs:
            continue
        # Avoid stepping away when already on target-like position
        d = dist8(nx, ny, tx, ty)
        # deterministic tie: ordered moves already gives consistent selection when d equal
        if d < bestd:
            bestd = d
            best_move = (dx, dy)

    return [int(best_move[0]), int(best_move[1])]
