def choose_move(observation):
    w = int(observation.get("grid_width") or 8)
    h = int(observation.get("grid_height") or 8)
    sp = observation.get("self_position")
    sx, sy = (sp[0], sp[1]) if isinstance(sp, (list, tuple)) and len(sp) >= 2 else (0, 0)
    obs_cells = set()
    for p in observation.get("obstacles") or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            obs_cells.add((int(p[0]), int(p[1])))
    unclaimed = observation.get("unclaimed_cells") or []
    un_set = set()
    for p in unclaimed:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            un_set.add((int(p[0]), int(p[1])))
    self_terr = set()
    for p in observation.get("self_territory") or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            self_terr.add((int(p[0]), int(p[1])))
    opp_terr = set()
    for p in observation.get("opponent_territory") or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            opp_terr.add((int(p[0]), int(p[1])))
    op = observation.get("opponent_position")
    ox, oy = (op[0], op[1]) if isinstance(op, (list, tuple)) and len(op) >= 2 else (w - 1, h - 1)

    dirs = [(-1,-1),(-1,0),(-1,1),(0,-1),(0,0),(0,1),(1,-1),(1,0),(1,1)]
    def inb(x, y):
        return 0 <= x < w and 0 <= y < h
    def dist(a, b):
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

    # Build frontier candidates from our territory (local only)
    candidates = []
    for (tx, ty) in self_terr:
        for dx, dy in dirs:
            nx, ny = tx + dx, ty + dy
            if not inb(nx, ny) or (nx, ny) in obs_cells:
                continue
            if (nx, ny) in self_terr:
                continue
            if (nx, ny) in opp_terr:
                candidates.append((nx, ny, 8))
            elif (nx, ny) in un_set:
                candidates.append((nx, ny, 5))
            else:
                # consider near-unknown as mild target if unclaimed list is incomplete
                candidates.append((nx, ny, 2))

    # Always include neighbors around current position
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if inb(nx, ny) and (nx, ny) not in obs_cells:
            if (nx, ny) in opp_terr:
                candidates.append((nx, ny, 10))
            elif (nx, ny) in un_set:
                candidates.append((nx, ny, 6))
            else:
                candidates.append((nx, ny, 1))

    if not candidates:
        return [0, 0]

    # Deterministic tie-break: prefer higher value, then closer to opponent, then closer to unclaimed center, then lexicographic
    best = None
    # Center of grid as stable anchor
    anchor = (w // 2, h // 2)
    for (tx, ty, basev) in candidates:
        if (tx, ty) in obs_cells or not inb(tx, ty):
            continue
        v = basev
        d_op = dist((tx, ty), (ox, oy))
        v += 0.05 * (-d_op)  # closer to opponent territory boundary is better
        v += 0.02 * (-dist((tx, ty), anchor))
        # Small bias toward continuing in same general direction (deterministic)
        v += 0.01 * (abs(tx - ox) < abs(sx - ox))
        key = (v, -d_op, -dist((tx, ty), anchor), tx, ty)
        if best is None or key > best[0]:
            best = (key, (tx, ty))

    tx, ty = best[1]
    dx = 0
    if tx > sx:
        dx = 1
    elif tx < sx:
        dx = -1
    dy = 0
