    for p in observation.get("obstacles") or []:

    for key in ("unclaimed_cells", "resources", "opponent_territory"):
        if targets:

    if not targets:
        targets = [(ox, oy), (w // 2, h // 2)]

        return abs(a[0] - b[0]) + abs(a[1] - b[1])
