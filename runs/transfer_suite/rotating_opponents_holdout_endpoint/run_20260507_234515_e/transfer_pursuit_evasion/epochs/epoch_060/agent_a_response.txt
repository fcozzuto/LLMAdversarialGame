    opp_role = (observation.get("opponent_role", "") or "").lower()

    def is_escape(role):
        return ("evader" in role) or ("escape" in role) or ("runner" in role)

    escape = is_escape(self_role) and not (("pursuer" in self_role) or ("hunter" in self_role))
    opp_escape = is_escape(opp_role) and not (("pursuer" in opp_role) or ("hunter" in opp_role))

    obstacles = observation.get("obstacles", []) or []
    oset = set()
    for p in obstacles:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            oset.add((int(p[0]), int(p[1])))

    resources = observation.get("resources", []) or []
    rset = set()
    for p in resources:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            rset.add((int(p[0]), int(p[1])))

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def inside(x, y):
        return 0 <= x < w and 0 <= y < h

    def occ(x, y):
        return (x, y) in oset

    def manh(x1, y1, x2, y2):
        return abs(x1 - x2) + abs(y1 - y2)

    def opp_next(px, py, tx, ty):
