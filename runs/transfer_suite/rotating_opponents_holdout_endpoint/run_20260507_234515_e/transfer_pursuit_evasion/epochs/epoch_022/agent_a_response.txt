def choose_move(observation):
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    srole = str(observation.get("self_role", "")).lower()
    is_pursuer = any(k in srole for k in ("purs", "catch", "seeker", "chase"))

    obstacles = observation.get("obstacles", []) or []
    oset = set()
    for p in obstacles:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            oset.add((int(p[0]), int(p[1])))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def cheb(x, y):
        dx = abs(x - ox)
        dy = abs(y - oy)
        return dx if dx > dy else dy

    cx = (w - 1) / 2.0
    cy = (h - 1) / 2.0

    deltas = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    best = None
    best_score = -10**18 if is_pursuer else 10**18
    best_corner = None
    corners = [(0, 0), (0, h - 1), (w - 1, 0), (w - 1, h - 1)]
    if not is_pursuer:
        # Prefer corner farthest from pursuer (opponent)
        best_corner = max(corners, key=lambda c: abs(c[0] - ox) + abs(c[1] - oy))

    for dx, dy in deltas:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in oset:
            continue
        d = cheb(nx, ny)
        # Tie-breakers: avoid edges for pursuer, for evader drift toward chosen corner
        edge = abs(nx - cx) + abs(ny - cy)
        center_bias = -edge
        if is_pursuer:
            score = (-d * 100.0) + (center_bias * 0.2)
            # Extra preference for moving in the direct direction to opponent
            score += (-(abs((nx - ox)) + abs((ny - oy))) - 0.01 * abs((nx - sx) - (ny - sy)))
        else:
            tx, ty = best_corner
            man_to_corner = abs(nx - tx) + abs(ny - ty)
            score = (d * 100.0) + (man_to_corner * 1.5) + (edge * 0.05)
        if is_pursuer:
            if score > best_score:
                best_score = score
                best = [dx, dy]
        else:
            if score < best_score:
                best_score = score
                best = [dx, dy]

    if best is None:
        return [0, 0]
    return [int(best[0]), int(best[1]]]
