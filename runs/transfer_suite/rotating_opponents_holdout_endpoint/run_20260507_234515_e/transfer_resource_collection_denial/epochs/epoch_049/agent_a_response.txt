def choose_move(observation):
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    w, h = observation["grid_width"], observation["grid_height"]
    resources = observation.get("resources") or []
    obstacles = observation.get("obstacles") or []
    obs_set = set()
    for p in obstacles:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            obs_set.add((p[0], p[1]))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obs_set

    def bfs(start):
        from collections import deque
        dist = {}
        q = deque([start])
        dist[start] = 0
        while q:
            x, y = q.popleft()
            d = dist[(x, y)]
            for dx in (-1, 0, 1):
                for dy in (-1, 0, 1):
                    if dx == 0 and dy == 0:
                        continue
                    nx, ny = x + dx, y + dy
                    if inb(nx, ny) and (nx, ny) not in dist:
                        dist[(nx, ny)] = d + 1
                        q.append((nx, ny))
        return dist

    legal = []
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            nx, ny = sx + dx, sy + dy
            if inb(nx, ny):
                legal.append((dx, dy))
    if not legal:
        return [0, 0]

    if resources:
        ds = bfs((sx, sy))
        do = bfs((ox, oy))
        best = None
        best_key = None
        for rx, ry in resources:
            rs = ds.get((rx, ry))
            ro = do.get((rx, ry))
            if rs is None and ro is None:
                continue
            if rs is None:
                key = (-10**9, 10**9)
            elif ro is None:
                key = (10**9, rs)
            else:
                key = (ro - rs, -rs)
            if best_key is None or key > best_key:
                best_key = key
                best = (rx, ry)
        if best is None:
            target = (w // 2, h // 2)
        else:
            target = best
    else:
        target = (w // 2, h // 2)

    # Pick move that maximizes advantage toward chosen target (using BFS distances from next cell)
    ds_next = None
    dmap_next = None
    if resources:
        d_s = bfs((sx, sy))
        d_o = bfs((ox, oy))
    best_move = (0, 0)
    best_score = -10**18
    for dx, dy in legal:
        nx, ny = sx + dx, sy + dy
        ns = bfs((nx, ny)) if resources else None
        if resources:
            rs = ns.get(target)
            ro = do.get(target)
            if rs is None:
                score = -10**12
            elif ro is None:
                score = 10**6 - rs
            else:
                score = (ro - rs) * 1000 - rs
        else:
            score = -max(abs(nx - target[0]), abs(ny - target[1])) + 0.1 * max(abs(nx - ox), abs(ny - oy))
        if score > best_score:
            best_score = score
            best_move = (dx, dy)
    return [int(best_move[0]), int(best_move[1])]

def choose_move(observation):
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    w, h = observation["grid_width"], observation["grid_height"]
    resources = observation.get("resources") or []
    obstacles = observation.get("obstacles") or []

    obs = set()
    for p in obstacles:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            obs.add((p[0], p[1]))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obs

    def man(a, b, c, d):
        return abs(a - c) + abs(b - d)

    def nearest_res_dist(x, y):
        if not resources:
            return 10**9
        md = 10**9
        for r in resources:
            if isinstance(r, (list, tuple)) and len(r) >= 2:
                rx, ry = r[0], r[1]
                if inb(rx, ry) or (rx, ry) == (x, y):
                    d = man(x, y, rx, ry)
                    if d < md:
                        md = d
        return md

    best_move = [0, 0]
    best_score = -10**18
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            if dx == 0 and dy == 0:
                nx, ny = sx, sy
            else:
                nx, ny = sx + dx, sy + dy
            if not inb(nx, ny):
                continue
            dres = nearest_res_dist(nx, ny)
            dop = man(nx, ny, ox, oy)
            # Encourage resource acquisition, keep distance from opponent.
            score = (-3.0 * dres) + (0.7 * dop)
            # Prefer moves that reduce resource distance if possible.
            cur_dres = nearest_res_dist(sx, sy)
            if dres < cur_dres:
                score += 1.0
            if dop > man(sx, sy, ox, oy):
                score += 0.5
            if score > best_score:
                best_score = score
                best_move = [dx, dy]
