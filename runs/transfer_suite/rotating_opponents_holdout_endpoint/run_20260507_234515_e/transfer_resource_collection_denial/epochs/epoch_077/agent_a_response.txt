def choose_move(observation):
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    w, h = observation["grid_width"], observation["grid_height"]
    resources = observation.get("resources") or []
    obstacles = observation.get("obstacles") or []
    obs = set()
    for p in obstacles:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            obs.add((int(p[0]), int(p[1])))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obs

    def cheb(ax, ay, bx, by):
        dx = ax - bx
        if dx < 0: dx = -dx
        dy = ay - by
        if dy < 0: dy = -dy
        return dx if dx > dy else dy

    def man(ax, ay, bx, by):
        dx = ax - bx
        if dx < 0: dx = -dx
        dy = ay - by
        if dy < 0: dy = -dy
        return dx + dy

    res_list = []
    for r in resources:
        if isinstance(r, (list, tuple)) and len(r) >= 2:
            x, y = int(r[0]), int(r[1])
            if inb(x, y) and (x != sx or y != sy):
                res_list.append((x, y))

    moves = []
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            nx, ny = sx + dx, sy + dy
            if inb(nx, ny):
                moves.append((dx, dy, nx, ny))
    if not moves:
        return [0, 0]

    # If resources exist, pick move that maximizes "chance to reach a resource before opponent".
    if res_list:
        best = None
        best_val = -10**18
        for dx, dy, nx, ny in moves:
            # Evaluate best resource for us from this next cell
            # Prefer resources where we are not farther than opponent.
            best_cell = -10**18
            for rx, ry in res_list:
                ds = cheb(nx, ny, rx, ry)
                do = cheb(ox, oy, rx, ry)
                # tie-break bias: being closer than opponent is high priority
                lead = do - ds
                # encourage moving toward nearby resources and away from extreme delay
                proximity = -man(nx, ny, rx, ry)
                value = lead * 1000 + proximity
                # small penalty if we are already behind by a lot
                if lead < 0:
                    value -= (-lead) * 50
                if value > best_cell:
                    best_cell = value
            # Also slightly avoid letting opponent potentially steal the closest "contested" resource
            contested = 0
            for rx, ry in res_list[:]:
                ds = cheb(nx, ny, rx, ry)
                do = cheb(ox, oy, rx, ry)
                if do <= ds and cheb(sx, sy, rx, ry) > do:
                    contested += 1
            total = best_cell - contested * 10
            if total > best_val:
                best_val = total
                best = (dx, dy)
        return [int(best[0]), int(best[1])]

    # No resources: move to center to be ready for respawns / reduce worst-case.
    tx, ty = (w - 1) // 2, (h - 1) // 2
    best = (0, 0, sx, sy)
    bestd = 10**9
    for dx, dy, nx, ny in moves:
        d = cheb(nx, ny, tx, ty)
        if d < bestd:
            bestd = d
            best = (dx, dy, nx, ny)
    return [int(best[0]), int(best[1])]
