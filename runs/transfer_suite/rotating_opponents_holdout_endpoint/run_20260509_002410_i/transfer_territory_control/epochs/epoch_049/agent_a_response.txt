        def adj_opp(p):
            x, y = p
            for ax in (-1, 0, 1):
                for ay in (-1, 0, 1):
                    if ax == 0 and ay == 0:
                        continue
                    if (x + ax, y + ay) in opp_t:
                        return 1
            return 0

        tx, ty = sorted(unclaimed, key=lambda p: (-adj_opp(p), abs(p[0] - sx) + abs(p[1] - sy), p[0], p[1]))[0]
    elif opp_t:
        tx, ty = sorted(opp_t, key=lambda p: (abs(p[0] - sx) + abs(p[1] - sy), p[0], p[1]))[0]
    else:
        tx, ty = (w - 1) // 2, (h - 1) // 2

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    best = (10**9, None)
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not ok(nx, ny):
            continue
        dist = (abs(nx - tx) + abs(ny - ty))
        near_opp = 0
        for ax in (-1, 0, 1):
            for ay in (-1, 0, 1):
                if ax == 0 and ay == 0:
                    continue
                if (nx + ax, ny + ay) in opp_t:
                    near_opp = 1
        score = (dist, -near_opp, nx, ny)
        if score < best[0]:
            best = (score, [dx, dy])

    if best[1] is None:
        return [0, 0]
    return best[1]
