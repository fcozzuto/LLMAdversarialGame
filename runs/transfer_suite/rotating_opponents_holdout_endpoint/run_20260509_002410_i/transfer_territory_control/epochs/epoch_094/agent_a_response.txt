def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position") or (0, 0)
    ox, oy = observation.get("opponent_position") or (sx, sy)
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    obstacles = set()
    for p in observation.get("obstacles") or []:
        if p and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obstacles

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    # candidates: frontier around opponent territory + center bias + unclaimed fallback
    opp_terr = [(int(x), int(y)) for x, y in (observation.get("opponent_territory") or [])]
    unclaimed = [(int(x), int(y)) for x, y in (observation.get("unclaimed_cells") or [])]
    opp_set = set(opp_terr)
    un_set = set(unclaimed)

    center = (w // 2, h // 2)
    cand = []
    if opp_terr:
        for x, y in opp_terr:
            for dx, dy in moves:
                nx, ny = x + dx, y + dy
                if inb(nx, ny) and ((nx, ny) in opp_set or (nx, ny) in un_set):
                    # value: prefer entering opponent territory, otherwise unclaimed; also prefer closeness to center
                    d = abs(nx - sx) + abs(ny - sy)
                    dc = abs(nx - center[0]) + abs(ny - center[1])
                    v = (5.0 if (nx, ny) in opp_set else 2.0) - 1.2 * d - 0.15 * dc
                    cand.append((v, nx, ny))
    if not cand:
        for x, y in unclaimed:
            if inb(x, y):
                d = abs(x - sx) + abs(y - sy)
                dc = abs(x - center[0]) + abs(y - center[1])
                cand.append((3.0 - 1.1 * d - 0.2 * dc, x, y))
    if not cand:
        # final fallback: approach center
        tx, ty = center
    else:
        cand.sort(key=lambda t: (-t[0], abs(t[1] - center[0]) + abs(t[2] - center[1]), t[1], t[2]))
        _, tx, ty = cand[0]

    # Choose a one-step move that reduces dist to chosen target (prefer obstacle-free)
    best = (10**9, 0, 0)
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny):
            continue
        d = abs(nx - tx) + abs(ny - ty)
        # prefer capturing immediate neighbor target types
        bonus = 0.0
        if (nx, ny) in opp_set:
            bonus += 2.0
        elif (nx, ny) in un_set:
            bonus += 0.6
        # tie-break: steer toward target deterministically
        key = (d - bonus, abs(nx - center[0]) + abs(ny - center[1]), nx, ny)
        if key < best:
            best = (key, dx, dy)
    if best[1] == 0 and best[2] == 0:
        return [0, 0]
    return [int(best[1]), int(best[2])]
