def choose_move(observation):
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    sx, sy = observation.get("self_position", (0, 0))
    ox, oy = observation.get("opponent_position", (w - 1, h - 1))

    obstacles = set()
    for p in observation.get("obstacles") or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            obstacles.add((int(p[0]), int(p[1])))

    self_t = set((int(p[0]), int(p[1])) for p in (observation.get("self_territory") or []))
    opp_t = set((int(p[0]), int(p[1])) for p in (observation.get("opponent_territory") or []))
    unclaimed = set((int(p[0]), int(p[1])) for p in (observation.get("unclaimed_cells") or []))

    cx, cy = (w - 1) / 2.0, (h - 1) / 2.0

    def ok(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obstacles

    def md(a, b):
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    best = (0, 0)
    bestv = -10**18

    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not ok(nx, ny):
            continue

        v = 0.0
        if (nx, ny) in opp_t:
            v += 12.0
        elif (nx, ny) in unclaimed:
            v += 5.0
        else:
            v += 0.5

        dcent_old = abs(sx - cx) + abs(sy - cy)
        dcent_new = abs(nx - cx) + abs(ny - cy)
        v += (dcent_old - dcent_new) * 0.6

        # If it helps our expansion (adjacent to our territory), value it more.
        neigh = 0
        opp_neigh = 0
        for ax in (-1, 0, 1):
            for ay in (-1, 0, 1):
                if ax == 0 and ay == 0:
                    continue
                px, py = nx + ax, ny + ay
                if 0 <= px < w and 0 <= py < h and (px, py) not in obstacles:
                    if (px, py) in self_t:
                        neigh += 1
                    if (px, py) in opp_t:
                        opp_neigh += 1
        v += neigh * 1.2
        v -= opp_neigh * 1.0

        # Prefer moves that increase distance from opponent unless we are invading.
        if (nx, ny) not in opp_t:
            v += (md((nx, ny), (ox, oy)) - md((sx, sy), (ox, oy))) * 0.35

        # Small deterministic tie-break: closer to center then lexicographic
        tie = (-(abs(nx - cx) + abs(ny - cy)), nx, ny)
        cand = (v, tie)
        if cand > (bestv, None):
            bestv = v
            best = (dx, dy)

    return [int(best[0]), int(best[1])]
