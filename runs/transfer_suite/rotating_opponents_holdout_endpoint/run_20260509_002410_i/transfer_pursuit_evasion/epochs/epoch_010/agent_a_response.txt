def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [w - 1, h - 1])
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    obs = set()
    for p in observation.get("obstacles", []) or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            obs.add((int(p[0]), int(p[1])))

    role = str(observation.get("self_role", "")).lower()
    pursuer = ("pursuer" in role) or ("pursue" in role) or ("pursuit" in role) or role == "pursuer"

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def bfs_dist(start, target):
        if start in obs or target in obs:
            return None
        q = [target]
        dist = {target: 0}
        qi = 0
        while qi < len(q):
            x, y = q[qi]
            qi += 1
            nd = dist[(x, y)] + 1
            for dx, dy in moves:
                nx, ny = x + dx, y + dy
                if not inb(nx, ny) or (nx, ny) in obs or (nx, ny) in dist:
                    continue
                dist[(nx, ny)] = nd
                if (nx, ny) == start:
                    return nd
                q.append((nx, ny))
        return dist.get(start)

    # Deterministic tie-break: prefer staying still last, then smallest dx, then smallest dy.
    best = (10**9, None, 10**9, 10**9)  # (score, keep, dist, manhattan)
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obs:
            continue
        d = bfs_dist((nx, ny), (ox, oy))
        if d is None:
            continue
        man = abs(nx - ox) + abs(ny - oy)
        if pursuer:
            score = d * 1000 + man
            key = (score, (dx == 0 and dy == 0), d, man)
            if key < best:
                best = key
                best_move = [dx, dy]
        else:
            # Evader: maximize shortest path distance to pursuer; if equal, maximize Euclidean/Manhattan.
            score = -d * 1000 - man
            key = (score, (dx == 0 and dy == 0), -d, -man)
            if key < best:
                best = key
                best_move = [dx, dy]

    # If BFS filtering removed everything (shouldn't), fall back to direct move heuristic.
    if 'best_move' not in locals():
        best_move = [0, 0]
        best_val = -10**18 if pursuer else 10**18
        for dx, dy in moves:
            nx, ny = sx + dx, sy + dy
            if not inb(nx, ny) or (nx, ny) in obs:
                continue
            dd = (nx - ox) * (nx - ox) + (ny - oy) * (ny - oy)
            val = -dd if pursuer else dd
            if (pursuer and val > best_val) or ((not pursuer) and val < best_val):
                best_val = val
                best_move = [dx, dy]

    return [int(best_move[0]), int(best_move[1])]
