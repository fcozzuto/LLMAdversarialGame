def choose_move(observation):
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    sx = int(sx); sy = int(sy); ox = int(ox); oy = int(oy)

    obstacles = set()
    for p in observation.get("obstacles", []) or []:
        if p and len(p) >= 2:
            x = int(p[0]); y = int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

    resources = []
    for r in observation.get("resources", []) or []:
        if r and len(r) >= 2:
            x = int(r[0]); y = int(r[1])
            if 0 <= x < w and 0 <= y < h and (x, y) not in obstacles:
                resources.append((x, y))
    if not resources:
        return [0, 0]

    def cheb(ax, ay, bx, by):
        dx = ax - bx
        if dx < 0: dx = -dx
        dy = ay - by
        if dy < 0: dy = -dy
        return dx if dx > dy else dy

    # Deterministic greedy target with opponent-aware bias.
    best = None
    best_key = None
    for rx, ry in resources:
        my_d = cheb(sx, sy, rx, ry)
        op_d = cheb(ox, oy, rx, ry)
        # Prefer resources we can reach earlier; also encourage shorter paths.
        # Key is minimized lexicographically.
        score = 1.6 * my_d - 0.6 * op_d
        tie = (my_d - op_d)  # more positive means further behind; prefer smaller
        key = (score, my_d, tie, rx, ry)
        if best_key is None or key < best_key:
            best_key = key
            best = (rx, ry)

    tx, ty = best

    def sign(v):
        if v > 0: return 1
        if v < 0: return -1
        return 0

    # Candidate moves preference: diagonal toward target, then axis toward target, then staying.
    cand = []
    dx0 = sign(tx - sx)
    dy0 = sign(ty - sy)
    if dx0 != 0 and dy0 != 0:
        cand.append((dx0, dy0))
    if dx0 != 0:
        cand.append((dx0, 0))
    if dy0 != 0:
        cand.append((0, dy0))
    cand.append((dx0, dy0) if (dx0, dy0) not in cand else (0, 0))
    cand.append((0, 0))
    # Fill remaining 8 dirs deterministically for obstacle avoidance.
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            if (dx, dy) not in cand:
                cand.append((dx, dy))

    def in_bounds(x, y):
        return 0 <= x < w and 0 <= y < h

    # Choose best candidate by minimizing objective at next step.
    best_move = (0, 0)
    best_val = None
    for dx, dy in cand[:9]:
        nx = sx + dx
        ny = sy + dy
        if not in_bounds(nx, ny) or (nx, ny) in obstacles:
            continue
        my_d = cheb(nx, ny, tx, ty)
        op_d = cheb(ox, oy, tx, ty)
        # Slightly penalize moving away from target.
        val = 1.6 * my_d - 0.6 * op_d + 0.05 * (abs(tx - nx) + abs(ty - ny))
        key = (val, abs(tx - nx) + abs(ty - ny), dx, dy)
        if best_val is None or key < best_val:
            best_val = key
            best_move = (dx, dy)

    dx, dy = best_move
    return [int(dx), int(dy)]
