def choose_move(observation):
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    w = observation.get("grid_width", 8)
    h = observation.get("grid_height", 8)

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    obstacles = set()
    for p in observation.get("obstacles", []) or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = p[0], p[1]
            if inb(x, y):
                obstacles.add((x, y))

    resources = []
    for p in observation.get("resources", []) or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = p[0], p[1]
            if inb(x, y) and (x, y) not in obstacles:
                resources.append((x, y))

    legal = []
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            nx, ny = sx + dx, sy + dy
            if inb(nx, ny) and (nx, ny) not in obstacles:
                legal.append((dx, dy))
    if not legal:
        return [0, 0]

    def mdist(x1, y1, x2, y2):
        d1 = x1 - x2
        if d1 < 0:
            d1 = -d1
        d2 = y1 - y2
        if d2 < 0:
            d2 = -d2
        return d1 + d2

    # If no visible resources, move toward a corner farthest from opponent to reduce their capture options.
    if not resources:
        corners = [(0, 0), (0, h - 1), (w - 1, 0), (w - 1, h - 1)]
        target = max(corners, key=lambda c: mdist(c[0], c[1], ox, oy))
        best = [0, 0]
        bestv = None
        for dx, dy in legal:
            nx, ny = sx + dx, sy + dy
            v = mdist(nx, ny, target[0], target[1])
            if bestv is None or v < bestv or (v == bestv and (dx, dy) < tuple(best)):
                bestv = v
                best = [dx, dy]
        return best

    # Choose move that maximizes capture advantage: (opp_dist - our_dist), with tie-break on earliest/closest.
    best_move = None
    best_score = None
    best_t = None
    # Deterministic tie-break order for moves
    move_order = sorted(legal, key=lambda m: (m[0], m[1]))

    for dx, dy in move_order:
        nx, ny = sx + dx, sy + dy
        worst_opponent = -10**9
        best_adv_for_move = -10**9
        best_res_t = None
        for rx, ry in resources:
            od = mdist(ox, oy, rx, ry)
            nd = mdist(nx, ny, rx, ry)
            adv = od - nd  # higher is better: we get there first (or opponent later)
            # small bias: avoid huge nd if adv ties; deterministic using resource ordering by coordinates
            tie_bias = 0
            if (nd, rx, ry) == (0, rx, ry):
                tie_bias = 0
            if adv > best_adv_for_move or (adv == best_adv_for_move and (nd < worst_opponent) ):
                best_adv_for_move = adv
                best_res_t = (nd, rx, ry)
            # Track "difficulty": reduce opponent's best possible advantage after our move
            if adv > worst_opponent:
                worst_opponent = adv

        # Primary: maximize our best advantage over resources; Secondary: minimize our distance to that resource
        if best_res_t is None:
            continue
        nd, rx, ry = best_res_t
        score = best_adv_for_move
        t = (nd, rx, ry)
        if best_score is None or score > best_score or (score == best_score and (t < best_t)):
            best_score = score
            best_t = t
            best_move = [dx, dy]

    if best_move is None:
        return [0, 0]
    return best_move
