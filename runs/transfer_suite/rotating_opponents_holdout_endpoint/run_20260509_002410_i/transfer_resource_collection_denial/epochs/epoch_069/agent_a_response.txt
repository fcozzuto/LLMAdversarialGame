def choose_move(observation):
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    sx = int(sx); sy = int(sy); ox = int(ox); oy = int(oy)

    obstacles = set()
    for p in observation.get("obstacles") or []:
        if p and len(p) >= 2:
            x = int(p[0]); y = int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

    resources = []
    for r in observation.get("resources") or []:
        if r and len(r) >= 2:
            x = int(r[0]); y = int(r[1])
            if 0 <= x < w and 0 <= y < h:
                resources.append((x, y))

    if not resources:
        return [0, 0]

    def cheb(ax, ay, bx, by):
        dx = ax - bx
        if dx < 0: dx = -dx
        dy = ay - by
        if dy < 0: dy = -dy
        return dx if dx > dy else dy

    best = None
    best_key = None
    # New policy: strongly prioritize resources I'm closer to; if all are contested, "intercept" the least-contested.
    for rx, ry in resources:
        my_d = cheb(sx, sy, rx, ry)
        op_d = cheb(ox, oy, rx, ry)
        contested = 1 if op_d < my_d else 0
        # Prefer: (op_d - my_d) large; then smaller my_d; then deterministic coords.
        key = (op_d - my_d, -contested, -my_d, -rx, -ry)
        if best_key is None or key > best_key:
            best_key = key
            best = (rx, ry)

    tx, ty = best
    def sign(a, b):
        if a == b: return 0
        return 1 if b > a else -1

    def is_blocked(nx, ny):
        return (nx, ny) in obstacles

    dx = sign(sx, tx)
    dy = sign(sy, ty)

    candidates = []
    candidates.append((dx, dy))
    candidates.append((dx, 0))
    candidates.append((0, dy))
    if dx != 0 and dy != 0:
        candidates.append((dx, -dy))
        candidates.append((-dx, dy))
    candidates.append((0, 0))

    # Deterministic tie-break: choose first candidate that reduces chebyshev distance and isn't blocked when possible.
    cur_d = cheb(sx, sy, tx, ty)
    for ddx, ddy in candidates:
        nx = sx + ddx
        ny = sy + ddy
        if nx < 0 or nx >= w or ny < 0 or ny >= h:
            continue
        nb = is_blocked(nx, ny)
        nd = cheb(nx, ny, tx, ty)
        if nb:
            continue
        if nd <= cur_d:
            return [ddx, ddy]

    # If all "good" steps are blocked, allow staying or a blocked step that maximizes distance reduction deterministically.
    best_move = (0, 0)
    best_nd = -10**9
    for ddx, ddy in candidates:
        nx = sx + ddx
        ny = sy + ddy
        if nx < 0 or nx >= w or ny < 0 or ny >= h:
            continue
        nd = cheb(nx, ny, tx, ty)
        val = cur_d - nd  # reduction
        if val > best_nd:
            best_nd = val
            best_move = (ddx, ddy)
    return [int(best_move[0]), int(best_move[1])]
