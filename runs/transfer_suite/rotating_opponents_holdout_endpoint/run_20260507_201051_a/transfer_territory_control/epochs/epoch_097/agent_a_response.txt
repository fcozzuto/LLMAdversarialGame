def choose_move(observation):
    w = int(observation.get("grid_width") or 8)
    h = int(observation.get("grid_height") or 8)
    sx, sy = observation.get("self_position") or (0, 0)
    ox, oy = observation.get("opponent_position") or (w - 1, h - 1)
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    obst = set()
    for p in observation.get("obstacles") or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obst.add((x, y))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obst

    unclaimed = observation.get("unclaimed_cells") or []
    cand = []
    for c in unclaimed:
        if isinstance(c, (list, tuple)) and len(c) >= 2:
            x, y = int(c[0]), int(c[1])
            if inb(x, y):
                edge = (x == 0 or x == w - 1 or y == 0 or y == h - 1)
                cand.append((x, y, edge))

    if not cand:
        # Fallback: go to nearest safe neighbor that reduces distance to opponent's position
        moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
        best = (10**18, (0, 0))
        for dx, dy in moves:
            nx, ny = sx + dx, sy + dy
            if inb(nx, ny):
                d = abs(nx - ox) + abs(ny - oy)
                if (d, nx, ny) < best:
                    best = (d, (dx, dy))
        return list(best[1])

    edges = [t for t in cand if t[2]]
    cand = edges if edges else cand

    best = None
    # Maximize: (distance from opponent) - (distance from self) + edge priority
    for x, y, edge in cand:
        dso = abs(x - sx) + abs(y - sy)
        dpo = abs(x - ox) + abs(y - oy)
        score = (dpo - dso) + (5 if edge else 0)
        key = (score, -dso, -dpo, x, y)
        if best is None or key > best[0]:
            best = (key, (x, y))

    tx, ty = best[1]
    dx = 0 if tx == sx else (1 if tx > sx else -1)
    dy = 0 if ty == sy else (1 if ty > sy else -1)

    nx, ny = sx + dx, sy + dy
    if not inb(nx, ny):
        # Deterministic obstacle handling: try axis-aligned alternatives
        opts = [(dx, 0), (0, dy), (0, 0)]
        if dx != 0 and dy != 0:
            opts = [(dx, dy), (dx, 0), (0, dy), (0, 0)]
        for mx, my in opts:
            tx2, ty2 = sx + mx, sy + my
            if inb(tx2, ty2):
                return [int(mx), int(my)]
        return [0, 0]
    return [int(dx), int(dy)]
