def choose_move(observation):
    w = int(observation["grid_width"]); h = int(observation["grid_height"])
    sx, sy = observation["self_position"]
    resources = observation.get("resources") or []
    obstacles = observation.get("obstacles") or []
    # parse obstacles/resources as (x,y)
    obs = set()
    for p in obstacles:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obs.add((x, y))
    res = []
    for r in resources:
        if isinstance(r, (list, tuple)) and len(r) >= 2:
            x, y = int(r[0]), int(r[1])
            if 0 <= x < w and 0 <= y < h:
                res.append((x, y))
        elif isinstance(r, dict):
            pos = r.get("position", r.get("pos", None))
            if isinstance(pos, (list, tuple)) and len(pos) >= 2:
                x, y = int(pos[0]), int(pos[1])
                if 0 <= x < w and 0 <= y < h:
                    res.append((x, y))
    if not res:
        return [0, 0]
    ox, oy = observation["opponent_position"]

    def dist(a, b):
        ax, ay = a; bx, by = b
        dx = abs(bx - ax); dy = abs(by - ay)
        return dx if dx > dy else dy  # Chebyshev for diagonal grid

    # choose best target resource considering who can reach sooner
    best_target = res[0]
    best_val = -10**9
    for tx, ty in res:
        if (tx, ty) in obs:
            continue
        ds = dist((sx, sy), (tx, ty))
        do = dist((ox, oy), (tx, ty))
        # Prefer resources where we are closer; tie-break toward closer absolute distance and higher "promise"
        val = (do - ds) * 10 - ds
        # discourage targets behind obstacles a bit (local heuristic: if direct line steps frequently hit obstacles)
        steps = max(abs(tx - sx), abs(ty - sy))
        if steps > 0:
            hits = 0
            cx, cy = sx, sy
            dx = 0 if tx == sx else (1 if tx > sx else -1)
            dy = 0 if ty == sy else (1 if ty > sy else -1)
            for _ in range(min(steps, 4)):
                nx, ny = cx + dx, cy + dy
                if (nx, ny) in obs:
                    hits += 1
                cx, cy = nx, ny
            val -= hits * 6
        if val > best_val or (val == best_val and (dist((sx, sy), (tx, ty)) < dist((sx, sy), best_target))):
            best_val = val
            best_target = (tx, ty)

    # move one step toward best_target; avoid obstacles if possible
    tx, ty = best_target
    best_moves = []
    best_score = -10**9
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            nx, ny = sx + dx, sy + dy
            if not (0 <= nx < w and 0 <= ny < h):
                continue
            if (nx, ny) in obs:
                continue
            # if we can stay on a resource, do it (collect)
            val = 0
            if (nx, ny) == (tx, ty):
                val += 1000
            # primary: reduce distance to target
            val += (dist((ox, oy), (tx, ty)) - dist((nx, ny), (tx, ty))) * 10
            val += -(dist((nx, ny), (tx, ty)))
            # secondary: keep moving generally toward target
            val += -(abs(tx - nx) + abs(ty - ny)) * 0.01
            if val > best_score:
                best_score = val
                best_moves = [[dx, dy]]
            elif val == best_score:
                best_moves.append([dx, dy])

    # deterministic tie-break: prefer diagonal, then right, then up, then stay
    if not best_moves:
        return [0, 0]
    def key(m):
        dx, dy = m
        diag = 1 if dx != 0 and dy != 0 else 0
        return (-diag, -dx, -dy)
    best_moves.sort(key=key)
    return best_moves[0]
