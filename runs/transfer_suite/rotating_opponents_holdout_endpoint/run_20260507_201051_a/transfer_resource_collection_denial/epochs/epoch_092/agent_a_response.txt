def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sp = observation.get("self_position", (0, 0))
    op = observation.get("opponent_position", (w - 1, h - 1))
    sx, sy = int(sp[0]), int(sp[1])
    ox, oy = int(op[0]), int(op[1])

    obstacles = set()
    for p in observation.get("obstacles") or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

    resources = []
    for p in observation.get("resources") or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h and (x, y) not in obstacles:
                resources.append((x, y))
    if not resources:
        return [0, 0]

    def man(a, b, c, d):
        x = a - c
        if x < 0: x = -x
        y = b - d
        if y < 0: y = -y
        return x + y

    # Choose target: maximize being closer than opponent; prefer nearer and corner-ish
    best = None
    best_key = None
    for rx, ry in resources:
        ds = man(sx, sy, rx, ry)
        do = man(ox, oy, rx, ry)
        lead = do - ds  # positive => we are closer
        corner = min(rx, w - 1 - rx) + min(ry, h - 1 - ry)
        # If opponent is much closer, deprioritize; still allow if we are not too far.
        key = (-(lead), ds, corner)
        # Prefer larger lead primarily:
        key = (-(lead), ds, corner)
        if best_key is None or key < best_key:
            best_key = key
            best = (rx, ry)

    tx, ty = best
    # Next step toward target, with local obstacle avoidance
    dx = 0 if tx == sx else (1 if tx > sx else -1)
    dy = 0 if ty == sy else (1 if ty > sy else -1)
    candidates = []
    # Preferred: diagonal/straight toward target
    candidates.append((dx, dy))
    # Then axis steps
    candidates.append((dx, 0))
    candidates.append((0, dy))
    # Then small detours
    candidates.append((dx, -dy))
    candidates.append((-dx, dy))
    candidates.append((-dx, 0))
    candidates.append((0, -dy))
    candidates.append((0, 0))

    def safe_step(step):
        nx, ny = sx + step[0], sy + step[1]
        if nx < 0 or nx >= w or ny < 0 or ny >= h:
            return False
        if (nx, ny) in obstacles:
            return False
        return True

    best_step = (0, 0)
    best_dist = None
    # Deterministically pick the step minimizing distance to target; tie-break by fewer moves (avoid 0)
    for step in candidates:
        if not safe_step(step):
            continue
        nx, ny = sx + step[0], sy + step[1]
        d = man(nx, ny, tx, ty)
        move_mag = abs(step[0]) + abs(step[1])
        key = (d, -move_mag)
        if best_dist is None or key < best_dist:
            best_dist = key
            best_step = step

    if not safe_step(best_step):
        return [0, 0]
    return [int(best_step[0]), int(best_step[1])]
