def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    self_role = (observation.get("self_role", "") or "").lower()
    opponent_role = (observation.get("opponent_role", "") or "").lower()
    self_is_pursuer = ("purs" in self_role) or ("hunter" in self_role) or ("chaser" in self_role)
    if not self_is_pursuer and (("purs" in opponent_role) or ("hunter" in opponent_role) or ("chaser" in opponent_role)):
        self_is_pursuer = False
    obstacles = observation.get("obstacles", []) or []
    obs = set()
    for p in obstacles:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            obs.add((int(p[0]), int(p[1])))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def local_block(x, y):
        c = 0
        for dx in (-1, 0, 1):
            for dy in (-1, 0, 1):
                if dx == 0 and dy == 0:
                    continue
                nx, ny = x + dx, y + dy
                if 0 <= nx < w and 0 <= ny < h and (nx, ny) in obs:
                    c += 1
        return c

    corner_targets = [(0, 0), (0, h - 1), (w - 1, 0), (w - 1, h - 1)]
    my_corner = min(corner_targets, key=lambda t: (sx - t[0]) * (sx - t[0]) + (sy - t[1]) * (sy - t[1]))
    opp_corner = min(corner_targets, key=lambda t: (ox - t[0]) * (ox - t[0]) + (oy - t[1]) * (oy - t[1]))

    best = None
    best_score = None
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny):
            continue
        if (nx, ny) in obs:
            continue
        dist2 = (nx - ox) * (nx - ox) + (ny - oy) * (ny - oy)
        block = local_block(nx, ny)
        wall_pull = 0
        if nx in (0, w - 1) or ny in (0, h - 1):
            wall_pull = 1
        if self_is_pursuer:
            score = dist2 * 100 + (10 - block) * 5 - wall_pull
        else:
            # Evader: run away from pursuer; prefer moving toward the farther corner while avoiding obstacle-dense neighborhoods
            target = my_corner if ((my_corner[0] - ox) ** 2 + (my_corner[1] - oy) ** 2) >= ((opp_corner[0] - ox) ** 2 + (opp_corner[1] - oy) ** 2) else opp_corner)
            to_target2 = (nx - target[0]) * (nx - target[0]) + (ny - target[1]) * (ny - target[1])
            score = -dist2 * 100 + block * 8 - wall_pull * 2 + to_target2
        if best is None or (score < best_score):
            best_score = score
            best = (dx, dy)

    if best is None:
        return [0, 0]
    return [best[0], best[1]]
