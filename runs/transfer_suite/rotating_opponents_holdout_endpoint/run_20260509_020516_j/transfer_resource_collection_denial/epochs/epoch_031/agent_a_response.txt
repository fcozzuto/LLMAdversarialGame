def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position", [0, 0]) or [0, 0]
    ox, oy = observation.get("opponent_position", [0, 0]) or [0, 0]
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    obstacles = set()
    for p in observation.get("obstacles", None) or []:
        if p is not None and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

    resources = []
    for r in observation.get("resources", None) or []:
        if r is not None and len(r) >= 2:
            x, y = int(r[0]), int(r[1])
            if 0 <= x < w and 0 <= y < h and (x, y) not in obstacles:
                resources.append((x, y))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def d(a, b):
        return max(abs(a[0] - b[0]), abs(a[1] - b[1]))  # king/diagonal steps

    # If no visible resources: keep distance from opponent while moving toward center.
    if not resources:
        cx, cy = w // 2, h // 2
        tx = cx if sx == ox else (sx - 1 if ox > sx else sx + 1)
        ty = cy if sy == oy else (sy - 1 if oy > sy else sy + 1)
        if tx < 0 or tx >= w: tx = cx
        if ty < 0 or ty >= h: ty = cy
        tx = max(0, min(w - 1, tx))
        ty = max(0, min(h - 1, ty))
        dx = 0 if tx == sx else (1 if tx > sx else -1)
        dy = 0 if ty == sy else (1 if ty > sy else -1)
        if inb(sx + dx, sy + dy):
            return [dx, dy]
        return [0, 0]

    # Strategy: avoid likely sweep-row contention by deprioritizing resources on opponent's current row.
    # Score favors: self can arrive sooner (lower d_self), and opponent later (higher d_opp),
    # plus a strong penalty if on opponent's row.
    best = None
    best_val = None
    for (rx, ry) in resources:
        ds = d((sx, sy), (rx, ry))
        do = d((ox, oy), (rx, ry))
        row_pen = 3.5 if ry == oy else 0.0  # sweep_rows tends to pressure current row
        col_pen = 1.0 if rx == ox else 0.0  # slight extra contention if same column
        val = (ds - do) + row_pen + col_pen
        # tie-break: smaller ds
        key = (val, ds)
        if best_val is None or key < best_val:
            best_val = key
            best = (rx, ry)

    rx, ry = best
    dx = 0 if rx == sx else (1 if rx > sx else -1)
    dy = 0 if ry == sy else (1 if ry > sy else -1)

    # If immediate diagonal step hits obstacle, try axis-aligned deterministic alternative.
    nx, ny = sx + dx, sy + dy
    if (nx, ny) in obstacles or not inb(nx, ny):
        if dx != 0:
            ax, ay = sx + dx, sy
            if inb(ax, ay) and (ax, ay) not in obstacles:
                return [dx, 0]
        if dy != 0:
            ax, ay = sx, sy + dy
            if inb(ax, ay) and (ax, ay) not in obstacles:
                return [0, dy]
        # last resort: move toward target without diagonals if possible
        if dx != 0:
            ax = sx + dx
