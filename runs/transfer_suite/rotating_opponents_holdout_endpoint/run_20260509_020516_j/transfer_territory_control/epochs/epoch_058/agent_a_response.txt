def choose_move(observation):
    W = int(observation.get("grid_width", 8) or 8)
    H = int(observation.get("grid_height", 8) or 8)
    sp = observation.get("self_position") or [0, 0]
    sx, sy = int(sp[0]), int(sp[1])

    def to_set(v):
        s = set()
        for p in v or []:
            if p is None or len(p) < 2:
                continue
            x, y = int(p[0]), int(p[1])
            if 0 <= x < W and 0 <= y < H:
                s.add((x, y))
        return s

    obstacles = to_set(observation.get("obstacles"))
    resources = to_set(observation.get("resources"))
    unclaimed = to_set(observation.get("unclaimed_cells"))
    self_t = to_set(observation.get("self_territory"))
    opp_t = to_set(observation.get("opponent_territory"))

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def inb(x, y): return 0 <= x < W and 0 <= y < H
    def free(x, y): return inb(x, y) and (x, y) not in obstacles
    def md(a, b): return abs(a[0] - b[0]) + abs(a[1] - b[1])

    # target preference: resources > unclaimed > opponent territory (to flip)
    if resources:
        targets = resources
        w_target = 1000
    elif unclaimed:
        targets = unclaimed
        w_target = 200
    else:
        targets = opp_t if opp_t else unclaimed
        w_target = 80

    if not targets:
        return [0, 0]

    # Prefer attacking opponent near the border: choose closest opponent-territory cell
    opp_border = set()
    if self_t and opp_t:
        for (x, y) in self_t:
            for dx, dy in dirs:
                nx, ny = x + dx, y + dy
                if (nx, ny) in opp_t:
                    opp_border.add((nx, ny))
        if opp_border:
            border_t = opp_border
        else:
            border_t = opp_t
    else:
        border_t = opp_t

    best = None
    best_score = -10**18
    # Deterministic tie-break: prefer (dx,dy) order as listed, and higher score
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not free(nx, ny):
            continue
        cell = (nx, ny)

        # Feature scores
        if cell in resources:
            base = 100000
        elif cell in unclaimed:
            base = 5000
        elif cell in opp_t:
            base = 3200
        elif cell in self_t:
            base = 1200
        else:
            base = 800

        # Distance-to-target term (maximize negative distance)
        # For opponent targets, use border targets if available.
        if targets is resources:
            d = min(md((nx, ny), t) for t in targets)
        elif targets is unclaimed:
            d = min(md((nx, ny), t) for t in targets)
        else:
            d = min(md((nx, ny), t) for t in border_t) if border_t else min(md((nx, ny), t) for t in targets)

        # Encourage getting closer fast; discourage hovering far from best target
        score = base + w_target * (-d) - 5 * (dx == 0 and dy == 0)

        # Minor preference to keep moving toward target direction
        if dx != 0 or dy != 0:
            score += 3

        if score > best_score:
            best_score = score
            best = (dx, dy)

    if best is None:
        return [0, 0]
    return [int(best[0]), int(best[1])]
