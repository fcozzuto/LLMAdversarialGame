    H = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position", (0, 0))
    ox, oy = observation.get("opponent_position", (0, 0))
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    obstacles = set()
    for p in observation.get("obstacles") or []:
        if p and len(p) >= 2:
            obstacles.add((int(p[0]), int(p[1])))

    self_t = set(map(tuple, observation.get("self_territory") or []))
    opp_t = set(map(tuple, observation.get("opponent_territory") or []))
    unclaimed = set(map(tuple, observation.get("unclaimed_cells") or []))
    all_cells = list(unclaimed) if unclaimed else []

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    def inb(x, y): return 0 <= x < W and 0 <= y < H
    def cheb(a, b): return max(abs(a[0] - b[0]), abs(a[1] - b[1]))

    # Pick a deterministic target: unclaimed adjacent to our territory; else nearest unclaimed to us;
    # else nearest edge unclaimed; else opponent territory.
    frontier = []
    if self_t:
        for (x, y) in self_t:
            for dx, dy in dirs:
                nx, ny = x + dx, y + dy
                if inb(nx, ny) and (nx, ny) in unclaimed:
                    frontier.append((nx, ny))
    frontier = sorted(set(frontier))
    candidates = frontier if frontier else (sorted(all_cells) if all_cells else [])
    if not candidates:
        edge = []
        for (x, y) in unclaimed:
            if x in (0, W - 1) or y in (0, H - 1):
                edge.append((x, y))
        candidates = sorted(set(edge)) if edge else sorted(list(opp_t)) if opp_t else []

    if not candidates:
        return [0, 0]

    target = min(candidates, key=lambda p: (cheb((sx, sy), p), abs(p[0] - (W - 1 - ox)) + abs(p[1] - (H - 1 - oy)), p[0], p[1]))

    def obstacle_penalty(x, y):
        if not obstacles:
            return 0
        best = 10
        for (ox1, oy1) in obstacles:
            d = cheb((x, y), (ox1, oy1))
            if d < best:
                best = d
        return 2 if best == 1 else (1 if best == 0 else 0)
