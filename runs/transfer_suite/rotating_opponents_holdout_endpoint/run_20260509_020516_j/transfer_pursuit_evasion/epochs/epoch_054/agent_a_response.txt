def choose_move(observation):
    w = int(observation.get("grid_width") or 8)
    h = int(observation.get("grid_height") or 8)
    s = observation.get("self_position") or [0, 0]
    o = observation.get("opponent_position") or [0, 0]
    sx, sy = int(s[0]), int(s[1])
    ox, oy = int(o[0]), int(o[1])

    obstacles = set()
    for it in observation.get("obstacles") or []:
        try:
            if isinstance(it, dict):
                x, y = it.get("x"), it.get("y")
            else:
                x, y = it[0], it[1]
            if x is None or y is None:
                continue
            x, y = int(x), int(y)
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))
        except Exception:
            continue

    role = (observation.get("self_role") or "").lower()
    is_evader = any(k in role for k in ("evader", "runner", "escape", "evasion"))
    is_pursuer = any(k in role for k in ("pursuer", "chaser", "catcher"))
    if not is_evader and not is_pursuer:
        is_pursuer = True

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def in_bounds(x, y):
        return 0 <= x < w and 0 <= y < h

    def blocked(x, y):
        return (x, y) in obstacles

    def d2(a, b, c, d):
        dx, dy = a - c, b - d
        return dx * dx + dy * dy

    def best_move_toward():
        best = None
        best_sc = -10**18
        for dx, dy in dirs:
            nx, ny = sx + dx, sy + dy
            if not in_bounds(nx, ny) or blocked(nx, ny):
                continue
            # score: prefer closer, penalize giving opponent immediate escape by local max distance
            my = -d2(nx, ny, ox, oy)
            opp_next = 0
            # opponent likely moves similarly (pursuit/evasion unknown), so assume it can choose one step that helps it:
            for odx, ody in dirs:
                tx, ty = ox + odx, oy + ody
                if not in_bounds(tx, ty) or blocked(tx, ty):
                    continue
                # if we are pursuer: opponent wants maximize distance; approximate by max distance after their move
                opp_next = max(opp_next, d2(nx, ny, tx, ty))
            sc = my - 0.15 * opp_next
            if sc > best_sc:
                best_sc, best = sc, [dx, dy]
        return best if best is not None else [0, 0]

    def best_move_away():
        # Move to increase distance, also bias toward farthest corner to reduce running in circles.
        corners = [(0, 0), (0, h - 1), (w - 1, 0), (w - 1, h - 1)]
        far_corner = max(corners, key=lambda c: d2(sx, sy, c[0], c[1]))
        fx, fy = far_corner

        best = None
        best_sc = -10**18
        for dx, dy in dirs:
            nx, ny = sx + dx, sy + dy
            if not in_bounds(nx, ny) or blocked(nx, ny):
                continue
            dist_from_opp = d2(nx, ny, ox, oy)
            # keep moving generally toward farthest corner
            toward_corner = -d2(nx, ny, fx, fy)
            # avoid stepping into "trap" near obstacles by discouraging adjacency to them (local heuristic)
            adj_obs = 0
            for ax in (-1, 0, 1):
                for ay in (-1, 0, 1):
                    if ax == 0 and ay == 0:
                        continue
                    if blocked(nx + ax, ny + ay):
                        adj_obs += 1
            sc = 1.0 * dist_from_opp + 0.06 * toward_corner - 0.08 * adj_obs
            if sc > best_sc:
                best_sc, best = sc, [dx, dy]
        return best if best is not None else [0, 0]

    if is_pursuer:
        return best_move_toward()
    return best_move_away()
