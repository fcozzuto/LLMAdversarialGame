        if isinstance(it, dict):

    resources = observation.get("resources") or []

            x, y = r[0], r[1]
        if x is None or y is None:

    if res_pts:

            tx, ty = min(res_pts, key=lambda p: (abs(p[0] - sx) + abs(p[1] - sy), p[0], p[1]))
            mode = "to_resource"
        else:
            tx, ty = min(res_pts, key=lambda p: (-((abs(p[0] - sx) + abs(p[1] - sy))), p[0], p[1]))
            mode = "from_resource"
