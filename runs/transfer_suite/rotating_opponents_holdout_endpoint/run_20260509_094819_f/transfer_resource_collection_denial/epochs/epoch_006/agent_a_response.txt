    h = observation.get("grid_height", 0)

    ox, oy = observation.get("opponent_position", (0, 0))

    obst = set()
    for p in obstacles:
        try:
            obst.add((p[0], p[1]))
        except Exception:
            pass

    deltas = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def man(a, b, c, d):
        dx = a - c
        if dx < 0:
            dx = -dx
        dy = b - d
        if dy < 0:
            dy = -dy
        return dx + dy

    target = None
    bestt = 10**18
    for rx, ry in resources:
        d = man(sx, sy, rx, ry)
        if d < bestt:
            bestt = d
            target = (rx, ry)

    if target is None:

    best_move = (0, 0)
