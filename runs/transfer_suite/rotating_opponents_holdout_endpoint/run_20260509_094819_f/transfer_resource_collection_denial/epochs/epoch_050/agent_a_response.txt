def choose_move(observation):

    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position", [0, 0]) or [0, 0]
    ox, oy = observation.get("opponent_position", [0, 0]) or [0, 0]
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    obstacles = set()
    for p in observation.get("obstacles") or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

    resources = []
    for p in observation.get("resources") or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h and (x, y) not in obstacles:
                resources.append((x, y))

    if not resources:
        return [0, 0]

    def d2(ax, ay, bx, by):
        dx = ax - bx
        dy = ay - by
        return dx * dx + dy * dy

    rem = int(observation.get("remaining_resource_count", len(resources)) or 0)
    mydamp = 0 if rem <= 3 else 1  # later: focus more on immediate pickups

    best_r = None
    best_key = None
    for rx, ry in resources:
        myd = d2(sx, sy, rx, ry)
        opd = d2(ox, oy, rx, ry)
        # Primary: maximize our lead (opd - myd); Secondary: smaller myd; Tertiary: prefer lower x+y
        lead = opd - myd
        key = (lead, -mydamp * myd, -(rx + ry))
        if best_key is None or key > best_key:
            best_key = key
            best_r = (rx, ry)

    rx, ry = best_r
    if sx == rx and sy == ry:
        return [0, 0]

    def step_towards(tx, ty):
        dx = 0 if tx == sx else (1 if tx > sx else -1)
        dy = 0 if ty == sy else (1 if ty > sy else -1)
        return dx, dy

    # Try candidate moves in a deterministic order: closer to target first, then safer.
    tx_dx, tx_dy = step_towards(rx, ry)
    candidates = []
    for dx in (tx_dx, 0, -tx_dx):
        for dy in (tx_dy, 0, -tx_dy):
            if dx == 0 and dy == 0:
                candidates.append((0, 0))
            else:
                candidates.append((dx, dy))
    # Deduplicate while preserving order
    seen = set()
    ordered = []
    for c in candidates:
        if c not in seen:
            seen.add(c)
            ordered.append(c)
