def choose_move(observation):
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    obst = set()
    for p in observation.get("obstacles") or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obst.add((x, y))

    res = []
    for p in observation.get("resources") or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h and (x, y) not in obst:
                res.append((x, y))
    if not res:
        return [0, 0]

    res = list(set(res))

    dirs = [(dx, dy) for dx in (-1, 0, 1) for dy in (-1, 0, 1)]
    def cheb(ax, ay, bx, by):
        a = ax - bx
        if a < 0: a = -a
        b = ay - by
        if b < 0: b = -b
        return a if a > b else b

    def valid_step(dx, dy):
        nx, ny = sx + dx, sy + dy
        if not (0 <= nx < w and 0 <= ny < h): 
            return False
        return (nx, ny) not in obst

    # Tie-break against opponent: prefer resources where we are closer than opponent.
    best = (0, 0)
    best_val = -10**18

    # Deterministic ordering for tie-breaking: by (dx,dy) sorted.
    ordered_dirs = sorted(dirs, key=lambda t: (t[0], t[1]))

    for dx, dy in ordered_dirs:
        if not valid_step(dx, dy):
            continue
        nx, ny = sx + dx, sy + dy

        my_capture = 0
        opp_capture = 0
        deny = 0

        # Score each candidate move by evaluating the best resource after stepping.
        for (rx, ry) in res:
            myd = cheb(nx, ny, rx, ry)
            opd = cheb(ox, oy, rx, ry)
            # If already on resource, prioritize strongly.
            if nx == rx and ny == ry:
                myd = 0
                my_capture += 50
            # Similar for opponent immediate capture
            if ox == rx and oy == ry:
                opd = 0
                opp_capture += 10

            # Advantage term: closer than opponent is good; being further is bad.
            # Use small weights to avoid unstable jumps.
            advantage = (opd - myd)
            # Encourage picking closer resources while also denying those opponent can reach first.
            if myd <= 1:
                myd_term = 6 - myd
            else:
                myd_term = 3 / (1 + myd)
            if opd <= 1:
                oppd_term = 4 - opd
            else:
                oppd_term = 2 / (1 + opd)

            if myd < opd:
                deny += 2 * advantage + myd_term
            elif myd > opd:
                deny += 1.5 * advantage - oppd_term
            else:
                deny += 0.25 * advantage

        # Also slightly prefer moving toward the opponent if we're behind (avoid being trapped far).
        approach_opp = -cheb(nx, ny, ox, oy) * 0.01
        val = my_capture + deny + approach_opp - opp_capture * 0.2

        # Minimal deterministic tie-break: prefer non-staying when values equal.
        if val > best_val or (val == best_val and (dx, dy) != (0, 0) and best == (0, 0)):
            best_val = val
            best = (dx, dy)

    return [int(best[0]), int(best[1])]
