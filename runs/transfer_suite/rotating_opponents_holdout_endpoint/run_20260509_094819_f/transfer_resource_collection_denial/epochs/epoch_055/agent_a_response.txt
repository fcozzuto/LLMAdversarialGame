def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    s = observation.get("self_position") or [0, 0]
    o = observation.get("opponent_position") or [0, 0]
    sx, sy = int(s[0]), int(s[1])
    ox, oy = int(o[0]), int(o[1])

    obstacles = set()
    for p in observation.get("obstacles") or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

    resources = []
    for p in observation.get("resources") or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h and (x, y) not in obstacles:
                resources.append((x, y))
    if not resources:
        return [0, 0]

    def cheb(ax, ay, bx, by):
        dx = ax - bx
        if dx < 0: dx = -dx
        dy = ay - by
        if dy < 0: dy = -dy
        return dx if dx > dy else dy

    def best_target(fromx, fromy, avoid_cell=None):
        best = None
        best_key = None
        for r in resources:
            if avoid_cell is not None and r == avoid_cell:
                continue
            d = cheb(fromx, fromy, r[0], r[1])
            key = (d, r[0], r[1])
            if best_key is None or key < best_key:
                best_key = key
                best = r
        return best

    opp_target = best_target(ox, oy)
    # Prefer denying opponent's likely target (switch strategy if possible)
    candidates = []
    for r in resources:
        if r == opp_target:
            continue
        myd = cheb(sx, sy, r[0], r[1])
        opd = cheb(ox, oy, r[0], r[1])
        # Advantage first; also prefer cells that are "ahead" in our sweep direction vs opponent
        adv = opd - myd
        bias = 0
        if ox < sx:
            bias = r[0]  # closer to our side (left-biased)
        elif ox > sx:
            bias = (w - 1 - r[0])
        else:
            bias = abs(r[1] - sy)
        key = (-adv, -bias, myd, r[0], r[1])
        candidates.append((key, r))

    if candidates:
        candidates.sort(key=lambda x: x[0])
        target = candidates[0][1]
    else:
        target = opp_target

    # Choose move that decreases chebyshev distance to target, while not stepping into obstacle
    tx, ty = target
    best_move = [0, 0]
    best_key = None
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            if dx == 0 and dy == 0:
                nx, ny = sx, sy
            else:
                nx, ny = sx + dx, sy + dy
            if not (0 <= nx < w and 0 <= ny < h):
                continue
            if (nx, ny) in obstacles:
                continue
            d = cheb(nx, ny, tx, ty)
            # discourage being too close to opponent target to reduce immediate contest losses
            contest_pen = 0
            if opp_target is not None:
                contest_pen = -cheb(nx, ny, opp_target[0], opp_target[1])
            key = (d, contest_pen, nx, ny)
            if best_key is None or key < best_key:
                best_key = key
                best_move = [dx, dy]

    return [int(best_move[0]), int(best_move[1])]
