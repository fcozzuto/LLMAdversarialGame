def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position") or [0, 0]
    ox, oy = observation.get("opponent_position") or [0, 0]
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    resources = observation.get("resources") or []
    obstacles = observation.get("obstacles") or []
    if not resources:
        return [0, 0]

    obs = set()
    for p in obstacles:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obs.add((x, y))

    def sign(z):
        return 0 if z == 0 else (1 if z > 0 else -1)

    def cheb(x1, y1, x2, y2):
        dx = x1 - x2
        if dx < 0: dx = -dx
        dy = y1 - y2
        if dy < 0: dy = -dy
        return dx if dx > dy else dy

    def around_obst(x, y):
        for dx in (-1, 0, 1):
            nx = x + dx
            if nx < 0 or nx >= w: 
                continue
            for dy in (-1, 0, 1):
                if dx == 0 and dy == 0:
                    continue
                ny = y + dy
                if ny < 0 or ny >= h:
                    continue
                if (nx, ny) in obs:
                    return 1
        return 0

    res = []
    for p in resources:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h and (x, y) not in obs:
                res.append((x, y))
    if not res:
        return [0, 0]

    # Target selection: deterministic scoring biased to beat opponent.
    best = None
    best_sc = None
    for (rx, ry) in res:
        ds = cheb(sx, sy, rx, ry)
        do = cheb(ox, oy, rx, ry)
        # Prefer resources closer to self; penalize adjacency to obstacles; small anti-race boost.
        adj = around_obst(rx, ry)
        sc = ds + 0.7 * adj - 0.35 * (do - ds)
        if best is None or sc < best_sc or (sc == best_sc and (rx, ry) < best):
            best = (rx, ry)
            best_sc = sc

    tx, ty = best

    # Candidate moves: small local evaluation with obstacle avoidance.
    deltas = [(-1,-1),(-1,0),(-1,1),(0,-1),(0,0),(0,1),(1,-1),(1,0),(1,1)]
    # Deterministic preference ordering toward target to break ties early.
    pref = []
    for dx, dy in deltas:
        nx, ny = sx + dx, sy + dy
        if not (0 <= nx < w and 0 <= ny < h):
            continue
        if (nx, ny) in obs:
            continue
        pref.append((dx, dy))
    if not pref:
        return [0, 0]

    # Sort pref by distance to target, then by tuple delta to keep deterministic.
    def move_score(dx, dy):
        nx, ny = sx + dx, sy + dy
        return (cheb(nx, ny, tx, ty) + 0.9 * around_obst(nx, ny), dx, dy)

    pref.sort(key=move_score)
    dx, dy = pref[0]
    return [int(dx), int(dy)]
