def choose_move(observation):
    w = int(observation.get("grid_width") or 8)
    h = int(observation.get("grid_height") or 8)
    sx, sy = observation.get("self_position") or (0, 0)
    ox, oy = observation.get("opponent_position") or (w - 1, h - 1)
    obstacles = set(tuple(p) for p in (observation.get("obstacles") or []))
    self_set = set(tuple(p) for p in (observation.get("self_territory") or []))
    opp_set = set(tuple(p) for p in (observation.get("opponent_territory") or []))
    unclaimed = set(tuple(p) for p in (observation.get("unclaimed_cells") or []))
    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def inb(x, y): return 0 <= x < w and 0 <= y < h

    def best_unclaimed_score():
        best = None
        # local: consider unclaimed or opponent cells within 5 of us
        for (x, y) in unclaimed:
            dx, dy = x - sx, y - sy
            d2 = dx * dx + dy * dy
            if d2 <= 25 and (best is None or d2 < best[0]):
                best = (d2, (x, y))
        if best is None:
            for (x, y) in opp_set:
                dx, dy = x - sx, y - sy
                d2 = dx * dx + dy * dy
                if d2 <= 25 and (best is None or d2 < best[0]):
                    best = (d2, (x, y))
        if best is None:
            return None
        return best[1]

    target = best_unclaimed_score()

    def eval_cell(nx, ny):
        if not inb(nx, ny) or (nx, ny) in obstacles:
            return -10**18
        if (nx, ny) in self_set:
            base = 0.0
        elif (nx, ny) in opp_set:
            base = 2.0  # flipping into opponent
        elif (nx, ny) in unclaimed:
            base = 1.3
        else:
            base = 0.7  # fallback
        # prefer frontier: cells that are adjacent to our territory but not already ours (encourage expansion)
        adj_self = 0
        for dx, dy in moves:
            x2, y2 = nx + dx, ny + dy
            if inb(x2, y2) and (x2, y2) in self_set:
                adj_self += 1
        frontier = adj_self * 0.35
        # avoid getting too close to opponent (reduce chance of being flipped back)
        d2o = (nx - ox) * (nx - ox) + (ny - oy) * (ny - oy)
        safety = d2o * 0.02
        # if have a target, move closer to it
        if target is not None:
            tx, ty = target
            dt = (nx - tx) * (nx - tx) + (ny - ty) * (ny - ty)
            dt0 = (sx - tx) * (sx - tx) + (sy - ty) * (sy - ty)
            approach = (dt0 - dt) * 0.12
        else:
            approach = 0.0
        # slight preference to center (avoid edge-locking by edge-claim opponent)
        cx, cy = (w - 1) / 2.0, (h - 1) / 2.0
        cent = -((nx - cx) * (nx - cx) + (ny - cy) * (ny - cy)) * 0.01
        # bias: don't chase opponent directly; prefer diagonals that increase safety
        return base + frontier + safety + approach + cent

    best_move = (0, 0)
    best_val = -10**19
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        v = eval_cell(nx, ny)
        if v > best_val:
            best_val = v
            best_move = (dx, dy)
        elif v == best_val:
