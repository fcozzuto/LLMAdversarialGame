def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sp = observation.get("self_position") or [0, 0]
    op = observation.get("opponent_position") or [w - 1, h - 1]
    sx, sy = int(sp[0]), int(sp[1])
    ox, oy = int(op[0]), int(op[1])

    obstacles = set()
    for p in observation.get("obstacles") or []:
        if p and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

    def neigh8(x, y):
        for dx in (-1, 0, 1):
            for dy in (-1, 0, 1):
                if dx == 0 and dy == 0:
                    continue
                nx, ny = x + dx, y + dy
                if 0 <= nx < w and 0 <= ny < h:
                    yield nx, ny

    self_cells = set((int(x), int(y)) for x, y in (observation.get("self_territory") or []))
    opp_cells = set((int(x), int(y)) for x, y in (observation.get("opponent_territory") or []))
    unclaimed = observation.get("unclaimed_cells") or []

    # Candidate set: unclaimed adjacent to our territory (frontier pressure).
    frontier = set()
    if self_cells:
        for x, y in self_cells:
            for nx, ny in neigh8(x, y):
                if (nx, ny) in obstacles:
                    continue
                if (nx, ny) not in self_cells and (nx, ny) not in opp_cells:
                    frontier.add((nx, ny))

    unclaimed_set = set((int(x), int(y)) for x, y in unclaimed if x is not None and y is not None)
    candidates = [c for c in frontier if c in unclaimed_set]
    if not candidates:
        candidates = [c for c in unclaimed_set if c not in obstacles]

    # If still empty, just head toward center.
    if not candidates:
        cx, cy = (w - 1) // 2, (h - 1) // 2
        if sx < cx:
            return [1, 0]
        if sx > cx:
            return [-1, 0]
        if sy < cy:
            return [0, 1]
        if sy > cy:
            return [0, -1]
        return [0, 0]

    def md(x1, y1, x2, y2):
        return abs(x1 - x2) + abs(y1 - y2)

    cx, cy = (w - 1) / 2.0, (h - 1) / 2.0
    best = None
    for tx, ty in candidates:
        if (tx, ty) in obstacles:
            continue
        ds = md(sx, sy, tx, ty)
        do = md(ox, oy, tx, ty)
        # Prefer cells we can secure sooner; boost frontier adjacency; avoid giving opponent close access.
        adv = do - ds
        center = -(abs(tx - cx) + abs(ty - cy))
        edge = 1 if (tx == 0 or tx == w - 1 or ty == 0 or ty == h - 1) else 0
        # Penalize cells adjacent to obstacles to reduce blocked movement.
        obs_adj = 0
        for nx, ny in neigh8(tx, ty):
            if (nx, ny) in obstacles:
                obs_adj += 1
        score = adv * 120 + center * 2.0 + edge * 1.0 - obs_adj * 0.6
        if best is None or score > best[0] or (score == best[0] and ds < best[1]):
            best = (score, ds, tx, ty)

    tx, ty = best[2], best[3]

    dx = 0 if tx == sx else (1 if tx > sx else -1)
    dy = 0 if ty == sy else (1 if ty > sy else -1)

    # If diagonal is blocked by obstacle (moving into obstacle), fall back to axis move or stay.
    nx, ny = sx + dx, sy + dy
    if (nx, ny) in obstacles:
        if dx != 0 and (sx + dx, sy) not in obstacles:
            return [dx, 0]
        if dy != 0 and (sx, sy + dy) not in obstacles:
            return [0, dy]
