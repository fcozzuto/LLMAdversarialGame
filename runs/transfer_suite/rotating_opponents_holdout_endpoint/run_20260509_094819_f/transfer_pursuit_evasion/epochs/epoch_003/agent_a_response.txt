def choose_move(observation):
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    x, y = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    x, y, ox, oy = int(x), int(y), int(ox), int(oy)

    obstacles = observation.get("obstacles", []) or []
    obs = set((int(px), int(py)) for px, py in obstacles)

    self_role = (observation.get("self_role", "") or "").lower()
    opponent_role = (observation.get("opponent_role", "") or "").lower()
    is_pursuer = ("purs" in self_role) or ("purs" in opponent_role and "evad" not in self_role)

    deltas = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def valid(nx, ny):
        return 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obs

    corners = [(0, 0), (0, h - 1), (w - 1, 0), (w - 1, h - 1)]

    def corner_goal(nx, ny):
        return max(abs(nx - cx) + abs(ny - cy) for cx, cy in corners)

    def degree(nx, ny):
        c = 0
        for ddx, ddy in deltas:
            tx, ty = nx + ddx, ny + ddy
            if valid(tx, ty):
                c += 1
        return c

    def safe_bonus(nx, ny):
        # Prefer positions with fewer immediate obstacle "pressure" (avoid tight pockets)
        near_block = 0
        for ddx, ddy in [(-1, 0), (1, 0), (0, -1), (0, 1)]:
            tx, ty = nx + ddx, ny + ddy
            if not (0 <= tx < w and 0 <= ty < h) or (tx, ty) in obs:
                near_block += 1
        return -0.35 * near_block

    best = None
    best_score = None
    for dx, dy in deltas:
        nx, ny = x + dx, y + dy
        if not valid(nx, ny):
            continue

        man = abs(nx - ox) + abs(ny - oy)
        d2 = (nx - ox) * (nx - ox) + (ny - oy) * (ny - oy)
        deg = degree(nx)
        cg = corner_goal(nx)

        if is_pursuer:
            # Chase: minimize distance, avoid dead ends, slightly bias toward trapping corners
            score = (-man) + (-0.01 * d2) + (0.18 * deg) + (0.03 * cg) + safe_bonus(nx, ny)
        else:
            # Evade: maximize distance and mobility, deterministically drift to farthest corner
            score = (man) + (0.003 * d2) + (0.22 * deg) + (0.10 * cg) + safe_bonus(nx, ny)

        if best is None or (score > best_score) or (score == best_score and (dx, dy) < best):
            best = (dx, dy)
            best_score = score

    if best is None:
        return [0, 0]
    return [int(best[0]), int(best[1])]

    x, y = observation.get("self_position", [0, 0])

    obs = observation.get("obstacles", []) or []
    blocked = set((int(a), int(b)) for a, b in obs)

    self_role = (observation.get("self_role", "") or "").lower()
    opponent_role = (observation.get("opponent_role", "") or "").lower()
    pursuit = ("purs" in self_role) or ("purs" in opponent_role and "evad" not in self_role)
    aim_resource = True

    resources = observation.get("resources", []) or []
    res_list = []
    for r in resources:
        if isinstance(r, (list, tuple)) and len(r) >= 2:
            res_list.append((int(r[0]), int(r[1])))
    if not res_list:
        aim_resource = False

    deltas = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    best = None
    best_sc = None

    for dx, dy in deltas:
        nx, ny = x + dx, y + dy
        if not (0 <= nx < w and 0 <= ny < h):
            continue
        if (nx, ny) in blocked:
            continue

        dist = abs(nx - ox) + abs(ny - oy)
        if pursuit:
            sc = -dist
        else:
            sc = dist

        if aim_resource:
            dmin = None
            for rx, ry in res_list:
                d = abs(nx - rx) + abs(ny - ry)
                if dmin is None or d < dmin:
                    dmin = d
            if dmin is not None:
                sc += (-dmin if pursuit else dmin) * 2

        if best_sc is None or sc > best_sc:
            best_sc = sc
            best = (dx, dy)

    if best is None:
        return [0, 0]
    return [best[0], best[1]]
