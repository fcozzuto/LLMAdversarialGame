        dy = abs(ay - by)
        return dx if dx > dy else dy

        for r in res_list:
            if r and len(r) >= 2:
                rx, ry = int(r[0]), int(r[1])
                if 0 <= rx < w and 0 <= ry < h:
                    d = cheb(x, y, rx, ry)
                    if d < best:
                        best = d
        return best if best != 10**9 else 0
