def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sp = observation.get("self_position", [0, 0])
    op = observation.get("opponent_position", [0, 0])
    sx, sy = int(sp[0]), int(sp[1])
    ox, oy = int(op[0]), int(op[1])

    obstacles = observation.get("obstacles") or []
    blocked = set()
    for p in obstacles:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                blocked.add((x, y))

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def cheb(x1, y1, x2, y2):
        dx = x1 - x2
        if dx < 0: dx = -dx
        dy = y1 - y2
        if dy < 0: dy = -dy
        return dx if dx > dy else dy

    def neighbors(x, y):
        out = []
        for dx, dy in moves:
            nx, ny = x + dx, y + dy
            if 0 <= nx < w and 0 <= ny < h and (nx, ny) not in blocked:
                out.append((nx, ny, dx, dy))
        return out

    my_next = neighbors(sx, sy)
    if not my_next:
        return [0, 0]

    opp_moves = neighbors(ox, oy)  # opponent current possible moves
    if not opp_moves:
        opp_moves = [(ox, oy, 0, 0)]

    best = None
    best_val = 10**18

    for nx, ny, mdx, mdy in my_next:
        # Worst-case opponent response: maximize distance from our next position
        worst = -1
        best_for_opp = None
        for tx, ty, _, _ in opp_moves:
            d = cheb(nx, ny, tx, ty)
            if d > worst:
                worst = d
                best_for_opp = (tx, ty)
        # Tie-break: favor moves that increase flexibility and head directly toward current opponent
        flex = sum(1 for _, _, _, _ in neighbors(nx, ny))
        direct = cheb(nx, ny, ox, oy)
        val = (worst, direct, -flex)
        if val < best_val or best is None:
            best_val = val
            best = (mdx, mdy)

    return [int(best[0]), int(best[1])]
