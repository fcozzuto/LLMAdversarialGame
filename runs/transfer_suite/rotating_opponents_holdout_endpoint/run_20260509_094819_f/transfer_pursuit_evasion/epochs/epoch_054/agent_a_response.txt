    h = int(observation.get("grid_height", 8) or 8)
    s = observation.get("self_position") or [0, 0]
    o = observation.get("opponent_position") or [0, 0]
    sx, sy = int(s[0]), int(s[1])
    ox, oy = int(o[0]), int(o[1])

    obstacles = observation.get("obstacles") or []
    blocked = set()
    for p in obstacles:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            blocked.add((int(p[0]), int(p[1])))

    def ok(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in blocked

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    role = str(observation.get("self_role") or "")
    is_pursuer = ("purs" in role.lower()) or (role.lower() == "pursuer")

    def cheb(x1, y1, x2, y2):
        dx = x1 - x2
        if dx < 0: dx = -dx
        dy = y1 - y2
        if dy < 0: dy = -dy
        return dx if dx > dy else dy

    def corner_far_from(px, py):
        corners = [(0, 0), (0, h - 1), (w - 1, 0), (w - 1, h - 1)]
        # farthest by chebyshev with deterministic tie-break
        return max(corners, key=lambda c: (cheb(c[0], c[1], px, py), -c[0], -c[1]))

    tx, ty = corner_far_from(ox, oy) if is_pursuer else corner_far_from(ox, oy)
    best = None

            if ok(nx, ny):
                cnt += 1
