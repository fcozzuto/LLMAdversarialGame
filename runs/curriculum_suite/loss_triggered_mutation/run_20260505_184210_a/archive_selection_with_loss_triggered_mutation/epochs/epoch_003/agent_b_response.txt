def choose_move(observation):
    x, y = observation["self_position"]
    width = observation["grid_width"]
    height = observation["grid_height"]
    target_y = y
    target_x = width - 1 if y % 2 == 0 else 0
    if x != target_x:
        return [1 if target_x > x else -1, 0]
    if y < height - 1:
        return [0, 1]
    return [0, 0]
