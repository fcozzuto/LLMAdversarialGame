def choose_move(observation):
    w = observation["grid_width"]
    h = observation["grid_height"]
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    resources = observation.get("resources", [])
    obstacles = set(tuple(p) for p in observation.get("obstacles", []))

    def dist(a, b):
        return max(abs(a[0] - b[0]), abs(a[1] - b[1]))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def best_step(tx, ty):
        best = (10**9, 0, 0)
        for dx in (-1, 0, 1):
            for dy in (-1, 0, 1):
                if dx == 0 and dy == 0:
                    nx, ny = sx, sy
                else:
                    nx, ny = sx + dx, sy + dy
                if not inb(nx, ny):
                    continue
                if (nx, ny) in obstacles:
                    continue
                d = max(abs(nx - tx), abs(ny - ty))
                opp_d = dist((nx, ny), (ox, oy))
                step_bias = -(abs(nx - (w - 1) / 2) + abs(ny - (h - 1) / 2)) * 1e-6
                # Prefer closing on target; if contested, also avoid getting too close to opponent (deny less valuable).
                score = (d, -opp_d, step_bias)
                if score < best:
                    best = score
                    best_move = [dx, dy]
        return best_move if "best_move" in locals() else [0, 0]

    # Choose target deterministically using advantage and contest dynamics.
    if resources:
        cand = []
        for rx, ry in resources:
            ds = dist((sx, sy), (rx, ry))
            do = dist((ox, oy), (rx, ry))
            adv = do - ds  # positive means we arrive first
            # Prefer positive advantage; if none, pick the resource opponent is closest to (deny).
            tie = (abs(rx - (w - 1) / 2) + abs(ry - (h - 1) / 2)) * 1e-6
            cand.append((-adv, ds + (0.01 * tie), rx, ry))
        cand.sort()
        _, _, tx, ty = cand[0]
    else:
        # No resources visible: intercept near opponent along their direction, but keep on-board.
        tx, ty = (ox + sx) // 2, (oy + sy) // 2

    # If opponent is adjacent, bias toward contesting their next neighborhood rather than pure chasing.
    opp_adj = dist((sx, sy), (ox, oy)) <= 1
    if opp_adj and resources:
        # pick a nearby resource that minimizes opponent's advantage
        best = None
        for rx, ry in resources:
            ds = dist((sx, sy), (rx, ry))
            do = dist((ox, oy), (rx, ry))
            threat = (do - ds, ds)
            if best is None or threat < best[0]:
                best = (threat, (rx, ry))
        tx, ty = best[1]

    return best_step(tx, ty)
