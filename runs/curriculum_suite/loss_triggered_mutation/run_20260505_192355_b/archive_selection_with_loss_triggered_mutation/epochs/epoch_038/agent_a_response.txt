def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position", (0, 0))
    ox, oy = observation.get("opponent_position", (0, 0))
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    obstacles = observation.get("obstacles", []) or []
    obst = set()
    for p in obstacles:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obst.add((x, y))

    resources = observation.get("resources", []) or []
    res = []
    for r in resources:
        if isinstance(r, (list, tuple)) and len(r) >= 2:
            x, y = int(r[0]), int(r[1])
            if 0 <= x < w and 0 <= y < h and (x, y) not in obst:
                res.append((x, y))

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def free(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obst

    def mdist(x1, y1, x2, y2):
        return abs(x1 - x2) + abs(y1 - y2)

    def clamp_step(tx, ty, x, y):
        dx = 0 if tx == x else (1 if tx > x else -1)
        dy = 0 if ty == y else (1 if ty > y else -1)
        return dx, dy

    # If no resources, drift toward the "front" between us (avoid getting stuck).
    if not res:
        tx, ty = (w - 1 + ox) // 2, (h - 1 + oy) // 2
        dx, dy = clamp_step(tx, ty, sx, sy)
        nx, ny = sx + dx, sy + dy
        if free(nx, ny):
            return [dx, dy]
        # fallback: pick any free move that reduces distance to the target
        best = None
        for dx, dy in moves:
            nx, ny = sx + dx, sy + dy
            if not free(nx, ny):
                continue
            key = (mdist(nx, ny, tx, ty), mdist(nx, ny, ox, oy))
            if best is None or key < best[0]:
                best = (key, dx, dy)
        if best is None:
            return [0, 0]
        return [best[1], best[2]]

    # Core change: choose resource where we can arrive earlier than opponent (lead bias).
    # Score tuple favors: largest lead (oppdist - mydist), then closeness to us, then tie-break by position.
    best_res = None
    best_key = None
    for rx, ry in res:
        myd = mdist(sx, sy, rx, ry)
        opd = mdist(ox, oy, rx, ry)
        lead = opd - myd
        # Prefer positive lead; also slightly prefer resources nearer the center to avoid edge traps.
        cx, cy = (w - 1) // 2, (h - 1) // 2
        center_bias = mdist(rx, ry, cx, cy) * 0.01
        key = (-lead, myd + center_bias, rx + ry * w)
        if best_key is None or key < best_key:
            best_key = key
            best_res = (rx, ry)

    tx, ty = best_res
    dx, dy = clamp_step(tx, ty, sx, sy)

    # If direct step is blocked, choose among legal moves the one that maximizes lead improvement.
    nx, ny = sx + dx, sy + dy
    if free(nx, ny):
        return [dx, dy]

    best = None
    for mdx, mdy in moves:
        nx, ny = sx + mdx, sy + mdy
        if not free(nx, ny):
            continue
        myd = mdist(nx, ny, tx, ty)
        opd = mdist(ox, oy, tx, ty)
        lead_key = (-(opd - myd), myd, mdx + mdy * w)
        if best is None or lead_key < best[0]:
            best = (lead_key, mdx, mdy)
    if best is None:
        return [0, 0]
    return [best[1], best[2]]
