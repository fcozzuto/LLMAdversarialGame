def choose_move(observation):
    w = observation.get("grid_width", 8)
    h = observation.get("grid_height", 8)
    sx, sy = observation.get("self_position", (0, 0))
    ox, oy = observation.get("opponent_position", (0, 0))
    resources = observation.get("resources", []) or []
    obstacles = set(tuple(p) for p in (observation.get("obstacles", []) or []))
    moves = [(-1, -1), (-1, 0), (-1, 1),
             (0, -1), (0, 0), (0, 1),
             (1, -1), (1, 0), (1, 1)]

    def inb(x, y): return 0 <= x < w and 0 <= y < h
    def ok(x, y): return inb(x, y) and (x, y) not in obstacles
    def md(ax, ay, bx, by):
        dx = ax - bx
        if dx < 0: dx = -dx
        dy = ay - by
        if dy < 0: dy = -dy
        return dx + dy

    if not resources:
        tx, ty = w // 2, h // 2
        best = None
        for dx, dy in moves:
            nx, ny = sx + dx, sy + dy
            if ok(nx, ny):
                d = md(nx, ny, tx, ty)
                if best is None or (d, dx, dy) < best:
                    best = (d, dx, dy)
        return [best[1], best[2]] if best else [0, 0]

    # Pick a resource where we are likely to arrive first; if none, pick a resource that forces
    # the opponent to waste time (either closest to them or closest overall).
    best_res = None
    for rx, ry in resources:
        ds = md(sx, sy, rx, ry)
        do = md(ox, oy, rx, ry)
        margin = do - ds  # positive => we are closer
        # Primary: maximize margin (prefer stealing). Secondary: prefer lower opponent distance.
        # Tertiary: prefer lower our distance.
        key = (-1 if margin <= 0 else 0, -margin, do, ds, rx, ry)
        if best_res is None or key < best_res[0]:
            best_res = (key, (rx, ry))
    rx, ry = best_res[1]

    # For each move, evaluate how much it improves our progress to the chosen resource,
    # and how it changes separation if the opponent is also racing that resource.
    best = None
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not ok(nx, ny):
            continue
        ds2 = md(nx, ny, rx, ry)
        do_now = md(ox, oy, rx, ry)
        # Separation after move: larger ds2-do_now means we're farther; want smaller ds2.
        # Add small term to slightly worsen opponent closeness if tie.
        score = (ds2, -((do_now - ds2) if do_now >= ds2 else -(ds2 - do_now)), abs(nx - rx) + abs(ny - ry))
        if best is None or score < best[0] or (score == best[0] and (dx, dy) < best[1]):
            best = (score, dx, dy)

    return [best[1], best[2]] if best else [0, 0]

def choose_move(observation):

    h = int(observation.get("grid_height", 8))
    sx, sy = observation.get("self_position", (0, 0))
    ox, oy = observation.get("opponent_position", (0, 0))
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    obstacles = set()
    for p in observation.get("obstacles", []) or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            obstacles.add((int(p[0]), int(p[1])))

    resources = observation.get("resources", []) or []

    moves = [(-1, -1), (-1, 0), (-1, 1),
             (0, -1), (0, 0), (0, 1),

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def ok(x, y):
        return inb(x, y) and (x, y) not in obstacles

    def md(ax, ay, bx, by):
        dx = ax - bx
        if dx < 0: dx = -dx
        dy = ay - by
        if dy < 0: dy = -dy
        return dx + dy

    def best_target():
        if not resources:
            return (w // 2, h // 2)
        best = None
        for r in resources:
            if not (isinstance(r, (list, tuple)) and len(r) >= 2):
                continue
            rx, ry = int(r[0]), int(r[1])
            d1 = md(sx, sy, rx, ry)
            d2 = md(ox, oy, rx, ry)
            adv = d2 - d1
            key = (-(adv), d1, rx, ry)  # minimize tuple to prefer larger adv
            if best is None or key < best[0]:
                best = (key, (rx, ry))
        return best[1] if best else (w // 2, h // 2)

    tx, ty = best_target()

        adv = opd - myd
        key = (-(adv), myd, abs(nx - tx) + abs(ny - ty), dx, dy)
        if best_key is None or key < best_key:
            best_key = key
            best_move = (dx, dy)

    if best_move is None:
