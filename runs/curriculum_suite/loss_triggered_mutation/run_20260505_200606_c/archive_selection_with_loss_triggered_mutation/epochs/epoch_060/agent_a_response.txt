def choose_move(observation):
    w = observation["grid_width"]
    h = observation["grid_height"]
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    resources = observation.get("resources", []) or []
    obstacles = set(tuple(p) for p in (observation.get("obstacles", []) or []))

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def bfs_dist(start, goal):
        if start == goal:
            return 0
        if goal in obstacles:
            return 10**9
        sx0, sy0 = start
        if (sx0, sy0) in obstacles:
            return 10**9
        qx, qy = [sx0], [sy0]
        dist = {(sx0, sy0): 0}
        qi = 0
        while qi < len(qx):
            x, y = qx[qi], qy[qi]
            qi += 1
            d = dist[(x, y)]
            for dx, dy in moves:
                nx, ny = x + dx, y + dy
                if not inb(nx, ny) or (nx, ny) in obstacles:
                    continue
                if (nx, ny) not in dist:
                    nd = d + 1
                    if (nx, ny) == goal:
                        return nd
                    dist[(nx, ny)] = nd
                    qx.append(nx)
                    qy.append(ny)
        return 10**9

    def best_move():
        if not resources:
            tx, ty = (w - 1) // 2, (h - 1) // 2
            best = (10**9, 10**9)
            bestm = (0, 0)
            for dx, dy in moves:
                nx, ny = sx + dx, sy + dy
                if not inb(nx, ny) or (nx, ny) in obstacles:
                    continue
                v = (abs(nx - tx) + abs(ny - ty), abs(nx - ox) + abs(ny - oy))
                if v < best:
                    best = v
                    bestm = (dx, dy)
            return list(bestm)

        best_val = (-10**18, 10**18, 10**18)
        best_move_delta = (0, 0)
        for dx, dy in moves:
            nx, ny = sx + dx, sy + dy
            if not inb(nx, ny) or (nx, ny) in obstacles:
                continue
            se_next = (nx, ny)
            local_best = (-10**18, 10**18)
            for rx, ry in resources:
                r = (rx, ry)
                se_d = bfs_dist(se_next, r)
                op_d = bfs_dist((ox, oy), r)
                # Prefer resources where we are closer; also prefer smaller opponent distance.
                # Primary: (op_d - se_d), Secondary: se_d, Tertiary: op_d.
                val = (op_d - se_d, se_d)
                if val > local_best:
                    local_best = val
            primary = local_best[0]
            se_d_best = local_best[1]
            op_d_best = bfs_dist((ox, oy), resources[[((bfs_dist((nx, ny), (rx, ry))) for rx, ry in resources).index(min((bfs_dist((nx, ny), (rx, ry)), 10**9) for rx, ry in resources)))] if resources else (ox, oy))
            # Above line is messy; simplify deterministically:
            op_d_best = min(bfs_dist((ox, oy), r) for r in resources)
            cand = (primary, -se_d_best, -op_d_best)
            if cand > best_val:
                best_val = cand
                best_move_delta = (dx, dy)
        return list(best_move_delta)

    return best_move()
