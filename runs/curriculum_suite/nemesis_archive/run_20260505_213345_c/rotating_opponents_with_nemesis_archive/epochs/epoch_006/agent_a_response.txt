def choose_move(observation):
    w = observation.get("grid_width", 8)
    h = observation.get("grid_height", 8)
    sx, sy = observation.get("self_position", (0, 0))
    ox, oy = observation.get("opponent_position", (0, 0))

    obstacles = observation.get("obstacles", []) or []
    obs = set()
    for p in obstacles:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            obs.add((int(p[0]), int(p[1])))

    resources = observation.get("resources", []) or []
    res = []
    for p in resources:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            res.append((int(p[0]), int(p[1])))

    def in_bounds(x, y):
        return 0 <= x < w and 0 <= y < h

    def dist2(a, b, c, d):
        dx = a - c
        dy = b - d
        return dx * dx + dy * dy

    moves = [(dx, dy) for dx in (-1, 0, 1) for dy in (-1, 0, 1)]
    if not res or w <= 0 or h <= 0:
        # still try to edge-patrol counter: increase distance from opponent safely
        best = (0, 0, -10**18)
        for dx, dy in moves:
            if dx == 0 and dy == 0:
                continue
            nx, ny = sx + dx, sy + dy
            if not in_bounds(nx, ny) or (nx, ny) in obs:
                continue
            val = dist2(nx, ny, ox, oy) - 0.1 * (dx * dx + dy * dy)
            if val > best[2]:
                best = (dx, dy, val)
        return [best[0], best[1]] if best[2] != -10**18 else [0, 0]

    # opponent greedy target estimate
    def greedy_next(px, py):
        best_t = None
        best_d = 10**18
        for rx, ry in res:
            d = dist2(px, py, rx, ry)
            if d < best_d:
                best_d = d
                best_t = (rx, ry)
        if best_t is None:
            return (px, py)
        tx, ty = best_t
        best = (0, 0)
        bestv = -10**18
        for dx, dy in moves:
            nx, ny = px + dx, py + dy
            if not in_bounds(nx, ny) or (nx, ny) in obs:
                continue
            v = -dist2(nx, ny, tx, ty)
            if v > bestv:
                bestv = v
                best = (nx, ny)
        return best

    opp_nx, opp_ny = greedy_next(ox, oy)

    # Our move scoring: go to nearest resource, deny opponent by increasing their future resource closeness and separation
    best_move = (0, 0, -10**18)
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not in_bounds(nx, ny) or (nx, ny) in obs:
            continue

        # resource gain: prioritize resource closest to next position
        my_best = 10**18
        their_best = 10**18
        for rx, ry in res:
            my_best = min(my_best, dist2(nx, ny, rx, ry))
            their_best = min(their_best, dist2(opp_nx, opp_ny, rx, ry))

        # immediate pickup incentive
        pickup = 1 if (nx, ny) in set(res) else 0

        # separation / interception: further from predicted opponent position is good
        sep = dist2(nx, ny, opp_nx, opp_ny)

        # also discourage stepping toward a spot they would likely reach quickly: compare our closeness vs theirs at same resource
        opp_here_best = 10**18
        for rx, ry in res:
            opp_here_best = min(opp_here_best, dist2(opp_nx, opp_ny, rx, ry))

        val = 2.0 * pickup + (-0.9 * my_best) + 0.15 * sep + (-0.2 * their_best) - 0.01 * (abs(dx) + abs(dy))
        if val > best_move[2]:
            best_move = (dx, dy, val)

    return [best_move[0], best_move[1]]
