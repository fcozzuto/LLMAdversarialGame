def choose_move(observation):
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    resources = []
    for p in observation.get("resources", []) or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            resources.append((int(p[0]), int(p[1])))

    obstacles = set()
    for p in observation.get("obstacles", []) or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            obstacles.add((int(p[0]), int(p[1])))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    if not (0 <= sx < w and 0 <= sy < h):
        return [0, 0]
    if not resources:
        return [0, 0]

    def man(x1, y1, x2, y2):
        return abs(x1 - x2) + abs(y1 - y2)

    # Predict opponent's greedy target (nearest resource).
    opp_target = None
    best_do = None
    for rx, ry in resources:
        if (rx, ry) in obstacles:
            continue
        d = man(ox, oy, rx, ry)
        if best_do is None or d < best_do or (d == best_do and (rx, ry) < opp_target):
            best_do = d
            opp_target = (rx, ry)

    # Choose our target:
    # - If opponent targets a resource we can also reach quickly, pick the best alternative that
    #   is farther from the opponent (interception-style switch).
    my_best = None
    my_best_score = None
    for rx, ry in resources:
        if (rx, ry) in obstacles:
            continue
        d_us = man(sx, sy, rx, ry)
        d_opp = man(ox, oy, rx, ry)
        reach_threat = d_us - d_opp  # negative => we are closer than opponent
        score = (d_us, -reach_threat, rx, ry)
        if my_best_score is None or score < my_best_score:
            my_best_score = score
            my_best = (rx, ry)

    target = my_best
    if opp_target is not None and target == opp_target:
        # pick another resource: prioritize that opponent is worse at, but keep us reasonably close
        alt = None
        alt_score = None
        for rx, ry in resources:
            if (rx, ry) in obstacles or (rx, ry) == opp_target:
                continue
            d_us = man(sx, sy, rx, ry)
            d_opp = man(ox, oy, rx, ry)
            score = (d_us + 2, -(d_opp - d_us), rx, ry)
            if alt_score is None or score < alt_score:
                alt_score = score
                alt = (rx, ry)
        if alt is not None:
            target = alt

    tx, ty = target

    # Move one step toward target with deterministic obstacle-aware fallback.
    dx = 0 if tx == sx else (1 if tx > sx else -1)
    dy = 0 if ty == sy else (1 if ty > sy else -1)

    candidates = [(dx, dy), (dx, 0), (0, dy), (dx, -dy), (-dx, dy), (0, 0), (-dx, 0), (0, -dy), (-dx, -dy)]
    seen = set()
    ordered = []
    for c in candidates:
        if c not in seen:
            seen.add(c)
            ordered.append(c)

    best = [0, 0]
    best_dist = None
    for mdx, mdy in ordered:
        nx, ny = sx + mdx, sy + mdy
        if not inb(nx, ny):
            continue
        if (nx, ny) in obstacles:
            continue
        d = man(nx, ny, tx, ty)
        # Prefer strict improvement; tie-break deterministically by move vector.
        key = (d, abs(mdx) + abs(mdy), mdx, mdy)
        if best_dist is None or key < best_dist:
            best_dist = key
            best = [mdx, mdy]

    return [int(best[0]), int(best[1])]
