def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    resources = observation.get("resources", []) or []
    if not resources:
        return [0, 0]

    obstacles = set()
    for p in (observation.get("obstacles", []) or []):
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

    res = []
    for p in resources:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h and (x, y) not in obstacles:
                res.append((x, y))
    if not res:
        return [0, 0]

    def man(a, b):
        return abs(a[0] - b[0]) + abs(a[1] - b[1])

    center = ((w - 1) / 2.0, (h - 1) / 2.0)

    scored = []
    for r in res:
        md = man((sx, sy), r)
        od = man((ox, oy), r)
        adv = od - md  # positive means we are closer
        # mild preference to stabilize toward board center (helps vs row-sweeps)
        cpen = abs(r[0] - center[0]) + abs(r[1] - center[1])
        # bonus if we can contest opponent's row/col direction
        rowcol = (1 if r[1] == oy else 0) + (1 if r[0] == ox else 0)
        val = (adv * 1000) + (-md) * 2 + (-cpen) * 0.5 + rowcol * 3
        scored.append((val, adv, -md, r[0], r[1]))

    scored.sort(reverse=True)
    target = scored[0][4], scored[0][5]

    tx, ty = target
    dx = 0 if tx == sx else (1 if tx > sx else -1)
    dy = 0 if ty == sy else (1 if ty > sy else -1)

    # If moving diagonally would increase our distance too much, force axis step deterministically
    if dx != 0 and dy != 0:
        if man((sx + dx, sy + dy), (tx, ty)) > man((sx + dx, sy), (tx, ty)) + 0:
            if man((sx + dx, sy), (tx, ty)) <= man((sx, sy + dy), (tx, ty)):
                dy = 0
            else:
                dx = 0

    return [dx, dy]

def choose_move(observation):
    w = int(observation.get("grid_width") or 8)
    h = int(observation.get("grid_height") or 8)

    sp = observation.get("self_position") or [0, 0]

    sx = int(sp[0]) if isinstance(sp, (list, tuple)) and len(sp) > 0 else 0
    sy = int(sp[1]) if isinstance(sp, (list, tuple)) and len(sp) > 1 else 0
    ox = int(op[0]) if isinstance(op, (list, tuple)) and len(op) > 0 else 0

    for p in observation.get("obstacles") or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x = int(p[0]); y = int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

        if isinstance(p, (list, tuple)) and len(p) >= 2:

            if 0 <= x < w and 0 <= y < h and (x, y) not in obstacles:
                resources.append((x, y))
    if not resources:
        return [0, 0]

    def man(ax, ay, bx, by):
        return abs(ax - bx) + abs(ay - by)

    cx = (w - 1) * 0.5
    cy = (h - 1) * 0.5

    moves = [(-1, 0), (1, 0), (0, -1), (0, 1), (0, 0)]
    best = None
    bestv = -10**18

    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not (0 <= nx < w and 0 <= ny < h) or (nx, ny) in obstacles:
            continue
        vd = 0
        for (rx, ry) in resources:

            od = man(ox, oy, rx, ry)
            center = abs(rx - cx) + abs(ry - cy)
            vd += (od - sd) - 0.01 * center
        if vd > bestv:
            bestv = vd
            best = [dx, dy]

    return best if best is not None else [0, 0]
