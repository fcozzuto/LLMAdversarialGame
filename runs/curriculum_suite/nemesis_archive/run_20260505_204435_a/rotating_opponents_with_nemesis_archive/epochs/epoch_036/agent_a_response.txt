def choose_move(observation):
    w = observation.get("grid_width", 8) or 8
    h = observation.get("grid_height", 8) or 8
    sx, sy = observation.get("self_position", (0, 0))
    ox, oy = observation.get("opponent_position", (0, 0))
    resources = observation.get("resources", []) or []
    obstacles = observation.get("obstacles", []) or []

    blocked = set()
    for p in obstacles:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            blocked.add((p[0], p[1]))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def legal(x, y):
        return inb(x, y) and (x, y) not in blocked

    def parse_pos(r):
        if isinstance(r, dict):
            pos = r.get("position")
            if isinstance(pos, (list, tuple)) and len(pos) >= 2:
                return (pos[0], pos[1])
        if isinstance(r, (list, tuple)) and len(r) >= 2:
            return (r[0], r[1])
        return None

    targets = []
    for r in resources:
        p = parse_pos(r)
        if p is not None and legal(p[0], p[1]):
            targets.append(p)

    moves = [(dx, dy) for dx in (-1, 0, 1) for dy in (-1, 0, 1)]
    if not legal(sx, sy):
        for dx, dy in moves:
            nx, ny = sx + dx, sy + dy
            if legal(nx, ny):
                return [dx, dy]
        return [0, 0]

    def man(x1, y1, x2, y2):
        return abs(x1 - x2) + abs(y1 - y2)

    if not targets:
        # deterministic safe-ish: move to increase distance from opponent along diagonal preference
        best = None
        bestv = -10**9
        for dx, dy in moves:
            nx, ny = sx + dx, sy + dy
            if not legal(nx, ny):
                continue
            v = man(nx, ny, sx, sy) - man(nx, ny, ox, oy) * 0.01
            if v > bestv:
                bestv = v
                best = (dx, dy)
        return list(best) if best is not None else [0, 0]

    # Evaluate: for each move, best target it can reach, but penalize targets the opponent can contest immediately.
    best_move = (0, 0)
    best_val = -10**18
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not legal(nx, ny):
            continue
        # if we step onto a resource, prefer heavily
        base = 0
        for t in targets:
            if t[0] == nx and t[1] == ny:
                base += 1000
        best_target_val = -10**18
        for tx, ty in targets:
            self_d = man(nx, ny, tx, ty)
            opp_d = man(ox, oy, tx, ty)
            # contest if opponent can reach in <=1 after its move (approximate)
            contest = 1 if opp_d <= 1 else 0
            # Prefer being closer than opponent, and avoid immediate contesting targets
            v = (opp_d - self_d) * 10 - self_d - contest * 80 + (base * 0.001)
            # small tie-breaker: move reduces distance to the chosen target in a "directional" way
            v -= abs((tx - nx) - (tx - sx)) * 0.01
            if v > best_target_val:
                best_target_val = v
        # If opponent could capture some nearby resource next, slightly bias away by maximizing min distance to resources
        opp_threat = 0
        for tx, ty in targets:
            if man(ox, oy, tx, ty) <= 1:
                opp_threat = 1
                break
        total = best_target_val - opp_threat * 20
        if total > best_val:
            best_val = total
            best_move = (dx, dy)

    return [best_move[0], best_move[1]]
