def choose_move(observation):
    w = observation.get("grid_width", 8)
    h = observation.get("grid_height", 8)
    sx, sy = observation.get("self_position", (0, 0))
    ox, oy = observation.get("opponent_position", (0, 0))
    resources = observation.get("resources") or []
    obstacles = observation.get("obstacles") or []

    obs = set()
    for o in obstacles:
        if isinstance(o, (list, tuple)) and len(o) == 2:
            x, y = o[0], o[1]
            if isinstance(x, int) and isinstance(y, int) and 0 <= x < w and 0 <= y < h:
                obs.add((x, y))

    res = []
    for r in resources:
        if isinstance(r, (list, tuple)) and len(r) == 2:
            x, y = r[0], r[1]
            if isinstance(x, int) and isinstance(y, int) and 0 <= x < w and 0 <= y < h and (x, y) not in obs:
                res.append((x, y))
    if not res:
        return [0, 0]

    def md(ax, ay, bx, by):
        return abs(ax - bx) + abs(ay - by)

    opp_best = min(res, key=lambda p: (md(ox, oy, p[0], p[1]), p[0], p[1]))
    ax0 = md(sx, sy, opp_best[0], opp_best[1])
    bx0 = md(ox, oy, opp_best[0], opp_best[1])

    moves = []
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            nx, ny = sx + dx, sy + dy
            if 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obs:
                moves.append((dx, dy, nx, ny))
    if not moves:
        return [0, 0]

    def best_score(nx, ny):
        # Aim: secure a resource before opponent, else block by moving toward opponent's nearest.
        s = -10**9
        for tx, ty in res:
            myd = md(nx, ny, tx, ty)
            opd = md(ox, oy, tx, ty)
            # parity/arrival bias: only meaningful if we can match distance class
            parity = 1 if (myd % 2) == (opd % 2) else 0
            lead = opd - myd
            # discourage helping opponent by letting them be closer by a lot
            score = lead * 6 + parity * 2 - myd * 0.2
            if (tx, ty) == opp_best:
                # intercept urgency: close the gap on opponent's nearest
                score += 30 if myd < opd else 10 - (myd - opd) * 2
            # prefer going where I become the closest (among all resources)

                score += 1
            # slight preference for nearer overall
            score -= 0.01 * myd
            if score > s:
                s = score
        # additional bias: if opponent is currently clearly closer to their best,
        # move to reduce that gap toward their best resource
        myd_to_oppbest = md(nx, ny, opp_best[0], opp_best[1])
        s += (bx0 - myd_to_oppbest) * 3
        return s

    best = None
    best_val = None
    for dx, dy, nx, ny in moves:
        v = best_score(nx, ny)
        # tie-break deterministically: prefer diagonal then right then up then smaller dx/dy magnitude
