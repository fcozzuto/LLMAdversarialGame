def choose_move(observation):
    w = observation.get("grid_width", 8)
    h = observation.get("grid_height", 8)
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    resources = observation.get("resources") or []
    obstacles = observation.get("obstacles") or []
    obs = set()
    for p in obstacles:
        if isinstance(p, (list, tuple)) and len(p) == 2:
            x, y = p[0], p[1]
            if isinstance(x, int) and isinstance(y, int) and 0 <= x < w and 0 <= y < h:
                obs.add((x, y))

    def md(ax, ay, bx, by):
        return abs(ax - bx) + abs(ay - by)

    moves = []
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            nx, ny = sx + dx, sy + dy
            if 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obs:
                moves.append((dx, dy, nx, ny))
    if not moves:
        return [0, 0]

    # Pick a target resource: prefer those where we are not too far behind opponent.
    best_t = None
    best_key = None
    for rx, ry in resources:
        if not (isinstance(rx, int) and isinstance(ry, int)):
            continue
        d1 = md(sx, sy, rx, ry)
        d2 = md(ox, oy, rx, ry)
        # smaller is better: we want to be closer (d1-d2 negative), but still progress to a nearby resource.
        key = (d1 - d2, d1 + 0.1 * (abs(rx - (w - 1)) + abs(ry - (h - 1))))
        if best_key is None or key < best_key:
            best_key = key
            best_t = (rx, ry)
    # If no resources, just drift away from opponent while staying in bounds
    if best_t is None:
        tx, ty = (w - 1, h - 1)
    else:
        tx, ty = best_t

    best_move = None
    best_score = None
    for dx, dy, nx, ny in moves:
        # Core: minimize distance to our target
        d_self = md(nx, ny, tx, ty)
        d_op = md(nx, ny, ox, oy)

        score = -1.25 * d_self + 0.15 * (d_op)  # keep some distance from denier

        # Block/interpose: if opponent is closer to the target, try to step into the Manhattan "corridor"
        # between opponent and target (only when aligned in x or y to keep it simple and deterministic).
        if best_t is not None:
            d_op_target = md(ox, oy, tx, ty)
            if d_op_target > 0 and md(sx, sy, tx, ty) > d_op_target:
                if nx == ox:
                    # moving on same column as opponent relative progress
                    score += 2.0 / (1 + abs(ny - oy)) - 0.05 * d_self
                elif ny == oy:
                    score += 2.0 / (1 + abs(nx - ox)) - 0.05 * d_self
                else:
                    # if not aligned, discourage moving away from a likely interpose line
                    score += -0.2 * (abs(nx - ox) + abs(ny - oy))

        # Obstacle proximity (soft): avoid stepping adjacent to obstacles
        adj_obs = 0
        for ax in (-1, 0, 1):
            for ay in (-1, 0, 1):
                if (nx + ax, ny + ay) in obs:
                    adj_obs += 1
        score += -0.35 * adj_obs

        # Small deterministic tie-break: prefer moves that increase x when scores equal, then increase y
        tie = (0.0,)
        if best_score is None or score > best_score:
            best_score = score
            best_move = (dx, dy)
        elif best_score is not None and score == best_score:
            cand = (nx > sx, ny > sy, dx, dy)
            cur = (sx + best_move[0] > sx, sy + best_move[1] > sy, best_move[0], best_move[1])
            if cand > cur:
                best_move = (dx, dy)

    if best_move is None:
        return [0, 0]
    return [int(best_move[0]), int(best_move[1])]
