def choose_move(observation):
    w = observation.get("grid_width", 8)
    h = observation.get("grid_height", 8)
    sx, sy = observation.get("self_position", (0, 0))
    ox, oy = observation.get("opponent_position", (0, 0))
    resources = observation.get("resources") or []
    obstacles = observation.get("obstacles") or []

    obs = set()
    for o in obstacles:
        if isinstance(o, (list, tuple)) and len(o) == 2:
            obs.add((o[0], o[1]))

    res = []
    for r in resources:
        if isinstance(r, (list, tuple)) and len(r) == 2:
            x, y = r[0], r[1]
            if 0 <= x < w and 0 <= y < h and (x, y) not in obs:
                res.append((x, y))

    if not res:
        return [0, 0]

    deltas = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    cx, cy = (w - 1) / 2.0, (h - 1) / 2.0

    def inside(x, y):
        return 0 <= x < w and 0 <= y < h

    def manh(ax, ay, bx, by):
        return abs(ax - bx) + abs(ay - by)

    opp_near = min(manh(sx, sy, rx, ry), manh(ox, oy, rx, ry) for rx, ry in res)

    best = (0, 0)
    best_val = -10**18
    for dx, dy in deltas:
        nx, ny = sx + dx, sy + dy
        if not inside(nx, ny) or (nx, ny) in obs:
            continue

        my_center = - (abs(nx - cx) + abs(ny - cy))
        move_val = my_center * 0.25  # tie-breaker

        # Evaluate best possible target after this move.
        local_best = -10**18
        for rx, ry in res:
            d_self = manh(nx, ny, rx, ry)
            d_opp = manh(ox, oy, rx, ry)

            # Core: steal resources where opponent is relatively farther.
            steal = (d_opp - d_self)

            # If we're behind, still allow "lunge" steals but heavily damped.
            behind_pen = 0
            if d_self > d_opp:
                behind_pen = (d_self - d_opp) * 0.9

            # Add a small preference to reduce distance once chosen.
            proximity = -d_self * 0.08

            # Bonus for cutting off toward the resource that is closest to us now.
            local = steal * 1.3 + proximity - behind_pen

            # If opponent is extremely close to some resources, avoid feeding:
            # prefer targets that keep opponent distance relatively high.
            if opp_near <= 3:
                local += (d_opp - d_self) * 0.15

            if local > local_best:
                local_best = local

        move_val += local_best
        # Deterministic preference ordering: on ties, pick lexicographically smaller (dx,dy) favoring staying.
        if move_val > best_val:
            best_val = move_val
            best = (dx, dy)
        elif move_val == best_val:
            if (dx, dy) < best:
                best = (dx, dy)

    return [best[0], best[1]]
