def choose_move(observation):
    w = observation["grid_width"]
    h = observation["grid_height"]
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    resources = observation.get("resources", [])
    obstacles = set(tuple(p) for p in observation.get("obstacles", []))
    deltas = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def inside(x, y):
        return 0 <= x < w and 0 <= y < h

    def bfs(startx, starty, maxd):
        INF = 10**9
        dist = [[INF] * h for _ in range(w)]
        first = [[None] * h for _ in range(w)]
        qx, qy = [startx], [starty]
        dist[startx][starty] = 0
        first[startx][starty] = (0, 0)
        qi = 0
        while qi < len(qx):
            x, y = qx[qi], qy[qi]
            qi += 1
            d = dist[x][y]
            if d >= maxd:
                continue
            for dx, dy in deltas:
                nx, ny = x + dx, y + dy
                if not inside(nx, ny) or (nx, ny) in obstacles:
                    continue
                nd = d + 1
                if nd < dist[nx][ny]:
                    dist[nx][ny] = nd
                    if d == 0:
                        first[nx][ny] = (dx, dy)
                    else:
                        first[nx][ny] = first[x][y]
                    qx.append(nx)
                    qy.append(ny)
        return dist, first

    if not resources:
        tx, ty = w // 2, h // 2
        best = (10**9, 10**9, 0, 0)
        for dx, dy in deltas:
            nx, ny = sx + dx, sy + dy
            if not inside(nx, ny) or (nx, ny) in obstacles:
                continue
            d = abs(nx - tx) + abs(ny - ty)
            key = (d, abs(dx) + abs(dy), dx, dy)
            if key < best:
                best = key
        return [best[2], best[3]]

    maxd = 10
    mydist, myfirst = bfs(sx, sy, maxd)
    oppdist, _ = bfs(ox, oy, maxd)

    best = None  # (adv, myd, tie, move)
    for rx, ry in resources:
        myd = mydist[rx][ry]
        oppd = oppdist[rx][ry]
        if myd >= 10**9 and oppd >= 10**9:
            continue
        adv = myd - oppd
        fx = myfirst[rx][ry]
        if fx is None:
            continue
        dx, dy = fx
        tie = (abs(rx - sx), abs(ry - sy), rx, ry)
        cand = (adv, myd, tie, dx, dy)
        if best is None or cand < best:
            best = cand

    if best is not None:
        return [best[3], best[4]]

    # Fallback: minimize distance to nearest resource while avoiding obstacles
    tx, ty = min(resources, key=lambda p: abs(p[0] - sx) + abs(p[1] - sy))
    best = (10**9, 10**9, 0, 0)
    for dx, dy in deltas:
        nx, ny = sx + dx, sy + dy
        if not inside(nx, ny) or (nx, ny) in obstacles:
            continue
        d = abs(nx - tx) + abs(ny - ty)
        key = (d, abs(dx) + abs(dy), dx, dy)
        if key < best:
            best = key
    return [best[2], best[3]]

def choose_move(observation):
    w = observation["grid_width"]
    h = observation["grid_height"]
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    resources = observation.get("resources") or []
    obs = observation.get("obstacles") or []
    obstacles = set(tuple(p) for p in obs)
    deltas = [(-1,-1),(-1,0),(-1,1),(0,-1),(0,0),(0,1),(1,-1),(1,0),(1,1)]

    def inside(x, y):
        return 0 <= x < w and 0 <= y < h

    def sqd(ax, ay, bx, by):
        dx = ax - bx
        dy = ay - by
        return dx*dx + dy*dy

    best = None
    best_score = None
    for dx, dy in deltas:
        nx, ny = sx + dx, sy + dy
        if not inside(nx, ny) or (nx, ny) in obstacles:
            continue
        if resources:
            rdist = min(sqd(nx, ny, rx, ry) for rx, ry in resources)
        else:
            rdist = 0
        odist = sqd(nx, ny, ox, oy)
        score = -rdist + 0.15 * odist
        if best is None or score > best_score or (score == best_score and (dx, dy) < best):
            best = (dx, dy)
            best_score = score

    if best is None:
        return [0, 0]
    return [best[0], best[1]]
