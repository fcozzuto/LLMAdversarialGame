def choose_move(observation):
    w = observation["grid_width"]
    h = observation["grid_height"]
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    resources = observation.get("resources") or []
    obstacles = set((p[0], p[1]) for p in (observation.get("obstacles") or []))
    if not resources:
        return [0, 0]

    deltas = [(-1, -1), (-1, 0), (-1, 1),
              (0, -1), (0, 0), (0, 1),
              (1, -1), (1, 0), (1, 1)]

    def inside(x, y):
        return 0 <= x < w and 0 <= y < h

    def manh(x1, y1, x2, y2):
        return abs(x1 - x2) + abs(y1 - y2)

    best = None
    # Heuristic: prefer a resource we can reach before opponent (lead), otherwise minimize opponent snatch threat.
    for dx, dy in deltas:
        nx, ny = sx + dx, sy + dy
        if not inside(nx, ny) or (nx, ny) in obstacles:
            continue

        our_best = None
        for (rx, ry) in resources:
            our_d = manh(nx, ny, rx, ry)
            op_d = manh(ox, oy, rx, ry)
            lead = op_d - our_d  # positive means we are closer than opponent
            # Score key: maximize lead; then prefer shorter our_d; then deterministic by resource coord.
            key = (-lead, our_d, rx, ry)
            if our_best is None or key < our_best[0]:
                our_best = (key, lead, our_d, rx, ry)

        # Secondary: reduce immediate overlap with opponent (shadow behavior), but deterministically.
        # Also prefer staying closer to the currently best target.
        lead = our_best[1]
        our_d = our_best[2]
        rx, ry = our_best[3]
        threat = manh(nx, ny, ox, oy)
        # Final move key: maximize lead, then our distance to target, then avoid proximity to opponent, then lexicographic move.
        move_key = (-lead, our_d, threat, dx, dy, rx, ry)
        if best is None or move_key < best[0]:
            best = (move_key, dx, dy)

    if best is None:
        return [0, 0]
    return [best[1], best[2]]

def choose_move(observation):
    w = observation.get("grid_width", 0)
    h = observation.get("grid_height", 0)
    sx, sy = observation.get("self_position", (0, 0))
    ox, oy = observation.get("opponent_position", (0, 0))

    obstacles = set()
    for p in observation.get("obstacles") or []:
        try:
            x, y = p
            obstacles.add((x, y))
        except Exception:
            pass

    resources = []
    for r in observation.get("resources") or []:
        try:
            x, y = r
            resources.append((x, y))
        except Exception:
            pass

    deltas = [(-1, -1), (-1, 0), (-1, 1),
              (0, -1), (0, 0), (0, 1),

    def inside(x, y):
        return 0 <= x < w and 0 <= y < h

        return abs(x1 - x2) + abs(y1 - y2)

    best_key = None

        if not inside(nx, ny) or (nx, ny) in obstacles:
            continue
        if not resources:
            key = (0, abs(nx - ox) + abs(ny - oy), dx, dy)
        else:
            # Choose a resource that maximizes how much closer we are than the opponent.
            # key: (best_lead, -our_dist, -opp_dist, rx, ry)
            best_for_move = None
            for rx, ry in resources:
                our_d = manh(nx, ny, rx, ry)
                op_d = manh(ox, oy, rx, ry)

        if best_key is None or key < best_key:
            best_key = key
            best_move = [dx, dy]

    return best_move
