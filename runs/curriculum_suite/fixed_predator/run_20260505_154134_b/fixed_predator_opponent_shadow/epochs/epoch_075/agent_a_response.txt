def choose_move(observation):
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    resources = observation.get("resources") or []
    obstacles = observation.get("obstacles") or []
    obs = set()
    for p in obstacles:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            obs.add((int(p[0]), int(p[1])))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def cheb(ax, ay, bx, by):
        dx = ax - bx
        if dx < 0:
            dx = -dx
        dy = ay - by
        if dy < 0:
            dy = -dy
        return dx if dx > dy else dy

    def opp_target(px, py):
        if not resources:
            return (px, py)
        best = None
        best_key = None
        for cx, cy in resources:
            ds = cheb(px, py, cx, cy)
            key = (ds, cx, cy)
            if best_key is None or key < best_key:
                best_key = key
                best = (cx, cy)
        return best

    def step_toward(px, py, tx, ty):
        dx = 0 if tx == px else (1 if tx > px else -1)
        dy = 0 if ty == py else (1 if ty > py else -1)
        # allow diagonal; if diagonal blocked, fall back deterministically
        candidates = [(dx, dy), (dx, 0), (0, dy), (0, 0)]
        for mx, my in candidates:
            nx, ny = px + mx, py + my
            if inb(nx, ny) and (nx, ny) not in obs:
                return [mx, my]
        # should not happen often
        return [0, 0]

    if not resources:
        return [0, 0]

    # Build candidate moves for self
    moves = []
    for mx in (-1, 0, 1):
        for my in (-1, 0, 1):
            nx, ny = sx + mx, sy + my
            if inb(nx, ny) and (nx, ny) not in obs:
                moves.append((mx, my))
    if not moves:
        return [0, 0]

    # Predict opponent target once (my action-agnostic) for a more decisive chase/intercept
    ttx, tty = opp_target(ox, oy)

    best_move = moves[0]
    best_val = None
    # Center preference to reduce dithering
    cx0, cy0 = (w - 1) / 2.0, (h - 1) / 2.0

    for mx, my in moves:
        nsx, nsy = sx + mx, sy + my
        # Opponent predicted one step toward its target
        pom = step_toward(ox, oy, ttx, tty)
        nox, noy = ox + pom[0], oy + pom[1]

        # Evaluate best resource we can secure relative to opponent after prediction
        val = None
        for rx, ry in resources:
            ds = cheb(nsx, nsy, rx, ry)
            do = cheb(nox, noy, rx, ry)
            # prioritize where we are closer (larger (do-ds)), then faster, then center-ish
            rel = do - ds
            cen = -(((rx - cx0) ** 2 + (ry - cy0) ** 2))
            key = (rel, -ds, cen, -rx, -ry)
            if val is None or key > val:
                val = key
        if best_val is None or val > best_val:
            best_val = val
            best_move = (mx, my)

    return [int(best_move[0]), int(best_move[1])]

    def dist(ax, ay, bx, by):
        dx = ax - bx
        if dx < 0: dx = -dx

        if dy < 0: dy = -dy
        return dx if dx > dy else dy

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    best = None
    best_val = None

    if resources:
        pts = []
        for r in resources:
            if isinstance(r, (list, tuple)) and len(r) >= 2:
                pts.append((int(r[0]), int(r[1])))
        if not pts:
            pts = [(sx, sy)]
    else:
        pts = [(sx, sy)]

    for dx, dy in moves:
        nx, ny = int(sx) + dx, int(sy) + dy
        if not inb(nx, ny) or (nx, ny) in obs:
            continue
        # Evaluate: prefer getting closer to a resource that opponent isn't as close to.
        val = None
        for rx, ry in pts:
            ds = dist(nx, ny, rx, ry)
            do = dist(ox, oy, rx, ry)
            score = ds - 0.6 * do
            if val is None or score < val:
                val = score
        # Fallback: if somehow all resources invalid, just move toward opponent.
        if val is None:
            val = dist(nx, ny, ox, oy)
        key = (val, dist(nx, ny, ox, oy), dx, dy)
        if best is None or key < best:
            best = key

            nx, ny = int(sx) + dx, int(sy) + dy
            if inb(nx, ny) and (nx, ny) not in obs:
