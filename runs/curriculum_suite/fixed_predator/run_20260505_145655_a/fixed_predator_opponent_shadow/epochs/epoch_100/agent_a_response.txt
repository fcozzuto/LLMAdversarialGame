def choose_move(observation):
    w = observation["grid_width"]
    h = observation["grid_height"]
    x, y = observation["self_position"]
    ox, oy = observation["opponent_position"]
    resources = observation.get("resources", []) or []
    obs = set((p[0], p[1]) for p in (observation.get("obstacles", []) or []))

    def in_bounds(nx, ny):
        return 0 <= nx < w and 0 <= ny < h

    def man(ax, ay, bx, by):
        return abs(ax - bx) + abs(ay - by)

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    if not resources:
        best = (None, -10**9)
        for dx, dy in dirs:
            nx, ny = x + dx, y + dy
            if not in_bounds(nx, ny) or (nx, ny) in obs:
                continue
            val = man(ox, oy, x, y) - man(ox, oy, nx, ny)
            if val > best[1]:
                best = ((dx, dy), val)
        return [best[0][0], best[0][1]] if best[0] is not None else [0, 0]

    best_move = (0, 0)
    best_score = -10**18

    def eval_move(nx, ny):
        best_contested = None  # (-d_me, d_opp)
        best_free = None       # (d_opp - d_me, -d_me)
        for rx, ry in resources:
            d_me = man(nx, ny, rx, ry)
            d_opp = man(ox, oy, rx, ry)
            if d_opp < d_me:
                cand = (-d_me, d_opp)
                if best_contested is None or cand > best_contested:
                    best_contested = cand
            else:
                cand = (d_opp - d_me, -d_me)
                if best_free is None or cand > best_free:
                    best_free = cand
        if best_contested is not None:
            # prioritize denying opponent-advantaged resources by minimizing my distance
            return (best_contested[0], best_contested[1], 1)  # lexicographic
        else:
            # otherwise race generally, with a bias to keep opponent far from that target
            return (best_free[0], -best_free[1], 0)

    for dx, dy in dirs:
        nx, ny = x + dx, y + dy
        if not in_bounds(nx, ny) or (nx, ny) in obs:
            continue
        sc = eval_move(nx, ny)
        if sc > best_score:
            best_score = sc
            best_move = (dx, dy)

    return [best_move[0], best_move[1]]
