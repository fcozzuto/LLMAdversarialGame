        return 0 <= nx < w and 0 <= ny < h

        return abs(a - c) + abs(b - d)

        nx, ny = x + dx, y + dy
        if inb(nx, ny) and (nx, ny) not in obstacles:
            safe.append((dx, dy, nx, ny))

    res_list = [tuple(r) for r in resources]
