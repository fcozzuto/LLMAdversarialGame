def choose_move(observation):
    w = observation["grid_width"]
    h = observation["grid_height"]
    x, y = observation["self_position"]
    ox, oy = observation["opponent_position"]
    resources = observation.get("resources", []) or []
    obstacles = set(tuple(p) for p in (observation.get("obstacles", []) or []))

    def sign(v):
        return 0 if v == 0 else (1 if v > 0 else -1)

    def in_bounds(nx, ny):
        return 0 <= nx < w and 0 <= ny < h

    def blocked(nx, ny):
        return (nx, ny) in obstacles

    def manh(ax, ay, bx, by):
        return abs(ax - bx) + abs(ay - by)

    def best_target(px, py):
        if not resources:
            return None
        best = resources[0]
        best_sc = None
        for tx, ty in resources:
            d_me = manh(px, py, tx, ty)
            d_opp = manh(ox, oy, tx, ty)
            sc = (d_opp - d_me) * 1000 - d_me + 0.001 * d_opp
            if best_sc is None or sc > best_sc:
                best, best_sc = [tx, ty], sc
            elif sc == best_sc:
                if (tx, ty) < (best[0], best[1]):
                    best = [tx, ty]
        return best

    if not resources:
        return [sign(ox - x), sign(oy - y)]

    target = best_target(x, y)
    tx, ty = target[0], target[1]
    dx0, dy0 = sign(tx - x), sign(ty - y)

    moves = []
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            nx, ny = x + dx, y + dy
            if in_bounds(nx, ny) and not blocked(nx, ny):
                moves.append((dx, dy))

    def cell_value(nx, ny):
        if (nx, ny) in resources:
            return 1e9
        # prefer cells that improve advantage toward target
        d_me = manh(nx, ny, tx, ty)
        d_opp = manh(ox, oy, tx, ty)
        # also lightly prefer moving closer to target and away from opponent
        return (d_opp - d_me) * 1000 - d_me - 0.01 * manh(nx, ny, ox, oy)

    # deterministic limited lookahead: try best immediate move; tie-break by lookahead one ply
    best_first = None
    best_score1 = None
    for dx, dy in moves:
        nx, ny = x + dx, y + dy
        s1 = cell_value(nx, ny)
        if best_score1 is None or s1 > best_score1 or (s1 == best_score1 and (dx, dy) < best_first):
            best_first, best_score1 = (dx, dy), s1

    # one-ply improvement for robustness
    best_move = best_first
    best_score2 = None
    for dx, dy in moves:
        nx, ny = x + dx, y + dy
        # simulate our best next step (opponent fixed)
        best_next = None
        best_s_next = None
        for ddx in (-1, 0, 1):
            for ddy in (-1, 0, 1):
                nnx, nny = nx + ddx, ny + ddy
                if in_bounds(nnx, nny) and not blocked(nnx, nny):
                    s = cell_value(nnx, nny)
                    if best_s_next is None or s > best_s_next or (s == best_s_next and (ddx, ddy) < best_next):
                        best_next, best_s_next = (ddx, ddy), s
        s2 = best_s_next if best_s_next is not None else -1e18
        if best_score2 is None or s2 > best_score2 or (s2 == best_score2 and (dx, dy) < best_move):
            best_move, best_score2 = (dx, dy), s2

    # avoid aimless wandering: if our target step is available, prefer it when close
    if (x + dx0, y + dy0) in [(x + m[0], y + m[1]) for m in moves] and manh(x, y, tx, ty) > 0:
        # prefer the target-directed move unless it is clearly dominated
        cand = (dx0, dy0)
        for dx, dy in moves:
            if (dx, dy) == cand:
                nx, ny = x + dx, y + dy
                if cell_value(nx, ny) >= best_score1:
                    return [dx, dy]

    return [best_move[0], best_move[1]]

        return inb(nx, ny) and (nx, ny) not in obstacles
