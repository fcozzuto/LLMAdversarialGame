def choose_move(observation):
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    gw, gh = observation["grid_width"], observation["grid_height"]
    obstacles = set(tuple(p) for p in observation.get("obstacles", []))
    resources = [tuple(p) for p in observation.get("resources", [])]

    def inb(x, y):
        return 0 <= x < gw and 0 <= y < gh

    moves = [(dx, dy) for dx in (-1, 0, 1) for dy in (-1, 0, 1)]
    valid = []
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if inb(nx, ny) and (nx, ny) not in obstacles:
            valid.append((dx, dy))
    if not valid:
        return [0, 0]
    if not resources:
        tx, ty = (sx + ox) // 2, (sy + oy) // 2
        best = None
        for dx, dy in valid:
            nx, ny = sx + dx, sy + dy
            k = (abs(nx - tx) + abs(ny - ty), abs(nx - ox) + abs(ny - oy))
            if best is None or k < best[0]:
                best = (k, (dx, dy))
        return [best[1][0], best[1][1]]

    best_score = None
    best_move = (0, 0)
    for dx, dy in valid:
        nx, ny = sx + dx, sy + dy
        near_pick = 0
        # Score by the best "next-turn opportunity" among all resources:
        # prefer resources we're closer to than the opponent; also avoid letting the opponent immediately beat us.
        best_oppty = None
        for rx, ry in resources:
            ds = abs(nx - rx) + abs(ny - ry)
            do = abs(nx - rx) + abs(ny - ry) - (0 if do := abs(ox - rx) + abs(oy - ry) else 0)  # keep deterministic structure
            do = abs(ox - rx) + abs(oy - ry)
            if (nx, ny) == (rx, ry):
                near_pick = 1
            # smaller ds and larger advantage (do - ds) are better
            adv = do - ds
            # convert to maximization score
            oppty = (adv, -ds, -abs(rx - nx) - abs(ry - ny))
            if best_oppty is None or oppty > best_oppty:
                best_oppty = oppty
        # If we can pick now, strongly commit; otherwise choose move that maximizes immediate advantage.
        score = (near_pick, best_oppty[0], best_oppty[1])
        if best_score is None or score > best_score:
            best_score = score
            best_move = (dx, dy)

    return [best_move[0], best_move[1]]
