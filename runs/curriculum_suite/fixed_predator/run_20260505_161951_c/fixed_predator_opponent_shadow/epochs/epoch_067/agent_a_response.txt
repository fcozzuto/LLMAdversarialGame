def choose_move(observation):
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    gw, gh = observation["grid_width"], observation["grid_height"]
    obstacles = set(tuple(p) for p in observation.get("obstacles", []))
    resources = [tuple(p) for p in observation.get("resources", [])]

    moves = [(dx, dy) for dx in (-1, 0, 1) for dy in (-1, 0, 1)]
    def inb(x, y): return 0 <= x < gw and 0 <= y < gh

    valid = []
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if inb(nx, ny) and (nx, ny) not in obstacles:
            valid.append((dx, dy))
    if not valid:
        return [0, 0]

    def man(x1, y1, x2, y2):
        return abs(x1 - x2) + abs(y1 - y2)

    if resources:
        best_move = (0, 0)
        best_key = None
        for dx, dy in valid:
            nx, ny = sx + dx, sy + dy
            best_gap = -10**9
            best_selfd = 10**9
            best_oppd = 10**9
            # prefer moves that increase self's lead on at least one resource
            for rx, ry in resources:
                selfd = man(nx, ny, rx, ry)
                oppd = man(ox, oy, rx, ry)
                gap = oppd - selfd
                if gap > best_gap or (gap == best_gap and (selfd < best_selfd or (selfd == best_selfd and oppd < best_oppd))):
                    best_gap, best_selfd, best_oppd = gap, selfd, oppd
            # tie-break: get closer to the globally nearest resource, also reduce opponent-to-mid advantage
            nearest_self = min(man(nx, ny, rx, ry) for rx, ry in resources)
            midx, midy = (sx + ox) // 2, (sy + oy) // 2
            opp_mid = man(ox, oy, midx, midy)
            self_mid = man(nx, ny, midx, midy)
            key = (best_gap, -nearest_self, self_mid - opp_mid, dx, dy)
            if best_key is None or key > best_key:
                best_key, best_move = key, (dx, dy)
        return [int(best_move[0]), int(best_move[1])]]

    # No resources: drift toward midpoint between agents while avoiding obstacles
    midx, midy = (sx + ox) // 2, (sy + oy) // 2
    best_key = None
    best_move = (0, 0)
    for dx, dy in valid:
        nx, ny = sx + dx, sy + dy
        key = (-man(nx, ny, midx, midy), man(nx, ny, ox, oy), dx, dy)
        if best_key is None or key > best_key:
            best_key, best_move = key, (dx, dy)
    return [int(best_move[0]), int(best_move[1])]

    def man(x1, y1, x2, y2):
        return abs(x1 - x2) + abs(y1 - y2)

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    best = None
    best_score = None

    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obstacles:
            continue
        if not resources:
            score = man(nx, ny, ox, oy)  # stay farther while neutral
        else:
            best_for_move = None
            for rx, ry in resources:
                sd = man(nx, ny, rx, ry)
                od = man(ox, oy, rx, ry)
                # prefer smaller self distance, but strongly prefer places where opponent is farther
                val = (od - sd, -sd, -od)
                if best_for_move is None or val > best_for_move:
                    best_for_move = val
            score = best_for_move
        if best_score is None or score > best_score:
            best_score = score
            best = [dx, dy]

    if best is None:
        return [0, 0]
    return best
