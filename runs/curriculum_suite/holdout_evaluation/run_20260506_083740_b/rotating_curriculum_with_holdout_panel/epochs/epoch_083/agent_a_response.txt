def choose_move(observation):
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    resources = observation.get("resources", []) or []
    obstacles = observation.get("obstacles", []) or []

    ob = set()
    for p in obstacles:
        if p and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                ob.add((x, y))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def valid(x, y):
        return inb(x, y) and (x, y) not in ob

    def man(a, b, c, d):
        return abs(a - c) + abs(b - d)

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    valid_resources = []
    for r in resources:
        if r and len(r) >= 2:
            rx, ry = int(r[0]), int(r[1])
            if valid(rx, ry):
                valid_resources.append((rx, ry))

    if not valid_resources:
        # Simple defensive step: maximize distance from opponent while staying valid
        best = (0, 0, -10**9)
        for dx, dy in moves:
            nx, ny = sx + dx, sy + dy
            if not valid(nx, ny):
                nx, ny = sx, sy
            score = man(nx, ny, ox, oy)
            if score > best[2]:
                best = (dx, dy, score)
        return [best[0], best[1]]

    # Choose a target resource for each agent (nearest by manhattan) to bias pursuit
    self_target = None
    opp_target = None
    best_sd = 10**9
    best_od = 10**9
    for rx, ry in valid_resources:
        sd = man(sx, sy, rx, ry)
        od = man(ox, oy, rx, ry)
        if sd < best_sd or (sd == best_sd and (rx, ry) < self_target):
            best_sd = sd
            self_target = (rx, ry)
        if od < best_od or (od == best_od and (rx, ry) < opp_target):
            best_od = od
            opp_target = (rx, ry)

    cx, cy = (w - 1) // 2, (h - 1) // 2  # slight center bias to stabilize routes

    best_dx, best_dy, best_val = 0, 0, -10**18
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not valid(nx, ny):
            nx, ny = sx, sy
            dx, dy = 0, 0

        # Attraction to own target
        sd_next = man(nx, ny, self_target[0], self_target[1])
        # "Interception": prevent opponent from getting closer to their target
        od_next = man(nx, ny, opp_target[0], opp_target[1])  # compare through our move as proxy
        opp_curr = man(ox, oy, opp_target[0], opp_target[1])
        opp_dist_gain_block = max(0, opp_curr - od_next)  # how much our position could block
        # Head-to-head separation
        sep = man(nx, ny, ox, oy)
        # Center bias (small)
        center = - (abs(nx - cx) + abs(ny - cy))

        # Score: prioritize reducing our distance to our target, while maximizing separation and blocking
        val = (-2.2 * sd_next) + (1.3 * sep) + (2.5 * opp_dist_gain_block) + (0.15 * center)
