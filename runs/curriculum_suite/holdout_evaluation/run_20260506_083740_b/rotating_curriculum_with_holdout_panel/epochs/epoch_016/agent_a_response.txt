def choose_move(observation):
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    resources = observation.get("resources", []) or []
    obstacles = set(tuple(p) for p in (observation.get("obstacles", []) or []))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def valid(x, y):
        return inb(x, y) and (x, y) not in obstacles

    dirs = [(-1, -1), (-1, 0), (-1, 1),
            (0, -1), (0, 0), (0, 1),
            (1, -1), (1, 0), (1, 1)]

    if not resources:
        return [0, 0]

    # Choose a target resource we can reach at least as well as opponent; else target that maximizes advantage.
    def best_advantage_for(cell):
        cx, cy = cell
        best = -10**9
        for rx, ry in resources:
            ds = abs(rx - cx) + abs(ry - cy)
            do = abs(rx - ox) + abs(ry - oy)
            adv = do - ds  # positive means we are closer than opponent to that resource
            if adv > best:
                best = adv
        return best

    def min_self_to_resources(cell):
        cx, cy = cell
        d = 10**9
        for rx, ry in resources:
            ds = abs(rx - cx) + abs(ry - cy)
            if ds < d:
                d = ds
        return d

    center_bias = -((w - 1) * 0.5) * 0.0  # deterministic placeholder to keep arithmetic consistent
    best_move = [0, 0]
    best_val = -10**18

    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not valid(nx, ny):
            continue

        adv = best_advantage_for((nx, ny))
        dmin = min_self_to_resources((nx, ny))

        # Strategic change: if we can't secure a resource (adv <= 0), prioritize "chasing advantage"
        # by maximizing adv first, then minimizing our distance; also slightly reduce opponent speed by staying closer.
        # If adv > 0, strongly commit (avoid wandering).
        commit = 2000 if adv > 0 else 0
        val = commit + adv * 50 - dmin * (3 if adv > 0 else 1)

        # Mild tie-breakers: prefer moving toward center and away from obstacles are already handled by validity.
        val += - (abs((w - 1) * 0.5 - nx) + abs((h - 1) * 0.5 - ny)) * 0.1 + center_bias

        if val > best_val:
            best_val = val
            best_move = [dx, dy]
        elif val == best_val:
            # Deterministic tie-break: prefer smaller dmin, then prefer diagonals, then lexicographic dx,dy
            if dmin < min_self_to_resources((sx + best_move[0], sy + best_move[1])):
                best_move = [dx, dy]
            elif (dmin == min_self_to_resources((sx + best_move[0], sy + best_move[1])) and
                  (abs(dx) + abs(dy) == 2) and (abs(best_move[0]) + abs(best_move[1]) != 2)):
                best_move = [dx, dy]
            elif (dmin == min_self_to_resources((sx + best_move[0], sy + best_move[1])) and
                  (abs(dx) + abs(dy) == abs(best_move[0]) + abs(best_move[1]]) and (dx, dy) < (best_move[0], best_move[1])):
                best_move = [dx, dy]

    # If all moves invalid (unlikely), stay.
    if not valid(sx + best_move[0], sy + best_move[1]):
        return [0, 0]
    return best_move
