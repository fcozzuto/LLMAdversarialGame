def choose_move(observation):
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    resources = observation.get("resources", []) or []
    obstacles = observation.get("obstacles", []) or []

    ob = set()
    for p in obstacles:
        if p and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                ob.add((x, y))

    def inb(x, y): return 0 <= x < w and 0 <= y < h
    def valid(x, y): return inb(x, y) and (x, y) not in ob
    moves = [(-1,-1),(-1,0),(-1,1),(0,-1),(0,0),(0,1),(1,-1),(1,0),(1,1)]

    # If no resources, drift toward center deterministically.
    if not resources:
        cx, cy = (w - 1) // 2, (h - 1) // 2
        best = (0, 0, -10**9)
        for dx, dy in moves:
            nx, ny = sx + dx, sy + dy
            if not valid(nx, ny): nx, ny = sx, sy
            center = - (abs(nx - cx) + abs(ny - cy))
            if center > best[2]:
                best = (dx, dy, center)
        return [best[0], best[1]]

    # Pick the resource the opponent is closest to; we counter it.
    best_res = None
    best_odo = 10**9
    for r in resources:
        if not r or len(r) < 2: 
            continue
        rx, ry = int(r[0]), int(r[1])
        if not valid(rx, ry): 
            continue
        d = abs(ox - rx) + abs(oy - ry)
        if d < best_odo:
            best_odo = d
            best_res = (rx, ry)

    if best_res is None:
        return [0, 0]

    rx, ry = best_res
    cx, cy = (w - 1) // 2, (h - 1) // 2

    # Evaluate move by (opp_dist - self_dist) to the counter-target, plus mild center bias,
    # plus avoid moving closer to a different resource that the opponent could take first.
    other_res = []
    for r in resources:
        if not r or len(r) < 2:
            continue
        tx, ty = int(r[0]), int(r[1])
        if valid((tx, ty)) and (tx, ty) != (rx, ry):
            other_res.append((tx, ty))

    best = None
    best_key = None
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not valid(nx, ny):
            nx, ny, dx, dy = sx, sy, 0, 0

        self_d = abs(nx - rx) + abs(ny - ry)
        opp_d = abs(ox - rx) + abs(oy - ry)

        # Encourage also to be at least not worse than opponent on the counter-target.
        advantage = (opp_d - self_d)

        # Center bias to disrupt edge patrol patterns.
        center_bias = -(abs(nx - cx) + abs(ny - cy)) * 0.02

        # Penalty if move makes a secondary resource much faster for opponent than us.
        penalty = 0.0
        for tx, ty in other_res[:6]:
            sd2 = abs(nx - tx) + abs(ny - ty)
            od2 = abs(ox - tx) + abs(oy - ty)
            # If opponent is already closer, avoid becoming even more behind.
            if od2 < sd2:
                penalty += (sd2 - od2) * 0.03
            else:
                penalty += (od2 - sd2) * 0.005

        key = (advantage + center_bias - penalty, -self_d, -abs(nx - ox) - abs(ny - oy), dx, dy)
        if best_key is None or key > best_key:
            best_key = key
            best = (dx, dy)

    return [int(best[0]), int(best[1])]
