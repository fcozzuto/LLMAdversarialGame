def choose_move(observation):
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    obs = set()
    for p in observation.get("obstacles", []) or []:
        if p and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if inb(x, y):
                obs.add((x, y))

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def ok(x, y):
        return inb(x, y) and (x, y) not in obs

    resources = []
    for r in observation.get("resources", []) or []:
        if r and len(r) >= 2:
            x, y = int(r[0]), int(r[1])
            if ok(x, y):
                resources.append((x, y))
    if not resources:
        return [0, 0]

    def dist(a, b, c, d):
        return abs(a - c) + abs(b - d)

    # Opponent's likely target (nearest resource)
    target = min(resources, key=lambda t: dist(ox, oy, t[0], t[1]))

    # Estimate opponent greedy step toward its nearest target, obstacle-aware
    best_opp_step = (0, 0)
    best_opp_d = 10**9
    for dx, dy in moves:
        nx, ny = ox + dx, oy + dy
        if not ok(nx, ny):
            continue
        d = dist(nx, ny, target[0], target[1])
        if d < best_opp_d or (d == best_opp_d and (dx, dy) < best_opp_step):
            best_opp_d = d
            best_opp_step = (dx, dy)
    ox2, oy2 = ox + best_opp_step[0], oy + best_opp_step[1]

    # Evaluate self moves by stealing potential, not just nearest-resource chasing
    best_move = (0, 0)
    best_val = -10**18
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not ok(nx, ny):
            continue
        d_self_t = dist(nx, ny, target[0], target[1])
        d_opp_t = dist(ox2, oy2, target[0], target[1])
        adv_target = d_opp_t - d_self_t  # higher is better

        # Diversify: also consider best advantage across all resources after our move
        max_adv_all = -10**9
        for rx, ry in resources:
            adv = dist(ox2, oy2, rx, ry) - dist(nx, ny, rx, ry)
            if adv > max_adv_all:
                max_adv_all = adv

        # Weight stealing opponent's likely target, but allow switching to a better one
        val = 10 * adv_target + max_adv_all
        if val > best_val or (val == best_val and (dx, dy) < best_move):
            best_val = val
            best_move = (dx, dy)
