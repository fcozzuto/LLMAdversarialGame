def choose_move(observation):
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    resources = observation.get("resources", []) or []
    obstacles = observation.get("obstacles", []) or []

    ob = set()
    for p in obstacles:
        if p and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                ob.add((x, y))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def valid(x, y):
        return inb(x, y) and (x, y) not in ob

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def md(a, b, c, d):
        return abs(a - c) + abs(b - d)

    # If no resources, drift to center lane to create future optionality
    if not resources:
        tx, ty = (w - 1) // 2, (h - 1) // 2
        dx = 0 if tx == sx else (1 if tx > sx else -1)
        dy = 0 if ty == sy else (1 if ty > sy else -1)
        nx, ny = sx + dx, sy + dy
        if valid(nx, ny):
            return [dx, dy]

        # fallback: best safe move towards center
        best = (0, 10**9, 0, 0)
        for dx, dy in moves:
            nx, ny = sx + dx, sy + dy
            if not valid(nx, ny):
                continue
            dcen = md(nx, ny, (w - 1) // 2, (h - 1) // 2)
            if (0, dcen, dx, dy) < best:
                best = (0, dcen, dx, dy)
        return [best[2], best[3]]

    # Choose a contested target: prefer cells where we are closer than opponent, then closer overall.
    best_cell = None
    best_val = (-10**9, 10**9)
    for r in resources:
        if not r or len(r) < 2:
            continue
        rx, ry = int(r[0]), int(r[1])
        if not valid(rx, ry):
            continue
        sd = md(sx, sy, rx, ry)
        od = md(ox, oy, rx, ry)
        # "contested advantage": positive when we are closer; also avoid extreme distance when advantage small
        adv = od - sd
        val = (adv, sd)
        if val > best_val:
            best_val = val
            best_cell = (rx, ry)

    rx, ry = best_cell
    # Move greedily but deterministically: first try step that minimizes our distance to target.
    best_move = (0, 0, 10**9, 10**9)
