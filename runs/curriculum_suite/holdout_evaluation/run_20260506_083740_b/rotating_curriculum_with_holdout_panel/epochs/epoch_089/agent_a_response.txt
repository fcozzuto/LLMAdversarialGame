def choose_move(observation):
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    obs = set()
    for p in observation.get("obstacles", []) or []:
        if p and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obs.add((x, y))
    def inb(x, y):
        return 0 <= x < w and 0 <= y < h
    def ok(x, y):
        return inb(x, y) and (x, y) not in obs
    resources = []
    for r in observation.get("resources", []) or []:
        if r and len(r) >= 2:
            x, y = int(r[0]), int(r[1])
            if ok(x, y):
                resources.append((x, y))
    if not resources:
        # deterministic drift toward center while avoiding obstacles
        tx, ty = (w - 1) / 2.0, (h - 1) / 2.0
        best = (0, 0)
        bestv = -10**9
        for dx, dy in moves:
            nx, ny = sx + dx, sy + dy
            if not ok(nx, ny):
                continue
            d = -((nx - tx) ** 2 + (ny - ty) ** 2)
            if d > bestv:
                bestv = d
                best = (dx, dy)
        return [int(best[0]), int(best[1])]
    def cheb(x1, y1, x2, y2):
        a = abs(x1 - x2)
        b = abs(y1 - y2)
        return a if a > b else b
    # nearest obstacle distance (as mild penalty/encouragement for spacing)
    obst = list(obs)
    def obst_dist(x, y):
        if not obst:
            return 99
        m = 99
        for ox2, oy2 in obst:
            d = abs(x - ox2) + abs(y - oy2)
            if d < m:
                m = d
        return m
    best_move = (0, 0)
    best_val = -10**18
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not ok(nx, ny):
            continue
        my_pos = (nx, ny)
        # evaluate best resource tradeoff: gain for being earlier, penalty for letting opponent be earlier
        val = 0
        my_best = 10**9
        opp_best = 10**9
        for rx, ry in resources:
            sd = cheb(nx, ny, rx, ry)
            od = cheb(ox, oy, rx, ry)
            if sd < my_best:
                my_best = sd
            if od < opp_best:
                opp_best = od
            # prefer resources where we are at least as fast; also include small preference for nearer ones
            diff = od - sd  # positive => we are faster
            # stronger incentive to flip "we are faster than opponent"
            val += (diff * 6) + (diff >= 0) * 8 - sd * 1
        # also deny opponent: make their nearest resource harder
        val += (opp_best - my_best) * 2
        # spacing from obstacles (avoid getting stuck behind them)
        val += min(6, obst_dist(nx, ny)) * 0.3
        if val > best_val:
            best_val = val
            best_move = (dx, dy)
        elif val == best_val:
            # deterministic tie-break: prefer staying still last (or first) based on turn parity
            if (dx == 0 and dy == 0) == ((observation.get("turn_index", 0) % 2) == 0):
                best_move = (dx, dy)
    return [int(best_move[0]), int(best_move[1])]
