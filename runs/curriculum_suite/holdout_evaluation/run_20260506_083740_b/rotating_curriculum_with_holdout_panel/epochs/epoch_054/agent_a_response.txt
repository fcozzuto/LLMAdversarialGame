def choose_move(observation):
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    resources = observation.get("resources", []) or []
    obstacles = observation.get("obstacles", []) or []

    ob = set()
    for p in obstacles:
        if p and len(p) >= 2:
            ob.add((int(p[0]), int(p[1])))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def valid(x, y):
        return inb(x, y) and (x, y) not in ob

    def md(ax, ay, bx, by):
        return abs(ax - bx) + abs(ay - by)

    if not resources:
        # deterministic fallback: move away from opponent if possible
        bestd = 1e9
        best = [0, 0]
        for dx in (-1, 0, 1):
            for dy in (-1, 0, 1):
                if dx == 0 and dy == 0:
                    continue
                nx, ny = sx + dx, sy + dy
                if not valid(nx, ny):
                    continue
                d = md(nx, ny, ox, oy)
                if d > bestd:
                    bestd = d
                    best = [dx, dy]
        return best if valid(sx + best[0], sy + best[1]) else [0, 0]

    cx, cy = (w - 1) / 2.0, (h - 1) / 2.0
    # Pick resource with highest advantage over opponent, tie-break by centrality and our distance
    best_r = None
    best_key = None
    for r in resources:
        if not r or len(r) < 2:
            continue
        rx, ry = int(r[0]), int(r[1])
        if not valid(rx, ry):
            continue
        d_me = md(sx, sy, rx, ry)
        d_opp = md(ox, oy, rx, ry)
        center_pen = (rx - cx) ** 2 + (ry - cy) ** 2
        # Encourage taking a resource we can reach strictly sooner; discourage near-contested
        contest_pen = 0
        if d_opp <= d_me:
            contest_pen = 50 + 5 * (d_me - d_opp)  # stronger when opponent not behind
        # Larger is better
        key = (d_opp - d_me) * 10 - contest_pen - 0.05 * d_me - 0.001 * center_pen
        if best_key is None or key > best_key:
            best_key = key
            best_r = (rx, ry)

    if best_r is None:
        return [0, 0]

    tx, ty = best_r
    # Greedy next step toward target with obstacle-avoid and slight anti-opponent pressure
    best_move = [0, 0]
    best_score = -1e18
    cur_do = md(sx, sy, ox, oy)
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            nx, ny = sx + dx, sy + dy
            if not valid(nx, ny):
                continue
            dn = md(nx, ny, tx, ty)
            do = md(nx, ny, ox, oy)
            # Prefer progress, avoid stepping into opponent proximity
            step_adv = (cur_do - do)
            opp_prox_pen = 0
            if do <= 2:
                opp_prox_pen = 80
            # Slightly prefer moves that reduce target distance and increase opponent distance
            score = (-dn) * 10 + step_adv * 2 - opp_prox_pen
            if score > best_score:
                best_score = score
                best_move = [dx, dy]
    return best_move
