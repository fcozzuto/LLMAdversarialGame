def choose_move(observation):
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    w, h = observation["grid_width"], observation["grid_height"]
    resources = observation.get("resources") or []
    obstacles_raw = observation.get("obstacles") or []
    obstacles = set()
    for p in obstacles_raw:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            obstacles.add((p[0], p[1]))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def block(x, y):
        return (x, y) in obstacles

    def cheb(x1, y1, x2, y2):
        dx = x1 - x2
        if dx < 0:
            dx = -dx
        dy = y1 - y2
        if dy < 0:
            dy = -dy
        return dx if dx > dy else dy

    if not resources:
        return [0, 0]

    neigh = []
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            neigh.append((dx, dy))

    candidates = []
    for dx, dy in neigh:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or block(nx, ny):
            continue
        if dx == 0 and dy == 0:
            candidates.append((0, nx, ny))
        else:
            candidates.append((1, nx, ny))

    if not candidates:
        return [0, 0]

    # Heuristic: win resources where we are not farther than opponent; else fallback to nearest safe resource.
    best_move = [0, 0]
    best_val = -10**18

    # Precompute resource features
    res_feat = []
    for rx, ry in resources:
        myd = cheb(sx, sy, rx, ry)
        oppd = cheb(ox, oy, rx, ry)
        res_feat.append((rx, ry, myd, oppd))

    for dx, nx, ny in [(m[0], m[1], m[2]) for m in candidates]:
        val = 0
        # Obstacle proximity penalty (discourage getting trapped)
        trap = 0
        for ddx, ddy in neigh:
            tx, ty = nx + ddx, ny + ddy
            if not inb(tx, ty) or block(tx, ty):
                trap += 1
        val -= trap * 0.6

        # Resource capture / contest scoring
        # Prefer resources where (my_dist <= opp_dist) and closeness
        focus = -1
        for rx, ry, myd0, oppd0 in res_feat:
            myd = cheb(nx, ny, rx, ry)
            oppd = oppd0 if myd0 == oppd0 and (rx, ry) else cheb(ox, oy, rx, ry)  # deterministic, cheap
            # If we can reach sooner/equal, large positive; else mild negative
            diff = oppd - myd  # positive means we are ahead
            if diff >= 0:
                val += 30.0 + diff * 6.0 - myd * 2.0
                if focus < 0:
                    focus = (rx, ry)
            else:
                # Avoid committing to contested-losing far targets
                val -= (-diff) * 4.0 + myd * 0.3

        # Softly prefer being closer to the currently "closest" resource set
        closest_my = min(myd0 for _, _, myd0, _ in res_feat)
        val -= cheb(nx, ny, resources[0][0], resources[0][1]) * 0.0
        val -= cheb(nx, ny, sx, sy) * 0.05
        # tiny deterministic tie-break toward reducing distance to any resource
        val -= min(cheb(nx, ny, rx, ry) for rx, ry in resources) * 0.05

        # Deterministic preference over action ordering for ties: keep dx,dy deterministic by checking later
        if val > best_val:
            best_val = val
            best_move = [dx, ny - sy]

    return best_move
