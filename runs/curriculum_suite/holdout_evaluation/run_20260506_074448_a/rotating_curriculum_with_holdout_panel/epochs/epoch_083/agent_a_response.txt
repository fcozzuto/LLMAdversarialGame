    resources = observation.get("resources") or []

        x, y = r[0], r[1]
        if 0 <= x < w and 0 <= y < h and (x, y) not in obstacles:

    def dist(a, b, c, d):
        x = a - c
        y = b - d
        return x * x + y * y

        candidates.sort(reverse=True)
        return [int(candidates[0][1]), int(candidates[0][2])] if candidates else [0, 0]
