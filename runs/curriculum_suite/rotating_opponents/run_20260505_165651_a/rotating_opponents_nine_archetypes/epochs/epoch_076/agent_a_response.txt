def choose_move(observation):
    w = observation.get("grid_width", 8)
    h = observation.get("grid_height", 8)
    sx, sy = observation.get("self_position", (0, 0))
    ox, oy = observation.get("opponent_position", (0, 0))
    resources = observation.get("resources") or []
    obstacles = observation.get("obstacles") or []
    obs = {(p[0], p[1]) for p in obstacles if isinstance(p, (list, tuple)) and len(p) >= 2}
    def inb(x, y): return 0 <= x < w and 0 <= y < h
    def blocked(x, y): return (not inb(x, y)) or ((x, y) in obs)
    def man(a, b, c, d):
        t = a - c
        if t < 0: t = -t
        u = b - d
        if u < 0: u = -u
        return t + u
    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    if not resources:
        # flee from opponent a bit (deterministic)
        best = None
        for dx, dy in moves:
            nx, ny = sx + dx, sy + dy
            if blocked(nx, ny): continue
            v = (man(ox, oy, nx, ny), -abs(nx - (w-1)/2) - abs(ny - (h-1)/2))
            if best is None or v > best[0]:
                best = (v, dx, dy)
        return [best[1], best[2]] if best else [0, 0]

    # Choose an opponent-leaning target to contest (center-rush tends to claim early)
    opp_target = None
    opp_best = None
    for r in resources:
        x, y = r[0], r[1]
        if not inb(x, y) or (x, y) in obs: continue
        d = man(ox, oy, x, y)
        if opp_best is None or d < opp_best:
            opp_best = d
            opp_target = (x, y)

    # Competitive resource scoring: maximize (opp_d - self_d), but also consider absolute self distance.
    best = None
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if blocked(nx, ny): continue
        # Prefer increasing distance from opponent while moving toward a contested resource.
        best_res = None
        for r in resources:
            x, y = r[0], r[1]
            if not inb(x, y) or (x, y) in obs: continue
            self_d = man(nx, ny, x, y)
            opp_d = man(ox, oy, x, y)
            adv = opp_d - self_d  # larger => we are relatively closer
            # Boost contesting the opponent's likely target
            boost = -self_d if opp_target == (x, y) else 0
            cell_pref = (adv + boost, -self_d, -man(ox, oy, nx, ny), x, y)
            if best_res is None or cell_pref > best_res:
                best_res = cell_pref
        if best_res is None:
            cell_pref = (0, -1, man(ox, oy, nx, ny), nx, ny)
        # Primary: maximize advantage; secondary: maximize opponent distance; tertiary: deterministic tie-break.
        overall = (best_res[0], best_res[1], best_res[2], best_res[3], best_res[4])
        if best is None or overall > best[0]:
            best = (overall, dx, dy)

def choose_move(observation):

    raw_obs = observation.get("obstacles") or []
    obs = set()
    for p in raw_obs:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            obs.add((p[0], p[1]))
        elif isinstance(p, dict) and "x" in p and "y" in p:
            obs.add((p["x"], p["y"]))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

        return (not inb(x, y)) or ((x, y) in obs)
