def choose_move(observation):
    w = observation.get("grid_width", 8)
    h = observation.get("grid_height", 8)
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    resources = observation.get("resources", [])
    obstacles = set(tuple(p) for p in observation.get("obstacles", []))
    if not resources:
        # No resources visible: move away from opponent toward center deterministically
        dxs = [0]
        dys = [0]
        target = (w // 2, h // 2)
    else:
        def dist(a, b):
            return max(abs(a[0] - b[0]), abs(a[1] - b[1]))
        # Choose closest resource to us (tie-break by coords)
        target = min(resources, key=lambda r: (dist((sx, sy), r), r[1], r[0]))

    target_x, target_y = target
    deltas = [(dx, dy) for dx in (-1, 0, 1) for dy in (-1, 0, 1)]
    # Deterministic preference order: toward target, then away from opponent, then lower dx/dy lexicographically
    def score_move(dx, dy):
        nx, ny = sx + dx, sy + dy
        if nx < 0 or nx >= w or ny < 0 or ny >= h:
            return (10**9, 10**9, 10**9, 10**9)
        if (nx, ny) in obstacles:
            return (10**8, 10**8, 10**8, 10**8)
        if (nx, ny) == (ox, oy):
            # Avoid collisions by penalizing
            pen = 3
        else:
            pen = 0
        d_to_target = dist((nx, ny), (target_x, target_y))
        d_to_opp = dist((nx, ny), (ox, oy))
        # Prefer smaller distance to target, larger distance from opponent
        return (d_to_target + pen, -d_to_opp, dx, dy)

    best = None
    best_s = None
    for dx, dy in deltas:
        s = score_move(dx, dy)
        if best_s is None or s < best_s:
            best_s = s
            best = (dx, dy)
    return [int(best[0]), int(best[1])]
