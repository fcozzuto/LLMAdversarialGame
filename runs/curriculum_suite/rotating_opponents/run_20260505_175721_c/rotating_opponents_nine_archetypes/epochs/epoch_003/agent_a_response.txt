def choose_move(observation):
    w = observation.get("grid_width", 8)
    h = observation.get("grid_height", 8)
    sx, sy = observation.get("self_position", (0, 0))
    ox, oy = observation.get("opponent_position", (0, 0))
    obstacles = set(tuple(p) for p in observation.get("obstacles", []))
    resources = observation.get("resources", [])

    def dist(ax, ay, bx, by):
        return abs(ax - bx) + abs(ay - by)

    if not resources:
        tx, ty = w // 2, h // 2
    else:
        # Choose target deterministically: resource where opponent is closest but I'm not too far behind
        # (helps break ties vs naive nearest-resource behavior).
        tx = sy = None
        best = None
        for rx, ry in resources:
            myd = dist(sx, sy, rx, ry)
            opd = dist(ox, oy, rx, ry)
            # Prefer making delta (opd-myd) as positive as possible; tie-break by smaller myd.
            key = (-(opd - myd), myd, ry, rx)
            if best is None or key < best:
                best = key
                tx, ty = rx, ry

    best_move = [0, 0]
    best_u = None
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            nx, ny = sx + dx, sy + dy
            if nx < 0 or nx >= w or ny < 0 or ny >= h:
                continue
            if (nx, ny) in obstacles:
                continue

            if resources:
                # Utility: how much more quickly I can reach the same resource than the opponent.
                best_delta = None
                best_myd = None
                for rx, ry in resources:
                    myd = dist(nx, ny, rx, ry)
                    opd = dist(ox, oy, rx, ry)
                    delta = opd - myd
                    if best_delta is None or (delta, -myd) > (best_delta, -best_myd):
                        best_delta = delta
                        best_myd = myd
                u = best_delta * 100 - best_myd
            else:
                u = -(dist(nx, ny, tx, ty))

            if (nx, ny) == (ox, oy):
                u += 10**6

            if best_u is None or u > best_u or (u == best_u and (dx, dy) < (best_move[0], best_move[1])):
                best_u = u
                best_move = [dx, dy]

    return best_move
