def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position", (0, 0)) or (0, 0)
    ox, oy = observation.get("opponent_position", (0, 0)) or (0, 0)
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    obstacles = set()
    for p in observation.get("obstacles") or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

    resources = []
    for r in observation.get("resources") or []:
        if isinstance(r, (list, tuple)) and len(r) >= 2:
            x, y = int(r[0]), int(r[1])
            if 0 <= x < w and 0 <= y < h and (x, y) not in obstacles:
                resources.append((x, y))
    if not resources:
        return [0, 0]

    def cheb(ax, ay, bx, by):
        dx = ax - bx
        if dx < 0: dx = -dx
        dy = ay - by
        if dy < 0: dy = -dy
        return dx if dx > dy else dy

    def in_bounds(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obstacles

    # Choose a resource where we are not worse than opponent; otherwise pick the one that maximizes (d2 - d1).
    best = None
    best_val = -10**18
    for tx, ty in resources:
        d1 = cheb(sx, sy, tx, ty)
        d2 = cheb(ox, oy, tx, ty)
        # Prefer being closer or at least tied; also slightly prefer farther-away opponent (to deny future captures).
        val = (d2 - d1) * 100 - d1
        if d1 == 0:
            val = 10**9
        if d2 < d1:
            val -= 2000  # strongly discourage hopeless grabs
        if val > best_val or (val == best_val and (tx, ty) < best):
            best_val = val
            best = (tx, ty)
    tx, ty = best

    # Move one step toward target, but avoid squares where opponent is closer/equal next turn when possible.
    moves = []
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            if dx == 0 and dy == 0:
                nx, ny = sx, sy
            else:
                nx, ny = sx + dx, sy + dy
            if in_bounds(nx, ny):
                moves.append((dx, dy, nx, ny))
    if not moves:
        return [0, 0]

    # Opponent one-step reach estimate (chebyshev): if opponent can reach the target in <= our resulting distance, avoid.
    best_m = None
    best_mval = -10**18
    for dx, dy, nx, ny in moves:
        myd = cheb(nx, ny, tx, ty)
        opd = cheb(ox, oy, tx, ty)
        # Penalize staying too far when opponent can contest; encourage reducing distance and centrality.
        val = (opd - myd) * 120 - myd
        # Slightly bias away from edges only if opponent is edge-patrol style (opponent near border).

        if ox == 0 or oy == 0 or ox == w - 1 or oy == h - 1:
            edge = 1
        cx, cy = (nx - (w - 1) / 2.0), (ny - (h - 1) / 2.0)
