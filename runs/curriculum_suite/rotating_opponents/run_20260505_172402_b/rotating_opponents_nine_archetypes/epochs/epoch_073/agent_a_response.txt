                resources.append((x, y))

    def inside(nx, ny):
        return 0 <= nx < w and 0 <= ny < h and (nx, ny) not in obstacles

    from collections import deque
    q = deque()
    q.append((sx, sy))

    parent = {(sx, sy): None}
    while q:

                parent[(nx, ny)] = (x, y)

    def cheb(ax, ay, bx, by):

        dy = ay - by
