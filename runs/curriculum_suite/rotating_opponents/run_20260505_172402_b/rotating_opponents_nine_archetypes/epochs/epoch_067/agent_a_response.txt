            for rx, ry in resources:

                ddy = y - ry

                if best is None or d < best:

            return best * 10.0 + d_op
        return d_op

    target_y = sy

        mx, my = resources[0]
        for rx, ry in resources[1:]:
            if abs(rx - sx) + abs(ry - sy) < abs(mx - sx) + abs(my - sy):

        target_x, target_y = mx, my
