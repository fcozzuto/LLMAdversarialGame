def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sp = observation.get("self_position") or [0, 0]
    op = observation.get("opponent_position") or [w - 1, h - 1]
    sx, sy = int(sp[0]), int(sp[1])
    ox, oy = int(op[0]), int(op[1])

    obstacles = set()
    for p in observation.get("obstacles") or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

    resources = []
    for p in observation.get("resources") or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h and (x, y) not in obstacles:
                resources.append((x, y))

    def cheb(ax, ay, bx, by):
        dx = ax - bx
        if dx < 0: dx = -dx
        dy = ay - by
        if dy < 0: dy = -dy
        return dx if dx > dy else dy

    def valid(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obstacles

    if not valid(sx, sy):
        sx = 0 if sx < 0 else (w - 1 if sx >= w else sx)
        sy = 0 if sy < 0 else (h - 1 if sy >= h else sy)
        if not valid(sx, sy):
            for y in range(h):
                for x in range(w):
                    if valid(x, y):
                        sx, sy = x, y
                        break
                else:
                    continue
                break

    if not resources:
        return [0, 0]

    cx, cy = w // 2, h // 2

    # Pick a target where we are relatively closer than opponent (diagonal-friendly).
    best_t = None
    best_key = None
    for rx, ry in resources:
        sd = cheb(sx, sy, rx, ry)
        od = cheb(ox, oy, rx, ry)
        # Prefer strong advantage; break ties by closer to us, then closer to center.
        key = (-(od - sd), sd, abs(rx - cx) + abs(ry - cy))
        if best_key is None or key < best_key:
            best_key = key
            best_t = (rx, ry)
    tx, ty = best_t

    moves = []
    for dy in (-1, 0, 1):
        for dx in (-1, 0, 1):
            nx, ny = sx + dx, sy + dy
            # Prefer valid cells; invalid move will be rejected by engine, but we avoid it.
            if valid(nx, ny):
                moves.append((dx, dy, nx, ny))
    if not moves:
        return [0, 0]

    # Score next state by change in "advantage to target" plus anti-trap slight center pull.
    best = None
    best_m = (0, 0)
    for dx, dy, nx, ny in moves:
        sd_now = cheb(sx, sy, tx, ty)
        od_now = cheb(ox, oy, tx, ty)
        sd_next = cheb(nx, ny, tx, ty)
        # Opponent likely moves; approximate by their current distance (no prediction data).
        od_next = od_now
        adv_now = od_now - sd_now
        adv_next = od_next - sd_next
        # Use improvement first; then reduce our distance; then prefer center.
        imp = adv_next - adv_now
        key = (-imp, sd_next, abs(nx - cx) + abs(ny - cy), dx, dy)
        if best is None or key < best:
            best = key
            best_m = (dx, dy)

    return [int(best_m[0]), int(best_m[1])]
