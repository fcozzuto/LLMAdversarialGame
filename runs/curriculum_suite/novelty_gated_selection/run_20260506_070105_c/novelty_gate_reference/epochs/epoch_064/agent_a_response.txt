def choose_move(observation):
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    sp = observation.get("self_position", [0, 0])
    op = observation.get("opponent_position", [0, 0])
    sx, sy = int(sp[0]), int(sp[1])
    ox, oy = int(op[0]), int(op[1])

    obstacles = set()
    for p in observation.get("obstacles") or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

    resources = []
    for r in observation.get("resources") or []:
        if isinstance(r, (list, tuple)) and len(r) >= 2:
            x, y = int(r[0]), int(r[1])
            if 0 <= x < w and 0 <= y < h and (x, y) not in obstacles:
                resources.append((x, y))

    def cheb(ax, ay, bx, by):
        dx = abs(ax - bx)
        dy = abs(ay - by)
        return dx if dx > dy else dy

    def in_bounds(x, y):
        return 0 <= x < w and 0 <= y < h

    def best_target():
        if not resources:
            return None
        # Prefer resources we can reach sooner than opponent; lightly bias against late-game far targets.
        best = None
        best_sc = -10**18
        for rx, ry in resources:
            ds = cheb(sx, sy, rx, ry)
            do = cheb(ox, oy, rx, ry)
            tempo = do - ds  # positive => we are closer
            y_contest = 1 if ry == oy else 0
            # Mild preference for resources on "our sweep" direction to avoid dithering near edges.
            # If opponent is below us, we push upward sweep; otherwise push downward.
            sweep_dir = -1 if oy > sy else 1
            sweep = -(ry - sy) * sweep_dir
            sc = 2.6 * tempo + 0.9 * y_contest + 0.15 * sweep - 0.55 * ds
            if sc > best_sc:
                best_sc = sc
                best = (rx, ry)
        return best

    target = best_target()

    # If no resources, move toward center while still respecting bounds/obstacles.
    if target is None:
        cx, cy = w // 2, h // 2
        tx, ty = cx, cy
    else:
        tx, ty = target

    dx = 0
    if tx > sx:
        dx = 1
    elif tx < sx:
        dx = -1
    dy = 0
    if ty > sy:
        dy = 1
    elif ty < sy:
        dy = -1

    # Candidate moves: primary diagonal/axis step toward target, then alternatives avoiding obstacles.
    candidates = []
    candidates.append((dx, dy))
    if dx != 0 and dy != 0:
        candidates.append((dx, 0))
        candidates.append((0, dy))
    candidates.append((0, 0))
    # Deterministic tie-breakers: prefer moves that reduce cheb distance to target.
    uniq = []
    seen = set()
    for c in candidates:
        if c not in seen:
            seen.add(c)
            uniq.append(c)

    best_m = (0, 0)
    best_v = -10**18
    for mx, my in uniq:
        nx, ny = sx + mx, sy + my
        if not in_bounds(nx, ny):
            continue
        if (nx, ny) in obstacles:
            continue
        # Heuristic value: get closer to target; also prevent stepping onto opponent (slight).
        dist_now = cheb(nx, ny, tx, ty)
        dist_op = cheb(nx, ny, ox, oy)
        v = -dist_now + 0.08 * dist_op
        if v > best_v:
            best_v = v
            best_m = (mx, my)

    mx, my = best_m
    if mx < -1:
        mx = -1
    if mx > 1:
        mx = 1
    if my < -1:
        my = -1
    if my > 1:
        my = 1
    return [int(mx), int(my)]

    res = []
    for r in observation.get("resources") or []:
        if isinstance(r, (list, tuple)) and len(r) >= 2:

            if 0 <= x < w and 0 <= y < h and (x, y) not in obs:
                res.append((x, y))

    def cheb(ax, ay, bx, by):
        dx = ax - bx
        if dx < 0: dx = -dx
        dy = ay - by
        if dy < 0: dy = -dy
        return dx if dx > dy else dy

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    target = None
    if res:
        best = None
        for rx, ry in res:
            ds = cheb(sx, sy, rx, ry)
            do = cheb(ox, oy, rx, ry)
            sc = do - ds
            key = (sc, -ds, -rx, -ry)
            if best is None or key > best[0]:
                best = (key, (rx, ry))
        target = best[1]

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    best_move = (0, 0)
    best_key = None

    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obs:
            continue
        if target is not None:
            tx, ty = target
            ds2 = cheb(nx, ny, tx, ty)
            do2 = cheb(ox, oy, tx, ty)
            key = (do2 - ds2, -ds2, -nx, -ny)
        else:
            do2 = cheb(nx, ny, ox, oy)
            ds2 = cheb(nx, ny, sx, sy)
            key = (-do2, -ds2, -nx, -ny)
        if best_key is None or key > best_key:
            best_key = key
            best_move = (dx, dy)
