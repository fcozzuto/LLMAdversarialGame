def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)

    def parse_pos(p):
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            try:
                x, y = int(p[0]), int(p[1])
            except Exception:
                return None
        elif isinstance(p, dict):
            x = p.get("x", p.get("X", p.get("col", p.get("cx", 0))))
            y = p.get("y", p.get("Y", p.get("row", p.get("cy", 0))))
            try:
                x, y = int(x), int(y)
            except Exception:
                return None
        else:
            return None
        if 0 <= x < w and 0 <= y < h:
            return (x, y)
        return None

    sp = parse_pos(observation.get("self_position", (0, 0)))
    op = parse_pos(observation.get("opponent_position", (w - 1, h - 1)))
    if sp is None:
        sp = (0, 0)
    if op is None:
        op = (w - 1, h - 1)
    sx, sy = sp
    ox, oy = op

    resources = []
    for r in (observation.get("resources", []) or []):
        rp = parse_pos(r)
        if rp is not None:
            resources.append(rp)

    if not resources:
        return [0, 0]

    obstacles = set()
    for o in (observation.get("obstacles", []) or []):
        op2 = parse_pos(o)
        if op2 is not None:
            obstacles.add(op2)

    def md(x1, y1, x2, y2):
        return abs(x1 - x2) + abs(y1 - y2)

    best = None
    best_key = None
    for tx, ty in resources:
        sd = md(sx, sy, tx, ty)
        od = md(ox, oy, tx, ty)
        key = (od - sd, -sd)  # maximize advantage; then minimize self distance
        if best is None or key > best_key:
            best = (tx, ty)
            best_key = key
    tx, ty = best

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    best_move = (0, 0)
    best_score = None
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not (0 <= nx < w and 0 <= ny < h):
            continue
        obstacle_pen = 1
        if (nx, ny) in obstacles:
            continue
        # Prefer reducing distance; secondarily move to contest (closer than opponent)
        nd = md(nx, ny, tx, ty)
        pod = md(nx, ny, ox, oy)
        score = (-nd, pod, 0)  # deterministic order via tuple
        if best_score is None or score > best_score:
            best_score = score
            best_move = (dx, dy)

    if best_score is None:
        # Fallback: allow obstacle cells if all are blocked (engine would keep us anyway if invalid)
        best_score = None
        for dx, dy in dirs:
            nx, ny = sx + dx, sy + dy
            if not (0 <= nx < w and 0 <= ny < h):
                continue
            nd = md(nx, ny, tx, ty)
            score = (-nd, 0)
            if best_score is None or score > best_score:
                best_score = score
                best_move = (dx, dy)

    return [int(best_move[0]), int(best_move[1])]
