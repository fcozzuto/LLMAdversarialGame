def choose_move(observation):
    sx, sy = observation.get("self_position", (0, 0))
    ox, oy = observation.get("opponent_position", (sx, sy))
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    resources = observation.get("resources", []) or []
    obstacles = observation.get("obstacles", []) or []

    obs = set()
    for p in obstacles:
        try:
            obs.add((int(p[0]), int(p[1])))
        except Exception:
            pass

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def man(x1, y1, x2, y2):
        return abs(x1 - x2) + abs(y1 - y2)

    dirs = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    res = []
    for r in resources:
        try:
            rx, ry = int(r[0]), int(r[1])
            if inb(rx, ry):
                res.append((rx, ry))
        except Exception:
            pass

    if not res:
        dx, dy = (1 if ox > sx else -1 if ox < sx else 0), (1 if oy > sy else -1 if oy < sy else 0)
        nx, ny = sx + dx, sy + dy
        if inb(nx, ny) and (nx, ny) not in obs:
            return [dx, dy]
        return [0, 0]

    best_score = -10**18
    best_move = [0, 0]

    # Precompute for opponent-denial pressure
    opp_best_d = 10**9
    opp_best_r = None
    for rx, ry in res:
        d = man(ox, oy, rx, ry)
        if d < opp_best_d:
            opp_best_d = d
            opp_best_r = (rx, ry)

    my_best_r = None
    my_best_d = 10**9
    for rx, ry in res:
        d = man(sx, sy, rx, ry)
        if d < my_best_d:
            my_best_d = d
            my_best_r = (rx, ry)

    # Candidate target is either my nearest or the one opponent is nearer to (deny race)
    deny_target = opp_best_r
    if opp_best_r is not None and my_best_r is not None:
        my_d_to_opp = man(sx, sy, opp_best_r[0], opp_best_r[1])
        if my_d_to_opp > opp_best_d:
            # If I'm behind, still move to reduce the gap; avoid overcommitting.
            pass

    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obs:
            continue

        # Primary: progress to my nearest
        d_my = 10**9
        for rx, ry in res:
            d = man(nx, ny, rx, ry)
            if d < d_my:
                d_my = d

        # Secondary: if opponent is closer to some resource, try to reduce its reachability
        d_denied = 10**9
        if deny_target is not None:
            d_denied = man(nx, ny, deny_target[0], deny_target[1])

        # Tertiary: avoid getting too close to opponent (resource deniers often "shade" access)
        d_opp = man(nx, ny, ox, oy)

        score = 0
        score += -2.2 * d_my
        if deny_target is not None:
            # Encourage moving toward denied target mainly when opponent is likely to grab it
            gap = opp_best_d - my_best_d
            score += -1.4 * d_denied + 0.25 * (gap)
        score += 0.06 * d_opp

        # Deterministic tie-break: prefer moves with smallest (abs(dx)+abs(dy)) then lexicographic
        t = (abs(dx) + abs(dy), dx, dy)
        if score > best_score or (score == best_score and t < (abs(best_move[0]) + abs(best_move[1]), best_move[0], best_move[1])):
            best_score = score

        return 0 <= x < w and 0 <= y < h

    best = None
    best_sc = -10**18
    cur_near = min((md(sx, sy, rx, ry) for rx, ry in res), default=10**9)
    for dx, dy in dirs:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obs:
