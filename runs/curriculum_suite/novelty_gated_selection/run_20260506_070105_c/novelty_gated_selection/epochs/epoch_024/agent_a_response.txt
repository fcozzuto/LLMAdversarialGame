def choose_move(observation):
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    w, h = observation["grid_width"], observation["grid_height"]
    resources = observation.get("resources", []) or []
    obstacles = set(tuple(p) for p in (observation.get("obstacles", []) or []))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def man(x1, y1, x2, y2):
        return abs(x1 - x2) + abs(y1 - y2)

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    if not resources:
        return [0, 0]

    # Focus on resources that opponent is likely to grab soon; we contest those (denial).
    rs = [(man(ox, oy, rx, ry), rx, ry) for rx, ry in resources]
    rs.sort()
    top = rs[:3]

    best_delta = (0, 0)
    best_val = -10**18
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obstacles:
            nx, ny = sx, sy

        # Primary: maximize closeness advantage vs opponent for soonest-opponent resources.
        val = 0
        for opp_d, rx, ry in top:
            my_d = man(nx, ny, rx, ry)
            gap = opp_d - my_d  # positive => we are closer
            if my_d == 0:
                val += 20  # likely pick up immediately
            val += min(8, gap) * 2.0

        # Secondary: reduce overall opponent reach (even to non-top resources).
        # (Small deterministic "sweep" pressure.)
        for opp_d, rx, ry in rs[:6]:
            my_d = man(nx, ny, rx, ry)
            val += 0.25 * min(4, opp_d - my_d)

        # Tertiary: avoid being trapped by favoring moves that decrease min distance to any resource.
        min_my_d = min(m for _, _, _ in rs)
        if rs:
            min_my_d = min(man(nx, ny, rx, ry) for _, rx, ry in rs)
        val -= 0.05 * min_my_d

        if val > best_val:
            best_val = val
            best_delta = (nx - sx, ny - sy)

    return [int(best_delta[0]), int(best_delta[1])]

def choose_move(observation):
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    w, h = observation["grid_width"], observation["grid_height"]
    resources = observation.get("resources") or []
    obs_list = observation.get("obstacles") or []
    obstacles = set(tuple(p) for p in obs_list)

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    moves = [(-1, -1), (-1, 0), (-1, 1),
             (0, -1), (0, 0), (0, 1),
             (1, -1), (1, 0), (1, 1)]

    if not resources:
        return [0, 0]

    def man(x1, y1, x2, y2):
        return abs(x1 - x2) + abs(y1 - y2)

    scored = []
    for rx, ry in resources:
        od = man(ox, oy, rx, ry)
        scored.append((od, rx, ry))
    scored.sort()
    top = scored[:3]

    best = (float("-inf"), 0, 0)
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obstacles:
            nx, ny = sx, sy

        val = 0.0
        for od, rx, ry in top:
            md = man(nx, ny, rx, ry)
            if md == 0:
                val += 1000.0
            if md <= od:
                val += 80.0 - md
            else:
                val += max(-50.0, (od - md) * 3.0)
            val += 1.0 / (1.0 + md)

        if val > best[0]:
            best = (val, dx, dy)
    return [int(best[1]), int(best[2])]
