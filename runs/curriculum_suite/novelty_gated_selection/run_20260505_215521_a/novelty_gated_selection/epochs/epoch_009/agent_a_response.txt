def choose_move(observation):

    ox, oy = observation.get("opponent_position", [0, 0])
    w = observation.get("grid_width", 8)
    h = observation.get("grid_height", 8)

    resources = observation.get("resources", []) or []
    obstacles = observation.get("obstacles", []) or []
    obst = set()
    for p in obstacles:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            obst.add((p[0], p[1]))

    def man(x1, y1, x2, y2):
        dx = x1 - x2
        if dx < 0: dx = -dx
        dy = y1 - y2
        if dy < 0: dy = -dy
        return dx + dy

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    # Candidate next positions (including stay)
    moves = []
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            nx, ny = sx + dx, sy + dy
            if inb(nx, ny) and (nx, ny) not in obst:
                moves.append((dx, dy, nx, ny))

    if not moves:
        return [0, 0]

    # If we can beat opponent to any resource, go for the best "race" target.
    # Otherwise, posture defensively toward center and away from opponent's closest resource.
    cx, cy = (w - 1) / 2.0, (h - 1) / 2.0
    opp_best = None
    for r in resources:
        if not (isinstance(r, (list, tuple)) and len(r) >= 2):

        d = man(ox, oy, tx, ty)

                if ax == 0 and ay == 0:

                px, py = nx + ax, ny + ay
                if (px, py) in obst:
                    adj_pen += 1
