def choose_move(observation):
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))

    obst = set()
    for p in observation.get("obstacles", []) or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            obst.add((int(p[0]), int(p[1])))

    resources = []
    for r in observation.get("resources", []) or []:
        if isinstance(r, (list, tuple)) and len(r) >= 2:
            resources.append((int(r[0]), int(r[1])))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def md(x1, y1, x2, y2):
        dx = x1 - x2
        if dx < 0:
            dx = -dx
        dy = y1 - y2
        if dy < 0:
            dy = -dy
        return dx + dy

    moves = []
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            nx, ny = sx + dx, sy + dy
            if inb(nx, ny) and (nx, ny) not in obst:
                moves.append((dx, dy, nx, ny))
    if not moves:
        return [0, 0]
    if not resources:
        # Static-ish guard: head toward center to be ready
        tx, ty = (w - 1) // 2, (h - 1) // 2
        best = None
        for dx, dy, nx, ny in moves:
            key = (md(nx, ny, tx, ty), md(ox, oy, nx, ny))
            if best is None or key < best[0]:
                best = (key, dx, dy)
        return [best[1], best[2]]

    # Predict opponent's nearest resource and contest it
    opp_target = min(resources, key=lambda t: md(ox, oy, t[0], t[1]))
    tx, ty = opp_target
    opp_d0 = md(ox, oy, tx, ty)

    # Also consider our nearest resource as fallback
    my_target = min(resources, key=lambda t: md(sx, sy, t[0], t[1]))
    mx, my = my_target

    cx, cy = (w - 1) // 2, (h - 1) // 2

    # Deterministic evaluation
    best_key = None
    best_move = (0, 0)
    for dx, dy, nx, ny in moves:
        d_self_opp = md(nx, ny, tx, ty)
        d_opp_after = md(ox, oy, tx, ty)  # unchanged by our move (no opponent model step)
        d_self_my = md(nx, ny, mx, my)

        # "Block" heuristic: reduce distance to a point on the line to opponent target using step sign
        sx1 = 0
        if tx > ox:
            sx1 = 1
        elif tx < ox:
            sx1 = -1
        sy1 = 0
        if ty > oy:
            sy1 = 1
        elif ty < oy:
            sy1 = -1
        bx, by = ox + sx1, oy + sy1
        d_block = md(nx, ny, bx, by)

        # Center pressure for center_rush matchup
        d_center = md(nx, ny, cx, cy)

        # Key: primary minimize our distance to contested target,
        # secondary deny via being closer to the block point,
        # tertiary progress on our own resource, and stay involved in mid-board.
        key = (
            d_self_opp,
            -min(opp_d0, d_block),   # higher block closeness wins
            d_self_my,
            d_center,
            dx * 10 + dy            # deterministic tie-break
        )
        if best_key is None or key < best_key:
            best_key = key
            best_move = (dx, dy)

    return [best_move[0], best_move[1]]
