def choose_move(observation):
    w = int(observation.get("grid_width", 0) or 0)
    h = int(observation.get("grid_height", 0) or 0)
    sp = observation.get("self_position", [0, 0]) or [0, 0]
    op = observation.get("opponent_position", [0, 0]) or [0, 0]
    sx, sy = int(sp[0]), int(sp[1])
    ox, oy = int(op[0]), int(op[1])

    obst = set()
    for p in observation.get("obstacles", []) or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            obst.add((int(p[0]), int(p[1])))

    resources = []
    for r in observation.get("resources", []) or []:
        if isinstance(r, (list, tuple)) and len(r) >= 2:
            resources.append((int(r[0]), int(r[1])))

    if w <= 0 or h <= 0:
        return [0, 0]

    moves = [(-1, -1), (-1, 0), (-1, 1),
             (0, -1), (0, 0), (0, 1),
             (1, -1), (1, 0), (1, 1)]

    def in_bounds(x, y):
        return 0 <= x < w and 0 <= y < h

    def best_distance_to_resources(x, y):
        if not resources:
            return 10**9
        md = 10**9
        for rx, ry in resources:
            d = abs(x - rx) + abs(y - ry)
            if d < md:
                md = d
        return md

    def min_dist_to_opponent(x, y):
        return abs(x - ox) + abs(y - oy)

    # Bias: opponent tends toward center; if we're not close, slightly prefer progressing toward resources on our side.
    cx, cy = (w - 1) // 2, (h - 1) // 2
    my_center_adv = -(abs(sx - cx) + abs(sy - cy))

    best_move = (0, 0)
    best_val = -10**18
    close_opp = (min_dist_to_opponent(sx, sy) <= 3)

    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not in_bounds(nx, ny) or (nx, ny) in obst:
            continue

        dres = best_distance_to_resources(nx, ny)
        dop = min_dist_to_opponent(nx, ny)

        # Resource value: closer is better, with strong reward for stepping on a resource.
        res_val = 0
        if resources:
            if any((nx, ny) == (rx, ry) for rx, ry in resources):
                res_val += 10**6
            res_val += (200 - dres) * 1000

        # Opponent avoidance: only strongly when close, otherwise mild.
        if close_opp:
            opp_val = dop * 900
            opp_pen = - (3 - min(dop, 3)) * 8000 if dop < 3 else 0
            opp_val += opp_pen
        else:
            # Mildly favor staying away and not walking into center-rush collision.
            opp_val = dop * 150 + (my_center_adv - (-(abs(nx - cx) + abs(ny - cy)))) * 40

        # Obstacle-edge preference: prefer moves that reduce chance of getting blocked next step.
        edge_pen = 0
        # Try to see if next move options shrink too much.
        free_next = 0
        for ddx, ddy in moves:
            tx, ty = nx + ddx, ny + ddy
            if in_bounds(tx, ty) and (tx, ty) not in obst:
                free_next += 1
        edge_pen = (4 - min(free_next, 4)) * 60

        val = res_val + opp_val - edge_pen

        if val > best_val:
            best_val = val
            best_move = (dx, dy)
        elif val == best_val:
            # Deterministic tie-breaker: prefer moves that reduce dist to nearest resource; then increase opponent distance.
            cur_dres = best_distance_to_resources(sx + best_move[0], sy + best_move[1])
            if dres < cur_dres:
                best_move = (dx, dy)
            elif dres == cur_dres:
                cur_dop = min_dist_to_opponent(sx + best_move[0], sy + best_move[1])
                if dop > cur_dop:
                    best_move = (dx, dy)

    return [int(best_move[0]), int(best_move[1])]
