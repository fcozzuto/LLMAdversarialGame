def choose_move(observation):
    w = int(observation.get("grid_width", 0) or 0)
    h = int(observation.get("grid_height", 0) or 0)
    sp = observation.get("self_position", [0, 0]) or [0, 0]
    op = observation.get("opponent_position", [0, 0]) or [0, 0]
    sx, sy = int(sp[0]), int(sp[1])
    ox, oy = int(op[0]), int(op[1])

    if w <= 0 or h <= 0:
        return [0, 0]

    obst = set()
    for p in observation.get("obstacles", []) or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            obst.add((int(p[0]), int(p[1])))

    resources = []
    for r in observation.get("resources", []) or []:
        if isinstance(r, (list, tuple)) and len(r) >= 2:
            resources.append((int(r[0]), int(r[1])))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def cheb(a, b):
        ax, ay = a
        bx, by = b
        da = abs(ax - bx)
        db = abs(ay - by)
        return da if da >= db else db

    moves = [(-1, -1), (-1, 0), (-1, 1),
             (0, -1), (0, 0), (0, 1),
             (1, -1), (1, 0), (1, 1)]

    # Choose target: maximize "steal priority" = opponent farther than us.
    if resources:
        best = None
        for r in resources:
            sd = cheb((sx, sy), r)
            od = cheb((ox, oy), r)
            adv = od - sd  # larger => we are closer
            # tie-break deterministically: smaller sd, then x,y
            cand = (-(adv), sd, r[0], r[1])
            if best is None or cand < best:
                best = cand
        tx, ty = best[2], best[3]
    else:
        # No visible resources: drift toward center to reduce opponent reach.
        tx, ty = (w // 2), (h // 2)

    # Greedy step toward best target with obstacle-aware fallback.
    best_move = (10**9, 10**9, 0, 0)  # (score, dist, dx, dy)
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obst:
            continue
        nd = cheb((nx, ny), (tx, ty))
        # Evaluate "steal" at next step (also reduces giving opponent shorter access).
        opp_d = cheb((ox, oy), (tx, ty))
        self_d = nd
        steal = opp_d - self_d
        # Also consider moving away from opponent slightly if stealing is not improving.
        away = cheb((nx, ny), (ox, oy))
        score = (-(steal), self_d, -away)
        # Convert lexicographic tuple to comparable deterministic ranking
        rank = (score[0], score[1], -score[2], dx, dy)
        if rank < (best_move[0], best_move[1], best_move[2], best_move[3], best_move[4]):
            best_move = (rank[0], rank[1], rank[3], rank[4])
    # If all moves invalid, stay.
    if best_move[2] == 0 and best_move[3] == 0:
        return [0, 0]
    return [best_move[2], best_move[3]]

    obst = set()
    for p in observation.get("obstacles") or []:

            obst.add((x, y))
        except:
            pass

    def parse_xy(item):

            return int(item[0]), int(item[1])
        except:
            return None

    resources = []
    for r in observation.get("resources") or []:
        xy = parse_xy(r)
        if xy is not None:
            resources.append(xy)

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def cheb(ax, ay, bx, by):
        dx = ax - bx
        if dx < 0:
            dx = -dx
        dy = ay - by
        if dy < 0:
            dy = -dy
        return dx if dx >= dy else dy

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    target = None

        best = None
        for rx, ry in resources:
            sd = cheb(sx, sy, rx, ry)
            od = cheb(ox, oy, rx, ry)
            # Prefer resources closer to us; break ties by making opponent farther.
            key = (sd, -od, rx, ry)
            if best is None or key < best:
                best = key
                target = (rx, ry)

    def valid_step(dx, dy):
        nx, ny = sx + dx, sy + dy
        return inb(nx, ny) and (nx, ny) not in obst

    if target is None:
        for dx, dy in moves:
            if valid_step(dx, dy):
                return [dx, dy]
        return [0, 0]

    rx, ry = target
    best_move = None
    best_key = None
    for dx, dy in moves:
        if not valid_step(dx, dy):
            continue
        nx, ny = sx + dx, sy + dy
        dist = cheb(nx, ny, rx, ry)
