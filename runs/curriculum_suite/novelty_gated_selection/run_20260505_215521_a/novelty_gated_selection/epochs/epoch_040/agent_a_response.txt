    ox, oy = observation.get("opponent_position", (0, 0))

    h = int(observation.get("grid_height", 1)) or 1

    def to_xy(v):

            return int(v[0]), int(v[1])
        return None

        d = ax - bx
        if d < 0:

        e = ay - by
        if e < 0:
            e = -e
        return d + e

    best_move = (0, 0)
