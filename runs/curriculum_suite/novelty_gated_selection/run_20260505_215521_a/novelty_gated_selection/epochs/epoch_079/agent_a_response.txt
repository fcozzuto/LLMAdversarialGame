def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sp = observation.get("self_position") or [0, 0]
    op = observation.get("opponent_position") or [0, 0]
    sx, sy = int(sp[0]), int(sp[1])
    ox, oy = int(op[0]), int(op[1])

    obst = set()
    for p in observation.get("obstacles") or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            obst.add((int(p[0]), int(p[1])))

    resources = []
    for r in observation.get("resources") or []:
        if isinstance(r, (list, tuple)) and len(r) >= 2:
            resources.append((int(r[0]), int(r[1])))

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def man(a, b, c, d):
        return abs(a - c) + abs(b - d)

    # Evaluate moves by "resource contest margin": maximize (opponent_distance - my_distance).
    # This steers toward resources we can reach first, otherwise toward those that still deny.
    best = (-10**9, 0, 0)
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if not inb(nx, ny) or (nx, ny) in obst:
            continue

        if resources:
            best_margin = -10**9
            best_selfd = 10**9
            for rx, ry in resources:
                myd = man(nx, ny, rx, ry)
                opd = man(ox, oy, rx, ry)
                margin = opd - myd
                if margin > best_margin or (margin == best_margin and myd < best_selfd):
                    best_margin = margin
                    best_selfd = myd
            # Tie-break: prefer not staying if equal contest, and keep away from opponent a bit.
            opp_close = man(nx, ny, ox, oy)
            score = (best_margin * 1000) + (10 if (dx != 0 or dy != 0) else 0) - opp_close
            cand = (score, best_margin, -best_selfd)
        else:
            # No resources: maximize distance from opponent while avoiding obstacles (greedy).
            d = max(abs(nx - ox), abs(ny - oy))
            score = d * 1000 + (10 if (dx != 0 or dy != 0) else 0)
            cand = (score, d, -man(nx, ny, ox, oy))

        if cand > best:
            best = cand
            best_move = [dx, dy]

    # Always return something valid; if all moves blocked, stay.
    return best_move if 'best_move' in locals() else [0, 0]
