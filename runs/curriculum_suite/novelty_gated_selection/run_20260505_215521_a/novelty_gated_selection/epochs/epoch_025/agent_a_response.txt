def choose_move(observation):
    sx, sy = observation["self_position"]
    ox, oy = observation["opponent_position"]
    w = int(observation["grid_width"])
    h = int(observation["grid_height"])
    resources = observation.get("resources", []) or []
    obstacles = observation.get("obstacles", []) or []

    obst = set()
    for p in obstacles:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            obst.add((int(p[0]), int(p[1])))

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def man(x1, y1, x2, y2):
        dx = x1 - x2
        if dx < 0:
            dx = -dx
        dy = y1 - y2
        if dy < 0:
            dy = -dy
        return dx + dy

    def blocked(x, y):
        return not inb(x, y) or (x, y) in obst

    if not resources:
        # Fallback: deterministic drift to center-right to create pressure.
        tx, ty = w - 2, h // 2
        best = None
        for dx, dy in moves:
            nx, ny = sx + dx, sy + dy
            if blocked(nx, ny):
                continue
            key = (man(nx, ny, tx, ty), dx, dy)
            if best is None or key < best[0]:
                best = (key, [dx, dy])
        return best[1] if best else [0, 0]

    # Resource contest: pick a resource that is currently closer to opponent than to us; if none, pick our nearest.
    chosen = None
    chosen_score = None
    for rx, ry in resources:
        rx, ry = int(rx), int(ry)
        ds = man(sx, sy, rx, ry)
        do = man(ox, oy, rx, ry)
        # Prefer denying: opponent advantage, then overall closeness.
        denom = ds + do + 1
        adv = do - ds  # positive means opponent closer
        score = (-adv, do, denom)  # lexicographic
        if adv > 0:
            if chosen_score is None or score < chosen_score:
                chosen_score = score
                chosen = (rx, ry)
    if chosen is None:
        # No clear deny target; go for best resource by simple deterministic tie-break.
        bestd = None
        for rx, ry in resources:
            rx, ry = int(rx), int(ry)
            ds = man(sx, sy, rx, ry)
            key = (ds, rx, ry)
            if bestd is None or key < bestd[0]:
                bestd = (key, (rx, ry))
        chosen = bestd[1]

    tx, ty = chosen

    # Greedy step that improves distance to target, but also discourages walking into the opponent's reach next turn.
    # Compute opponent "reach" heuristic by measuring distance reduction potential.
    best = None
    for dx, dy in moves:
        nx, ny = sx + dx, sy + dy
        if blocked(nx, ny):
            continue
        ds_next = man(nx, ny, tx, ty)
        # Deny pressure: if this move also keeps opponent farther from the same target, it's better.
        do_now = man(ox, oy, tx, ty)
        # Opponent best response one step (deterministic min over valid moves).
        best_do = None
        for odx, ody in moves:
            px, py = ox + odx, oy + ody
            if blocked(px, py):
                continue
            d = man(px, py, tx, ty)
            if best_do is None or d < best_do:
                best_do = d
        if best_do is None:
            best_do = do_now
        # Lower ds_next and higher (best_do) to delay their capture.
        key = (ds_next, best_do, dx, dy)
        if best is None or key < best[0]:
            best = (key, [dx, dy])
    return best[1] if best else [0, 0]
