def choose_move(observation):
    sx, sy = observation.get("self_position", [0, 0])
    ox, oy = observation.get("opponent_position", [0, 0])
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    resources = observation.get("resources", []) or []
    obstacles = observation.get("obstacles", []) or []

    obst = set()
    for p in obstacles:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            obst.add((int(p[0]), int(p[1])))

    def inb(x, y):
        return 0 <= x < w and 0 <= y < h

    def dist(x1, y1, x2, y2):
        dx = x1 - x2
        if dx < 0:
            dx = -dx
        dy = y1 - y2
        if dy < 0:
            dy = -dy
        return dx + dy

    moves = [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 0), (0, 1), (1, -1), (1, 0), (1, 1)]
    if not resources:
        # Drift to corner opposite opponent with obstacle avoidance
        tx = 0 if ox > (w - 1) // 2 else (w - 1)
        ty = 0 if oy > (h - 1) // 2 else (h - 1)
        best_move = (0, 0)
        best_val = -10**9
        for dx, dy in moves:
            nx, ny = sx + dx, sy + dy
            if not inb(nx, ny) or (nx, ny) in obst:
                continue
            val = -(dist(nx, ny, tx, ty)) - 0.01 * dist(nx, ny, ox, oy)
            if val > best_val:
                best_val = val
                best_move = (dx, dy)
        return [best_move[0], best_move[1]]

    # Pick target resource where we have strongest advantage (opponent farther than us),
    # but if opponent is currently much closer to everything, choose the least-contested.
    best = None
    top_adv = -10**9
    for rx, ry in resources:
        ds = dist(sx, sy, rx, ry)
        do = dist(ox, oy, rx, ry)
        if do < ds:
            # still consider, but penalize if opponent is ahead
            pass
        # advantage + tie-breakers: urgency to us, and prefer farther from opponent
        adv = (do - ds) * 10.0 + (do - ds >= 2) * 5.0
        # small deterministic bias to avoid ties: favor lower (rx,ry)
        tie = -(ds) * 1.0 - (do) * 0.05 - (rx + ry) * 0.0001
        val = adv + tie
        if val > top_adv:
            top_adv = val
            best = (rx, ry, ds, do)

    if best is None:
        return [0, 0]
    rx, ry, ds, do = best

    # If opponent is ahead significantly even on best target, flip to least contested among resources
    if do - ds >= 1:
        min_contest = 10**9
        chosen = None
        for r2x, r2y in resources:
            ds2 = dist(sx, sy, r2x, r2y)
            do2 = dist(ox, oy, r2x, r2y)
            # contest measured by how close opponent is relative to us
            contest = do2 - ds2 + 0.5 * do2
            # keep some progress: don't pick too far for us
            if ds2 > ds + 4:
                contest += 6
            if contest < min_contest or (contest == min_contest and (r2x, r2y) < (chosen[0], chosen[1])):
                min_contest = contest
                chosen = (r2x, r2y, ds2, do2)
