def choose_move(observation):
    w = int(observation.get("grid_width", 8) or 8)
    h = int(observation.get("grid_height", 8) or 8)
    sx, sy = observation.get("self_position", (0, 0)) or (0, 0)
    ox, oy = observation.get("opponent_position", (0, 0)) or (0, 0)
    sx, sy, ox, oy = int(sx), int(sy), int(ox), int(oy)

    obstacles = set()
    for p in observation.get("obstacles", []) or []:
        if isinstance(p, (list, tuple)) and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

    resources = []
    for r in observation.get("resources", []) or []:
        if isinstance(r, (list, tuple)) and len(r) >= 2:
            x, y = int(r[0]), int(r[1])
            if 0 <= x < w and 0 <= y < h and (x, y) not in obstacles:
                resources.append((x, y))
    if not resources:
        return [0, 0]

    def cheb(ax, ay, bx, by):
        dx = ax - bx
        if dx < 0:
            dx = -dx
        dy = ay - by
        if dy < 0:
            dy = -dy
        return dx if dx > dy else dy

    def valid(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obstacles

    target = None
    best = -10**18
    # Prefer resources we can reach at least as fast, with a bonus for being the closest.
    for rx, ry in resources:
        myd = cheb(sx, sy, rx, ry)
        opd = cheb(ox, oy, rx, ry)
        # Higher is better. Deterministic tie-break later.
        val = (opd - myd) * 10 - myd
        if opd >= myd:
            val += 50
        # Slight preference for near-center resources (reduce symmetry stalls).
        cx, cy = (w - 1) / 2.0, (h - 1) / 2.0
        val -= int(abs(rx - cx) + abs(ry - cy) + 0.5)
        if val > best:
            best = val
            target = (rx, ry)
        elif val == best and target is not None:
            # Deterministic: lexicographically smaller target.
            if (rx, ry) < target:
                target = (rx, ry)

    tx, ty = target

    moves = []
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            nx, ny = sx + dx, sy + dy
            if valid(nx, ny):
                moves.append((dx, dy, nx, ny))

    if not moves:
        return [0, 0]

    curd = cheb(sx, sy, tx, ty)
    # Choose move that minimizes distance to target; break ties by preferring dx then dy toward target direction.
    bestm = None
    bestv = 10**18
    pref_dx = 0 if tx == sx else (1 if tx > sx else -1)
    pref_dy = 0 if ty == sy else (1 if ty > sy else -1)
    for dx, dy, nx, ny in moves:
        nd = cheb(nx, ny, tx, ty)
        # Penalize worsening, strongly discourage increasing distance.
        inc_pen = (nd - curd)
        v = nd * 100 + inc_pen * 1000
        # Bonus if move aligns with preferred direction components.
        if dx == pref_dx:
            v -= 2
        if dy == pref_dy:
            v -= 2
        # Deterministic tie-break by move ordering.
        if v < bestv:
            bestv = v
            bestm = (dx, dy)
        elif v == bestv and bestm is not None:
            if (dx, dy) < bestm:
                bestm = (dx, dy)

    return [int(bestm[0]), int(bestm[1])]
