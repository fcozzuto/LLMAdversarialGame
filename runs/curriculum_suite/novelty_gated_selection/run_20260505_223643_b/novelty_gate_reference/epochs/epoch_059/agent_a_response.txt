def choose_move(observation):
    w = int(observation.get("grid_width", 8))
    h = int(observation.get("grid_height", 8))
    sp = observation.get("self_position", [0, 0])
    op = observation.get("opponent_position", [w - 1, h - 1])
    sx, sy = int(sp[0]), int(sp[1])
    ox, oy = int(op[0]), int(op[1])

    obstacles = set()
    for p in observation.get("obstacles") or []:
        if p and len(p) >= 2:
            x, y = int(p[0]), int(p[1])
            if 0 <= x < w and 0 <= y < h:
                obstacles.add((x, y))

    resources = []
    for r in observation.get("resources") or []:
        if r and len(r) >= 2:
            x, y = int(r[0]), int(r[1])
            if 0 <= x < w and 0 <= y < h and (x, y) not in obstacles:
                resources.append((x, y))

    def cheb(a, b, c, d):
        dx = a - c
        if dx < 0: dx = -dx
        dy = b - d
        if dy < 0: dy = -dy
        return dx if dx > dy else dy

    def inside(x, y):
        return 0 <= x < w and 0 <= y < h and (x, y) not in obstacles

    if not resources:
        # drift to reduce opponent dominance
        tx, ty = 0, 0
        best = None
        bestv = -10**18
        for dx in (-1, 0, 1):
            for dy in (-1, 0, 1):
                nx, ny = sx + dx, sy + dy
                if inside(nx, ny):
                    v = -(abs(nx - ox) + abs(ny - oy)) - (abs(nx - tx) + abs(ny - ty))
                    if v > bestv:
                        bestv = v
                        best = (dx, dy)
        return list(best if best is not None else (0, 0))

    # Target: maximize how much closer we are than opponent, with slight preference for nearer overall
    best_t = resources[0]
    best_score = -10**18
    for (rx, ry) in resources:
        sd = cheb(sx, sy, rx, ry)
        od = cheb(ox, oy, rx, ry)
        # If opponent is closer, still allow if we're close behind but prefer "swingy" leads
        lead = od - sd
        score = lead * 10 - sd
        # Prefer resources that are not perfectly "diagonal mirror" aligned with opponent (reduces head-to-head)
        if (rx - ox) == (ry - oy) or (rx - ox) == -(ry - oy):
            score -= 1.5
        if score > best_score:
            best_score = score
            best_t = (rx, ry)

    rx, ry = best_t
    # Move: choose among valid deltas the one that most reduces distance to target, while avoiding stepping adjacent to obstacles
    deltas = [(-1,-1),(-1,0),(-1,1),(0,-1),(0,0),(0,1),(1,-1),(1,0),(1,1)]
    def obst_adj(x, y):
        c = 0
        for dx in (-1,0,1):
            for dy in (-1,0,1):
                if dx == 0 and dy == 0: 
                    continue
                if (x+dx, y+dy) in obstacles:
                    c += 1
        return c

    best_d = (0, 0)
    bestv = -10**18
    for dx, dy in deltas:
        nx, ny = sx + dx, sy + dy
        if not inside(nx, ny):
            continue
        nd = cheb(nx, ny, rx, ry)
        # Encourage progress and deny giving opponent too much closeness improvement
        opp_new = cheb(ox, oy, rx, ry)
        self_gain = cheb(sx, sy, rx, ry) - nd
        v = self_gain * 5 - nd
        v -= obst_adj(nx, ny) * 0.7
        # If stepping keeps us from being overtaken, add small bonus
        if self_gain > 0 and nd <= opp_new:
            v += 0.8
        # Deterministic tie-breaker: closer to target in manhattan, then prefer diagonal, then lower dx/dy ordering
        man = abs(nx - rx) + abs(ny - ry)
        v -= man * 0.01
        if v > bestv:
            bestv = v
            best_d = (dx, dy)

    return [int(best_d[0]), int(best_d[1])]
